    38→    38→def cmd_compile(args):
    39→    39→    """Compile an AICL source file into executable code with Proof of Origin."""
    40→    40→    source = read_source(args.source)
    41→    41→    compiler = Compiler(target_language=args.target)
    42→    42→    result = compiler.compile_to_file(source, args.output_dir, source_path=args.source)
    43→    43→
    44→    44→    if result.success:
    45→    45→        print("Compilation successful")
    46→    46→        print(f"  Stages completed: {', '.join(result.stages_completed)}")
    47→    47→        print(f"  Output directory: {args.output_dir}")
    48→    48→        print(f"  Main file: {os.path.join(args.output_dir, 'main.py')}")
    49→    49→        print(f"  Test file: {os.path.join(args.output_dir, 'test_main.py')}")
    50→    50→        print(f"  Architecture tree: {os.path.join(args.output_dir, 'architecture_tree.txt')}")
    51→    51→        print(f"  Proof of Origin: {os.path.join(args.output_dir, 'main.aicl-proof')}")
    52→    52→        print(f"  TODOs remaining: {result.todo_count}")
    53→    53→        print(f"  Fully compiled: {'Yes' if result.fully_compiled else 'No'}")
    54→    54→
    55→    55→        # Show audit coverage summary
    56→    56→        if result.provenance:
    57→    57→            audit = result.provenance.compute_audit_coverage()
    58→    58→            print(f"  Audit coverage: {audit['audit_coverage']:.1%} ({audit['auditable_artifacts']}/{audit['total_artifacts']} artifacts)")
    59→    59→
    60→    60→        # Show Proof of Origin status
    61→    61→        if result.proof:
    62→    62→            verification = result.proof.verify()
    63→    63→            print(f"  Proof of Origin: {'VALID' if verification['valid'] else 'INVALID'}")
    64→    64→
    65→    65→        if result.warnings:
    66→    66→            print(f"\n  Warnings ({len(result.warnings)}):")
    67→    67→            for w in result.warnings:
    68→    68→                print(f"    - {w}")
    69→    69→    else:
    70→    70→        print("Compilation failed")
    71→    71→        for e in result.errors:
    72→    72→            print(f"  Error: {e}")
    73→    73→        sys.exit(1)

    76→    76→def cmd_parse(args):

    86→    86→    print(f"  Constraints: {len(program.constraints)}")
    87→    87→    for c in program.constraints:

    89→    89→    print(f"  Risks: {len(program.risks)}")

   109→   109→        print(f"    - {b.name}")

   117→   117→    warnings = program.validate()

   119→   119→        print(f"\n  Warnings ({len(warnings)}):")
   120→   120→        for w in warnings:

   156→   156→def cmd_explain(args):
   157→   157→    """
   158→   158→    Explain the compilation of an AICL source file or Proof of Origin.
   159→   159→
   160→   160→    Shows the provenance chain for every generated piece of code,
   161→   161→    answering the question: "Why did the compiler generate this line?"
   162→   162→
   163→   163→    Two modes:
   164→   164→        1. From source: aicl explain pong.aicl
   165→   165→           Compiles the source and explains the compilation.
   166→   166→        2. From proof:  aicl explain --proof pong.aicl-proof
   167→   167→           Reads the Proof of Origin directly — no compiler needed.
   168→   168→           This proves the proof file is self-contained.
   169→   169→
   170→   170→    Examples:
   171→   171→        aicl explain pong.aicl                          # Full compilation trace
   172→   172→        aicl explain pong.aicl --behavior MovePaddle    # Specific behavior trace
   173→   173→        aicl explain pong.aicl --coverage               # Explicability coverage report
   174→   174→        aicl explain --proof output/main.aicl-proof     # Explain from proof file
   175→   175→    """
   176→   176→    # Mode 1: Read from proof file
   177→   177→    if args.proof or args.source.endswith('.aicl-proof'):
   178→   178→        proof_path = args.source
   179→   179→        if not os.path.exists(proof_path):
   180→   180→            print(f"Error: Proof file not found: {proof_path}")
   181→   181→            sys.exit(1)
   182→   182→
   183→   183→        proof = ProofOfOrigin.from_file(proof_path)
   184→   184→
   185→   185→        if args.coverage:
   186→   186→            print(proof.coverage_report())
   187→   187→        elif args.behavior:
   188→   188→            print(proof.explain_behavior(args.behavior))
   189→   189→        else:
   190→   190→            print(proof.explain())
   191→   191→
   192→   192→        # Show proof metadata
   193→   193→        print()
   194→   194→        print(f"Source: Proof of Origin ({proof_path})")
   195→   195→        print(f"Compiled: {proof.timestamp}")
   196→   196→        print(f"Compiler: AICL v{proof.compiler_version}")
   197→   197→        print(f"Provenance records: {len(proof.records)}")
   198→   198→        print(f"Artifacts: {len(proof.artifacts)}")
   199→   199→
   200→   200→        audit_cov = proof.audit_coverage.get('audit_coverage', 0)
   201→   201→        print(f"Audit coverage: {audit_cov:.1%}")
   202→   202→
   203→   203→        exp_cov = proof.explicability_coverage.get('coverage_ratio', 1.0)
   204→   204→        exp_total = proof.explicability_coverage.get('total_lines', 0)
   205→   205→        exp_accounted = proof.explicability_coverage.get('accounted_lines', 0)
   206→   206→        print(f"Explicability coverage: {exp_cov:.1%} ({exp_accounted}/{exp_total} lines)")
   207→   207→
   208→   208→        return
   209→   209→
   210→   210→    # Mode 2: Compile from source
   211→   211→    source = read_source(args.source)
   212→   212→    compiler = Compiler()
   213→   213→    result = compiler.compile(source)
   214→   214→
   215→   215→    if not result.success:
   216→   216→        print("Compilation failed — cannot explain")
   217→   217→        for e in result.errors:
   218→   218→            print(f"  Error: {e}")
   219→   219→        sys.exit(1)
   220→   220→
   221→   221→    if not result.provenance or not result.provenance.records:
   222→   222→        print("No provenance records available. Compilation may have been cached.")
   223→   223→        return
   224→   224→
   225→   225→    if args.coverage:
   226→   226→        # Show explicability coverage report
   227→   227→        print(result.provenance.explain_coverage(result.source_code, result.test_code))
   228→   228→    elif args.behavior:
   229→   229→        # Explain a specific behavior
   230→   230→        print(result.provenance.explain_behavior(args.behavior))
   231→   231→    else:
   232→   232→        # Full explanation
   233→   233→        print(result.provenance.explain())
   234→   234→
   235→   235→    # Show compilation summary
   236→   236→    print()
   237→   237→    print(f"Compilation: {'SUCCESS' if result.fully_compiled else 'PARTIAL'}")
   238→   238→    print(f"TODOs remaining: {result.todo_count}")
   239→   239→    stats = result.provenance.get_statistics()
   240→   240→    print(f"Patterns used: {', '.join(stats.get('patterns_used', []))}")
   241→   241→    print(f"Provenance records: {stats.get('total', 0)}")
   242→   242→
   243→   243→    # Always show audit coverage
   244→   244→    audit = result.provenance.compute_audit_coverage()
   245→   245→    print(f"Audit coverage: {audit['audit_coverage']:.1%} ({audit['auditable_artifacts']}/{audit['total_artifacts']} artifacts)")
   246→   246→
   247→   247→    # Show line-level coverage
   248→   248→    coverage = result.provenance.compute_explicability_coverage(result.source_code, result.test_code)
   249→   249→    print(f"Explicability coverage: {coverage['coverage_ratio']:.1%} ({coverage['accounted_lines']}/{coverage['total_lines']} lines)")
   250→   250→

   252→   252→def cmd_audit(args):
   253→   253→    """
   254→   254→    Audit the compilation of an AICL source file or Proof of Origin.
   255→   255→
   256→   256→    Verifies the Auditability property:
   257→   257→        "A program is auditable if every generated artifact
   258→   258→         can be traced to its originating specification
   259→   259→         through a complete provenance chain."
   260→   260→
   261→   261→    Two modes:
   262→   262→        1. From source: aicl audit pong.aicl
   263→   263→           Compiles and audits.
   264→   264→        2. From proof:  aicl audit --proof pong.aicl-proof
   265→   265→           Audits the Proof of Origin directly — no compiler needed.
   266→   266→
   267→   267→    The audit computes:
   268→   268→        - Audit Coverage = Auditable Artifacts / Generated Artifacts
   269→   269→        - Orphan Artifacts = Artifacts without provenance
   270→   270→        - Audit Status = VERIFIED / PARTIAL / FAILED
   271→   271→        - Proof Verification = hash binding, linkage integrity
   272→   272→
   273→   273→    Examples:
   274→   274→        aicl audit pong.aicl               # Full audit report
   275→   275→        aicl audit pong.aicl --strict       # Fail (exit 1) if coverage < 100%
   276→   276→        aicl audit --proof output/main.aicl-proof  # Audit from proof file
   277→   277→    """
   278→   278→    # Mode 1: Read from proof file
   279→   279→    if args.proof or args.source.endswith('.aicl-proof'):
   280→   280→        proof_path = args.source
   281→   281→        if not os.path.exists(proof_path):
   282→   282→            print(f"Error: Proof file not found: {proof_path}")
   283→   283→            sys.exit(1)
   284→   284→
   285→   285→        proof = ProofOfOrigin.from_file(proof_path)
   286→   286→
   287→   287→        # Print the audit report reconstructed from proof
   288→   288→        print(proof.audit_report())
   289→   289→
   290→   290→        # Strict mode: fail if audit doesn't pass
   291→   291→        if args.strict:
   292→   292→            if not proof.audit_passed():
   293→   293→                print()
   294→   294→                print("STRICT MODE: Audit did not pass.")
   295→   295→                coverage_val = proof.audit_coverage.get('audit_coverage', 0)
   296→   296→                if coverage_val < 1.0:
   297→   297→                    print(f"  Audit coverage: {coverage_val:.1%} (target: 100%)")
   298→   298→                verification = proof.verify()
   299→   299→                for check in verification['checks']:
   300→   300→                    if not check['passed']:
   301→   301→                        print(f"  Failed check: {check['name']}: {check['description']}")
   302→   302→                sys.exit(1)
   303→   303→            else:
   304→   304→                print()
   305→   305→                print("STRICT MODE: Audit PASSED. Coverage is 100%. Proof is valid.")
   306→   306→
   307→   307→        return
   308→   308→
   309→   309→    # Mode 2: Compile from source
   310→   310→    source = read_source(args.source)
   311→   311→    compiler = Compiler()
   312→   312→    result = compiler.compile(source)
   313→   313→
   314→   314→    if not result.success:
   315→   315→        print("Compilation failed — cannot audit")
   316→   316→        for e in result.errors:
   317→   317→            print(f"  Error: {e}")
   318→   318→        sys.exit(1)
   319→   319→
   320→   320→    if not result.provenance:
   321→   321→        print("No provenance data available.")
   322→   322→        sys.exit(1)
   323→   323→
   324→   324→    # Generate and print the full audit report
   325→   325→    report = result.provenance.audit(result.source_code, result.test_code)
   326→   326→    print(report)
   327→   327→
   328→   328→    # Strict mode: fail if audit doesn't pass
   329→   329→    if args.strict:
   330→   330→        if not result.provenance.audit_passed(result.source_code, result.test_code):
   331→   331→            print()
   332→   332→            print("STRICT MODE: Audit did not pass.")
   333→   333→            audit = result.provenance.compute_audit_coverage()
   334→   334→            if audit['audit_coverage'] < 1.0:
   335→   335→                print(f"  Audit coverage: {audit['audit_coverage']:.1%} (target: 100%)")
   336→   336→                print(f"  Orphan artifacts: {len(audit['orphan_artifacts'])}")
   337→   337→                for orphan in audit['orphan_artifacts']:
   338→   338→                    print(f"    - [{orphan['type']}] {orphan['name']}")
   339→   339→            sys.exit(1)
   340→   340→        else:
   341→   341→            print()
   342→   342→            print("STRICT MODE: Audit PASSED. Coverage is 100%.")
   343→   343→

   345→   345→def cmd_proof(args):
   346→   346→    """
   347→   347→    Inspect and verify a Proof of Origin file.
   348→   348→
   349→   349→    The Proof of Origin is the central artifact of auditable compilation.
   350→   350→    It is a self-contained, verifiable record that proves:
   351→   351→        - Every generated artifact has a traceable provenance chain
   352→   352→        - No artifact exists without justification (No Orphan Property)
   353→   353→        - Audit coverage = 1.0 (Complete Coverage Property)
   354→   354→        - The proof is cryptographically bound to the generated code
   355→   355→
   356→   356→    Examples:
   357→   357→        aicl proof output/main.aicl-proof              # Inspect proof
   358→   358→        aicl proof output/main.aicl-proof --verify     # Verify proof integrity
   359→   359→        aicl proof output/main.aicl-proof --explain    # Show explanation from proof
   360→   360→        aicl proof output/main.aicl-proof --audit      # Show audit report from proof
   361→   361→    """
   362→   362→    proof_path = args.proof_file
   363→   363→    if not os.path.exists(proof_path):
   364→   364→        print(f"Error: Proof file not found: {proof_path}")
   365→   365→        sys.exit(1)
   366→   366→
   367→   367→    proof = ProofOfOrigin.from_file(proof_path)
   368→   368→
   369→   369→    # Default: show proof summary
   370→   370→    if not args.verify and not args.explain and not args.audit:
   371→   371→        # Show proof summary
   372→   372→        print("=" * 70)
   373→   373→        print("AICL PROOF OF ORIGIN")
   374→   374→        print("=" * 70)
   375→   375→        print()
   376→   376→        print(f"  File:           {proof_path}")
   377→   377→        print(f"  Format version: {proof.format_version}")
   378→   378→        print(f"  Compiler:       AICL v{proof.compiler_version}")
   379→   379→        print(f"  Timestamp:      {proof.timestamp}")
   380→   380→        print(f"  Source:         {proof.source_path or '<unknown>'}")
   381→   381→        print(f"  Target:         {proof.target_language}")
   382→   382→        print()
   383→   383→        print(f"  Source hash:    {proof.source_hash[:32]}...")
   384→   384→        print(f"  Program hash:   {proof.program_hash[:32]}...")
   385→   385→        print(f"  Test hash:      {proof.test_hash[:32]}...")
   386→   386→        print()
   387→   387→        print(f"  Provenance records: {len(proof.records)}")
   388→   388→        print(f"  Generated artifacts: {len(proof.artifacts)}")
   389→   389→        print()
   390→   390→
   391→   391→        # Formal properties
   392→   392→        print("FORMAL PROPERTIES")
   393→   393→        print("-" * 70)
   394→   394→        for prop_name, prop_data in proof.formal_properties.items():
   395→   395→            label = prop_name.replace("_", " ").title()
   396→   396→            verified = prop_data.get('verified', False)
   397→   397→            status = "VERIFIED" if verified else "FAILED"
   398→   398→            print(f"  {label}: {status}")
   399→   399→            if 'statement' in prop_data:
   400→   400→                print(f"    {prop_data['statement']}")
   401→   401→        print()
   402→   402→
   403→   403→        # Audit summary
   404→   404→        audit_summary = proof.audit_coverage
   405→   405→        total = audit_summary.get('total_artifacts', len(proof.artifacts))
   406→   406→        auditable = audit_summary.get('auditable_artifacts',
   407→   407→            sum(1 for a in proof.artifacts if not a.get('is_orphan', True)))
   408→   408→        coverage = audit_summary.get('audit_coverage',
   409→   409→            auditable / total if total > 0 else 1.0)
   410→   410→        orphans = len(audit_summary.get('orphan_artifacts', []))
   411→   411→
   412→   412→        print("AUDIT SUMMARY")
   413→   413→        print("-" * 70)
   414→   414→        print(f"  Artifacts:      {total}")
   415→   415→        print(f"  Auditable:      {auditable}")
   416→   416→        print(f"  Orphans:        {orphans}")
   417→   417→        print(f"  Audit Coverage: {coverage:.1%}")
   418→   418→        print(f"  Audit Status:   {'VERIFIED' if coverage >= 1.0 else 'FAILED'}")
   419→   419→        print()
   420→   420→
   421→   421→        # Verification
   422→   422→        verification = proof.verify()
   423→   423→        print("VERIFICATION")
   424→   424→        print("-" * 70)
   425→   425→        for check in verification['checks']:
   426→   426→            status = "PASS" if check['passed'] else "FAIL"
   427→   427→            print(f"  [{status:4s}] {check['name']}: {check['description']}")
   428→   428→        print()
   429→   429→
   430→   430→        print("=" * 70)
   431→   431→        if verification['valid'] and coverage >= 1.0:
   432→   432→            print("PROOF STATUS: VALID")
   433→   433→            print("  This Proof of Origin is valid and demonstrates full auditability.")
   434→   434→        else:
   435→   435→            print("PROOF STATUS: INVALID")
   436→   436→            print("  This Proof of Origin has verification failures.")
   437→   437→        print("=" * 70)
   438→   438→        return
   439→   439→
   440→   440→    # --verify: detailed verification
   441→   441→    if args.verify:

   443→   443→        print("=" * 70)

   449→   449→            print(f"  [{status:4s}] {check['name']}")
   450→   450→            print(f"         {check['description']}")

   497→   497→def cmd_verify(args):
   498→   498→    """
   499→   499→    Verify an AICL specification for completeness, coherence, and satisfaction.
   500→   500→
   501→   501→    Runs three levels of verification:
   502→   502→        1. Completeness: all required elements present
   503→   503→        2. Coherence: no contradictions or dangling references
   504→   504→        3. Satisfaction: specification is implementable
   505→   505→
   506→   506→    Examples:
   507→   507→        aicl verify pong.aicl                    # All three checks
   508→   508→        aicl verify pong.aicl --completeness     # Completeness only
   509→   509→        aicl verify pong.aicl --coherence        # Coherence only
   510→   510→        aicl verify pong.aicl --satisfaction     # Satisfaction only
   511→   511→    """
   512→   512→    source = read_source(args.source)
   513→   513→    from aicl.parser import Parser as P
   514→   514→    parser = P(source)
   515→   515→    program = parser.parse()
   516→   516→    verifier = SpecificationVerifier(program)
   517→   517→
   518→   518→    if args.completeness:
   519→   519→        report = verifier.completeness_only(source_name=args.source)
   520→   520→    elif args.coherence:
   521→   521→        report = verifier.coherence_only(source_name=args.source)
   522→   522→    elif args.satisfaction:
   523→   523→        report = verifier.satisfaction_only(source_name=args.source)
   524→   524→    else:
   525→   525→        report = verifier.verify(source_name=args.source)
   526→   526→
   527→   527→    print(report.summary())
   528→   528→
   529→   529→    if not report.passed:
   530→   530→        sys.exit(1)

   538→   538→        - Layer consolidation
   539→   539→        - Entity extraction
   540→   540→        - Behavior inlining
   541→   541→        - Risk distribution
   542→   542→        - Parallelization
   543→   543→        - Dependency reduction
   544→   544→        - Validation strengthening
   545→   545→
   546→   546→    Every optimization is recorded in the provenance chain.
   547→   547→
   548→   548→    Examples:
   549→   549→        aicl optimize pong.aicl
   550→   550→        aicl optimize pong.aicl --max-iterations 5
   551→   551→    """
   552→   552→    source = read_source(args.source)
   553→   553→    from aicl.parser import Parser as P
   554→   554→    parser = P(source)
   555→   555→    program = parser.parse()
   556→   556→
   557→   557→    from aicl.auto_optimizer import ArchitectureOptimizer
   558→   558→    optimizer = ArchitectureOptimizer(program)
   559→   559→    result = optimizer.optimize(max_iterations=args.max_iterations)
   560→   560→
