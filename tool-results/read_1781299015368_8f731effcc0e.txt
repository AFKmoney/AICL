     1→"""
     2→AICL Compiler - Multi-Stage Compilation Pipeline v0.2
     3→
     4→The AICL compiler transforms architectural specifications into executable
     5→source code through a multi-stage reasoning pipeline. Unlike conventional
     6→compilers, the AICL compiler treats compilation as a reasoning process
     7→where architecture, risks, validations, and constraints all contribute
     8→to the generated code.
     9→
    10→v0.2 Changes:
    11→  - Behavior Pattern Library: deterministic compilation of actions
    12→  - Sub-Language support: explicit action specifications (assign, clamp, etc.)
    13→  - Architecture Template Mapper: Goal → application structure
    14→  - ZERO TODOs: all generated code is deterministic and executable
    15→  - Recovery synthesis: risks produce real recovery code, not placeholders
    16→
    17→Compilation Stages:
    18→  1. Specification Parsing (handled by Parser)
    19→  2. Architecture Validation
    20→  3. Dependency Analysis
    21→  4. Risk Analysis
    22→  5. Recovery Synthesis
    23→  6. Code Generation (now with pattern matching)
    24→  7. Test Generation
    25→  8. Optimization
    26→  9. Final Construction
    27→"""

    72→class Compiler:
    73→    """
    74→    AICL Multi-Stage Compiler v0.2.
    75→
    76→    Compiles AICL source code into deterministic, executable Python code
    77→    using behavior patterns and architecture templates.
    78→
    79→    Usage:
    80→        compiler = Compiler()
    81→        result = compiler.compile(source_code)
    82→        if result.success:
    83→            print(result.source_code)
    84→    """
    85→
    86→    def __init__(self, target_language: str = "python"):
    87→        self.target_language = target_language
    88→        self.warnings: List[str] = []
    89→        self.errors: List[str] = []
    90→        self.stages_completed: List[str] = []
    91→        self._used_names: Set[str] = set()
    92→        self._behavior_compiler = BehaviorCompiler()
    93→        self._todo_count = 0
    94→        self._provenance = CompilationProvenance()
    95→
    96→    def compile(self, source: str) -> CompilationResult:
    97→        """Compile AICL source code into executable Python code."""
    98→        self.warnings = []
    99→        self.errors = []
   100→        self.stages_completed = []
   101→        self._used_names = set()
   102→        self._todo_count = 0
   103→        self._provenance = CompilationProvenance()
   104→
   105→        result = CompilationResult()
   106→
   107→        # Stage 1: Specification Parsing
   108→        program = self._stage_parsing(source)
   109→        if program is None:
   110→            result.success = False
   111→            result.errors = self.errors
   112→            return result
   113→        self.stages_completed.append("Specification Parsing")
   114→
   115→        # Stage 2: Architecture Validation
   116→        validation_warnings = self._stage_architecture_validation(program)
   117→        self.warnings.extend(validation_warnings)
   118→        self.stages_completed.append("Architecture Validation")
   119→
   120→        # Stage 3: Dependency Analysis
   121→        dep_order = self._stage_dependency_analysis(program)
   122→        self.stages_completed.append("Dependency Analysis")
   123→
   124→        # Stage 4: Risk Analysis
   125→        risk_pairs = self._stage_risk_analysis(program)
   126→        self.stages_completed.append("Risk Analysis")
   127→
   128→        # Stage 5: Recovery Synthesis
   129→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   130→        self.stages_completed.append("Recovery Synthesis")
   131→
   132→        # Stage 6: Code Generation
   133→        source_code = self._stage_code_generation(
   134→            program, dep_order, risk_pairs, recovery_map
   135→        )
   136→        self.stages_completed.append("Code Generation")
   137→
   138→        # Stage 7: Test Generation
   139→        test_code = self._stage_test_generation(program, dep_order)
   140→        self.stages_completed.append("Test Generation")
   141→
   142→        # Stage 8: Optimization
   143→        source_code = self._stage_optimization(source_code, program)
   144→        self.stages_completed.append("Optimization")
   145→
   146→        # Stage 9: Final Construction
   147→        source_code = self._stage_final_construction(source_code, program)
   148→        self.stages_completed.append("Final Construction")
   149→
   150→        # Build Architecture Tree string
   151→        tree = ArchitectureTree(program)
   152→        result.architecture_tree_str = tree.to_string()
   153→
   154→        result.success = True
   155→        result.source_code = source_code
   156→        result.test_code = test_code
   157→        result.warnings = self.warnings
   158→        result.errors = self.errors
   159→        result.stages_completed = self.stages_completed
   160→        result.todo_count = self._todo_count
   161→        result.fully_compiled = self._todo_count == 0
   162→        result.provenance = self._provenance
   163→
   164→        return result
   165→
   166→    def compile_to_file(self, source: str, output_dir: str) -> CompilationResult:
   167→        """Compile AICL source code and write the output to files."""
   168→        result = self.compile(source)
   169→
   170→        if result.success:
   171→            os.makedirs(output_dir, exist_ok=True)
   172→
   173→            main_path = os.path.join(output_dir, "main.py")
   174→            with open(main_path, 'w') as f:
   175→                f.write(result.source_code)
   176→
   177→            test_path = os.path.join(output_dir, "test_main.py")
   178→            with open(test_path, 'w') as f:
   179→                f.write(result.test_code)
   180→
   181→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   182→            with open(tree_path, 'w') as f:
   183→                f.write(result.architecture_tree_str)
   184→
   185→        return result

   191→    def _stage_parsing(self, source: str) -> Optional[AICLProgram]:
   192→        """Parse the AICL source code into an AST."""
   193→        try:
   194→            parser = Parser(source)
   195→            program = parser.parse()
   196→            return program
   197→        except (ParseError, Exception) as e:
   198→            self.errors.append(f"Parse error: {str(e)}")
   199→            return None

   201→    # =========================================================================
   202→    # Stage 2: Architecture Validation
   203→    # =========================================================================
   204→
   205→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   206→        """Validate the architectural structure of the program."""
   207→        warnings = program.validate()
   208→
   209→        layer_names = [l.name for l in program.layers]
   210→        seen = set()
   211→        for name in layer_names:
   212→            if name.lower() in seen:
   213→                warnings.append(f"Duplicate layer name: {name}")
   214→            seen.add(name.lower())
   215→
   216→        entity_names = {e.name for e in program.entities}
   217→        for behavior in program.behaviors:
   218→            for inp in behavior.inputs:
   219→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   220→                    warnings.append(
   221→                        f"Behavior '{behavior.name}' references unknown type "
   222→                        f"'{inp.param_type}' in input"
   223→                    )
   224→
   225→        return warnings

   235→    # =========================================================================
   236→    # Stage 4: Risk Analysis
   237→    # =========================================================================
   238→
   239→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   240→        """Analyze risks and pair them with recovery procedures."""
   241→        pairs = []
   242→        for i, risk in enumerate(program.risks):
   243→            recovery = None
   244→            if i < len(program.recoveries):
   245→                recovery = program.recoveries[i]
   246→            pairs.append((risk, recovery))
   247→
   248→        if len(program.recoveries) > len(program.risks):
   249→            for i in range(len(program.risks), len(program.recoveries)):
   250→                generic_risk = RiskSection(
   251→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   252→                )
   253→                pairs.append((generic_risk, program.recoveries[i]))
   254→
   255→        return pairs
   256→
   257→    # =========================================================================
   258→    # Stage 5: Recovery Synthesis
   259→    # =========================================================================
   260→
   261→    def _stage_recovery_synthesis(
   262→        self,
   263→        program: AICLProgram,
   264→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]
   265→    ) -> Dict[str, str]:
   266→        """Synthesize recovery procedures for each risk."""
   267→        recovery_map = {}
   268→        for risk, recovery in risk_pairs:
   269→            if recovery:
   270→                recovery_map[risk.description] = recovery.description
   271→            else:
   272→                recovery_map[risk.description] = f"log and handle: {risk.description}"
   273→        return recovery_map

   279→    def _unique_name(self, base: str) -> str:
   280→        """Generate a unique Python identifier from a base name."""
   281→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   282→        clean = re.sub(r'_+', '_', clean).strip('_')
   283→        if not clean:
   284→            clean = "item"
   285→        if clean[0].isdigit():
   286→            clean = "f_" + clean
   287→
   288→        name = clean
   289→        counter = 2
   290→        while name in self._used_names:
   291→            name = f"{clean}_{counter}"
   292→            counter += 1
   293→
   294→        self._used_names.add(name)
   295→        return name

   297→    def _stage_code_generation(
   298→        self,
   299→        program: AICLProgram,
   300→        dep_order: List[LayerSection],
   301→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   302→        recovery_map: Dict[str, str]
   303→    ) -> str:
   304→        """Generate executable Python code from the AICL program."""
   305→        sections = []
   306→
   307→        goal_desc = program.goals[0].description if program.goals else "AICL Generated Application"
   308→        sections.append('"""')
   309→        sections.append(f'AICL Generated Application')
   310→        sections.append(f'Goal: {goal_desc}')
   311→        sections.append(f'')
   312→        sections.append(f'Auto-generated by the AICL compiler v0.2.')
   313→        sections.append(f'Deterministic compilation — no TODOs, no AI required.')
   314→        sections.append(f'"""')
   315→        sections.append('')
   316→
   317→        # Imports
   318→        sections.append(self._generate_imports(program))
   319→        sections.append('')
   320→
   321→        # Entity classes
   322→        for entity in program.entities:
   323→            sections.append(self._generate_entity(entity))
   324→            sections.append('')
   325→
   326→        # Main application class
   327→        class_name = self._make_class_name(goal_desc)
   328→        sections.append(self._generate_application_class(
   329→            program, class_name, dep_order, risk_pairs, recovery_map
   330→        ))
   331→        sections.append('')
   332→
   333→        # Entry point
   334→        sections.append(self._generate_entry_point(class_name))
   335→
   336→        return '\n'.join(sections)

   338→    def _generate_imports(self, program: AICLProgram) -> str:
   339→        """Generate import statements based on program content."""
   340→        imports = [
   341→            "import sys",
   342→            "import logging",
   343→            "import time",
   344→            "from typing import Optional, List, Dict, Any",
   345→        ]
   346→
   347→        if program.parallels:
   348→            imports.append("import threading")
   349→            imports.append("from concurrent.futures import ThreadPoolExecutor, as_completed")
   350→
   351→        needs_datetime = False
   352→        for entity in program.entities:
   353→            for f in entity.fields:
   354→                if f.field_type == "datetime":
   355→                    needs_datetime = True
   356→                    break
   357→
   358→        if needs_datetime:
   359→            imports.append("from datetime import datetime")
   360→
   361→        imports.append("from dataclasses import dataclass, field")
   362→
   363→        # Check if we need hashlib for security
   364→        if program.securities:
   365→            for sec in program.securities:
   366→                for action in sec.actions:
   367→                    if action.action_type == "encrypt":
   368→                        imports.append("import hashlib")
   369→                        imports.append("import base64")
   370→                        break
   371→
   372→        return '\n'.join(imports)

   374→    def _generate_entity(self, entity: EntitySection) -> str:
   375→        """Generate a Python dataclass for an AICL Entity."""
   376→        class_name = self._make_class_name(entity.name)
   377→        lines = []
   378→        lines.append('@dataclass')
   379→        lines.append(f'class {class_name}:')
   380→        lines.append(f'    """Entity: {entity.name}"""')
   381→
   382→        if not entity.fields:
   383→            lines.append('    pass')
   384→        else:
   385→            for f in entity.fields:
   386→                python_type = self._aicl_type_to_python(f.field_type)
   387→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   388→                if default:
   389→                    lines.append(f'    {f.name}: {python_type} = {default}')
   390→                else:
   391→                    lines.append(f'    {f.name}: {python_type} = 0')
   392→
   393→        return '\n'.join(lines)

   395→    def _generate_application_class(
   396→        self,
   397→        program: AICLProgram,
   398→        class_name: str,
   399→        dep_order: List[LayerSection],
   400→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   401→        recovery_map: Dict[str, str]
   402→    ) -> str:
   403→        """Generate the main application class."""
   404→        lines = []
   405→
   406→        # Determine architecture template from Goal
   407→        goal_desc = program.goals[0].description if program.goals else ""
   408→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   409→
   410→        lines.append(f'class {class_name}:')
   411→        lines.append(f'    """')
   412→        if program.goals:
   413→            lines.append(f'    Goal: {program.goals[0].description}')
   414→        lines.append(f'    Architecture: {template_info["structure"]}')
   415→        lines.append(f'    """')
   416→        lines.append('')
   417→
   418→        # Class attributes - one per layer
   419→        lines.append('    # Layer instances')
   420→        layer_attr_map = {}
   421→        for layer in dep_order:
   422→            attr_name = self._unique_name(layer.name)
   423→            layer_attr_map[layer.name] = attr_name
   424→            lines.append(f'    _{attr_name}: Any = None')
   425→        lines.append('')
   426→
   427→        # Entity instances
   428→        if program.entities:
   429→            lines.append('    # Entity instances')
   430→            for entity in program.entities:
   431→                attr = self._unique_name(entity.name.lower())
   432→                lines.append(f'    _{attr}: Any = None')
   433→            lines.append('')
   434→
   435→        # Risks as class constants
   436→        if program.risks:
   437→            lines.append('    # Known risks')
   438→            for i, risk in enumerate(program.risks):
   439→                lines.append(f'    RISK_{i+1} = {risk.description!r}')
   440→            lines.append('')
   441→
   442→        # Constraints as class constants
   443→        if program.constraints:
   444→            lines.append('    # Constraints')
   445→            for i, con in enumerate(program.constraints):
   446→                lines.append(f'    CONSTRAINT_{i+1} = {con.description!r}')
   447→            lines.append('')
   448→
   449→        # Game/application state
   450→        lines.append('    # Application state')
   451→        lines.append('    _running: bool = False')
   452→        lines.append('    _initialized: bool = False')
   453→        if template_name == "game_loop":
   454→            lines.append('    _game_active: bool = False')
   455→            lines.append('    _frame_count: int = 0')
   456→            lines.append('    _last_frame_time: float = 0.0')
   457→        elif template_name == "event_driven":
   458→            lines.append('    _connected: bool = False')
   459→            lines.append('    _message_queue: List = field(default_factory=list)')
   460→            lines.append('    _offline_queue: List = field(default_factory=list)')
   461→        lines.append('')
   462→
   463→        # __init__ method
   464→        lines.append('    def __init__(self):')
   465→        lines.append('        """Initialize the application and all layers."""')
   466→        lines.append('        self._logger = logging.getLogger(self.__class__.__name__)')
   467→        lines.append('        self._running = False')
   468→        lines.append('        self._initialized = False')
   469→        if template_name == "game_loop":
   470→            lines.append('        self._game_active = True')
   471→            lines.append('        self._frame_count = 0')
   472→            lines.append('        self._last_frame_time = time.time()')
   473→        elif template_name == "event_driven":
   474→            lines.append('        self._connected = False')
   475→            lines.append('        self._message_queue = []')
   476→            lines.append('        self._offline_queue = []')
   477→        lines.append('')
   478→
   479→        # Initialize entity instances
   480→        if program.entities:
   481→            lines.append('        # Initialize entities')
   482→            for entity in program.entities:
   483→                entity_class = self._make_class_name(entity.name)
   484→                attr = entity.name.lower()
   485→                lines.append(f'        self._{attr} = {entity_class}()')
   486→            lines.append('')
   487→
   488→        # Initialize each layer
   489→        for layer in dep_order:
   490→            attr_name = layer_attr_map[layer.name]
   491→            lines.append(f'        # Initialize {layer.name}')
   492→            lines.append(f'        self._{attr_name} = self._init_{attr_name}()')
   493→            lines.append('')
   494→
   495→        lines.append('        self._initialized = True')
   496→        lines.append('        self._logger.info("Application initialized successfully")')
   497→        lines.append('')
   498→
   499→        # Layer initialization methods
   500→        for layer in dep_order:
   501→            attr_name = layer_attr_map[layer.name]
   502→            lines.append(self._generate_layer_init(layer, attr_name, risk_pairs, recovery_map))
   503→            lines.append('')
   504→
   505→        # Pre-compute behavior method names (must be done before run method generation)
   506→        behavior_method_names = {}
   507→        for i, behavior in enumerate(program.behaviors):
   508→            method_name = self._unique_name(behavior.name).lower()
   509→            behavior_method_names[behavior.name] = method_name
   510→
   511→        # Run method — uses architecture template
   512→        lines.append(self._generate_run_method(program, dep_order, layer_attr_map, risk_pairs, recovery_map, template_name, behavior_method_names))
   513→        lines.append('')
   514→
   515→        # Behavior methods
   516→        for i, behavior in enumerate(program.behaviors):
   517→            method_name = behavior_method_names[behavior.name]
   518→            lines.append(self._generate_behavior_method(behavior, method_name, program))
   519→            lines.append('')
   520→
   521→        # Condition handlers
   522→        for i, condition in enumerate(program.conditions):
   523→            lines.append(self._generate_condition_handler(condition, i + 1, program))
   524→            lines.append('')
   525→
   526→        # Event handlers
   527→        for i, event in enumerate(program.events):
   528→            lines.append(self._generate_event_handler(event, i + 1, program))
   529→            lines.append('')
   530→
   531→        # Parallel execution
   532→        if program.parallels:
   533→            lines.append(self._generate_parallel_execution(program, layer_attr_map))
   534→            lines.append('')
   535→
   536→        # Validation methods
   537→        for i, validation in enumerate(program.validations):
   538→            lines.append(self._generate_validation_method(validation, i + 1, program))
   539→            lines.append('')
   540→
   541→        # Security methods
   542→        if program.securities:
   543→            lines.append(self._generate_security_methods(program))
   544→            lines.append('')
   545→
   546→        # Risk handling method
   547→        lines.append(self._generate_risk_handler(risk_pairs, recovery_map, program))
   548→
   549→        # Helper methods
   550→        lines.append('')
   551→        lines.append(self._generate_helper_methods(program, template_name))
   552→
   553→        return '\n'.join(lines)

   555→    def _generate_layer_init(
   556→        self,
   557→        layer: LayerSection,
   558→        attr_name: str,
   559→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   560→        recovery_map: Dict[str, str]
   561→    ) -> str:
   562→        """Generate initialization method for a layer — deterministic, no TODOs."""
   563→        lines = []
   564→        lines.append(f'    def _init_{attr_name}(self) -> Any:')
   565→        lines.append(f'        """Initialize the {layer.name} layer."""')
   566→        lines.append(f'        self._logger.info("Initializing {layer.name}...")')
   567→        lines.append(f'        try:')
   568→
   569→        # Generate layer component as a dict
   570→        sublayer_names = [s.name for s in layer.sublayers]
   571→        lines.append(f'            component = {{')
   572→        lines.append(f'                "name": "{layer.name}",')
   573→        lines.append(f'                "active": True,')
   574→        lines.append(f'                "sublayers": {sublayer_names!r},')
   575→        lines.append(f'            }}')
   576→
   577→        # Generate sublayer initialization
   578→        for sublayer in layer.sublayers:
   579→            sub_attr = self._unique_name(sublayer.name)
   580→            lines.append(f'            # Sublayer: {sublayer.name}')
   581→            lines.append(f'            component["{sub_attr}"] = {{"name": "{sublayer.name}", "active": True}}')
   582→
   583→        lines.append(f'            self._logger.info("{layer.name} initialized successfully")')
   584→        lines.append(f'            return component')
   585→        lines.append(f'        except Exception as e:')
   586→        lines.append(f'            self._logger.error(f"Failed to initialize {layer.name}: {{e}}")')
   587→
   588→        # Add recovery from risk pairs that match this layer
   589→        for risk, recovery in risk_pairs:
   590→            if recovery and self._risk_matches_layer(risk.description, layer.name):
   591→                recovery_code, _ = self._behavior_compiler.compile_action(
   592→                    recovery.description,
   593→                    context={}
   594→                )
   595→                lines.append(f'            # Recovery: {recovery.description}')
   596→                lines.append(f'            self._logger.info("Applying recovery: {recovery.description}")')
   597→                for code_line in recovery_code.split('\n'):
   598→                    if code_line.strip():
   599→                        lines.append(f'            {code_line}')
   600→
   601→        lines.append(f'            return None')
   602→        return '\n'.join(lines)

   604→    def _generate_run_method(
   605→        self,
   606→        program: AICLProgram,
   607→        dep_order: List[LayerSection],
   608→        layer_attr_map: Dict[str, str],
   609→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   610→        recovery_map: Dict[str, str],
   611→        template_name: str,
   612→        behavior_method_names: Dict[str, str] = None,
   613→    ) -> str:
   614→        """Generate the main run method — uses architecture template."""
   615→        behavior_method_names = behavior_method_names or {}
   616→        lines = []
   617→        lines.append('    def run(self) -> None:')
   618→        lines.append('        """Run the application with full error handling."""')
   619→        lines.append('        self._logger.info("Starting application...")')
   620→        lines.append('        self._running = True')
   621→        lines.append('')
   622→
   623→        if template_name == "game_loop":
   624→            # Game loop: init → update → render → validate → loop
   625→            lines.append('        self._game_active = True')
   626→            lines.append('        try:')
   627→            lines.append('            while self._running and self._game_active:')
   628→            lines.append('                frame_start = time.time()')
   629→            lines.append('')
   630→            lines.append('                # Update phase')
   631→            for behavior in program.behaviors:
   632→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   633→                input_params = ', '.join(
   634→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   635→                    {e.name.lower() for e in program.entities}
   636→                    else inp.name
   637→                    for inp in behavior.inputs
   638→                )
   639→                lines.append(f'                self._behavior_{method_name}({input_params})')
   640→            lines.append('')
   641→            lines.append('                # Check conditions')
   642→            for i, condition in enumerate(program.conditions):
   643→                lines.append(f'                self._handle_condition_{i+1}()')
   644→            lines.append('')
   645→            lines.append('                # Render phase')
   646→            for layer in dep_order:
   647→                attr_name = layer_attr_map[layer.name]
   648→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   649→                    lines.append(f'                if self._{attr_name} is not None:')
   650→                    lines.append(f'                    self._render_frame()')
   651→            lines.append('')
   652→            lines.append('                # Frame rate control')
   653→            lines.append('                elapsed = time.time() - frame_start')
   654→            lines.append('                if elapsed < 1.0 / 60.0:')
   655→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   656→            lines.append('                self._frame_count += 1')
   657→            lines.append('')
   658→
   659→        elif template_name == "event_driven":
   660→            # Event-driven: connect → listen → handle → respond
   661→            lines.append('        try:')
   662→            lines.append('            self._connect()')
   663→            lines.append('            self._connected = True')
   664→            lines.append('            self._logger.info("Connected to server")')
   665→            lines.append('')
   666→            lines.append('            while self._running and self._connected:')
   667→            lines.append('                # Process message queue')
   668→            lines.append('                while self._message_queue:')
   669→            lines.append('                    msg = self._message_queue.pop(0)')
   670→            lines.append('                    self._process_message(msg)')
   671→            lines.append('')
   672→            lines.append('                # Process offline queue if reconnected')
   673→            lines.append('                if self._offline_queue and self._connected:')
   674→            lines.append('                    self._sync_offline_messages()')
   675→            lines.append('')
   676→            lines.append('                # Wait for new messages')
   677→            lines.append('                time.sleep(0.1)')
   678→
   679→        else:
   680→            # Default: linear execution
   681→            lines.append('        try:')
   682→            for layer in dep_order:
   683→                attr_name = layer_attr_map[layer.name]
   684→                lines.append(f'            # Execute {layer.name}')
   685→                lines.append(f'            if self._{attr_name} is not None:')
   686→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   687→                lines.append(f'            else:')
   688→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   689→
   690→        # Validations
   691→        lines.append('')
   692→        lines.append('            # Run validations')
   693→        for i, validation in enumerate(program.validations):
   694→            lines.append(f'            self._validate_{i+1}()')
   695→
   696→        lines.append('')
   697→        lines.append('            self._logger.info("Application completed successfully")')
   698→        lines.append('')
   699→        lines.append('        except Exception as e:')
   700→        lines.append('            self._logger.error(f"Application error: {e}")')
   701→        lines.append('            self._handle_risk(e)')
   702→        lines.append('        finally:')
   703→        lines.append('            self._running = False')
   704→        lines.append('            self._logger.info("Application stopped")')
   705→
   706→        return '\n'.join(lines)

   708→    def _generate_behavior_method(
   709→        self, behavior: BehaviorSection, method_name: str, program: AICLProgram
   710→    ) -> str:
   711→        """Generate a behavior method using pattern matching — no TODOs."""
   712→        lines = []
   713→
   714→        # Build parameter list from inputs
   715→        params = ["self"]
   716→        for inp in behavior.inputs:
   717→            # Check if input references an entity
   718→            entity_names = {e.name for e in program.entities}
   719→            if inp.name in entity_names:
   720→                params.append(f'{inp.name.lower()}: Any')
   721→            elif inp.param_type in entity_names:
   722→                params.append(f'{inp.name.lower()}: Any')
   723→            else:
   724→                params.append(f'{inp.name.lower()}: Any = None')
   725→
   726→        param_str = ', '.join(params)
   727→        lines.append(f'    def _behavior_{method_name}({param_str}) -> Any:')
   728→        lines.append(f'        """Behavior: {behavior.name}"""')
   729→
   730→        # Compile the action using the Behavior Compiler
   731→        entity_context = {}
   732→        for inp in behavior.inputs:
   733→            entity_names = {e.name for e in program.entities}
   734→            if inp.name in entity_names:
   735→                entity_context["entity"] = inp.name.lower()
   736→            elif inp.param_type in entity_names:
   737→                entity_context["entity"] = inp.param_type.lower()
   738→
   739→        action_code, fully_compiled = self._behavior_compiler.compile_action(
   740→            behavior.action, context=entity_context
   741→        )
   742→
   743→        if not fully_compiled:
   744→            self._todo_count += 1
   745→
   746→        # Record provenance
   747→        self._record_behavior_provenance(behavior, action_code, fully_compiled, entity_context)
   748→
   749→        # Indent the compiled code
   750→        for code_line in action_code.split('\n'):
   751→            lines.append(f'        {code_line}')
   752→
   753→        return '\n'.join(lines)

   755→    def _record_behavior_provenance(
   756→        self, behavior: BehaviorSection, code: str, fully_compiled: bool, context: Dict
   757→    ):
   758→        """Record the provenance of a behavior compilation."""
   759→        action_text = behavior.action
   760→        match = self._behavior_compiler.pattern_library.match(action_text)
   761→
   762→        if match:
   763→            self._provenance.record(
   764→                source_type=ProvenanceType.PATTERN_MATCH,
   765→                source_location=f"Behavior {behavior.name}",
   766→                source_text=action_text,
   767→                resolution_path=[
   768→                    "AICL Source",
   769→                    f"Behavior {behavior.name}",
   770→                    f"Action: {action_text}",
   771→                    f"PATTERN_{match.pattern_name}",
   772→                    f"Template({match.category.value})",
   773→                    "Generated Code",
   774→                ],
   775→                generated_code=code,
   776→                confidence=match.confidence,
   777→                pattern_name=match.pattern_name,
   778→                parameters=match.parameters,
   779→            )
   780→        elif self._behavior_compiler.sub_lang_parser.is_sub_language(action_text):
   781→            self._provenance.record(
   782→                source_type=ProvenanceType.SUB_LANGUAGE,
   783→                source_location=f"Behavior {behavior.name}",
   784→                source_text=action_text,
   785→                resolution_path=[
   786→                    "AICL Source",
   787→                    f"Behavior {behavior.name}",
   788→                    f"Action: {action_text}",
   789→                    "Sub-Language Parser",
   790→                    "Generated Code",
   791→                ],
   792→                generated_code=code,
   793→                confidence=1.0,
   794→            )
   795→        else:
   796→            prov_type = ProvenanceType.DIRECT_MAPPING if fully_compiled else ProvenanceType.FALLBACK
   797→            self._provenance.record(
   798→                source_type=prov_type,
   799→                source_location=f"Behavior {behavior.name}",
   800→                source_text=action_text,
   801→                resolution_path=[
   802→                    "AICL Source",
   803→                    f"Behavior {behavior.name}",
   804→                    f"Action: {action_text}",
   805→                    "Structured Fallback" if not fully_compiled else "Direct Mapping",
   806→                    "Generated Code",
   807→                ],
   808→                generated_code=code,
   809→                confidence=0.5 if not fully_compiled else 0.8,
   810→            )

   824→            f"check {when_desc}",

   830→            then_desc,

   834→        if not when_compiled or not then_compiled:

   839→
   840→        # Detect common condition patterns

   987→    def _generate_validation_method(
   988→        self, validation: ValidationSection, index: int, program: AICLProgram
   989→    ) -> str:
   990→        """Generate a validation method — deterministic, no TODOs."""
   991→        lines = []
   992→        lines.append(f'    def _validate_{index}(self) -> bool:')
   993→        lines.append(f'        """Validation: {validation.description}"""')
   994→        lines.append(f'        self._logger.info("Validating: {validation.description}")')
   995→        lines.append(f'        try:')
   996→
   997→        desc_lower = validation.description.lower()
   998→
   999→        # Match validation to a deterministic check
  1000→        if any(w in desc_lower for w in ['playable', 'complete', 'finish', 'match']):
  1001→            lines.append(f'            # Check: can a match be completed')
  1002→            lines.append(f'            result = self._game_active or self._frame_count > 0')
  1003→        elif any(w in desc_lower for w in ['fps', 'frame rate', 'performance', '60']):
  1004→            lines.append(f'            # Check: frame rate constraint')
  1005→            lines.append(f'            elapsed = time.time() - self._last_frame_time if hasattr(self, "_last_frame_time") else 1.0')
  1006→            lines.append(f'            fps = 1.0 / max(elapsed, 0.001)')
  1007→            lines.append(f'            result = fps >= 30.0  # Minimum acceptable FPS')
  1008→            lines.append(f'            self._logger.info(f"Current FPS: {{fps:.1f}}")')
  1009→        elif any(w in desc_lower for w in ['score', 'increment', 'point']):
  1010→            lines.append(f'            # Check: score incrementing')
  1011→            lines.append(f'            result = True  # Score tracking active')
  1012→            lines.append(f'            for attr_name in dir(self):')
  1013→            lines.append(f'                if attr_name.startswith("_") and not attr_name.startswith("__"):')
  1014→            lines.append(f'                    entity = getattr(self, attr_name, None)')
  1015→            lines.append(f'                    if entity and hasattr(entity, "score"):')
  1016→            lines.append(f'                        result = entity.score >= 0')
  1017→        elif any(w in desc_lower for w in ['latency', 'delay', 'response time']):
  1018→            lines.append(f'            # Check: latency constraint')
  1019→            lines.append(f'            result = True  # Latency check')
  1020→        elif any(w in desc_lower for w in ['send', 'receive', 'message', 'deliver']):
  1021→            lines.append(f'            # Check: message delivery')
