     1→     1→"""
     2→     2→AICL Compiler - Multi-Stage Compilation Pipeline
     3→     3→
     4→     4→The AICL compiler transforms architectural specifications into executable
     5→     5→source code through a multi-stage reasoning pipeline. Unlike conventional
     6→     6→compilers, the AICL compiler treats compilation as a reasoning process
     7→     7→where architecture, risks, validations, and constraints all contribute
     8→     8→to the generated code.
     9→     9→
    10→    10→Compilation Stages:
    11→    11→  1. Specification Parsing (handled by Parser)
    12→    12→  2. Architecture Validation
    13→    13→  3. Dependency Analysis
    14→    14→  4. Risk Analysis
    15→    15→  5. Recovery Synthesis
    16→    16→  6. Code Generation
    17→    17→  7. Test Generation
    18→    18→  8. Optimization
    19→    19→  9. Final Output Construction
    20→    20→"""

    22→    22→import os

    59→    59→class Compiler:
    60→    60→    """
    61→    61→    AICL Multi-Stage Compiler.
    62→    62→
    63→    63→    Compiles AICL source code into executable Python code.
    64→    64→    The compilation pipeline includes architecture validation,
    65→    65→    dependency analysis, risk analysis, recovery synthesis,
    66→    66→    code generation, test generation, and optimization.
    67→    67→
    68→    68→    Usage:
    69→    69→        compiler = Compiler()
    70→    70→        result = compiler.compile(source_code)
    71→    71→        if result.success:
    72→    72→            print(result.source_code)
    73→    73→    """
    74→    74→
    75→    75→    def __init__(self, target_language: str = "python"):
    76→    76→        self.target_language = target_language
    77→    77→        self.warnings: List[str] = []
    78→    78→        self.errors: List[str] = []
    79→    79→        self.stages_completed: List[str] = []
    80→    80→        self._used_names: Set[str] = set()
    81→    81→
    82→    82→    def compile(self, source: str) -> CompilationResult:
    83→    83→        """
    84→    84→        Compile AICL source code into executable Python code.
    85→    85→
    86→    86→        Args:
    87→    87→            source: AICL source code string.
    88→    88→
    89→    89→        Returns:
    90→    90→            CompilationResult containing the generated code and metadata.
    91→    91→        """
    92→    92→        self.warnings = []
    93→    93→        self.errors = []
    94→    94→        self.stages_completed = []
    95→    95→        self._used_names = set()
    96→    96→
    97→    97→        result = CompilationResult()
    98→    98→
    99→    99→        # Stage 1: Specification Parsing
   100→   100→        program = self._stage_parsing(source)
   101→   101→        if program is None:
   102→   102→            result.success = False
   103→   103→            result.errors = self.errors
   104→   104→            return result
   105→   105→        self.stages_completed.append("Specification Parsing")
   106→   106→
   107→   107→        # Stage 2: Architecture Validation
   108→   108→        validation_warnings = self._stage_architecture_validation(program)
   109→   109→        self.warnings.extend(validation_warnings)
   110→   110→        self.stages_completed.append("Architecture Validation")
   111→   111→
   112→   112→        # Stage 3: Dependency Analysis
   113→   113→        dep_order = self._stage_dependency_analysis(program)
   114→   114→        self.stages_completed.append("Dependency Analysis")
   115→   115→
   116→   116→        # Stage 4: Risk Analysis
   117→   117→        risk_pairs = self._stage_risk_analysis(program)
   118→   118→        self.stages_completed.append("Risk Analysis")
   119→   119→
   120→   120→        # Stage 5: Recovery Synthesis
   121→   121→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   122→   122→        self.stages_completed.append("Recovery Synthesis")
   123→   123→
   124→   124→        # Stage 6: Code Generation
   125→   125→        source_code = self._stage_code_generation(
   126→   126→            program, dep_order, risk_pairs, recovery_map
   127→   127→        )
   128→   128→        self.stages_completed.append("Code Generation")
   129→   129→
   130→   130→        # Stage 7: Test Generation
   131→   131→        test_code = self._stage_test_generation(program, dep_order)
   132→   132→        self.stages_completed.append("Test Generation")
   133→   133→
   134→   134→        # Stage 8: Optimization
   135→   135→        source_code = self._stage_optimization(source_code, program)
   136→   136→        self.stages_completed.append("Optimization")
   137→   137→
   138→   138→        # Stage 9: Final Construction
   139→   139→        source_code = self._stage_final_construction(source_code, program)
   140→   140→        self.stages_completed.append("Final Construction")
   141→   141→
   142→   142→        # Build Architecture Tree string
   143→   143→        tree = ArchitectureTree(program)
   144→   144→        result.architecture_tree_str = tree.to_string()
   145→   145→
   146→   146→        result.success = True
   147→   147→        result.source_code = source_code
   148→   148→        result.test_code = test_code
   149→   149→        result.warnings = self.warnings
   150→   150→        result.errors = self.errors
   151→   151→        result.stages_completed = self.stages_completed
   152→   152→
   153→   153→        return result

   155→   155→    def compile_to_file(self, source: str, output_dir: str) -> CompilationResult:

   157→   157→        Compile AICL source code and write the output to files.

   168→   168→        if result.success:
   169→   169→            os.makedirs(output_dir, exist_ok=True)
   170→   170→
   171→   171→            # Write main source file
   172→   172→            main_path = os.path.join(output_dir, "main.py")
   173→   173→            with open(main_path, 'w') as f:
   174→   174→                f.write(result.source_code)
   175→   175→
   176→   176→            # Write test file
   177→   177→            test_path = os.path.join(output_dir, "test_main.py")
   178→   178→            with open(test_path, 'w') as f:
   179→   179→                f.write(result.test_code)
   180→   180→
   181→   181→            # Write architecture tree
   182→   182→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   183→   183→            with open(tree_path, 'w') as f:
   184→   184→                f.write(result.architecture_tree_str)
   185→   185→
   186→   186→        return result

   192→   192→    def _stage_parsing(self, source: str) -> Optional[AICLProgram]:
   193→   193→        """Parse the AICL source code into an AST."""
   194→   194→        try:
   195→   195→            parser = Parser(source)
   196→   196→            program = parser.parse()
   197→   197→            return program
   198→   198→        except (ParseError, Exception) as e:
   199→   199→            self.errors.append(f"Parse error: {str(e)}")
   200→   200→            return None

   202→   202→    # =========================================================================
   203→   203→    # Stage 2: Architecture Validation
   204→   204→    # =========================================================================
   205→   205→
   206→   206→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   207→   207→        """Validate the architectural structure of the program."""
   208→   208→        warnings = program.validate()
   209→   209→
   210→   210→        # Check for duplicate layer names
   211→   211→        layer_names = [l.name for l in program.layers]
   212→   212→        seen = set()
   213→   213→        for name in layer_names:
   214→   214→            if name.lower() in seen:
   215→   215→                warnings.append(f"Duplicate layer name: {name}")
   216→   216→            seen.add(name.lower())
   217→   217→
   218→   218→        # Check entity field references
   219→   219→        entity_names = {e.name for e in program.entities}
   220→   220→        for behavior in program.behaviors:
   221→   221→            for inp in behavior.inputs:
   222→   222→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   223→   223→                    warnings.append(
   224→   224→                        f"Behavior '{behavior.name}' references unknown type "
   225→   225→                        f"'{inp.param_type}' in input"
   226→   226→                    )
   227→   227→
   228→   228→        return warnings

   238→   238→    # =========================================================================
   239→   239→    # Stage 4: Risk Analysis
   240→   240→    # =========================================================================
   241→   241→
   242→   242→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   243→   243→        """Analyze risks and pair them with recovery procedures."""
   244→   244→        pairs = []
   245→   245→        for i, risk in enumerate(program.risks):
   246→   246→            recovery = None
   247→   247→            if i < len(program.recoveries):
   248→   248→                recovery = program.recoveries[i]
   249→   249→            pairs.append((risk, recovery))
   250→   250→
   251→   251→        # Handle orphaned recoveries
   252→   252→        if len(program.recoveries) > len(program.risks):
   253→   253→            for i in range(len(program.risks), len(program.recoveries)):
   254→   254→                generic_risk = RiskSection(
   255→   255→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   256→   256→                )
   257→   257→                pairs.append((generic_risk, program.recoveries[i]))
   258→   258→
   259→   259→        return pairs
   260→   260→
   261→   261→    # =========================================================================
   262→   262→    # Stage 5: Recovery Synthesis
   263→   263→    # =========================================================================
   264→   264→
   265→   265→    def _stage_recovery_synthesis(
   266→   266→        self,
   267→   267→        program: AICLProgram,
   268→   268→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]
   269→   269→    ) -> Dict[str, str]:
   270→   270→        """Synthesize recovery procedures for each risk."""
   271→   271→        recovery_map = {}
   272→   272→        for risk, recovery in risk_pairs:
   273→   273→            if recovery:
   274→   274→                recovery_map[risk.description] = recovery.description
   275→   275→            else:
   276→   276→                recovery_map[risk.description] = f"log and handle: {risk.description}"
   277→   277→        return recovery_map

   283→   283→    def _unique_name(self, base: str) -> str:
   284→   284→        """Generate a unique Python identifier from a base name."""
   285→   285→        # Clean the base name
   286→   286→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   287→   287→        clean = re.sub(r'_+', '_', clean).strip('_')
   288→   288→        if not clean:
   289→   289→            clean = "item"
   290→   290→        if clean[0].isdigit():
   291→   291→            clean = "f_" + clean
   292→   292→
   293→   293→        # Make unique
   294→   294→        name = clean
   295→   295→        counter = 2
   296→   296→        while name in self._used_names:
   297→   297→            name = f"{clean}_{counter}"
   298→   298→            counter += 1
   299→   299→
   300→   300→        self._used_names.add(name)
   301→   301→        return name

   303→   303→    def _stage_code_generation(
   304→   304→        self,
   305→   305→        program: AICLProgram,
   306→   306→        dep_order: List[LayerSection],
   307→   307→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   308→   308→        recovery_map: Dict[str, str]
   309→   309→    ) -> str:
   310→   310→        """Generate executable Python code from the AICL program."""
   311→   311→        sections = []
   312→   312→
   313→   313→        # Module docstring
   314→   314→        goal_desc = program.goals[0].description if program.goals else "AICL Generated Application"
   315→   315→        sections.append('"""')
   316→   316→        sections.append(f'AICL Generated Application')
   317→   317→        sections.append(f'Goal: {goal_desc}')
   318→   318→        sections.append(f'')
   319→   319→        sections.append(f'This file was automatically generated by the AICL compiler.')
   320→   320→        sections.append(f'Do not edit manually - regenerate from .aicl source.')
   321→   321→        sections.append(f'"""')
   322→   322→        sections.append('')
   323→   323→
   324→   324→        # Imports
   325→   325→        sections.append(self._generate_imports(program))
   326→   326→        sections.append('')
   327→   327→
   328→   328→        # Entity classes
   329→   329→        for entity in program.entities:
   330→   330→            sections.append(self._generate_entity(entity))
   331→   331→            sections.append('')
   332→   332→
   333→   333→        # Main application class
   334→   334→        class_name = self._make_class_name(goal_desc)
   335→   335→        sections.append(self._generate_application_class(
   336→   336→            program, class_name, dep_order, risk_pairs, recovery_map
   337→   337→        ))
   338→   338→        sections.append('')
   339→   339→
   340→   340→        # Entry point
   341→   341→        sections.append(self._generate_entry_point(class_name))
   342→   342→
   343→   343→        return '\n'.join(sections)

   345→   345→    def _generate_imports(self, program: AICLProgram) -> str:
   346→   346→        """Generate import statements based on program content."""
   347→   347→        imports = [
   348→   348→            "import sys",
   349→   349→            "import logging",
   350→   350→            "import time",
   351→   351→            "from typing import Optional, List, Dict, Any",
   352→   352→        ]
   353→   353→
   354→   354→        # Check if we need threading for Parallel sections
   355→   355→        if program.parallels:
   356→   356→            imports.append("import threading")
   357→   357→            imports.append("from concurrent.futures import ThreadPoolExecutor, as_completed")
   358→   358→
   359→   359→        # Check if we need datetime for Entity fields
   360→   360→        needs_datetime = False
   361→   361→        for entity in program.entities:
   362→   362→            for f in entity.fields:
   363→   363→                if f.field_type == "datetime":
   364→   364→                    needs_datetime = True
   365→   365→                    break
   366→   366→
   367→   367→        if needs_datetime:
   368→   368→            imports.append("from datetime import datetime")
   369→   369→
   370→   370→        # Always need dataclass for entities
   371→   371→        imports.append("from dataclasses import dataclass, field")
   372→   372→
   373→   373→        return '\n'.join(imports)

   375→   375→    def _generate_entity(self, entity: EntitySection) -> str:
   376→   376→        """Generate a Python dataclass for an AICL Entity."""
   377→   377→        class_name = self._make_class_name(entity.name)
   378→   378→        lines = []
   379→   379→        lines.append('@dataclass')
   380→   380→        lines.append(f'class {class_name}:')
   381→   381→        lines.append(f'    """Entity: {entity.name}"""')
   382→   382→
   383→   383→        if not entity.fields:
   384→   384→            lines.append('    pass')
   385→   385→        else:
   386→   386→            for f in entity.fields:
   387→   387→                python_type = self._aicl_type_to_python(f.field_type)
   388→   388→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   389→   389→                if default:
   390→   390→                    lines.append(f'    {f.name}: {python_type} = {default}')
   391→   391→                else:
   392→   392→                    lines.append(f'    {f.name}: {python_type} = 0')
   393→   393→
   394→   394→        return '\n'.join(lines)

   396→   396→    def _generate_application_class(
   397→   397→        self,
   398→   398→        program: AICLProgram,
   399→   399→        class_name: str,
   400→   400→        dep_order: List[LayerSection],
   401→   401→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   402→   402→        recovery_map: Dict[str, str]
   403→   403→    ) -> str:
   404→   404→        """Generate the main application class."""
   405→   405→        lines = []
   406→   406→
   407→   407→        lines.append(f'class {class_name}:')
   408→   408→        lines.append(f'    """')
   409→   409→
   410→   410→        if program.goals:
   411→   411→            lines.append(f'    Goal: {program.goals[0].description}')
   412→   412→        lines.append(f'    """')
   413→   413→        lines.append('')
   414→   414→
   415→   415→        # Class attributes - one per layer with unique names
   416→   416→        lines.append('    # Layer instances')
   417→   417→        layer_attr_map = {}  # Map from layer name to attribute name
   418→   418→        for layer in dep_order:
   419→   419→            attr_name = self._unique_name(layer.name)
   420→   420→            layer_attr_map[layer.name] = attr_name
   421→   421→            lines.append(f'    _{attr_name}: Any = None')
   422→   422→        lines.append('')
   423→   423→
   424→   424→        # Risks as class constants
   425→   425→        if program.risks:
   426→   426→            lines.append('    # Known risks')
   427→   427→            for i, risk in enumerate(program.risks):
   428→   428→                const_name = self._unique_name(f"risk_{i+1}")
   429→   429→                lines.append(f'    RISK_{i+1} = {risk.description!r}')
   430→   430→            lines.append('')
   431→   431→
   432→   432→        # Constraints as class constants
   433→   433→        if program.constraints:
   434→   434→            lines.append('    # Constraints')
   435→   435→            for i, con in enumerate(program.constraints):
   436→   436→                lines.append(f'    CONSTRAINT_{i+1} = {con.description!r}')
   437→   437→            lines.append('')
   438→   438→
   439→   439→        # __init__ method
   440→   440→        lines.append('    def __init__(self):')
   441→   441→        lines.append('        """Initialize the application and all layers."""')
   442→   442→        lines.append('        self._logger = logging.getLogger(self.__class__.__name__)')
   443→   443→        lines.append('        self._running = False')
   444→   444→        lines.append('        self._initialized = False')
   445→   445→        lines.append('')
   446→   446→
   447→   447→        # Initialize each layer with error handling
   448→   448→        for layer in dep_order:
   449→   449→            attr_name = layer_attr_map[layer.name]
   450→   450→            lines.append(f'        # Initialize {layer.name}')
   451→   451→            lines.append(f'        self._{attr_name} = self._init_{attr_name}()')
   452→   452→            lines.append('')
   453→   453→
   454→   454→        lines.append('        self._initialized = True')
   455→   455→        lines.append('        self._logger.info("Application initialized successfully")')
   456→   456→        lines.append('')
   457→   457→
   458→   458→        # Layer initialization methods
   459→   459→        for layer in dep_order:
   460→   460→            attr_name = layer_attr_map[layer.name]
   461→   461→            lines.append(self._generate_layer_init(layer, attr_name, risk_pairs, recovery_map))
   462→   462→            lines.append('')
   463→   463→
   464→   464→        # Run method
   465→   465→        lines.append(self._generate_run_method(program, dep_order, layer_attr_map, risk_pairs, recovery_map))
   466→   466→        lines.append('')
   467→   467→
   468→   468→        # Validation methods
   469→   469→        for i, validation in enumerate(program.validations):
   470→   470→            lines.append(self._generate_validation_method(validation, i + 1))
   471→   471→            lines.append('')
   472→   472→
   473→   473→        # Condition handlers
   474→   474→        for i, condition in enumerate(program.conditions):
   475→   475→            lines.append(self._generate_condition_handler(condition, i + 1))
   476→   476→            lines.append('')
   477→   477→
   478→   478→        # Event handlers
   479→   479→        for i, event in enumerate(program.events):
   480→   480→            lines.append(self._generate_event_handler(event, i + 1))
   481→   481→            lines.append('')
   482→   482→
   483→   483→        # Parallel execution
   484→   484→        if program.parallels:
   485→   485→            lines.append(self._generate_parallel_execution(program, layer_attr_map))
   486→   486→            lines.append('')
   487→   487→
   488→   488→        # Security methods
   489→   489→        if program.securities:
   490→   490→            lines.append(self._generate_security_methods(program))
   491→   491→            lines.append('')
   492→   492→
   493→   493→        # Risk handling method
   494→   494→        lines.append(self._generate_risk_handler(risk_pairs, recovery_map))
   495→   495→
   496→   496→        return '\n'.join(lines)

   498→   498→    def _generate_layer_init(
   499→   499→        self,
