    54→    54→const AICL_KEYWORDS = [
    55→    55→  'Goal', 'Layer', 'Sublayer', 'Validation', 'Risk', 'Recovery',
    56→    56→  'Constraint', 'Entity', 'Behavior', 'Input', 'Output', 'Action',
    57→    57→  'Condition', 'When', 'Then', 'Event', 'On', 'Parallel', 'Optimize',

    59→    59→  'Protect', 'Native', 'Import'
    60→    60→];
    61→    61→
    62→    62→const AICL_TYPES = [
    63→    63→  'string', 'integer', 'float', 'boolean', 'datetime',
    64→    64→  'list', 'dict', 'set', 'any', 'void', 'bytes'
    65→    65→];
    66→    66→
    67→    67→function highlightAICL(code: string): string {
    68→    68→  const lines = code.split('\n');
    69→    69→  return lines.map(line => {
    70→    70→    // Comment
    71→    71→    const commentIdx = line.indexOf('#');
    72→    72→    if (commentIdx !== -1) {
    73→    73→      const before = line.substring(0, commentIdx);
    74→    74→      const comment = line.substring(commentIdx);
    75→    75→      return highlightLine(before) + `<span class="aicl-comment">${escapeHtml(comment)}</span>`;
    76→    76→    }
    77→    77→    return highlightLine(line);
    78→    78→  }).join('\n');
    79→    79→}
    80→    80→
    81→    81→function highlightLine(line: string): string {
    82→    82→  let result = escapeHtml(line);
    83→    83→
    84→    84→  // Keywords at start of line or after whitespace (word boundary)
    85→    85→  for (const kw of AICL_KEYWORDS) {
    86→    86→    const regex = new RegExp(`(^|\\s)(${kw})(?=[:\\s]|$)`, 'g');
    87→    87→    result = result.replace(regex, `$1<span class="aicl-keyword">${kw}</span>`);
    88→    88→  }
    89→    89→
    90→    90→  // Types
    91→    91→  for (const t of AICL_TYPES) {
    92→    92→    const regex = new RegExp(`(\\s|^)(${t})(?=\\s|:|$)`, 'g');
    93→    93→    result = result.replace(regex, `$1<span class="aicl-type">${t}</span>`);
    94→    94→  }
    95→    95→
    96→    96→  return result;
    97→    97→}
    98→    98→
    99→    99→function escapeHtml(str: string): string {
   100→   100→  return str
   101→   101→    .replace(/&/g, '&amp;')
   102→   102→    .replace(/</g, '&lt;')
   103→   103→    .replace(/>/g, '&gt;')
   104→   104→    .replace(/"/g, '&quot;');
   105→   105→}
