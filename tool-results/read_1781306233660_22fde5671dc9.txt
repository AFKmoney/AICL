     1→"""
     2→AICL Compiler - Multi-Stage Compilation Pipeline v0.4
     3→
     4→Architecture Compilation with Explicable Compilation.
     5→
     6→Every generated line of code must have a traceable provenance chain.
     7→If the compiler cannot explain why it generated a line, it should not
     8→generate it. This principle is enforced through the provenance system
     9→which records every compilation decision.
    10→
    11→v0.4 Changes:
    12→  - Full provenance coverage: every code generation point records provenance
    13→  - Explicability coverage metric: ratio of provenance records to generated lines
    14→  - Provenance for entities, layers, conditions, events, validations, recovery,
    15→    security, helpers, architecture template, and test generation
    16→  - aicl explain now reports coverage ratio and lists unaccounted lines
    17→
    18→Compilation Stages:
    19→  1. Specification Parsing (handled by Parser)
    20→  2. Architecture Validation
    21→  3. Dependency Analysis
    22→  4. Risk Analysis
    23→  5. Recovery Synthesis
    24→  6. Code Generation (with full provenance)
    25→  7. Test Generation (with provenance)
    26→  8. Optimization
    27→  9. Final Construction
    28→"""

    73→class Compiler:
    74→    """
    75→    AICL Multi-Stage Compiler v0.2.
    76→
    77→    Compiles AICL source code into deterministic, executable Python code
    78→    using behavior patterns and architecture templates.
    79→
    80→    Usage:
    81→        compiler = Compiler()
    82→        result = compiler.compile(source_code)
    83→        if result.success:
    84→            print(result.source_code)
    85→    """
    86→
    87→    def __init__(self, target_language: str = "python"):
    88→        self.target_language = target_language
    89→        self.warnings: List[str] = []
    90→        self.errors: List[str] = []
    91→        self.stages_completed: List[str] = []
    92→        self._used_names: Set[str] = set()
    93→        self._behavior_compiler = BehaviorCompiler()
    94→        self._todo_count = 0
    95→        self._provenance = CompilationProvenance()
    96→
    97→    def compile(self, source: str) -> CompilationResult:
    98→        """Compile AICL source code into executable Python code."""
    99→        self.warnings = []
   100→        self.errors = []
   101→        self.stages_completed = []
   102→        self._used_names = set()
   103→        self._todo_count = 0
   104→        self._provenance = CompilationProvenance()
   105→
   106→        result = CompilationResult()
   107→
   108→        # Stage 1: Specification Parsing
   109→        program = self._stage_parsing(source)
   110→        if program is None:
   111→            result.success = False
   112→            result.errors = self.errors
   113→            return result
   114→        self.stages_completed.append("Specification Parsing")
   115→
   116→        # Stage 2: Architecture Validation
   117→        validation_warnings = self._stage_architecture_validation(program)
   118→        self.warnings.extend(validation_warnings)
   119→        self.stages_completed.append("Architecture Validation")
   120→
   121→        # Stage 3: Dependency Analysis
   122→        dep_order = self._stage_dependency_analysis(program)
   123→        self.stages_completed.append("Dependency Analysis")
   124→
   125→        # Stage 4: Risk Analysis
   126→        risk_pairs = self._stage_risk_analysis(program)
   127→        self.stages_completed.append("Risk Analysis")
   128→
   129→        # Stage 5: Recovery Synthesis
   130→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   131→        self.stages_completed.append("Recovery Synthesis")
   132→
   133→        # Stage 6: Code Generation
   134→        source_code = self._stage_code_generation(
   135→            program, dep_order, risk_pairs, recovery_map
   136→        )
   137→        self.stages_completed.append("Code Generation")
   138→
   139→        # Stage 7: Test Generation
   140→        test_code = self._stage_test_generation(program, dep_order)
   141→        self.stages_completed.append("Test Generation")
   142→
   143→        # Stage 8: Optimization
   144→        source_code = self._stage_optimization(source_code, program)
   145→        self.stages_completed.append("Optimization")
   146→
   147→        # Stage 9: Final Construction
   148→        source_code = self._stage_final_construction(source_code, program)
   149→        self.stages_completed.append("Final Construction")
   150→
   151→        # Build Architecture Tree string
   152→        tree = ArchitectureTree(program)
   153→        result.architecture_tree_str = tree.to_string()
   154→
   155→        result.success = True
   156→        result.source_code = source_code
   157→        result.test_code = test_code
   158→        result.warnings = self.warnings
   159→        result.errors = self.errors
   160→        result.stages_completed = self.stages_completed
   161→        result.todo_count = self._todo_count
   162→        result.fully_compiled = self._todo_count == 0
   163→        result.provenance = self._provenance
   164→
   165→        return result
   166→
   167→    def compile_to_file(self, source: str, output_dir: str) -> CompilationResult:
   168→        """Compile AICL source code and write the output to files."""
   169→        result = self.compile(source)
   170→
   171→        if result.success:
   172→            os.makedirs(output_dir, exist_ok=True)
   173→
   174→            main_path = os.path.join(output_dir, "main.py")
   175→            with open(main_path, 'w') as f:
   176→                f.write(result.source_code)
   177→
   178→            test_path = os.path.join(output_dir, "test_main.py")
   179→            with open(test_path, 'w') as f:
   180→                f.write(result.test_code)
   181→
   182→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   183→            with open(tree_path, 'w') as f:
   184→                f.write(result.architecture_tree_str)
   185→
   186→        return result

   188→    # =========================================================================
   189→    # Stage 1: Specification Parsing
   190→    # =========================================================================
   191→
   192→    def _stage_parsing(self, source: str) -> Optional[AICLProgram]:
   193→        """Parse the AICL source code into an AST."""
   194→        try:
   195→            parser = Parser(source)
   196→            program = parser.parse()
   197→            return program
   198→        except (ParseError, Exception) as e:
   199→            self.errors.append(f"Parse error: {str(e)}")
   200→            return None

   202→    # =========================================================================
   203→    # Stage 2: Architecture Validation
   204→    # =========================================================================
   205→
   206→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   207→        """Validate the architectural structure of the program."""
   208→        warnings = program.validate()
   209→
   210→        layer_names = [l.name for l in program.layers]
   211→        seen = set()
   212→        for name in layer_names:
   213→            if name.lower() in seen:
   214→                warnings.append(f"Duplicate layer name: {name}")
   215→            seen.add(name.lower())
   216→
   217→        entity_names = {e.name for e in program.entities}
   218→        for behavior in program.behaviors:
   219→            for inp in behavior.inputs:
   220→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   221→                    warnings.append(
   222→                        f"Behavior '{behavior.name}' references unknown type "
   223→                        f"'{inp.param_type}' in input"
   224→                    )
   225→
   226→        return warnings
   227→

   232→    def _stage_dependency_analysis(self, program: AICLProgram) -> List[LayerSection]:
   233→        """Analyze dependencies between layers."""
   234→        return program.layers

   236→    # =========================================================================
   237→    # Stage 4: Risk Analysis
   238→    # =========================================================================
   239→
   240→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   241→        """Analyze risks and pair them with recovery procedures."""
   242→        pairs = []
   243→        for i, risk in enumerate(program.risks):
   244→            recovery = None
   245→            if i < len(program.recoveries):
   246→                recovery = program.recoveries[i]
   247→            pairs.append((risk, recovery))
   248→
   249→        if len(program.recoveries) > len(program.risks):
   250→            for i in range(len(program.risks), len(program.recoveries)):
   251→                generic_risk = RiskSection(
   252→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   253→                )
   254→                pairs.append((generic_risk, program.recoveries[i]))
   255→
   256→        return pairs
   257→
   258→    # =========================================================================
   259→    # Stage 5: Recovery Synthesis
   260→    # =========================================================================
   261→
   262→    def _stage_recovery_synthesis(
   263→        self,
   264→        program: AICLProgram,
   265→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]
   266→    ) -> Dict[str, str]:
   267→        """Synthesize recovery procedures for each risk."""
   268→        recovery_map = {}
   269→        for risk, recovery in risk_pairs:
   270→            if recovery:
   271→                recovery_map[risk.description] = recovery.description
   272→            else:
   273→                recovery_map[risk.description] = f"log and handle: {risk.description}"
   274→        return recovery_map

   276→    # =========================================================================
   277→    # Stage 6: Code Generation (with Pattern Matching)
   278→    # =========================================================================
   279→
   280→    def _unique_name(self, base: str) -> str:
   281→        """Generate a unique Python identifier from a base name."""
   282→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   283→        clean = re.sub(r'_+', '_', clean).strip('_')
   284→        if not clean:
   285→            clean = "item"
   286→        if clean[0].isdigit():
   287→            clean = "f_" + clean
   288→
   289→        name = clean
   290→        counter = 2
   291→        while name in self._used_names:
   292→            name = f"{clean}_{counter}"
   293→            counter += 1
   294→
   295→        self._used_names.add(name)
   296→        return name
   297→
   298→    def _stage_code_generation(
   299→        self,
   300→        program: AICLProgram,
   301→        dep_order: List[LayerSection],
   302→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   303→        recovery_map: Dict[str, str]
   304→    ) -> str:
   305→        """Generate executable Python code from the AICL program."""
   306→        sections = []
   307→
   308→        goal_desc = program.goals[0].description if program.goals else "AICL Generated Application"
   309→
   310→        # Record architecture template provenance
   311→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   312→        self._provenance.record(
   313→            source_type=ProvenanceType.ARCHITECTURE_TEMPLATE,
   314→            source_location="Goal",
   315→            source_text=goal_desc,
   316→            resolution_path=[
   317→                "AICL Source",
   318→                f"Goal: {goal_desc}",
   319→                "ArchitectureTemplateMapper.match()",
   320→                f"Template: {template_name}",
   321→                f"Structure: {template_info['structure']}",
   322→                "Generated Code",
   323→            ],
   324→            generated_code=f"# Architecture template: {template_name} ({template_info['structure']})",
   325→            confidence=1.0,
   326→            template_name=template_name,
   327→        )
   328→
   329→        sections.append('"""')
   330→        sections.append(f'AICL Generated Application')
   331→        sections.append(f'Goal: {goal_desc}')
   332→        sections.append(f'')
   333→        sections.append(f'Auto-generated by the AICL compiler v0.4.')
   334→        sections.append(f'Explicable compilation — every line has provenance.')
   335→        sections.append(f'"""')
   336→        sections.append('')
   337→
   338→        # Imports
   339→        sections.append(self._generate_imports(program))
   340→        sections.append('')
   341→
   342→        # Entity classes
   343→        for entity in program.entities:
   344→            sections.append(self._generate_entity(entity))
   345→            sections.append('')
   346→
   347→        # Main application class
   348→        class_name = self._make_class_name(goal_desc)
   349→        sections.append(self._generate_application_class(
   350→            program, class_name, dep_order, risk_pairs, recovery_map
   351→        ))
   352→        sections.append('')
   353→
   354→        # Entry point
   355→        sections.append(self._generate_entry_point(class_name))
   356→
   357→        return '\n'.join(sections)

   359→    def _generate_imports(self, program: AICLProgram) -> str:
   360→        """Generate import statements based on program content."""
   361→        imports = [
   362→            "import sys",
   363→            "import logging",
   364→            "import time",
   365→            "from typing import Optional, List, Dict, Any",
   366→        ]
   367→
   368→        if program.parallels:
   369→            imports.append("import threading")
   370→            imports.append("from concurrent.futures import ThreadPoolExecutor, as_completed")
   371→
   372→        needs_datetime = False
   373→        for entity in program.entities:
   374→            for f in entity.fields:
   375→                if f.field_type == "datetime":
   376→                    needs_datetime = True
   377→                    break
   378→
   379→        if needs_datetime:
   380→            imports.append("from datetime import datetime")
   381→
   382→        imports.append("from dataclasses import dataclass, field")
   383→
   384→        # Check if we need hashlib for security
   385→        if program.securities:
   386→            for sec in program.securities:
   387→                for action in sec.actions:
   388→                    if action.action_type == "encrypt":
   389→                        imports.append("import hashlib")
   390→                        imports.append("import base64")
   391→                        break
   392→
   393→        return '\n'.join(imports)

   395→    def _generate_entity(self, entity: EntitySection) -> str:
   396→        """Generate a Python dataclass for an AICL Entity."""
   397→        class_name = self._make_class_name(entity.name)
   398→        lines = []
   399→        lines.append('@dataclass')
   400→        lines.append(f'class {class_name}:')
   401→        lines.append(f'    """Entity: {entity.name}"""')
   402→
   403→        if not entity.fields:
   404→            lines.append('    pass')
   405→        else:
   406→            for f in entity.fields:
   407→                python_type = self._aicl_type_to_python(f.field_type)
   408→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   409→                if default:
   410→                    lines.append(f'    {f.name}: {python_type} = {default}')
   411→                else:
   412→                    lines.append(f'    {f.name}: {python_type} = 0')
   413→
   414→        code = '\n'.join(lines)
   415→
   416→        # Record entity generation provenance
   417→        fields_desc = ', '.join(f'{f.name}: {f.field_type}' for f in entity.fields) if entity.fields else 'no fields'
   418→        self._provenance.record(
   419→            source_type=ProvenanceType.ENTITY_GENERATION,
   420→            source_location=f"Entity {entity.name}",
   421→            source_text=fields_desc,
   422→            resolution_path=[
   423→                "AICL Source",
   424→                f"Entity {entity.name}",
   425→                f"Fields: {fields_desc}",
   426→                "Dataclass Generation",
   427→                "Generated Code",
   428→            ],
   429→            generated_code=code,
   430→            confidence=1.0,
   431→            parameters={"class_name": class_name, "fields": str(len(entity.fields))},
   432→        )
   433→
   434→        return code

   436→    def _generate_application_class(
   437→        self,
   438→        program: AICLProgram,
   439→        class_name: str,
   440→        dep_order: List[LayerSection],
   441→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   442→        recovery_map: Dict[str, str]
   443→    ) -> str:
   444→        """Generate the main application class."""
   445→        lines = []
   446→
   447→        # Determine architecture template from Goal
   448→        goal_desc = program.goals[0].description if program.goals else ""
   449→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   450→
   451→        lines.append(f'class {class_name}:')
   452→        lines.append(f'    """')
   453→        if program.goals:
   454→            lines.append(f'    Goal: {program.goals[0].description}')
   455→        lines.append(f'    Architecture: {template_info["structure"]}')
   456→        lines.append(f'    """')
   457→        lines.append('')
   458→
   459→        # Class attributes - one per layer
   460→        lines.append('    # Layer instances')
   461→        layer_attr_map = {}
   462→        for layer in dep_order:
   463→            attr_name = self._unique_name(layer.name)
   464→            layer_attr_map[layer.name] = attr_name
   465→            lines.append(f'    _{attr_name}: Any = None')
   466→        lines.append('')
   467→
   468→        # Entity instances
   469→        if program.entities:
   470→            lines.append('    # Entity instances')
   471→            for entity in program.entities:
   472→                attr = self._unique_name(entity.name.lower())
   473→                lines.append(f'    _{attr}: Any = None')
   474→            lines.append('')
   475→
   476→        # Risks as class constants
   477→        if program.risks:
   478→            lines.append('    # Known risks')
   479→            for i, risk in enumerate(program.risks):
   480→                lines.append(f'    RISK_{i+1} = {risk.description!r}')
   481→            lines.append('')
   482→
   483→        # Constraints as class constants
   484→        if program.constraints:
   485→            lines.append('    # Constraints')
   486→            for i, con in enumerate(program.constraints):
   487→                lines.append(f'    CONSTRAINT_{i+1} = {con.description!r}')
   488→            lines.append('')
   489→
   490→        # Game/application state
   491→        lines.append('    # Application state')
   492→        lines.append('    _running: bool = False')
   493→        lines.append('    _initialized: bool = False')
   494→        if template_name == "game_loop":
   495→            lines.append('    _game_active: bool = False')
   496→            lines.append('    _frame_count: int = 0')
   497→            lines.append('    _last_frame_time: float = 0.0')
   498→        elif template_name == "event_driven":
   499→            lines.append('    _connected: bool = False')
   500→            lines.append('    _message_queue: List = field(default_factory=list)')
   501→            lines.append('    _offline_queue: List = field(default_factory=list)')
   502→        lines.append('')
   503→
   504→        # __init__ method
   505→        lines.append('    def __init__(self):')
   506→        lines.append('        """Initialize the application and all layers."""')
   507→        lines.append('        self._logger = logging.getLogger(self.__class__.__name__)')
   508→        lines.append('        self._running = False')
   509→        lines.append('        self._initialized = False')
   510→        if template_name == "game_loop":
   511→            lines.append('        self._game_active = True')
   512→            lines.append('        self._frame_count = 0')
   513→            lines.append('        self._last_frame_time = time.time()')
   514→        elif template_name == "event_driven":
   515→            lines.append('        self._connected = False')
   516→            lines.append('        self._message_queue = []')
   517→            lines.append('        self._offline_queue = []')
   518→        lines.append('')
   519→
   520→        # Initialize entity instances
   521→        if program.entities:
   522→            lines.append('        # Initialize entities')
   523→            for entity in program.entities:
   524→                entity_class = self._make_class_name(entity.name)
   525→                attr = entity.name.lower()
   526→                lines.append(f'        self._{attr} = {entity_class}()')
   527→            lines.append('')
   528→
   529→        # Initialize each layer
   530→        for layer in dep_order:
   531→            attr_name = layer_attr_map[layer.name]
   532→            lines.append(f'        # Initialize {layer.name}')
   533→            lines.append(f'        self._{attr_name} = self._init_{attr_name}()')
   534→            lines.append('')
   535→
   536→        lines.append('        self._initialized = True')
   537→        lines.append('        self._logger.info("Application initialized successfully")')
   538→        lines.append('')
   539→
   540→        # Layer initialization methods
   541→        for layer in dep_order:
   542→            attr_name = layer_attr_map[layer.name]
   543→            lines.append(self._generate_layer_init(layer, attr_name, risk_pairs, recovery_map))
   544→            lines.append('')
   545→
   546→        # Pre-compute behavior method names (must be done before run method generation)
   547→        behavior_method_names = {}
   548→        for i, behavior in enumerate(program.behaviors):
   549→            method_name = self._unique_name(behavior.name).lower()
   550→            behavior_method_names[behavior.name] = method_name
   551→
   552→        # Run method — uses architecture template
   553→        lines.append(self._generate_run_method(program, dep_order, layer_attr_map, risk_pairs, recovery_map, template_name, behavior_method_names))
   554→        lines.append('')
   555→
   556→        # Behavior methods
   557→        for i, behavior in enumerate(program.behaviors):
   558→            method_name = behavior_method_names[behavior.name]
   559→            lines.append(self._generate_behavior_method(behavior, method_name, program))
   560→            lines.append('')
   561→
   562→        # Condition handlers
   563→        for i, condition in enumerate(program.conditions):
   564→            lines.append(self._generate_condition_handler(condition, i + 1, program))
   565→            lines.append('')
   566→
   567→        # Event handlers
   568→        for i, event in enumerate(program.events):
   569→            lines.append(self._generate_event_handler(event, i + 1, program))
   570→            lines.append('')
   571→
   572→        # Parallel execution
   573→        if program.parallels:
   574→            lines.append(self._generate_parallel_execution(program, layer_attr_map))
   575→            lines.append('')
   576→
   577→        # Validation methods
   578→        for i, validation in enumerate(program.validations):
   579→            lines.append(self._generate_validation_method(validation, i + 1, program))
   580→            lines.append('')
   581→
   582→        # Security methods
   583→        if program.securities:
   584→            lines.append(self._generate_security_methods(program))
   585→            lines.append('')

   666→    def _generate_run_method(
   667→        self,
   668→        program: AICLProgram,
   669→        dep_order: List[LayerSection],
   670→        layer_attr_map: Dict[str, str],
   671→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   672→        recovery_map: Dict[str, str],
   673→        template_name: str,
   674→        behavior_method_names: Dict[str, str] = None,
   675→    ) -> str:
   676→        """Generate the main run method — uses architecture template."""
   677→        behavior_method_names = behavior_method_names or {}
   678→        lines = []
   679→        lines.append('    def run(self) -> None:')
   680→        lines.append('        """Run the application with full error handling."""')
   681→        lines.append('        self._logger.info("Starting application...")')
   682→        lines.append('        self._running = True')
   683→        lines.append('')
   684→
   685→        if template_name == "game_loop":
   686→            # Game loop: init → update → render → validate → loop
   687→            lines.append('        self._game_active = True')
   688→            lines.append('        try:')
   689→            lines.append('            while self._running and self._game_active:')
   690→            lines.append('                frame_start = time.time()')
   691→            lines.append('')
   692→            lines.append('                # Update phase')
   693→            for behavior in program.behaviors:
   694→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   695→                input_params = ', '.join(
   696→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   697→                    {e.name.lower() for e in program.entities}
   698→                    else inp.name
   699→                    for inp in behavior.inputs
   700→                )
   701→                lines.append(f'                self._behavior_{method_name}({input_params})')
   702→            lines.append('')
   703→            lines.append('                # Check conditions')
   704→            for i, condition in enumerate(program.conditions):
   705→                lines.append(f'                self._handle_condition_{i+1}()')
   706→            lines.append('')
   707→            lines.append('                # Render phase')
   708→            for layer in dep_order:
   709→                attr_name = layer_attr_map[layer.name]
   710→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   711→                    lines.append(f'                if self._{attr_name} is not None:')
   712→                    lines.append(f'                    self._render_frame()')
   713→            lines.append('')
   714→            lines.append('                # Frame rate control')
   715→            lines.append('                elapsed = time.time() - frame_start')
   716→            lines.append('                if elapsed < 1.0 / 60.0:')
   717→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   718→            lines.append('                self._frame_count += 1')
   719→            lines.append('')
   720→
   721→        elif template_name == "event_driven":
   722→            # Event-driven: connect → listen → handle → respond
   723→            lines.append('        try:')
   724→            lines.append('            self._connect()')
   725→            lines.append('            self._connected = True')
   726→            lines.append('            self._logger.info("Connected to server")')
   727→            lines.append('')
   728→            lines.append('            while self._running and self._connected:')
   729→            lines.append('                # Process message queue')
   730→            lines.append('                while self._message_queue:')
   731→            lines.append('                    msg = self._message_queue.pop(0)')
   732→            lines.append('                    self._process_message(msg)')
   733→            lines.append('')
   734→            lines.append('                # Process offline queue if reconnected')
   735→            lines.append('                if self._offline_queue and self._connected:')
   736→            lines.append('                    self._sync_offline_messages()')
   737→            lines.append('')
   738→            lines.append('                # Wait for new messages')
   739→            lines.append('                time.sleep(0.1)')
   740→
   741→        else:
   742→            # Default: linear execution
   743→            lines.append('        try:')
   744→            for layer in dep_order:
   745→                attr_name = layer_attr_map[layer.name]
   746→                lines.append(f'            # Execute {layer.name}')
   747→                lines.append(f'            if self._{attr_name} is not None:')
   748→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   749→                lines.append(f'            else:')
   750→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   751→
   752→        # Validations
   753→        lines.append('')
   754→        lines.append('            # Run validations')
   755→        for i, validation in enumerate(program.validations):
   756→            lines.append(f'            self._validate_{i+1}()')
   757→
   758→        lines.append('')
   759→        lines.append('            self._logger.info("Application completed successfully")')
   760→        lines.append('')
   761→        lines.append('        except Exception as e:')
   762→        lines.append('            self._logger.error(f"Application error: {e}")')
   763→        lines.append('            self._handle_risk(e)')
   764→        lines.append('        finally:')
   765→        lines.append('            self._running = False')
   766→        lines.append('            self._logger.info("Application stopped")')
   767→
   768→        return '\n'.join(lines)

   770→    def _generate_behavior_method(
   771→        self, behavior: BehaviorSection, method_name: str, program: AICLProgram
   772→    ) -> str:
   773→        """Generate a behavior method using pattern matching — no TODOs."""
   774→        lines = []
   775→
   776→        # Build parameter list from inputs
   777→        params = ["self"]
   778→        for inp in behavior.inputs:
   779→            # Check if input references an entity
   780→            entity_names = {e.name for e in program.entities}
   781→            if inp.name in entity_names:
   782→                params.append(f'{inp.name.lower()}: Any')
   783→            elif inp.param_type in entity_names:
   784→                params.append(f'{inp.name.lower()}: Any')
   785→            else:
   786→                params.append(f'{inp.name.lower()}: Any = None')
   787→
   788→        param_str = ', '.join(params)
   789→        lines.append(f'    def _behavior_{method_name}({param_str}) -> Any:')
   790→        lines.append(f'        """Behavior: {behavior.name}"""')
   791→
   792→        # Compile the action using the Behavior Compiler
   793→        entity_context = {}
   794→        for inp in behavior.inputs:
   795→            entity_names = {e.name for e in program.entities}
   796→            if inp.name in entity_names:
   797→                entity_context["entity"] = inp.name.lower()
   798→            elif inp.param_type in entity_names:
   799→                entity_context["entity"] = inp.param_type.lower()
   800→
   801→        action_code, fully_compiled = self._behavior_compiler.compile_action(
   802→            behavior.action, context=entity_context
   803→        )
   804→
   805→        if not fully_compiled:
   806→            self._todo_count += 1
   807→
   808→        # Record provenance
   809→        self._record_behavior_provenance(behavior, action_code, fully_compiled, entity_context)
   810→
   811→        # Indent the compiled code
   812→        for code_line in action_code.split('\n'):
   813→            lines.append(f'        {code_line}')
   814→
   815→        return '\n'.join(lines)

   833→                    f"PATTERN_{match.pattern_name}",
   834→                    f"Template({match.category.value})",

   837→                generated_code=code,
   838→                confidence=match.confidence,
   839→                pattern_name=match.pattern_name,

   855→                confidence=1.0,

   868→                    "Generated Code",
   869→                ],
   870→                generated_code=code,
   871→                confidence=0.5 if not fully_compiled else 0.8,

   874→    def _generate_condition_handler(

   883→
   884→        # Compile the WHEN clause
   885→        when_code, when_compiled = self._behavior_compiler.compile_action(
   886→            f"check {when_desc}",
   887→            context={}
   888→        )
   889→
   890→        # Compile the THEN clause
   891→        then_code, then_compiled = self._behavior_compiler.compile_action(
   892→            then_desc,
   893→            context={}
   894→        )
   895→
   896→        if not when_compiled or not then_compiled:
   897→            self._todo_count += 1
   898→
   899→        # Generate the condition check
   900→        when_lower = when_desc.lower()
   901→
   902→        # Detect common condition patterns
   903→        if any(w in when_lower for w in ['hit', 'collision', 'collide', 'contact']):
   904→            lines.append(f'        # When: {when_desc}')
   905→            lines.append(f'        collision = self._check_collisions()')
   906→            lines.append(f'        if collision:')

   908→            for code_line in then_code.split('\n'):
   909→                lines.append(f'            {code_line}')
   910→        elif any(w in when_lower for w in ['score', 'reach', 'point']):
   911→            lines.append(f'        # When: {when_desc}')
   912→            # Extract threshold from description
   913→            import re as _re
   914→            num_match = _re.search(r'\d+', when_desc)
   915→            threshold = num_match.group() if num_match else "10"
   916→            lines.append(f'        for player in [self._player]:')
   917→            lines.append(f'            if player.score >= {threshold}:')
   918→            lines.append(f'                self._logger.info("Condition triggered: {when_desc}")')

   920→                lines.append(f'                {code_line}')

   930→            lines.append(f'            self._logger.info("Condition triggered: {when_desc}")')
   931→            for code_line in then_code.split('\n'):
   932→                lines.append(f'            {code_line}')
   933→        elif any(w in when_lower for w in ['restore', 'reconnect', 'network back']):
   934→            lines.append(f'        # When: {when_desc}')
   935→            lines.append(f'        if self._connected and self._offline_queue:')
   936→            lines.append(f'            self._logger.info("Condition triggered: {when_desc}")')
   937→            for code_line in then_code.split('\n'):
   938→                lines.append(f'            {code_line}')

   966→                "Keyword Pattern Match",
   967→                "Generated Code",
   968→            ],

   970→            confidence=0.9,

  1590→    def _make_class_name(self, name: str) -> str:
  1591→        """Convert a description to a valid Python class name."""
  1592→        words = name.replace('-', ' ').replace('_', ' ').split()
  1593→        if not words:
  1594→            return "AiclApplication"
  1595→
  1596→        filtered = [w for w in words if w.lower() not in ('a', 'an', 'the', 'on', 'in', 'of')]
  1597→        if not filtered:
  1598→            filtered = words
  1599→
  1600→        return ''.join(w.capitalize() for w in filtered)
