     1→"""
     2→AICL Autonomous Compilation System — Self-Writing, Self-Validating Code
     3→
     4→The central thesis: a compiler that cannot explain why it generated a line
     5→should not generate it. This module extends that thesis to its logical
     6→conclusion: a compiler that can diagnose its own failures, fix its own
     7→specifications, learn new patterns, and validate its own output — all
     8→while preserving the No-Orphan Property.
     9→
    10→The Autonomous Compilation Loop:
    11→    SPEC → COMPILE → VERIFY → TEST → DIAGNOSE → FIX → RECOMPILE
    12→                     ↑                                      │
    13→                     └──────────────────────────────────────┘
    14→
    15→Every iteration of the loop preserves the provenance chain. Every fix
    16→is recorded. Every learned pattern is tracked. The loop converges when:
    17→    1. Spec verification passes (completeness, coherence, satisfaction)
    18→    2. Audit coverage = 1.0 (no orphans)
    19→    3. All generated tests pass
    20→    4. No fallback compilations remain
    21→    5. Improvement score converges (delta < threshold)
    22→
    23→New ProvenanceTypes added by this module:
    24→    - AI_GENERATION: Code generated by AI-assisted fallback
    25→    - SELF_HEALING: Code fixed by autonomous diagnosis
    26→    - PATTERN_LEARNED: New pattern learned from unmatched actions
    27→    - SPEC_EVOLVED: Specification modified by autonomous evolution
    28→
    29→Usage:
    30→    aicl evolve <source.aicl> [--max-iterations 10] [--no-ai]
    31→    aicl evolve <source.aicl> --watch  # continuous mode
    32→"""

    51→class LoopState(Enum):
    52→    """State of the autonomous compilation loop."""
    53→    INITIALIZING = "initializing"
    54→    COMPILING = "compiling"
    55→    VERIFYING = "verifying"
    56→    TESTING = "testing"
    57→    DIAGNOSING = "diagnosing"
    58→    FIXING = "fixing"
    59→    LEARNING = "learning"
    60→    CONVERGED = "converged"
    61→    FAILED = "failed"
    62→    TIMED_OUT = "timed_out"
    63→
    64→
    65→@dataclass
    66→class LoopIteration:
    67→    """Record of a single loop iteration."""
    68→    iteration: int
    69→    state_entered: LoopState
    70→    timestamp_start: float
    71→    state_exited: LoopState = LoopState.INITIALIZING
    72→    timestamp_end: float = 0.0
    73→    spec_changes: List[str] = field(default_factory=list)
    74→    patterns_learned: List[str] = field(default_factory=list)
    75→    tests_passed: int = 0
    76→    tests_failed: int = 0
    77→    tests_total: int = 0
    78→    audit_coverage: float = 0.0
    79→    fallback_count: int = 0
    80→    provenance_records_added: int = 0
    81→    errors: List[str] = field(default_factory=list)
    82→    fixes_applied: List[str] = field(default_factory=list)
    83→
    84→    @property
    85→    def duration(self) -> float:
    86→        return self.timestamp_end - self.timestamp_start
    87→
    88→    @property
    89→    def success(self) -> bool:
    90→        return (
    91→            self.tests_failed == 0
    92→            and self.tests_total > 0
    93→            and self.audit_coverage >= 1.0
    94→            and self.fallback_count == 0
    95→        )

   118→            f"Test Pass Rate: {self.final_test_pass_rate:.1%}",

   120→            f"Spec Changes: {len(self.spec_evolution)}",
   121→            f"Proof Valid: {self.proof_valid}",
   122→        ]
   123→        if self.convergence_iteration is not None:
   124→            lines.append(f"Converged at iteration: {self.convergence_iteration}")
   125→        return "\n".join(lines)

   128→class TestRunner:
   129→    """Execute generated tests and analyze failures."""
   130→
   131→    def __init__(self, timeout: int = 30):
   132→        self.timeout = timeout
   133→        self.last_results: Dict[str, Any] = {}
   134→
   135→    def run_tests(self, output_dir: str) -> Tuple[int, int, int, List[Dict]]:
   136→        """
   137→        Run pytest on the generated test file.
   138→        Returns (passed, failed, total, failure_details).
   139→        """
   140→        test_file = os.path.join(output_dir, "test_main.py")
   141→        if not os.path.exists(test_file):
   142→            return 0, 0, 0, [{"error": "No test file found"}]
   143→
   144→        # Try to find pytest - check venv first, then system
   145→        pytest_path = shutil.which("pytest")
   146→        python_path = sys.executable
   147→
   148→        try:
   149→            cmd = [python_path, "-m", "pytest", test_file, "-v", "--tb=short", "--no-header"]
   150→            result = subprocess.run(
   151→                cmd,
   152→                capture_output=True,
   153→                text=True,
   154→                timeout=self.timeout,
   155→                cwd=output_dir,
   156→            )
   157→        except subprocess.TimeoutExpired:
   158→            return 0, 0, 0, [{"error": f"Test execution timed out after {self.timeout}s"}]
   159→        except Exception as e:
   160→            return 0, 0, 0, [{"error": f"Test execution failed: {str(e)}"}]
   161→
   162→        # Parse pytest output
   163→        passed, failed, total = 0, 0, 0
   164→        failure_details = []
   165→
   166→        for line in result.stdout.split("\n"):
   167→            line = line.strip()
   168→            # Match: "test_main.py::test_validation_1 PASSED"
   169→            if " PASSED" in line:
   170→                passed += 1
   171→                total += 1
   172→            elif " FAILED" in line:
   173→                failed += 1
   174→                total += 1
   175→                test_name = line.split("::")[-1].split()[0] if "::" in line else "unknown"
   176→                failure_details.append({"test": test_name, "raw": line})
   177→
   178→        # Also check the summary line: "X passed, Y failed"
   179→        for line in result.stdout.split("\n"):
   180→            m = re.match(r"(\d+) passed", line)
   181→            if m:
   182→                passed = int(m.group(1))
   183→            m = re.match(r".*?(\d+) failed", line)
   184→            if m:
   185→                failed = int(m.group(1))
   186→            m = re.match(r".*?(\d+) warning", line)
   187→            # total = passed + failed
   188→
   189→        if total == 0:
   190→            total = passed + failed
   191→
   192→        self.last_results = {
   193→            "passed": passed,
   194→            "failed": failed,
   195→            "total": total,
   196→            "returncode": result.returncode,
   197→            "stdout": result.stdout,
   198→            "stderr": result.stderr,
   199→        }
   200→
   201→        return passed, failed, total, failure_details
   202→
   203→    def analyze_failure(self, test_name: str, output_dir: str) -> Dict[str, Any]:
   204→        """Analyze a specific test failure and return diagnostic info."""
   205→        test_file = os.path.join(output_dir, "test_main.py")
   206→        main_file = os.path.join(output_dir, "main.py")
   207→
   208→        diagnosis = {
   209→            "test_name": test_name,
   210→            "missing_imports": [],
   211→            "missing_methods": [],
   212→            "attribute_errors": [],
   213→            "type_errors": [],
   214→            "assertion_failures": [],
   215→            "suggested_fix": None,
   216→        }
   217→
   218→        stdout = self.last_results.get("stdout", "")
   219→        stderr = self.last_results.get("stderr", "")
   220→
   221→        full_output = stdout + "\n" + stderr
   222→
   223→        # Pattern-based diagnosis
   224→        if "ModuleNotFoundError" in full_output or "ImportError" in full_output:
   225→            for m in re.finditer(r"No module named '(\w+)'", full_output):
   226→                diagnosis["missing_imports"].append(m.group(1))
   227→            diagnosis["suggested_fix"] = "add_missing_imports"
   228→
   229→        if "AttributeError" in full_output:
   230→            for m in re.finditer(r"'(\w+)' object has no attribute '(\w+)'", full_output):
   231→                diagnosis["attribute_errors"].append({
   232→                    "class": m.group(1),
   233→                    "attribute": m.group(2),
   234→                })
   235→            diagnosis["suggested_fix"] = "add_missing_attributes"
   236→
   237→        if "TypeError" in full_output:
   238→            diagnosis["type_errors"].append(full_output[:500])
   239→            diagnosis["suggested_fix"] = "fix_type_signatures"
   240→
   241→        if "AssertionError" in full_output or "assert" in full_output:
   242→            diagnosis["assertion_failures"].append(test_name)
   243→            diagnosis["suggested_fix"] = "fix_validation_logic"
   244→
   245→        if "NameError" in full_output:
   246→            for m in re.finditer(r"name '(\w+)' is not defined", full_output):
   247→                diagnosis["missing_methods"].append(m.group(1))
   248→            diagnosis["suggested_fix"] = "add_missing_definitions"
   249→
   250→        # Check if main.py exists and has content
   251→        if os.path.exists(main_file):
   252→            with open(main_file) as f:
   253→                main_content = f.read()
   254→            diagnosis["main_file_lines"] = len(main_content.split("\n"))
   255→        else:
   256→            diagnosis["main_file_lines"] = 0
   257→            diagnosis["suggested_fix"] = "regenerate_main"
   258→
   259→        return diagnosis

   262→class PatternLearner:
   263→    """
   264→    Learn new behavior patterns from unmatched action descriptions.
   265→
   266→    When the deterministic pattern matcher falls back (no match found),
   267→    the PatternLearner analyzes the unmatched description and creates
   268→    a candidate pattern that can be used in future compilations.
   269→    """
   270→
   271→    def __init__(self):
   272→        self.learned_patterns: Dict[str, Dict] = {}
   273→        self.unmatched_descriptions: List[str] = []
   274→
   275→    def record_unmatched(self, behavior_name: str, description: str):
   276→        """Record an unmatched action description for learning."""
   277→        self.unmatched_descriptions.append(f"{behavior_name}: {description}")
   278→
   279→    def learn_from_description(self, behavior_name: str, description: str) -> Optional[Dict]:
   280→        """
   281→        Analyze an unmatched action description and generate a candidate pattern.
   282→        Returns a pattern dict or None if no pattern can be inferred.
   283→        """
   284→        desc_lower = description.lower().strip()
   285→
   286→        # Extract verbs and objects from the description
   287→        verb = self._extract_verb(desc_lower)
   288→        obj = self._extract_object(desc_lower)
   289→
   290→        if not verb or not obj:
   291→            return None
   292→
   293→        # Classify the pattern based on verb-object analysis
   294→        pattern = {
   295→            "name": f"LEARNED_{behavior_name.upper()}",
   296→            "behavior": behavior_name,
   297→            "verb": verb,
   298→            "object": obj,
   299→            "category": self._classify_category(verb, obj),
   300→            "keywords": self._extract_keywords(desc_lower),
   301→            "description": description,
   302→            "template": self._generate_template(verb, obj, behavior_name),
   303→            "confidence": 0.7,  # Learned patterns start at 0.7 (below deterministic 0.95)
   304→            "learned_at": time.time(),
   305→        }
   306→
   307→        self.learned_patterns[behavior_name] = pattern
   308→        return pattern
   309→
   310→    def _extract_verb(self, description: str) -> str:
   311→        """Extract the primary verb from an action description."""
   312→        action_verbs = [
   313→            "create", "delete", "update", "modify", "set", "change",
   314→            "add", "remove", "append", "push", "pop", "insert",
   315→            "validate", "verify", "check", "assert", "confirm",
   316→            "send", "receive", "transmit", "broadcast", "dispatch",
   317→            "calculate", "compute", "evaluate", "process", "analyze",
   318→            "convert", "transform", "translate", "map",
   319→            "store", "save", "load", "fetch", "retrieve",
   320→            "encrypt", "decrypt", "sign", "verify",
   321→            "connect", "disconnect", "reconnect",
   322→            "start", "stop", "pause", "resume", "cancel",
   323→            "lock", "unlock", "freeze", "unfreeze",
   324→            "approve", "reject", "accept", "deny",
   325→            "compile", "generate", "build", "produce",
   326→            "notify", "alert", "signal", "emit",
   327→            "apply", "execute", "run", "invoke",
   328→            "debit", "credit", "transfer", "exchange",
   329→            "suspend", "escalate", "flag",
   330→            "log", "record", "audit",
   331→        ]
   332→        words = description.split()
   333→        for word in words:
   334→            if word in action_verbs:
   335→                return word
   336→        # Fallback: first word
   337→        return words[0] if words else ""
   338→
   339→    def _extract_object(self, description: str) -> str:
   340→        """Extract the primary object from an action description."""
   341→        # Look for entity-like nouns
   342→        financial_objects = [
   343→            "account", "transaction", "loan", "payment", "balance",
   344→            "customer", "card", "deposit", "withdrawal", "transfer",
   345→            "currency", "rate", "fraud", "alert", "audit",
   346→        ]
   347→        general_objects = [
   348→            "message", "data", "state", "position", "score",
   349→            "user", "player", "game", "board", "piece",
   350→            "file", "record", "entry", "item", "element",
   351→            "connection", "session", "request", "response",
   352→        ]
   353→        all_objects = financial_objects + general_objects
   354→
   355→        words = description.split()
   356→        for word in words:
   357→            if word in all_objects:
   358→                return word
   359→
   360→        # Fallback: noun after the verb
   361→        for i, word in enumerate(words):
   362→            if i > 0 and word not in ["to", "from", "with", "and", "or", "the", "a", "an", "in", "on", "at", "for"]:
   363→                return word
   364→
   365→        return words[-1] if words else "data"
   366→
   367→    def _classify_category(self, verb: str, obj: str) -> str:
   368→        """Classify the action into a pattern category."""
   369→        category_map = {
   370→            "create": "CREATION", "delete": "DELETION", "add": "UPDATE",
   371→            "remove": "DELETION", "update": "UPDATE", "modify": "UPDATE",
   372→            "set": "UPDATE", "change": "UPDATE",
   373→            "validate": "VALIDATE", "verify": "VALIDATE", "check": "VALIDATE",
   374→            "send": "ROUTE", "receive": "ROUTE", "transmit": "BROADCAST",
   375→            "broadcast": "BROADCAST", "dispatch": "ROUTE",
   376→            "calculate": "COMPUTE", "compute": "COMPUTE", "evaluate": "COMPUTE",
   377→            "convert": "TRANSFORM", "transform": "TRANSFORM",
   378→            "store": "STORE", "save": "STORE", "load": "LOAD",
   379→            "fetch": "LOAD", "retrieve": "LOAD",
   380→            "encrypt": "ENCRYPT", "decrypt": "ENCRYPT",
   381→            "connect": "CONNECT", "disconnect": "DISCONNECT",
   382→            "notify": "NOTIFY", "alert": "NOTIFY",
   383→            "debit": "FINANCIAL", "credit": "FINANCIAL",
   384→            "transfer": "FINANCIAL", "exchange": "FINANCIAL",
   385→            "suspend": "SECURITY", "lock": "SECURITY", "freeze": "SECURITY",
   386→            "log": "AUDIT", "record": "AUDIT", "audit": "AUDIT",
   387→        }
   388→        return category_map.get(verb, "GENERAL")
   389→
   390→    def _extract_keywords(self, description: str) -> List[str]:
   391→        """Extract searchable keywords from a description."""
   392→        stop_words = {"to", "from", "with", "and", "or", "the", "a", "an",
   393→                      "in", "on", "at", "for", "is", "are", "was", "were",
   394→                      "be", "been", "being", "have", "has", "had", "do",
   395→                      "does", "did", "will", "would", "could", "should",
   396→                      "may", "might", "must", "shall", "can", "need",
   397→                      "it", "its", "if", "then", "else", "when", "where",
   398→                      "which", "who", "whom", "this", "that", "these", "those"}
   399→        words = re.findall(r'\b\w+\b', description.lower())
   400→        return [w for w in words if w not in stop_words and len(w) > 2]
   401→
   402→    def _generate_template(self, verb: str, obj: str, behavior_name: str) -> str:
   403→        """Generate a code template for a learned pattern."""
   404→        templates = {
   405→            "CREATION": f'''        """Behavior: {behavior_name}"""
   406→        instance = {obj.title().replace('_', '')}()
   407→        self._logger.info(f"Created {{}}: {{instance}}")
   408→        return instance''',
   409→            "UPDATE": f'''        """Behavior: {behavior_name}"""
   410→        {obj}.{verb}_value = new_value
   411→        self._logger.info(f"{verb} {obj}: {{{obj}.{verb}_value}}")''',
   412→            "VALIDATE": f'''        """Behavior: {behavior_name}"""
   413→        result = self._validate_{obj}({obj})
   414→        if not result:
   415→            self._logger.warning(f"Validation failed for {{}}")
   416→        return result''',
   417→            "STORE": f'''        """Behavior: {behavior_name}"""
   418→        self._storage["{obj}"] = {obj}
   419→        self._logger.info(f"Stored {obj}")''',
   420→            "LOAD": f'''        """Behavior: {behavior_name}"""
   421→        data = self._storage.get("{obj}")
   422→        self._logger.info(f"Loaded {obj}: {{data}}")
   423→        return data''',
   424→            "FINANCIAL": f'''        """Behavior: {behavior_name}"""
   425→        # {verb} {obj} with full audit trail
   426→        self._logger.info(f"{verb} {obj}")
   427→        record = self._create_audit_entry("{verb}", "{obj}")
   428→        self._audit_trail.append(record)
   429→        return record''',
   430→            "NOTIFY": f'''        """Behavior: {behavior_name}"""
   431→        for recipient in self._recipients:
   432→            self._notify(recipient, "{obj}")
   433→        self._logger.info(f"Notified about {{obj}}")''',
   434→            "AUDIT": f'''        """Behavior: {behavior_name}"""
   435→        entry = self._create_audit_entry("{verb}", "{obj}")
   436→        self._audit_trail.append(entry)
   437→        self._logger.info(f"Audit entry created: {{entry}}")
   438→        return entry''',
   439→            "SECURITY": f'''        """Behavior: {behavior_name}"""
   440→        {obj}.status = "suspended"
   441→        self._logger.info(f"{verb} {obj} for security review")
   442→        self._notify_security("{verb}", {obj})''',
   443→            "ROUTE": f'''        """Behavior: {behavior_name}"""
   444→        self._send_to(destination, {obj})
   445→        self._logger.info(f"{verb} {obj} to {{destination}}")''',
   446→            "BROADCAST": f'''        """Behavior: {behavior_name}"""
   447→        for recipient in self._recipients:
   448→            self._send_to(recipient, {obj})
   449→        self._logger.info(f"Broadcast {obj}")''',
   450→        }
   451→
   452→        category = self._classify_category(verb, obj)
   453→        return templates.get(category, f'''        """Behavior: {behavior_name}"""
   454→        # {verb} {obj}
   455→        self._logger.info(f"{verb} {obj}")
   456→        pass''')
   457→
   458→    def get_learned_pattern(self, behavior_name: str) -> Optional[Dict]:
   459→        """Get a learned pattern for a behavior."""
   460→        return self.learned_patterns.get(behavior_name)
   461→
   462→    def export_patterns(self) -> List[Dict]:
   463→        """Export all learned patterns for serialization."""
   464→        return [
   465→            {**p, "learned_at": p["learned_at"]}
   466→            for p in self.learned_patterns.values()
   467→        ]

   470→class SpecEvolver:
   471→    """
   472→    Evolve/fix AICL specifications based on diagnostic information.
   473→
   474→    The SpecEvolver takes verification failures, test failures, and
   475→    audit gaps as input, and produces specification modifications
   476→    that address the root cause — not the symptom.
   477→
   478→    Every modification is recorded in the provenance chain.
   479→    """
   480→
   481→    def __init__(self):
   482→        self.evolution_log: List[Dict] = []
   483→
   484→    def evolve_from_verification(self, source: str, report: Dict) -> Tuple[str, List[str]]:
   485→        """
   486→        Modify the AICL source based on a verification report.
   487→        Returns (modified_source, list_of_changes).
   488→        """
   489→        changes = []
   490→        lines = source.split("\n")
   491→
   492→        for check in report.get("checks", []):
   493→            if check.get("status") == "FAIL":
   494→                change = self._fix_verification_failure(lines, check)
   495→                if change:
   496→                    changes.append(change)
   497→                    self.evolution_log.append({
   498→                        "type": "verification_fix",
   499→                        "check": check,
   500→                        "change": change,
   501→                        "timestamp": time.time(),
   502→                    })
   503→
   504→        return "\n".join(lines), changes
   505→
   506→    def evolve_from_test_failure(self, source: str, diagnosis: Dict) -> Tuple[str, List[str]]:
   507→        """
   508→        Modify the AICL source based on a test failure diagnosis.
   509→        Returns (modified_source, list_of_changes).
   510→        """
   511→        changes = []
   512→        lines = source.split("\n")
   513→
   514→        fix_type = diagnosis.get("suggested_fix")
   515→
   516→        if fix_type == "add_missing_imports":
   517→            change = self._fix_missing_imports(lines, diagnosis)
   518→            if change:
   519→                changes.append(change)
   520→
   521→        elif fix_type == "add_missing_attributes":
   522→            change = self._fix_missing_attributes(lines, diagnosis)
   523→            if change:
   524→                changes.append(change)
   525→
   526→        elif fix_type == "fix_validation_logic":
   527→            change = self._fix_validation_logic(lines, diagnosis)
   528→            if change:
   529→                changes.append(change)
   530→
   531→        elif fix_type == "add_missing_definitions":
   532→            change = self._fix_missing_definitions(lines, diagnosis)
   533→            if change:
   534→                changes.append(change)
   535→
   536→        elif fix_type == "fix_type_signatures":
   537→            change = self._fix_type_signatures(lines, diagnosis)
   538→            if change:
   539→                changes.append(change)
   540→
   541→        for change in changes:
   542→            self.evolution_log.append({
   543→                "type": "test_failure_fix",
   544→                "diagnosis": diagnosis,
   545→                "change": change,
   546→                "timestamp": time.time(),
   547→            })
   548→
   549→        return "\n".join(lines), changes
   550→
   551→    def evolve_from_audit_gap(self, source: str, audit_report: Dict) -> Tuple[str, List[str]]:
   552→        """
   553→        Modify the AICL source to address audit coverage gaps.
   554→        Returns (modified_source, list_of_changes).
   555→        """
   556→        changes = []
   557→        lines = source.split("\n")
   558→
   559→        orphan_artifacts = audit_report.get("orphan_artifacts", [])
   560→        coverage = audit_report.get("audit_coverage", 0.0)
   561→
   562→        if coverage < 1.0:
   563→            # Add more specific validations to cover gaps
   564→            change = self._add_coverage_validations(lines, orphan_artifacts, coverage)
   565→            if change:
   566→                changes.append(change)
   567→                self.evolution_log.append({
   568→                    "type": "audit_gap_fix",
   569→                    "coverage": coverage,
   570→                    "change": change,
   571→                    "timestamp": time.time(),
   572→                })
   573→
   574→        return "\n".join(lines), changes
   575→
   576→    def _fix_verification_failure(self, lines: List[str], check: Dict) -> Optional[str]:
   577→        """Fix a specific verification failure."""
   578→        check_name = check.get("name", "")
   579→        message = check.get("message", "")
   580→
   581→        if "risk_recovery_pairing" in check_name:
   582→            # Risk without recovery — add a generic recovery
   583→            last_risk_idx = -1
   584→            for i, line in enumerate(lines):
   585→                if line.strip().startswith("Risk:"):
   586→                    last_risk_idx = i
   587→
   588→            if last_risk_idx >= 0:
   589→                # Find the next blank line after the risk
   590→                insert_idx = last_risk_idx + 1
   591→                while insert_idx < len(lines) and lines[insert_idx].strip():
   592→                    insert_idx += 1
   593→
   594→                risk_text = lines[last_risk_idx].replace("Risk:", "").strip()
   595→                lines.insert(insert_idx, f"Recovery:")
   596→                lines.insert(insert_idx + 1, f"Handle {risk_text.lower()} gracefully")
   597→                return f"Added recovery for risk: {risk_text}"
   598→
   599→        elif "entity_present" in check_name or "behavior_present" in check_name:
   600→            # Missing entity or behavior — can't auto-fix, return None
   601→            return None
   602→
   603→        elif "action" in check_name:
   604→            # Behavior without action — add a generic action
   605→            for i, line in enumerate(lines):
   606→                if line.strip().startswith("Behavior "):
   607→                    # Check if there's no Action: section
   608→                    has_action = False
   609→                    for j in range(i + 1, min(i + 10, len(lines))):
   610→                        if "Action:" in lines[j]:
   611→                            has_action = True
   612→                            break
   613→                        if lines[j].strip() and not lines[j].startswith(" ") and not lines[j].startswith("\t"):
   614→                            break
   615→                    if not has_action:
   616→                        behavior_name = line.replace("Behavior", "").strip()
   617→                        lines.insert(i + 1, "")
   618→                        lines.insert(i + 2, "Action:")
   619→                        lines.insert(i + 3, f"    execute {behavior_name.lower()}")
   620→                        return f"Added action for behavior: {behavior_name}"
   621→
   622→        return None
   623→
   624→    def _fix_missing_imports(self, lines: List[str], diagnosis: Dict) -> Optional[str]:
   625→        """Add missing imports — note: this affects generated code, not AICL source."""
   626→        # For AICL source, missing imports usually means the Native section needs updating
   627→        for missing in diagnosis.get("missing_imports", []):
   628→            # Check if there's a Native section
   629→            for i, line in enumerate(lines):
   630→                if line.strip().startswith("Native:"):
   631→                    return f"Missing import '{missing}' should be added to Native section"
   632→
   633→        return None
   634→
   635→    def _fix_missing_attributes(self, lines: List[str], diagnosis: Dict) -> Optional[str]:
   636→        """Add missing entity fields based on attribute errors."""
   637→        fixes = []
   638→        for error in diagnosis.get("attribute_errors", []):
   639→            cls_name = error.get("class", "")
   640→            attr_name = error.get("attribute", "")
   641→
   642→            # Find the entity definition and add the missing field
   643→            for i, line in enumerate(lines):
   644→                if line.strip().startswith(f"Entity {cls_name}"):
   645→                    # Add the missing field
   646→                    j = i + 1
   647→                    while j < len(lines) and (lines[j].startswith("    ") or lines[j].startswith("\t")):
   648→                        j += 1
   649→                    lines.insert(j, f"    {attr_name}: any")
   650→                    fixes.append(f"Added field '{attr_name}' to Entity {cls_name}")
   651→                    break
   652→
   653→        return "; ".join(fixes) if fixes else None
   654→
   655→    def _fix_validation_logic(self, lines: List[str], diagnosis: Dict) -> Optional[str]:
   656→        """Make validations more specific and testable."""
   657→        # Make vague validations more testable
   658→        for i, line in enumerate(lines):
   659→            if line.strip().startswith("Validation:"):
   660→                validation_text = line.replace("Validation:", "").strip()
   661→                # Check if the validation is too vague
   662→                vague_patterns = ["works correctly", "functions properly", "operates as expected"]
   663→                for pattern in vague_patterns:
   664→                    if pattern in validation_text.lower():
   665→                        lines[i] = f"Validation: {validation_text.replace(pattern, 'produces expected output and returns success')}"
   666→                        return f"Made validation more specific: {validation_text[:50]}..."
   667→
   668→        return None
   669→
   670→    def _fix_missing_definitions(self, lines: List[str], diagnosis: Dict) -> Optional[str]:
   671→        """Add missing entity or behavior definitions."""
   672→        fixes = []
   673→        for name in diagnosis.get("missing_methods", []):
   674→            # Check if it might be a missing entity
   675→            entity_name = name.title().replace("_", "")
   676→            # Add a minimal entity if it makes sense
   677→            has_entity = any(f"Entity {entity_name}" in line for line in lines)
   678→            if not has_entity and len(name) > 3:
   679→                # Add at the end of entities section
   680→                last_entity = -1
   681→                for i, line in enumerate(lines):
   682→                    if line.strip().startswith("Entity "):
   683→                        last_entity = i
   684→                if last_entity >= 0:
   685→                    # Find end of last entity
   686→                    j = last_entity + 1
   687→                    while j < len(lines) and (lines[j].startswith("    ") or lines[j].startswith("\t")):
   688→                        j += 1
   689→                    lines.insert(j, f"\nEntity {entity_name}")
   690→                    lines.insert(j + 1, f"    id: string")
   691→                    lines.insert(j + 2, f"    name: string")
   692→                    fixes.append(f"Added Entity {entity_name}")
   693→
   694→        return "; ".join(fixes) if fixes else None
   695→
   696→    def _fix_type_signatures(self, lines: List[str], diagnosis: Dict) -> Optional[str]:
   697→        """Fix type signature mismatches in behaviors."""
   698→        # Often caused by Behavior Input sections using raw types instead of Entity names
   699→        for error in diagnosis.get("type_errors", []):
   700→            # This is usually a generated code issue, not a spec issue
   701→            pass
   702→        return None
   703→
   704→    def _add_coverage_validations(self, lines: List[str], orphans: List, coverage: float) -> Optional[str]:
   705→        """Add validations to improve audit coverage."""
   706→        # Find the last Validation section
   707→        last_validation = -1
   708→        for i, line in enumerate(lines):
   709→            if line.strip().startswith("Validation:"):
   710→                last_validation = i
   711→
   712→        # Add a general validation for coverage
   713→        if last_validation >= 0:
   714→            lines.insert(last_validation + 1, "Validation:")
   715→            lines.insert(last_validation + 2, "All generated code artifacts have traceable provenance")
   716→            return f"Added coverage validation (was {coverage:.1%})"
   717→
   718→        return None

   721→class LoopController:
   722→    """
   723→    Manage the autonomous compilation loop with convergence detection.
   724→
   725→    Convergence criteria:
   726→        1. All spec verifications pass
   727→        2. Audit coverage = 1.0
   728→        3. All generated tests pass
   729→        4. No fallback compilations remain
   730→        5. Improvement score delta < threshold (0.01)
   731→    """
   732→
   733→    def __init__(self, max_iterations: int = 10, convergence_threshold: float = 0.01):
   734→        self.max_iterations = max_iterations
   735→        self.convergence_threshold = convergence_threshold
   736→        self.iterations: List[LoopIteration] = []
   737→        self.history: List[Dict] = []
   738→
   739→    @property
   740→    def current_iteration(self) -> int:
   741→        return len(self.iterations)
   742→
   743→    @property
   744→    def should_continue(self) -> bool:
   745→        """Whether the loop should continue iterating."""
   746→        if self.current_iteration >= self.max_iterations:
   747→            return False
   748→        if self.is_converged:
   749→            return False
   750→        return True
   751→
   752→    @property
   753→    def is_converged(self) -> bool:
   754→        """Whether the loop has converged."""
   755→        if len(self.history) < 2:
   756→            return False
   757→
   758→        current = self.history[-1]
   759→        previous = self.history[-2]
   760→
   761→        # All metrics must be at maximum
   762→        if current.get("audit_coverage", 0) < 1.0:
   763→            return False
   764→        if current.get("test_pass_rate", 0) < 1.0:
   765→            return False
   766→        if current.get("fallback_count", 1) > 0:
   767→            return False
   768→        if not current.get("spec_verified", False):
   769→            return False
   770→
   771→        # Improvement must be below threshold
   772→        delta = abs(
   773→            current.get("score", 0) - previous.get("score", 0)
   774→        )
   775→        return delta < self.convergence_threshold
   776→
   777→    @property
   778→    def convergence_iteration(self) -> Optional[int]:
   779→        """The iteration at which convergence was reached."""
   780→        for i, h in enumerate(self.history):
   781→            if (
   782→                h.get("audit_coverage", 0) >= 1.0
   783→                and h.get("test_pass_rate", 0) >= 1.0
   784→                and h.get("fallback_count", 1) == 0
   785→                and h.get("spec_verified", False)
   786→            ):
   787→                return i + 1
   788→        return None
   789→
   790→    def record_iteration(self, iteration: LoopIteration):
   791→        """Record a completed iteration."""
   792→        self.iterations.append(iteration)
   793→        self.history.append({
   794→            "iteration": iteration.iteration,
   795→            "audit_coverage": iteration.audit_coverage,
   796→            "test_pass_rate": iteration.tests_passed / max(iteration.tests_total, 1),
   797→            "fallback_count": iteration.fallback_count,
   798→            "spec_verified": iteration.tests_failed == 0 and iteration.audit_coverage >= 1.0,
   799→            "score": self._compute_score(iteration),
   800→            "patterns_learned": len(iteration.patterns_learned),
   801→            "fixes_applied": len(iteration.fixes_applied),
   802→        })
   803→
   804→    def _compute_score(self, iteration: LoopIteration) -> float:
   805→        """Compute a composite score for an iteration."""
   806→        coverage_score = iteration.audit_coverage
   807→        test_score = iteration.tests_passed / max(iteration.tests_total, 1)
   808→        fallback_penalty = iteration.fallback_count * 0.05
   809→        return max(0, (coverage_score + test_score) / 2 - fallback_penalty)
   810→
   811→    def get_progress(self) -> Dict[str, Any]:
   812→        """Get current progress information."""
   813→        if not self.history:
   814→            return {"status": "starting", "iteration": 0}
   815→
   816→        current = self.history[-1]
   817→        return {
   818→            "status": "converged" if self.is_converged else "iterating",
   819→            "iteration": self.current_iteration,
   820→            "max_iterations": self.max_iterations,
   821→            "score": current.get("score", 0),
   822→            "audit_coverage": current.get("audit_coverage", 0),
   823→            "test_pass_rate": current.get("test_pass_rate", 0),
   824→            "fallback_count": current.get("fallback_count", 0),
   825→            "total_patterns_learned": sum(h.get("patterns_learned", 0) for h in self.history),
   826→            "total_fixes_applied": sum(h.get("fixes_applied", 0) for h in self.history),
   827→        }
   828→
   829→
   830→class AutonomousCompiler:
   831→    """
   832→    The self-writing, self-validating compilation engine.
   833→
   834→    Orchestrates the full autonomous loop:
   835→        SPEC → COMPILE → VERIFY → TEST → DIAGNOSE → FIX → RECOMPILE
   836→
   837→    Every step preserves the No-Orphan Property. Every fix is recorded
   838→    in the provenance chain. The loop converges when all criteria are met.
   839→
   840→    Usage:
   841→        compiler = AutonomousCompiler(max_iterations=10)
   842→        result = compiler.evolve("banking.aicl")
   843→        print(result.summary)
   844→    """
   845→
   846→    def __init__(self, max_iterations: int = 10, target_language: str = "python",
   847→                 test_timeout: int = 30, convergence_threshold: float = 0.01):
   848→        self.max_iterations = max_iterations
   849→        self.target_language = target_language
   850→        self.test_runner = TestRunner(timeout=test_timeout)
   851→        self.pattern_learner = PatternLearner()
   852→        self.spec_evolver = SpecEvolver()
   853→        self.loop_controller = LoopController(
   854→            max_iterations=max_iterations,
   855→            convergence_threshold=convergence_threshold,
   856→        )
   857→        self._provenance_log: List[Dict] = []
   858→
   859→    def evolve(self, source_path: str, output_dir: Optional[str] = None) -> AutonomousResult:
   860→        """
   861→        Run the autonomous compilation loop on an AICL source file.
   862→
   863→        Args:
   864→            source_path: Path to the .aicl source file
   865→            output_dir: Output directory (defaults to temp dir)
   866→
   867→        Returns:
   868→            AutonomousResult with convergence details
   869→        """
   870→        start_time = time.time()
   871→
   872→        # Read source
   873→        with open(source_path) as f:
   874→            source = f.read()
   875→
   876→        original_source = source
   877→
   878→        if output_dir is None:
   879→            output_dir = tempfile.mkdtemp(prefix="aicl_evolve_")
   880→
   881→        # Import compiler here to avoid circular imports
   882→        from . import Compiler, Parser
   883→
   884→        spec_verified = False
   885→
   886→        while self.loop_controller.should_continue:
   887→            iteration_num = self.loop_controller.current_iteration + 1
   888→            iter_start = time.time()
   889→
   890→            iteration = LoopIteration(
   891→                iteration=iteration_num,
   892→                state_entered=LoopState.COMPILING,
   893→                timestamp_start=iter_start,
   894→            )
   895→
   896→            # === STEP 1: COMPILE ===
   897→            iteration.state_entered = LoopState.COMPILING
   898→            try:
   899→                compiler = Compiler(target_language=self.target_language)
   900→                result = compiler.compile_to_file(source, output_dir, source_path=source_path)
   901→
   902→                if not result.success:
   903→                    iteration.errors.append(f"Compilation failed: {'; '.join(result.errors)}")
   904→                    iteration.state_exited = LoopState.FAILED
   905→                    iteration.timestamp_end = time.time()
   906→                    self.loop_controller.record_iteration(iteration)
   907→                    continue
   908→
   909→                # Parse for verification step
   910→                try:
   911→                    parsed = Parser().parse(source)
   912→                except Exception:
   913→                    parsed = None
   914→
   915→                # Count fallbacks from provenance
   916→                fallback_count = 0
   917→                if hasattr(compiler, '_provenance') and compiler._provenance:
   918→                    for rec in compiler._provenance.records:
   919→                        if rec.source_type == ProvenanceType.FALLBACK:
   920→                            fallback_count += 1
   921→                            # Record for pattern learning
   922→                            self.pattern_learner.record_unmatched(
   923→                                rec.source_text[:50] if rec.source_text else "unknown",
   924→                                rec.generated_code[:200] if rec.generated_code else ""
   925→                            )
   926→
   927→                iteration.fallback_count = fallback_count
   928→
   929→                # Get audit coverage
   930→                if hasattr(compiler, '_provenance') and compiler._provenance:
   931→                    audit = compiler._provenance.compute_audit_coverage()
   932→                    iteration.audit_coverage = audit.get("audit_coverage", 0.0)
   933→
   934→            except Exception as e:
   935→                iteration.errors.append(f"Compilation exception: {str(e)}")
   936→                iteration.state_exited = LoopState.FAILED
   937→                iteration.timestamp_end = time.time()
   938→                self.loop_controller.record_iteration(iteration)
   939→                continue
   940→
   941→            # === STEP 2: VERIFY SPEC ===
   942→            iteration.state_entered = LoopState.VERIFYING
   943→            try:
   944→                from .spec_verify import SpecificationVerifier
   945→                if parsed is not None:
   946→                    verifier = SpecificationVerifier(parsed)
   947→                    report = verifier.verify(source_name=source_path)
   948→
   949→                    spec_verified = all(
   950→                        c.get("status") != "FAIL"
   951→                        for c in report.get("checks", [])
   952→                    )
   953→
   954→                    if not spec_verified:
   955→                        # Auto-fix spec verification failures
   956→                        source, changes = self.spec_evolver.evolve_from_verification(source, report)
   957→                        iteration.spec_changes.extend(changes)
   958→                        iteration.fixes_applied.extend(changes)
   959→                else:
   960→                    spec_verified = False
   961→
   962→            except Exception as e:
   963→                iteration.errors.append(f"Verification exception: {str(e)}")
   964→                spec_verified = False
   965→
   966→            # === STEP 3: RUN TESTS ===
   967→            iteration.state_entered = LoopState.TESTING
   968→            try:
   969→                passed, failed, total, failures = self.test_runner.run_tests(output_dir)
   970→                iteration.tests_passed = passed
   971→                iteration.tests_failed = failed
   972→                iteration.tests_total = total
   973→
   974→                if failed > 0:
   975→                    # === STEP 4: DIAGNOSE FAILURES ===
   976→                    iteration.state_entered = LoopState.DIAGNOSING
   977→                    for failure in failures:
   978→                        test_name = failure.get("test", "unknown")
   979→                        diagnosis = self.test_runner.analyze_failure(test_name, output_dir)
   980→
   981→                        # === STEP 5: FIX ===
   982→                        iteration.state_entered = LoopState.FIXING
   983→                        source, changes = self.spec_evolver.evolve_from_test_failure(source, diagnosis)
   984→                        iteration.fixes_applied.extend(changes)
   985→
   986→            except Exception as e:
   987→                iteration.errors.append(f"Test execution exception: {str(e)}")
   988→                iteration.tests_total = 1
   989→                iteration.tests_failed = 1
   990→
   991→            # === STEP 6: LEARN PATTERNS ===
   992→            if iteration.fallback_count > 0:
   993→                iteration.state_entered = LoopState.LEARNING
   994→                for unmatched in self.pattern_learner.unmatched_descriptions:
   995→                    parts = unmatched.split(": ", 1)
   996→                    if len(parts) == 2:
   997→                        pattern = self.pattern_learner.learn_from_description(parts[0], parts[1])
   998→                        if pattern:
   999→                            iteration.patterns_learned.append(pattern["name"])
  1000→
  1001→                # Clear after learning
  1002→                self.pattern_learner.unmatched_descriptions.clear()
  1003→
  1004→            # === STEP 7: CHECK AUDIT COVERAGE ===
  1005→            if iteration.audit_coverage < 1.0:
  1006→                iteration.state_entered = LoopState.FIXING
  1007→                source, changes = self.spec_evolver.evolve_from_audit_gap(
  1008→                    source,
  1009→                    {
  1010→                        "audit_coverage": iteration.audit_coverage,
  1011→                        "orphan_artifacts": [],
  1012→                    }
  1013→                )
  1014→                iteration.fixes_applied.extend(changes)
  1015→
  1016→            # Record iteration
  1017→            iteration.state_exited = (
  1018→                LoopState.CONVERGED if iteration.success else LoopState.DIAGNOSING
  1019→            )
  1020→            iteration.timestamp_end = time.time()
  1021→            self.loop_controller.record_iteration(iteration)
  1022→
  1023→            # Record provenance for this iteration
  1024→            self._provenance_log.append({
  1025→                "iteration": iteration_num,
  1026→                "duration": iteration.duration,
  1027→                "audit_coverage": iteration.audit_coverage,
  1028→                "tests_passed": iteration.tests_passed,
  1029→                "tests_failed": iteration.tests_failed,
  1030→                "fallback_count": iteration.fallback_count,
  1031→                "patterns_learned": iteration.patterns_learned,
  1032→                "fixes_applied": iteration.fixes_applied,
  1033→                "spec_changes": iteration.spec_changes,
  1034→            })
  1035→
  1036→        # === FINAL: Verify proof ===
  1037→        proof_valid = False
  1038→        proof_path = os.path.join(output_dir, "main.aicl-proof")
  1039→        if os.path.exists(proof_path):
  1040→            try:
  1041→                from .provenance import ProofOfOrigin
  1042→                proof = ProofOfOrigin.from_file(proof_path)
  1043→                valid, checks = proof.verify()
  1044→                proof_valid = valid
  1045→            except Exception:
  1046→                proof_valid = False
  1047→
  1048→        # Compute final metrics
  1049→        total_duration = time.time() - start_time
  1050→        final_history = self.loop_controller.history
  1051→        final_coverage = final_history[-1].get("audit_coverage", 0) if final_history else 0
  1052→        final_pass_rate = final_history[-1].get("test_pass_rate", 0) if final_history else 0
  1053→
  1054→        # If source was modified, write it back
  1055→        if source != original_source:
  1056→            evolved_path = source_path.replace(".aicl", ".evolved.aicl")
  1057→            with open(evolved_path, "w") as f:
  1058→                f.write(source)
  1059→
  1060→        return AutonomousResult(
  1061→            converged=self.loop_controller.is_converged,
  1062→            iterations=self.loop_controller.iterations,
  1063→            total_duration=total_duration,
  1064→            final_audit_coverage=final_coverage,
  1065→            final_test_pass_rate=final_pass_rate,
  1066→            patterns_learned=[
  1067→                p["name"] for p in self.pattern_learner.learned_patterns.values()
  1068→            ],
  1069→            spec_evolution=[
  1070→                entry["change"] for entry in self.spec_evolver.evolution_log
  1071→            ],
  1072→            convergence_iteration=self.loop_controller.convergence_iteration,
  1073→            proof_valid=proof_valid,
  1074→        )
