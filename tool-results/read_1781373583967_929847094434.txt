    69→    69→def highlight_aicl(source: str) -> Text:
    70→    70→    """Apply basic syntax highlighting to AICL source."""
    71→    71→    text = Text()
    72→    72→    for line in source.split("\n"):
    73→    73→        stripped = line.lstrip()
    74→    74→        if stripped.startswith("#"):
    75→    75→            text.append(line, style="dim italic")
    76→    76→        elif any(stripped.startswith(kw + ":") or stripped.startswith(kw + " ") for kw in AICL_KEYWORDS):
    77→    77→            # Find the keyword
    78→    78→            for kw in AICL_KEYWORDS:
    79→    79→                if stripped.startswith(kw + ":") or stripped.startswith(kw + " "):
    80→    80→                    indent = len(line) - len(stripped)
    81→    81→                    text.append(" " * indent)
    82→    82→                    text.append(kw, style="bold cyan")
    83→    83→                    rest = stripped[len(kw):]
    84→    84→                    # Color the colon
    85→    85→                    if rest.startswith(":"):
    86→    86→                        text.append(":", style="bold cyan")
    87→    87→                        rest = rest[1:]
    88→    88→                    text.append(rest, style="white")
    89→    89→                    break
    90→    90→            else:
    91→    91→                text.append(line, style="white")
    92→    92→        elif any(stripped.startswith(t + " ") or stripped == t for t in AICL_TYPES):
    93→    93→            text.append(line, style="green")
    94→    94→        else:
    95→    95→            text.append(line, style="white")
    96→    96→        text.append("\n")
    97→    97→    return text

   384→   384→    def _cmd_compile(self, args, output, editor):
   385→   385→        if not self.current_file:
   386→   386→            output.write("[red]No file loaded. Use :open <path> first.[/]")
   387→   387→            return
   388→   388→        target = "python"
   389→   389→        if args.strip() in ("rust", "javascript", "js", "go"):
   390→   390→            target = "javascript" if args.strip() == "js" else args.strip()
   391→   391→
   392→   392→        output.write(f"[bold]Compiling[/] {self.current_file} → {target}...")
   393→   393→        try:
   394→   394→            result = self.compiler.compile(self.current_file, target=target)
   395→   395→            self.last_output_dir = result.get("output_dir", "")
   396→   396→            coverage = result.get("audit_coverage", 0) * 100
   397→   397→            todos = result.get("todos_remaining", 0)
   398→   398→            valid = result.get("proof_valid", False)
   399→   399→
   400→   400→            status = "[green]VALID[/]" if valid else "[red]INVALID[/]"
   401→   401→            output.write(f"""[green]Compilation successful![/]
   402→   402→  Target:        {target}
   403→   403→  Output:        {self.last_output_dir}
   404→   404→  TODOs:         {todos}
   405→   405→  Audit:         {coverage:.1f}%
   406→   406→  Proof:         {status}""")
   407→   407→        except Exception as e:
   408→   408→            output.write(f"[red]Compilation failed:[/] {e}")

   410→   410→    def _cmd_verify(self, args, output, editor):
   411→   411→        if not self.current_file:
   412→   412→            output.write("[red]No file loaded. Use :open <path> first.[/]")
   413→   413→            return
   414→   414→        output.write("[bold]Verifying specification...[/]")
   415→   415→        try:
   416→   416→            with open(self.current_file) as f:
   417→   417→                source = f.read()
   418→   418→            result = verify_source(source)
   419→   419→            checks = result.get("checks", [])
   420→   420→            passed = sum(1 for c in checks if c.get("status") == "pass")
   421→   421→            failed = sum(1 for c in checks if c.get("status") == "fail")
   422→   422→            warned = sum(1 for c in checks if c.get("status") == "warn")
   423→   423→            overall = result.get("overall", "UNKNOWN")
   424→   424→            color = "green" if overall == "PASS" else "yellow" if overall == "WARN" else "red"
   425→   425→            output.write(f"[{color}]Overall: {overall}[/]  ({passed} passed, {warned} warnings, {failed} failed)")
   426→   426→            for check in checks:
   427→   427→                icon = {"pass": "[green]PASS[/]", "warn": "[yellow]WARN[/]", "fail": "[red]FAIL[/]"}
   428→   428→                status = icon.get(check["status"], check["status"])
   429→   429→                output.write(f"  {status}  {check.get('name', '')}")
   430→   430→        except Exception as e:
   431→   431→            output.write(f"[red]Verification failed:[/] {e}")

   453→   453→    def _cmd_proof(self, args, output, editor):
