   176→def make_table(data, col_widths=None, header=True):
   177→    """Create a styled table."""
   178→    if col_widths is None:
   179→        col_widths = [CONTENT_W / len(data[0])] * len(data[0])
   180→
   181→    t = Table(data, colWidths=col_widths, repeatRows=1 if header else 0)
   182→
   183→    style_cmds = [
   184→        ('FONTNAME', (0, 0), (-1, -1), 'Carlito'),
   185→        ('FONTSIZE', (0, 0), (-1, -1), 9),
   186→        ('LEADING', (0, 0), (-1, -1), 13),
   187→        ('TEXTCOLOR', (0, 0), (-1, -1), TEXT_PRIMARY),
   188→        ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
   189→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   190→        ('TOPPADDING', (0, 0), (-1, -1), 4),
   191→        ('BOTTOMPADDING', (0, 0), (-1, -1), 4),
   192→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   193→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   194→    ]
   195→
   196→    if header:
   197→        style_cmds.extend([
   198→            ('BACKGROUND', (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   199→            ('TEXTCOLOR', (0, 0), (-1, 0), TABLE_HEADER_TEXT),
   200→            ('FONTNAME', (0, 0), (-1, 0), 'CarlitoBold'),
   201→        ])
   202→
   203→    for i in range(1 if header else 0, len(data)):
   204→        bg = TABLE_ROW_EVEN if i % 2 == 0 else TABLE_ROW_ODD
   205→        style_cmds.append(('BACKGROUND', (0, i), (-1, i), bg))
   206→
   207→    style_cmds.append(('GRID', (0, 0), (-1, -1), 0.5, colors.HexColor('#cccccc')))
   208→    t.setStyle(TableStyle(style_cmds))
   209→    return t
