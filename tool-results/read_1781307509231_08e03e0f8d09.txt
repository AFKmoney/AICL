     1→     1→"""
     2→     2→AICL Compilation Provenance & Audit System
     3→     3→
     4→     4→Every line of generated code must have a traceable provenance chain:
     5→     5→    AICL Source → Parse → Pattern Match → Template → Generated Code
     6→     6→
     7→     7→This module records and reports WHY each line of code was generated,
     8→     8→and provides an audit system to verify that every generated artifact
     9→     9→is traceable to its originating specification.
    10→    10→
    11→    11→Core Concepts:
    12→    12→    - Provenance Record: Explains why a specific piece of code was generated
    13→    13→    - Generated Artifact: A named unit of generated code (class, method, function, test)
    14→    14→    - Audit Coverage: Ratio of auditable artifacts to total generated artifacts
    15→    15→    - Orphan Artifact: A generated artifact without provenance
    16→    16→
    17→    17→Formal Property (The Auditability Theorem):
    18→    18→    A program is auditable if every generated artifact
    19→    19→    can be traced to its originating specification
    20→    20→    through a complete provenance chain.
    21→    21→
    22→    22→    Audit Coverage = Auditable Artifacts / Generated Artifacts
    23→    23→    Target: Audit Coverage = 1.0
    24→    24→
    25→    25→Usage:
    26→    26→    aicl explain pong.aicl              # Full compilation trace
    27→    27→    aicl explain pong.aicl --behavior MovePaddle  # Specific behavior trace
    28→    28→    aicl audit pong.aicl                # Audit report with coverage
    29→    29→    aicl audit pong.aicl --strict       # Fail if coverage < 100%
    30→    30→"""
    31→    31→
    32→    32→from dataclasses import dataclass, field

    73→    73→@dataclass
    74→    74→class ProvenanceRecord:
    75→    75→    """
    76→    76→    A single provenance record explaining why a piece of code was generated.
    77→    77→
    78→    78→    This is the audit trail of the compiler. Every generated method,
    79→    79→    every line of behavior code, every recovery handler must have one.
    80→    80→    """
    81→    81→    source_type: ProvenanceType
    82→    82→    source_location: str          # Where in the AICL source (e.g., "Behavior MovePaddle")
    83→    83→    source_text: str              # The original AICL text
    84→    84→    resolution_path: List[str]    # Chain: ["AICL Source", "Parser", "Pattern Match", "MOVE", "Template"]
    85→    85→    generated_code: str           # The code that was generated

    87→    87→    pattern_name: str = ""        # Pattern that matched (if applicable)
    88→    88→    template_name: str = ""       # Architecture template (if applicable)
    89→    89→    parameters: Dict[str, str] = field(default_factory=dict)  # Parameters used
    90→    90→    artifact_names: List[str] = field(default_factory=list)    # Which artifacts this record covers

    93→    93→@dataclass
    94→    94→class GeneratedArtifact:
    95→    95→    """
    96→    96→    A named unit of generated code.
    97→    97→
    98→    98→    Every method, class, function, and dataclass produced by the compiler
    99→    99→    is registered as an artifact. The audit system verifies that each
   100→   100→    artifact has at least one provenance record explaining its existence.
   101→   101→
   102→   102→    An artifact without provenance is an "orphan" — it exists in the
   103→   103→    generated code but the compiler cannot explain why.
   104→   104→    """
   105→   105→    name: str                           # Human-readable name (e.g., "_behavior_move_paddle")
   106→   106→    artifact_type: ArtifactType         # Type of artifact
   107→   107→    source: str                         # Which AICL section triggered generation
   108→   108→    provenance_indices: List[int] = field(default_factory=list)  # Indices into records list
   109→   109→    code_snippet: str = ""              # Brief snippet of the generated code (first line)
   110→   110→
   111→   111→    @property
   112→   112→    def has_provenance(self) -> bool:
   113→   113→        """Whether this artifact has at least one provenance record."""
   114→   114→        return len(self.provenance_indices) > 0
   115→   115→
   116→   116→    @property
   117→   117→    def is_orphan(self) -> bool:
   118→   118→        """Whether this artifact is an orphan (no provenance)."""
   119→   119→        return not self.has_provenance

   122→   122→class CompilationProvenance:
   123→   123→    """
   124→   124→    Tracks the complete provenance of a compilation and provides
   125→   125→    audit capabilities.
   126→   126→
   127→   127→    Records every decision the compiler makes, creating an audit trail
   128→   128→    that can be queried to understand WHY any line of code exists.
   129→   129→
   130→   130→    This is the antidote to compiler complexity hiding. As AICL grows
   131→   131→    (30 patterns → 100 patterns → 500 patterns), the provenance tracker
   132→   132→    ensures every decision remains explainable.
   133→   133→
   134→   134→    v0.5: Added artifact tracking and audit coverage computation.
   135→   135→    """
   136→   136→
   137→   137→    def __init__(self):
   138→   138→        self.records: List[ProvenanceRecord] = []
   139→   139→        self.artifacts: List[GeneratedArtifact] = []
   140→   140→
   141→   141→    def record(
   142→   142→        self,
   143→   143→        source_type: ProvenanceType,
   144→   144→        source_location: str,
   145→   145→        source_text: str,
   146→   146→        resolution_path: List[str],
   147→   147→        generated_code: str,
   148→   148→        confidence: float = 1.0,
   149→   149→        pattern_name: str = "",
   150→   150→        template_name: str = "",
   151→   151→        parameters: Dict[str, str] = None,
   152→   152→        artifact_names: List[str] = None,
   153→   153→    ) -> ProvenanceRecord:
   154→   154→        """Record a compilation decision."""
   155→   155→        rec = ProvenanceRecord(
   156→   156→            source_type=source_type,
   157→   157→            source_location=source_location,
   158→   158→            source_text=source_text,
   159→   159→            resolution_path=resolution_path,
   160→   160→            generated_code=generated_code,
   161→   161→            confidence=confidence,
   162→   162→            pattern_name=pattern_name,
   163→   163→            template_name=template_name,
   164→   164→            parameters=parameters or {},
   165→   165→            artifact_names=artifact_names or [],
   166→   166→        )
   167→   167→        self.records.append(rec)
   168→   168→
   169→   169→        # Link this provenance record to any matching artifacts
   170→   170→        record_index = len(self.records) - 1
   171→   171→        for artifact_name in (artifact_names or []):
   172→   172→            for artifact in self.artifacts:
   173→   173→                if artifact.name == artifact_name:
   174→   174→                    artifact.provenance_indices.append(record_index)
   175→   175→
   176→   176→        return rec

   178→   178→    def register_artifact(
   179→   179→        self,
   180→   180→        name: str,
   181→   181→        artifact_type: ArtifactType,
   182→   182→        source: str,
   183→   183→        code_snippet: str = "",
   184→   184→    ) -> GeneratedArtifact:
   185→   185→        """
   186→   186→        Register a generated artifact.
   187→   187→
   188→   188→        Every method, class, function, and dataclass produced by the
   189→   189→        compiler should be registered. The audit system will verify
   190→   190→        that each artifact has at least one provenance record.
   191→   191→        """
   192→   192→        artifact = GeneratedArtifact(
   193→   193→            name=name,
   194→   194→            artifact_type=artifact_type,
   195→   195→            source=source,
   196→   196→            code_snippet=code_snippet,
   197→   197→        )
   198→   198→        self.artifacts.append(artifact)
   199→   199→
   200→   200→        # Check if any existing provenance records already cover this artifact
   201→   201→        for i, rec in enumerate(self.records):
   202→   202→            if name in rec.artifact_names:
   203→   203→                artifact.provenance_indices.append(i)
   204→   204→
   205→   205→        return artifact

   207→   207→    def link_provenance_to_artifact(self, record_index: int, artifact_name: str):
   208→   208→        """Explicitly link a provenance record to an artifact."""
   209→   209→        if record_index < len(self.records):
   210→   210→            if artifact_name not in self.records[record_index].artifact_names:
   211→   211→                self.records[record_index].artifact_names.append(artifact_name)
   212→   212→
   213→   213→        for artifact in self.artifacts:
   214→   214→            if artifact.name == artifact_name:
   215→   215→                if record_index not in artifact.provenance_indices:
   216→   216→                    artifact.provenance_indices.append(record_index)

   222→   222→    def explain(self, target: str = None) -> str:
   223→   223→        """
   224→   224→        Generate an explanation of the compilation.
   225→   225→
   226→   226→        Args:
   227→   227→            target: Optional filter (behavior name, pattern name, etc.)
   228→   228→
   229→   229→        Returns:
   230→   230→            Human-readable explanation of all compilation decisions.
   231→   231→        """
   232→   232→        lines = []
   233→   233→        lines.append("=" * 70)
   234→   234→        lines.append("AICL COMPILATION PROVENANCE TRACE")
   235→   235→        lines.append("=" * 70)
   236→   236→        lines.append("")
   237→   237→
   238→   238→        # Summary
   239→   239→        total = len(self.records)
   240→   240→        pattern_count = sum(1 for r in self.records if r.source_type == ProvenanceType.PATTERN_MATCH)
   241→   241→        sublang_count = sum(1 for r in self.records if r.source_type == ProvenanceType.SUB_LANGUAGE)
   242→   242→        fallback_count = sum(1 for r in self.records if r.source_type == ProvenanceType.FALLBACK)
   243→   243→        template_count = sum(1 for r in self.records if r.source_type == ProvenanceType.ARCHITECTURE_TEMPLATE)
   244→   244→        direct_count = sum(1 for r in self.records if r.source_type == ProvenanceType.DIRECT_MAPPING)
   245→   245→
   246→   246→        lines.append(f"Total decisions: {total}")
   247→   247→        lines.append(f"  Pattern matches:    {pattern_count}")
   248→   248→        lines.append(f"  Sub-language:       {sublang_count}")
   249→   249→        lines.append(f"  Direct mappings:    {direct_count}")
   250→   250→        lines.append(f"  Architecture tmpl:  {template_count}")
   251→   251→        lines.append(f"  Fallbacks:          {fallback_count}")
   252→   252→        lines.append("")
   253→   253→
   254→   254→        # Confidence score
   255→   255→        if total > 0:
   256→   256→            avg_confidence = sum(r.confidence for r in self.records) / total
   257→   257→            fully_deterministic = sum(1 for r in self.records if r.confidence >= 0.9)
   258→   258→            lines.append(f"Average confidence:    {avg_confidence:.2f}")
   259→   259→            lines.append(f"Fully deterministic:   {fully_deterministic}/{total} ({100*fully_deterministic//max(total,1)}%)")
   260→   260→            lines.append("")
   261→   261→
   262→   262→        # Group by source type
   263→   263→        by_type: Dict[ProvenanceType, List[ProvenanceRecord]] = {}
   264→   264→        for rec in self.records:
   265→   265→            if rec.source_type not in by_type:
   266→   266→                by_type[rec.source_type] = []
   267→   267→            by_type[rec.source_type].append(rec)
   268→   268→
   269→   269→        for source_type, records in by_type.items():
   270→   270→            type_label = source_type.value.replace("_", " ").title()
   271→   271→            lines.append("-" * 70)
   272→   272→            lines.append(f" {type_label} ({len(records)} decisions)")
   273→   273→            lines.append("-" * 70)
   274→   274→
   275→   275→            for i, rec in enumerate(records):
   276→   276→                # Filter by target if specified
   277→   277→                if target and target.lower() not in rec.source_location.lower() and target.lower() not in rec.pattern_name.lower():
   278→   278→                    continue
   279→   279→
   280→   280→                lines.append(f"")
   281→   281→                lines.append(f"  [{i+1}] {rec.source_location}")
   282→   282→                lines.append(f"  Source:  \"{rec.source_text}\"")
   283→   283→                lines.append(f"  Path:    {' → '.join(rec.resolution_path)}")
   284→   284→
   285→   285→                if rec.pattern_name:
   286→   286→                    lines.append(f"  Pattern: {rec.pattern_name}")
   287→   287→                if rec.template_name:
   288→   288→                    lines.append(f"  Template: {rec.template_name}")
   289→   289→                if rec.parameters:
   290→   290→                    params_str = ", ".join(f"{k}={v}" for k, v in rec.parameters.items() if not k.startswith("#"))
   291→   291→                    if params_str:
   292→   292→                        lines.append(f"  Params:  {params_str}")
   293→   293→
   294→   294→                lines.append(f"  Confidence: {rec.confidence:.2f}")
   295→   295→
   296→   296→                # Show generated code (truncated)
   297→   297→                code_lines = rec.generated_code.strip().split('\n')
   298→   298→                if len(code_lines) <= 4:
   299→   299→                    for cl in code_lines:
   300→   300→                        lines.append(f"  Code:    {cl}")
   301→   301→                else:
   302→   302→                    for cl in code_lines[:3]:
   303→   303→                        lines.append(f"  Code:    {cl}")
   304→   304→                    lines.append(f"  Code:    ... ({len(code_lines) - 3} more lines)")
   305→   305→
   306→   306→        lines.append("")
   307→   307→        lines.append("=" * 70)
   308→   308→        lines.append("END OF PROVENANCE TRACE")
   309→   309→        lines.append("=" * 70)
   310→   310→
   311→   311→        return '\n'.join(lines)

   370→   370→    def compute_explicability_coverage(self, generated_source: str, generated_tests: str = "") -> Dict:
   371→   371→        """
   372→   372→        Compute the explicability coverage ratio.
   373→   373→
   374→   374→        The explicability coverage measures what fraction of generated code
   375→   375→        lines are accounted for by provenance records. The target is 1.0
   376→   376→        (every line has provenance). Any deviation is a gap to investigate.
   377→   377→
   378→   378→        Returns:
   379→   379→            Dict with keys:
   380→   380→                - total_lines: total non-blank, non-comment lines in generated code
   381→   381→                - accounted_lines: lines that appear in at least one provenance record
   382→   382→                - coverage_ratio: accounted_lines / total_lines (target: 1.0)
   383→   383→                - unaccounted_lines: list of line numbers without provenance
   384→   384→                - by_type_coverage: coverage broken down by provenance type
   385→   385→        """
   386→   386→        # Combine source and test code
   387→   387→        all_code = generated_source
   388→   388→        if generated_tests:
   389→   389→            all_code += "\n" + generated_tests
   390→   390→
   391→   391→        # Count meaningful lines (non-blank, non-import, non-pure-comment)
   392→   392→        code_lines = []
   393→   393→        for i, line in enumerate(all_code.split('\n')):
   394→   394→            stripped = line.strip()
   395→   395→            if not stripped:
   396→   396→                continue
   397→   397→            # Skip pure comment lines and import lines for coverage
   398→   398→            if stripped.startswith('#') and not any(kw in stripped for kw in ['When:', 'On:', 'Risk:', 'Recovery:', 'Action:', 'Intent:']):
   399→   399→                continue
   400→   400→            if stripped.startswith(('import ', 'from ')):
   401→   401→                continue
   402→   402→            if stripped in ('"""', "'''", ''):
   403→   403→                continue
   404→   404→            code_lines.append((i + 1, stripped))
   405→   405→
   406→   406→        total_lines = len(code_lines)
   407→   407→        if total_lines == 0:
   408→   408→            return {
   409→   409→                "total_lines": 0,
   410→   410→                "accounted_lines": 0,
   411→   411→                "coverage_ratio": 1.0,
   412→   412→                "unaccounted_lines": [],
   413→   413→                "by_type_coverage": {},
   414→   414→            }
   415→   415→
   416→   416→        # Build set of code fragments from provenance records
   417→   417→        accounted_fragments = set()
   418→   418→        for rec in self.records:
   419→   419→            for code_line in rec.generated_code.strip().split('\n'):
   420→   420→                stripped = code_line.strip()
   421→   421→                if stripped:
   422→   422→                    # Normalize for matching (remove variable values that change)
   423→   423→                    normalized = re.sub(r'[\'"].*?[\'\"]', "'...'", stripped)
   424→   424→                    normalized = re.sub(r'\{[^}]*\}', '{...}', normalized)
   425→   425→                    accounted_fragments.add(normalized)
   426→   426→                    # Also add the original
   427→   427→                    accounted_fragments.add(stripped)
   428→   428→
   429→   429→        # Check which lines are accounted for
   430→   430→        accounted_lines = 0
   431→   431→        unaccounted = []
   432→   432→        for line_num, line_text in code_lines:
   433→   433→            stripped = line_text.strip()
   434→   434→            # Check if this line or a normalized version appears in provenance
   435→   435→            found = False
   436→   436→            if stripped in accounted_fragments:
   437→   437→                found = True
   438→   438→            else:
   439→   439→                # Try normalizing the source line too
   440→   440→                normalized = re.sub(r'[\'"].*?[\'\"]', "'...'", stripped)
   441→   441→                normalized = re.sub(r'\{[^}]*\}', '{...}', normalized)
   442→   442→                if normalized in accounted_fragments:
   443→   443→                    found = True
   444→   444→                else:
   445→   445→                    # Check if any provenance fragment is a prefix of this line
   446→   446→                    for frag in accounted_fragments:
   447→   447→                        if frag and stripped.startswith(frag.split('=')[0].split('(')[0].strip()):
   448→   448→                            found = True
   449→   449→                            break
   450→   450→
   451→   451→            if found:
   452→   452→                accounted_lines += 1
   453→   453→            else:
   454→   454→                # Only flag as unaccounted if it's a significant line
   455→   455→                if not stripped.startswith(('#', '"""', "'''", 'pass', 'else:', 'try:', 'except', 'finally:')):
   456→   456→                    unaccounted.append((line_num, stripped[:80]))
   457→   457→
   458→   458→        # Coverage by provenance type
   459→   459→        by_type_coverage = {}
   460→   460→        for ptype in ProvenanceType:
   461→   461→            count = len([r for r in self.records if r.source_type == ptype])
   462→   462→            if count > 0:
   463→   463→                by_type_coverage[ptype.value] = count
   464→   464→
   465→   465→        return {
   466→   466→            "total_lines": total_lines,
   467→   467→            "accounted_lines": accounted_lines,
   468→   468→            "coverage_ratio": accounted_lines / total_lines if total_lines > 0 else 1.0,
   469→   469→            "unaccounted_lines": unaccounted,
   470→   470→            "by_type_coverage": by_type_coverage,
   471→   471→        }
   472→   472→
   473→   473→    def compute_audit_coverage(self) -> Dict:
   474→   474→        """
   475→   475→        Compute the audit coverage of generated artifacts.
   476→   476→
   477→   477→        Audit Coverage = Auditable Artifacts / Generated Artifacts
   478→   478→
   479→   479→        An artifact is "auditable" if it has at least one provenance record.
   480→   480→        An artifact without provenance is an "orphan".
   481→   481→
   482→   482→        This is the key metric for the Auditability property:
   483→   483→            "A program is auditable if every generated artifact
   484→   484→             can be traced to its originating specification
   485→   485→             through a complete provenance chain."
   486→   486→
   487→   487→        Returns:
   488→   488→            Dict with:
   489→   489→                - total_artifacts: total number of generated artifacts
   490→   490→                - auditable_artifacts: artifacts with at least one provenance record
   491→   491→                - orphan_artifacts: list of artifacts without provenance
   492→   492→                - audit_coverage: ratio of auditable to total (target: 1.0)
   493→   493→                - by_type: breakdown by artifact type
   494→   494→        """
   495→   495→        total = len(self.artifacts)
   496→   496→        if total == 0:
   497→   497→            return {
   498→   498→                "total_artifacts": 0,
   499→   499→                "auditable_artifacts": 0,
   500→   500→                "orphan_artifacts": [],
   501→   501→                "audit_coverage": 1.0,
   502→   502→                "by_type": {},
   503→   503→            }
   504→   504→
   505→   505→        auditable = [a for a in self.artifacts if a.has_provenance]
   506→   506→        orphans = [a for a in self.artifacts if a.is_orphan]
   507→   507→
   508→   508→        # Breakdown by type
   509→   509→        by_type = {}
   510→   510→        for atype in ArtifactType:
   511→   511→            type_artifacts = [a for a in self.artifacts if a.artifact_type == atype]
   512→   512→            type_auditable = [a for a in type_artifacts if a.has_provenance]
   513→   513→            if type_artifacts:
   514→   514→                by_type[atype.value] = {
   515→   515→                    "total": len(type_artifacts),
   516→   516→                    "auditable": len(type_auditable),
   517→   517→                    "coverage": len(type_auditable) / len(type_artifacts),
   518→   518→                }
   519→   519→
   520→   520→        return {
   521→   521→            "total_artifacts": total,
   522→   522→            "auditable_artifacts": len(auditable),
   523→   523→            "orphan_artifacts": [{"name": a.name, "type": a.artifact_type.value, "source": a.source} for a in orphans],
   524→   524→            "audit_coverage": len(auditable) / total,
   525→   525→            "by_type": by_type,
   526→   526→        }

   528→   528→    def explain_coverage(self, generated_source: str, generated_tests: str = "") -> str:
   529→   529→        """Generate a coverage report for the explicable compilation property."""
   530→   530→        coverage = self.compute_explicability_coverage(generated_source, generated_tests)
   531→   531→        lines = []
   532→   532→        lines.append("=" * 70)
   533→   533→        lines.append("EXPLICABILITY COVERAGE REPORT")
   534→   534→        lines.append("=" * 70)
   535→   535→        lines.append("")
   536→   536→        lines.append(f"Total meaningful lines:   {coverage['total_lines']}")
   537→   537→        lines.append(f"Lines with provenance:    {coverage['accounted_lines']}")
   538→   538→        lines.append(f"Coverage ratio:           {coverage['coverage_ratio']:.3f}")
   539→   539→        lines.append("")
   540→   540→
   541→   541→        if coverage['coverage_ratio'] >= 0.95:
   542→   542→            lines.append("Status: PASS (coverage >= 95%)")
   543→   543→        elif coverage['coverage_ratio'] >= 0.80:
   544→   544→            lines.append("Status: PARTIAL (coverage 80-95%, investigation needed)")
   545→   545→        else:
   546→   546→            lines.append("Status: FAIL (coverage < 80%, provenance gaps detected)")
   547→   547→
   548→   548→        lines.append("")
   549→   549→
   550→   550→        if coverage['by_type_coverage']:
   551→   551→            lines.append("Provenance by type:")
   552→   552→            for ptype, count in sorted(coverage['by_type_coverage'].items(), key=lambda x: -x[1]):
   553→   553→                lines.append(f"  {ptype:30s} {count:4d} records")
   554→   554→            lines.append("")
   555→   555→
   556→   556→        if coverage['unaccounted_lines']:
   557→   557→            lines.append(f"Unaccounted lines ({len(coverage['unaccounted_lines'])}):")
   558→   558→            for line_num, text in coverage['unaccounted_lines'][:20]:
   559→   559→                lines.append(f"  L{line_num:4d}: {text}")
   560→   560→            if len(coverage['unaccounted_lines']) > 20:
   561→   561→                lines.append(f"  ... and {len(coverage['unaccounted_lines']) - 20} more")
   562→   562→        else:
   563→   563→            lines.append("All lines have provenance coverage.")
   564→   564→
   565→   565→        lines.append("")
   566→   566→        lines.append("=" * 70)
   567→   567→        return '\n'.join(lines)

   569→   569→    def audit(self, generated_source: str = "", generated_tests: str = "") -> str:
   570→   570→        """
   571→   571→        Generate a complete audit report.
   572→   572→
   573→   573→        This is the primary output of the audit system. It verifies the
   574→   574→        Auditability property:
   575→   575→
   576→   576→            "A program is auditable if every generated artifact
   577→   577→             can be traced to its originating specification
   578→   578→             through a complete provenance chain."
   579→   579→
   580→   580→        The audit report contains:
   581→   581→            1. Artifact summary (total, auditable, orphans)
   582→   582→            2. Audit coverage ratio (target: 1.0)
   583→   583→            3. Orphan artifact list (artifacts without provenance)
   584→   584→            4. Audit status (VERIFIED / PARTIAL / FAILED)
   585→   585→            5. Explicability coverage (line-level)
   586→   586→            6. Breakdown by artifact type
   587→   587→
   588→   588→        Returns:
   589→   589→            Human-readable audit report.
   590→   590→        """
   591→   591→        lines = []
   592→   592→        lines.append("=" * 70)
   593→   593→        lines.append("AICL AUDIT REPORT")
   594→   594→        lines.append("=" * 70)
   595→   595→        lines.append("")
   596→   596→
   597→   597→        # Artifact audit
   598→   598→        audit = self.compute_audit_coverage()
   599→   599→
   600→   600→        lines.append("ARTIFACT AUDIT")
   601→   601→        lines.append("-" * 70)
   602→   602→        lines.append(f"  Generated artifacts:     {audit['total_artifacts']}")
   603→   603→        lines.append(f"  Artifacts with provenance: {audit['auditable_artifacts']}")
   604→   604→        lines.append(f"  Orphan artifacts:        {len(audit['orphan_artifacts'])}")
   605→   605→        lines.append("")
   606→   606→
   607→   607→        coverage_pct = audit['audit_coverage'] * 100
   608→   608→        lines.append(f"  Audit Coverage:          {coverage_pct:.2f}%")
   609→   609→        lines.append("")
   610→   610→
   611→   611→        # Audit status
   612→   612→        if audit['audit_coverage'] >= 1.0:
   613→   613→            lines.append("  Audit Status: VERIFIED")
   614→   614→        elif audit['audit_coverage'] >= 0.90:
   615→   615→            lines.append("  Audit Status: PARTIAL (investigation needed)")
   616→   616→        else:
   617→   617→            lines.append("  Audit Status: FAILED (significant provenance gaps)")
   618→   618→        lines.append("")
   619→   619→
   620→   620→        # Orphan artifacts
   621→   621→        if audit['orphan_artifacts']:
   622→   622→            lines.append("ORPHAN ARTIFACTS (no provenance)")
   623→   623→            lines.append("-" * 70)
   624→   624→            for orphan in audit['orphan_artifacts']:
   625→   625→                lines.append(f"  [{orphan['type']:15s}] {orphan['name']}")
   626→   626→                lines.append(f"    Source: {orphan['source']}")
   627→   627→            lines.append("")
   628→   628→
   629→   629→        # Breakdown by artifact type
   630→   630→        if audit['by_type']:
   631→   631→            lines.append("COVERAGE BY ARTIFACT TYPE")
   632→   632→            lines.append("-" * 70)
   633→   633→            for atype, info in sorted(audit['by_type'].items(), key=lambda x: -x[1]['total']):
   634→   634→                cov_pct = info['coverage'] * 100
   635→   635→                status = "VERIFIED" if info['coverage'] >= 1.0 else "PARTIAL" if info['coverage'] >= 0.9 else "FAILED"
   636→   636→                lines.append(f"  {atype:15s}  {info['auditable']:3d}/{info['total']:3d}  {cov_pct:6.2f}%  [{status}]")
   637→   637→            lines.append("")
   638→   638→
   639→   639→        # Line-level explicability coverage
   640→   640→        if generated_source:
   641→   641→            exp_coverage = self.compute_explicability_coverage(generated_source, generated_tests)
   642→   642→            lines.append("LINE-LEVEL EXPLICABILITY")
   643→   643→            lines.append("-" * 70)
   644→   644→            lines.append(f"  Total meaningful lines:  {exp_coverage['total_lines']}")
   645→   645→            lines.append(f"  Lines with provenance:   {exp_coverage['accounted_lines']}")
   646→   646→            lines.append(f"  Explicability coverage:  {exp_coverage['coverage_ratio']:.2%}")
   647→   647→
   648→   648→            if exp_coverage['coverage_ratio'] >= 0.95:
   649→   649→                lines.append("  Line status: PASS (>= 95%)")
   650→   650→            elif exp_coverage['coverage_ratio'] >= 0.80:
   651→   651→                lines.append("  Line status: PARTIAL (80-95%)")
   652→   652→            else:
   653→   653→                lines.append("  Line status: FAIL (< 80%)")
   654→   654→
   655→   655→            if exp_coverage['unaccounted_lines']:
   656→   656→                lines.append(f"  Unaccounted lines: {len(exp_coverage['unaccounted_lines'])}")
   657→   657→            lines.append("")
   658→   658→
   659→   659→        # Provenance summary
   660→   660→        lines.append("PROVENANCE SUMMARY")
   661→   661→        lines.append("-" * 70)
   662→   662→        total_records = len(self.records)
   663→   663→        lines.append(f"  Total provenance records: {total_records}")
   664→   664→
   665→   665→        if total_records > 0:
   666→   666→            avg_conf = sum(r.confidence for r in self.records) / total_records
   667→   667→            lines.append(f"  Average confidence:       {avg_conf:.2f}")
   668→   668→            deterministic = sum(1 for r in self.records if r.confidence >= 0.9)
   669→   669→            lines.append(f"  Deterministic decisions:  {deterministic}/{total_records} ({100*deterministic//total_records}%)")
   670→   670→
   671→   671→            # By type summary
   672→   672→            by_type = {}
   673→   673→            for rec in self.records:
   674→   674→                label = rec.source_type.value.replace("_", " ").title()
   675→   675→                by_type[label] = by_type.get(label, 0) + 1
   676→   676→
   677→   677→            lines.append("  Provenance by type:")
   678→   678→            for label, count in sorted(by_type.items(), key=lambda x: -x[1]):
   679→   679→                lines.append(f"    {label:30s} {count:4d}")
   680→   680→        lines.append("")
   681→   681→
   682→   682→        # Final verdict
   683→   683→        lines.append("=" * 70)
   684→   684→        if audit['audit_coverage'] >= 1.0 and (not generated_source or exp_coverage['coverage_ratio'] >= 0.95):
   685→   685→            lines.append("AUDIT RESULT: PASSED")
   686→   686→            lines.append("  Every generated artifact has provenance.")
   687→   687→            lines.append("  Compilation is fully auditable.")
   688→   688→        elif audit['audit_coverage'] >= 0.90:
   689→   689→            lines.append("AUDIT RESULT: PARTIAL")
   690→   690→            lines.append("  Most artifacts have provenance.")
   691→   691→            lines.append("  Some orphan artifacts need investigation.")
   692→   692→        else:
   693→   693→            lines.append("AUDIT RESULT: FAILED")
   694→   694→            lines.append("  Significant provenance gaps detected.")
   695→   695→            lines.append("  Compilation is not fully auditable.")
   696→   696→        lines.append("=" * 70)
   697→   697→
   698→   698→        return '\n'.join(lines)

   700→   700→    def audit_passed(self, generated_source: str = "", generated_tests: str = "") -> bool:
   701→   701→        """
   702→   702→        Check whether the audit passes.
   703→   703→
   704→   704→        Audit passes if:
   705→   705→            1. Audit coverage = 1.0 (no orphan artifacts)
   706→   706→            2. Explicability coverage >= 0.95 (line-level)
   707→   707→
   708→   708→        Used by `aicl audit --strict` to determine exit code.
   709→   709→        """
   710→   710→        audit = self.compute_audit_coverage()
   711→   711→        if audit['audit_coverage'] < 1.0:
   712→   712→            return False
   713→   713→
   714→   714→        if generated_source:
   715→   715→            exp = self.compute_explicability_coverage(generated_source, generated_tests)
   716→   716→            if exp['coverage_ratio'] < 0.95:
   717→   717→                return False
   718→   718→
   719→   719→        return True
