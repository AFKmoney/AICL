    86→class Compiler:
    87→    """
    88→    AICL Multi-Stage Compiler v0.5.
    89→
    90→    Compiles AICL source code into deterministic, executable Python code
    91→    using behavior patterns and architecture templates.
    92→
    93→    Every generated artifact is tracked and must have provenance.
    94→    The audit system verifies: Audit Coverage = 1.0 (no orphans).
    95→
    96→    Usage:
    97→        compiler = Compiler()
    98→        result = compiler.compile(source_code)
    99→        if result.success:
   100→            print(result.source_code)
   101→    """
   102→
   103→    def __init__(self, target_language: str = "python"):
   104→        self.target_language = target_language
   105→        self.warnings: List[str] = []
   106→        self.errors: List[str] = []
   107→        self.stages_completed: List[str] = []
   108→        self._used_names: Set[str] = set()
   109→        self._behavior_compiler = BehaviorCompiler()
   110→        self._todo_count = 0
   111→        self._provenance = CompilationProvenance()
   112→
   113→    def compile(self, source: str) -> CompilationResult:
   114→        """Compile AICL source code into executable Python code."""
   115→        self.warnings = []
   116→        self.errors = []
   117→        self.stages_completed = []
   118→        self._used_names = set()
   119→        self._todo_count = 0
   120→        self._provenance = CompilationProvenance()
   121→
   122→        result = CompilationResult()
   123→
   124→        # Stage 1: Specification Parsing
   125→        program = self._stage_parsing(source)
   126→        if program is None:
   127→            result.success = False
   128→            result.errors = self.errors
   129→            return result
   130→        self.stages_completed.append("Specification Parsing")
   131→
   132→        # Stage 2: Architecture Validation
   133→        validation_warnings = self._stage_architecture_validation(program)
   134→        self.warnings.extend(validation_warnings)
   135→        self.stages_completed.append("Architecture Validation")
   136→
   137→        # Stage 3: Dependency Analysis
   138→        dep_order = self._stage_dependency_analysis(program)
   139→        self.stages_completed.append("Dependency Analysis")
   140→
   141→        # Stage 4: Risk Analysis
   142→        risk_pairs = self._stage_risk_analysis(program)
   143→        self.stages_completed.append("Risk Analysis")
   144→
   145→        # Stage 5: Recovery Synthesis
   146→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   147→        self.stages_completed.append("Recovery Synthesis")
   148→
   149→        # Stage 6: Code Generation
   150→        source_code = self._stage_code_generation(
   151→            program, dep_order, risk_pairs, recovery_map
   152→        )
   153→        self.stages_completed.append("Code Generation")
   154→
   155→        # Stage 7: Test Generation
   156→        test_code = self._stage_test_generation(program, dep_order)
   157→        self.stages_completed.append("Test Generation")
   158→
   159→        # Stage 8: Optimization
   160→        source_code = self._stage_optimization(source_code, program)
   161→        self.stages_completed.append("Optimization")
   162→
   163→        # Stage 9: Final Construction
   164→        source_code = self._stage_final_construction(source_code, program)
   165→        self.stages_completed.append("Final Construction")
   166→
   167→        # Build Architecture Tree string
   168→        tree = ArchitectureTree(program)
   169→        result.architecture_tree_str = tree.to_string()
   170→
   171→        result.success = True
   172→        result.source_code = source_code
   173→        result.test_code = test_code
   174→        result.warnings = self.warnings
   175→        result.errors = self.errors
   176→        result.stages_completed = self.stages_completed
   177→        result.todo_count = self._todo_count
   178→        result.fully_compiled = self._todo_count == 0
   179→        result.provenance = self._provenance
   180→
   181→        # Generate Proof of Origin — the central artifact
   182→        result.proof = self._provenance.to_proof(
   183→            source_text=source,
   184→            generated_source=source_code,
   185→            generated_tests=test_code,
   186→            compiler_version="0.6.0",
   187→            target_language=self.target_language,
   188→        )
   189→
   190→        return result
   191→
   192→    def compile_to_file(self, source: str, output_dir: str, source_path: str = "") -> CompilationResult:
   193→        """
   194→        Compile AICL source code and write the output to files.
   195→
   196→        Produces:
   197→            - main.py              (executable program)
   198→            - test_main.py         (generated tests)
   199→            - architecture_tree.txt (architecture visualization)
   200→            - main.aicl-proof      (Proof of Origin — first-class artifact)
   201→
   202→        The .aicl-proof file is a self-contained, verifiable record that
   203→        contains ALL information needed to reconstruct explain() and audit()
   204→        without the compiler. It is the central artifact of auditable compilation.
   205→        """
   206→        result = self.compile(source)
   207→
   208→        if result.success:
   209→            os.makedirs(output_dir, exist_ok=True)
   210→
   211→            main_path = os.path.join(output_dir, "main.py")
   212→            with open(main_path, 'w') as f:
   213→                f.write(result.source_code)
   214→
   215→            test_path = os.path.join(output_dir, "test_main.py")
   216→            with open(test_path, 'w') as f:
   217→                f.write(result.test_code)
   218→
   219→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   220→            with open(tree_path, 'w') as f:
   221→                f.write(result.architecture_tree_str)
   222→
   223→            # Write Proof of Origin — the central artifact
   224→            if result.proof:
   225→                # Update source_path in proof if provided
   226→                if source_path:
   227→                    result.proof.source_path = source_path
   228→                proof_path = os.path.join(output_dir, "main.aicl-proof")
   229→                result.proof.to_file(proof_path)
   230→
   231→        return result

   237→    def _stage_parsing(self, source: str) -> Optional[AICLProgram]:
   238→        """Parse the AICL source code into an AST."""
   239→        try:
   240→            parser = Parser(source)
   241→            program = parser.parse()
   242→            return program
   243→        except (ParseError, Exception) as e:
   244→            self.errors.append(f"Parse error: {str(e)}")
   245→            return None

   248→    # Stage 2: Architecture Validation
   249→    # =========================================================================
   250→
   251→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   252→        """Validate the architectural structure of the program."""
   253→        warnings = program.validate()
   254→
   255→        layer_names = [l.name for l in program.layers]
   256→        seen = set()
   257→        for name in layer_names:
   258→            if name.lower() in seen:
   259→                warnings.append(f"Duplicate layer name: {name}")
   260→            seen.add(name.lower())
   261→
   262→        entity_names = {e.name for e in program.entities}
   263→        for behavior in program.behaviors:
   264→            for inp in behavior.inputs:
   265→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   266→                    warnings.append(
   267→                        f"Behavior '{behavior.name}' references unknown type "
   268→                        f"'{inp.param_type}' in input"
   269→                    )
   270→
   271→        return warnings

   281→    # =========================================================================
   282→    # Stage 4: Risk Analysis
   283→    # =========================================================================
   284→
   285→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   286→        """Analyze risks and pair them with recovery procedures."""
   287→        pairs = []
   288→        for i, risk in enumerate(program.risks):
   289→            recovery = None
   290→            if i < len(program.recoveries):
   291→                recovery = program.recoveries[i]
   292→            pairs.append((risk, recovery))
   293→
   294→        if len(program.recoveries) > len(program.risks):
   295→            for i in range(len(program.risks), len(program.recoveries)):
   296→                generic_risk = RiskSection(
   297→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   298→                )
   299→                pairs.append((generic_risk, program.recoveries[i]))
   300→
   301→        return pairs
   302→
   303→    # =========================================================================
   304→    # Stage 5: Recovery Synthesis
   305→    # =========================================================================
   306→
   307→    def _stage_recovery_synthesis(
   308→        self,
   309→        program: AICLProgram,
   310→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]
   311→    ) -> Dict[str, str]:
   312→        """Synthesize recovery procedures for each risk."""
   313→        recovery_map = {}
   314→        for risk, recovery in risk_pairs:
   315→            if recovery:
   316→                recovery_map[risk.description] = recovery.description
   317→            else:
   318→                recovery_map[risk.description] = f"log and handle: {risk.description}"
   319→        return recovery_map

   325→    def _unique_name(self, base: str) -> str:
   326→        """Generate a unique Python identifier from a base name."""
   327→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   328→        clean = re.sub(r'_+', '_', clean).strip('_')
   329→        if not clean:
   330→            clean = "item"
   331→        if clean[0].isdigit():
   332→            clean = "f_" + clean
   333→
   334→        name = clean
   335→        counter = 2
   336→        while name in self._used_names:
   337→            name = f"{clean}_{counter}"
   338→            counter += 1
   339→
   340→        self._used_names.add(name)
   341→        return name

   343→    def _stage_code_generation(
   344→        self,
   345→        program: AICLProgram,
   346→        dep_order: List[LayerSection],
   347→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   348→        recovery_map: Dict[str, str]
   349→    ) -> str:
   350→        """Generate executable Python code from the AICL program."""
   351→        sections = []
   352→
   353→        goal_desc = program.goals[0].description if program.goals else "AICL Generated Application"
   354→
   355→        # Record architecture template provenance
   356→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   357→        self._provenance.record(
   358→            source_type=ProvenanceType.ARCHITECTURE_TEMPLATE,
   359→            source_location="Goal",
   360→            source_text=goal_desc,
   361→            resolution_path=[
   362→                "AICL Source",
   363→                f"Goal: {goal_desc}",
   364→                "ArchitectureTemplateMapper.match()",
   365→                f"Template: {template_name}",
   366→                f"Structure: {template_info['structure']}",
   367→                "Generated Code",
   368→            ],
   369→            generated_code=f"# Architecture template: {template_name} ({template_info['structure']})",
   370→            confidence=1.0,
   371→            template_name=template_name,
   372→            artifact_names=["architecture_template"],
   373→        )
   374→
   375→        sections.append('"""')
   376→        sections.append(f'AICL Generated Application')
   377→        sections.append(f'Goal: {goal_desc}')
   378→        sections.append(f'')
   379→        sections.append(f'Auto-generated by the AICL compiler v0.5.')
   380→        sections.append(f'Auditable compilation — every artifact has provenance.')
   381→        sections.append(f'"""')
   382→        sections.append('')
   383→
   384→        # Imports — with provenance
   385→        import_code = self._generate_imports(program)
   386→        sections.append(import_code)
   387→        sections.append('')
   388→
   389→        # Entity classes — with provenance (already implemented)
   390→        for entity in program.entities:
   391→            sections.append(self._generate_entity(entity))
   392→            sections.append('')
   393→
   394→        # Main application class
   395→        class_name = self._make_class_name(goal_desc)
   396→        sections.append(self._generate_application_class(
   397→            program, class_name, dep_order, risk_pairs, recovery_map
   398→        ))
   399→        sections.append('')
   400→
   401→        # Entry point — with provenance
   402→        sections.append(self._generate_entry_point(class_name))
   403→
   404→        return '\n'.join(sections)

   406→    def _generate_imports(self, program: AICLProgram) -> str:
   407→        """Generate import statements based on program content."""
   408→        imports = [
   409→            "import sys",
   410→            "import logging",
   411→            "import time",
   412→            "from typing import Optional, List, Dict, Any",
   413→        ]
   414→
   415→        if program.parallels:
   416→            imports.append("import threading")
   417→            imports.append("from concurrent.futures import ThreadPoolExecutor, as_completed")
   418→
   419→        needs_datetime = False
   420→        for entity in program.entities:
   421→            for f in entity.fields:
   422→                if f.field_type == "datetime":
   423→                    needs_datetime = True
   424→                    break
   425→
   426→        if needs_datetime:
   427→            imports.append("from datetime import datetime")
   428→
   429→        imports.append("from dataclasses import dataclass, field")
   430→
   431→        # Check if we need hashlib for security
   432→        if program.securities:
   433→            for sec in program.securities:
   434→                for action in sec.actions:
   435→                    if action.action_type == "encrypt":
   436→                        imports.append("import hashlib")
   437→                        imports.append("import base64")
   438→                        break
   439→
   440→        code = '\n'.join(imports)
   441→
   442→        # Register import artifact and record provenance
   443→        self._provenance.register_artifact(
   444→            name="imports",
   445→            artifact_type=ArtifactType.IMPORT_BLOCK,
   446→            source="Program structure",
   447→            code_snippet="import sys, logging, time, ...",
   448→        )
   449→        self._provenance.record(
   450→            source_type=ProvenanceType.IMPORT_GENERATION,
   451→            source_location="Program Structure",
   452→            source_text="Required imports based on program features",
   453→            resolution_path=[
   454→                "AICL Source",
   455→                "Program Analysis",
   456→                "Feature Detection (entities, security, parallel)",
   457→                "Import Resolution",
   458→                "Generated Code",
   459→            ],
   460→            generated_code=code,
   461→            confidence=1.0,
   462→            artifact_names=["imports"],
   463→        )
   464→
   465→        return code

   467→    def _generate_entity(self, entity: EntitySection) -> str:
   468→        """Generate a Python dataclass for an AICL Entity."""
   469→        class_name = self._make_class_name(entity.name)
   470→        artifact_name = f"Entity_{class_name}"
   471→        lines = []
   472→        lines.append('@dataclass')
   473→        lines.append(f'class {class_name}:')
   474→        lines.append(f'    """Entity: {entity.name}"""')
   475→
   476→        if not entity.fields:
   477→            lines.append('    pass')
   478→        else:
   479→            for f in entity.fields:
   480→                python_type = self._aicl_type_to_python(f.field_type)
   481→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   482→                if default:
   483→                    lines.append(f'    {f.name}: {python_type} = {default}')
   484→                else:
   485→                    lines.append(f'    {f.name}: {python_type} = 0')
   486→
   487→        code = '\n'.join(lines)
   488→
   489→        # Register artifact
   490→        self._provenance.register_artifact(
   491→            name=artifact_name,
   492→            artifact_type=ArtifactType.DATACLASS,
   493→            source=f"Entity {entity.name}",
   494→            code_snippet=f"class {class_name}:",
   495→        )
   496→
   497→        # Record entity generation provenance
   498→        fields_desc = ', '.join(f'{f.name}: {f.field_type}' for f in entity.fields) if entity.fields else 'no fields'
   499→        self._provenance.record(
   500→            source_type=ProvenanceType.ENTITY_GENERATION,
   501→            source_location=f"Entity {entity.name}",
   502→            source_text=fields_desc,
   503→            resolution_path=[
   504→                "AICL Source",
   505→                f"Entity {entity.name}",
   506→                f"Fields: {fields_desc}",
   507→                "Dataclass Generation",
   508→                "Generated Code",
   509→            ],
   510→            generated_code=code,
   511→            confidence=1.0,
   512→            parameters={"class_name": class_name, "fields": str(len(entity.fields))},
   513→            artifact_names=[artifact_name],
   514→        )
   515→
   516→        return code

   518→    def _generate_application_class(
   519→        self,
   520→        program: AICLProgram,
   521→        class_name: str,
   522→        dep_order: List[LayerSection],
   523→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   524→        recovery_map: Dict[str, str]
   525→    ) -> str:
   526→        """Generate the main application class."""
   527→        lines = []
   528→
   529→        # Determine architecture template from Goal
   530→        goal_desc = program.goals[0].description if program.goals else ""
   531→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   532→
   533→        # Register the class artifact
   534→        class_artifact = f"Class_{class_name}"
   535→        self._provenance.register_artifact(
   536→            name=class_artifact,
   537→            artifact_type=ArtifactType.CLASS,
   538→            source=f"Goal: {goal_desc}",
   539→            code_snippet=f"class {class_name}:",
   540→        )
   541→
   542→        lines.append(f'class {class_name}:')
   543→        lines.append(f'    """')
   544→        if program.goals:
   545→            lines.append(f'    Goal: {program.goals[0].description}')
   546→        lines.append(f'    Architecture: {template_info["structure"]}')
   547→        lines.append(f'    """')
   548→        lines.append('')
   549→
   550→        # Class attributes - one per layer
   551→        lines.append('    # Layer instances')
   552→        layer_attr_map = {}
   553→        for layer in dep_order:
   554→            attr_name = self._unique_name(layer.name)
   555→            layer_attr_map[layer.name] = attr_name
   556→            lines.append(f'    _{attr_name}: Any = None')
   557→        lines.append('')
   558→
   559→        # Entity instances
   560→        if program.entities:
   561→            lines.append('    # Entity instances')
   562→            for entity in program.entities:
   563→                attr = self._unique_name(entity.name.lower())
   564→                lines.append(f'    _{attr}: Any = None')
   565→            lines.append('')
   566→
   567→        # Risks as class constants
   568→        if program.risks:
   569→            lines.append('    # Known risks')
   570→            for i, risk in enumerate(program.risks):
   571→                lines.append(f'    RISK_{i+1} = {risk.description!r}')
   572→            lines.append('')
   573→
   574→        # Constraints as class constants
   575→        if program.constraints:
   576→            lines.append('    # Constraints')
   577→            for i, con in enumerate(program.constraints):
   578→                lines.append(f'    CONSTRAINT_{i+1} = {con.description!r}')
   579→            lines.append('')
   580→
   581→        # Game/application state
   582→        lines.append('    # Application state')
   583→        lines.append('    _running: bool = False')
   584→        lines.append('    _initialized: bool = False')
   585→        if template_name == "game_loop":
   586→            lines.append('    _game_active: bool = False')
   587→            lines.append('    _frame_count: int = 0')
   588→            lines.append('    _last_frame_time: float = 0.0')
   589→        elif template_name == "event_driven":

   598→            source_type=ProvenanceType.CLASS_STRUCTURE,

   611→            template_name=template_name,
   612→            artifact_names=[class_artifact],
   613→        )
   614→
   615→        # __init__ method
   616→        init_artifact = f"Method_{class_name}.__init__"
   617→        init_lines = []
   618→        init_lines.append('    def __init__(self):')
   619→        init_lines.append('        """Initialize the application and all layers."""')
   620→        init_lines.append('        self._logger = logging.getLogger(self.__class__.__name__)')
   621→        init_lines.append('        self._running = False')
   622→        init_lines.append('        self._initialized = False')
   623→        if template_name == "game_loop":
   624→            init_lines.append('        self._game_active = True')
   625→            init_lines.append('        self._frame_count = 0')
   626→            init_lines.append('        self._last_frame_time = time.time()')
   627→        elif template_name == "event_driven":
   628→            init_lines.append('        self._connected = False')
   629→            init_lines.append('        self._message_queue = []')
   630→            init_lines.append('        self._offline_queue = []')
   631→        init_lines.append('')
   632→
   633→        # Initialize entity instances
   634→        if program.entities:
   635→            init_lines.append('        # Initialize entities')
   636→            for entity in program.entities:
   637→                entity_class = self._make_class_name(entity.name)
   638→                attr = entity.name.lower()
   639→                init_lines.append(f'        self._{attr} = {entity_class}()')
   640→            init_lines.append('')
   641→
   642→        # Initialize each layer
   643→        for layer in dep_order:
   644→            attr_name = layer_attr_map[layer.name]
   645→            init_lines.append(f'        # Initialize {layer.name}')
   646→            init_lines.append(f'        self._{attr_name} = self._init_{attr_name}()')
   647→            init_lines.append('')
   648→
   649→        init_lines.append('        self._initialized = True')
   650→        init_lines.append('        self._logger.info("Application initialized successfully")')
   651→        init_lines.append('')
   652→
   653→        # Register __init__ artifact
   654→        self._provenance.register_artifact(
   655→            name=init_artifact,
   656→            artifact_type=ArtifactType.METHOD,
   657→            source=f"Class {class_name}",
   658→            code_snippet="def __init__(self):",
   659→        )
   660→
   661→        # Record __init__ provenance
   662→        init_code = '\n'.join(init_lines)
   663→        self._provenance.record(
   664→            source_type=ProvenanceType.CLASS_STRUCTURE,
   665→            source_location=f"Method __init__",
   666→            source_text=f"Initialize {len(dep_order)} layers, {len(program.entities)} entities",
   667→            resolution_path=[
   668→                "AICL Source",
   669→                f"Class {class_name}",

   674→            ],
   675→            generated_code=init_code,
   676→            confidence=1.0,
   677→            template_name=template_name,
   678→            artifact_names=[init_artifact],
   679→        )
   680→
   681→        lines.extend(init_lines)
   682→
   683→        # Layer initialization methods
   684→        for layer in dep_order:
   685→            attr_name = layer_attr_map[layer.name]
   686→            lines.append(self._generate_layer_init(layer, attr_name, risk_pairs, recovery_map))
   687→            lines.append('')
   688→
   689→        # Pre-compute behavior method names (must be done before run method generation)
   690→        behavior_method_names = {}
   691→        for i, behavior in enumerate(program.behaviors):
   692→            method_name = self._unique_name(behavior.name).lower()
   693→            behavior_method_names[behavior.name] = method_name
   694→

   696→        lines.append(self._generate_run_method(program, dep_order, layer_attr_map, risk_pairs, recovery_map, template_name, behavior_method_names))
   697→        lines.append('')
   698→
   699→        # Behavior methods

   701→            method_name = behavior_method_names[behavior.name]
   702→            lines.append(self._generate_behavior_method(behavior, method_name, program))

   720→        # Validation methods

   723→            lines.append('')
   724→
   725→        # Security methods
   726→        if program.securities:
   727→            lines.append(self._generate_security_methods(program))
   728→            lines.append('')
   729→
   730→        # Risk handling method
   731→        lines.append(self._generate_risk_handler(risk_pairs, recovery_map, program))
   732→
   733→        # Helper methods
   734→        lines.append('')
   735→        lines.append(self._generate_helper_methods(program, template_name))

   739→    def _generate_layer_init(
   740→        self,
   741→        layer: LayerSection,
   742→        attr_name: str,
   743→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   744→        recovery_map: Dict[str, str]
   745→    ) -> str:
   746→        """Generate initialization method for a layer — deterministic, no TODOs."""
   747→        lines = []
   748→        lines.append(f'    def _init_{attr_name}(self) -> Any:')
   749→        lines.append(f'        """Initialize the {layer.name} layer."""')
   750→        lines.append(f'        self._logger.info("Initializing {layer.name}...")')
   751→        lines.append(f'        try:')
   752→
   753→        # Generate layer component as a dict
   754→        sublayer_names = [s.name for s in layer.sublayers]
   755→        lines.append(f'            component = {{')
   756→        lines.append(f'                "name": "{layer.name}",')
   757→        lines.append(f'                "active": True,')
   758→        lines.append(f'                "sublayers": {sublayer_names!r},')
   759→        lines.append(f'            }}')
   760→
   761→        # Generate sublayer initialization
   762→        for sublayer in layer.sublayers:
   763→            sub_attr = self._unique_name(sublayer.name)
   764→            lines.append(f'            # Sublayer: {sublayer.name}')
   765→            lines.append(f'            component["{sub_attr}"] = {{"name": "{sublayer.name}", "active": True}}')
   766→
   767→        lines.append(f'            self._logger.info("{layer.name} initialized successfully")')
   768→        lines.append(f'            return component')
   769→        lines.append(f'        except Exception as e:')
   770→        lines.append(f'            self._logger.error(f"Failed to initialize {layer.name}: {{e}}")')
   771→
   772→        # Add recovery from risk pairs that match this layer
   773→        for risk, recovery in risk_pairs:
   774→            if recovery and self._risk_matches_layer(risk.description, layer.name):
   775→                recovery_code, _ = self._behavior_compiler.compile_action(
   776→                    recovery.description,
   777→                    context={}
   778→                )
   779→                lines.append(f'            # Recovery: {recovery.description}')
   780→                lines.append(f'            self._logger.info("Applying recovery: {recovery.description}")')
   781→                for code_line in recovery_code.split('\n'):
   782→                    if code_line.strip():
   783→                        lines.append(f'            {code_line}')
   784→
   785→        lines.append(f'            return None')
   786→
   787→        code = '\n'.join(lines)
   788→
   789→        # Register artifact
   790→        artifact_name = f"Method__init_{attr_name}"
   791→        self._provenance.register_artifact(
   792→            name=artifact_name,
   793→            artifact_type=ArtifactType.METHOD,
   794→            source=f"Layer {layer.name}",
   795→            code_snippet=f"def _init_{attr_name}(self):",
   796→        )
   797→
   798→        # Record layer initialization provenance
   799→        self._provenance.record(
   800→            source_type=ProvenanceType.LAYER_INITIALIZATION,
   801→            source_location=f"Layer {layer.name}",
   802→            source_text=f"Sublayers: {', '.join(sublayer_names)}" if sublayer_names else "No sublayers",
   803→            resolution_path=[
   804→                "AICL Source",
   805→                f"Layer {layer.name}",
   806→                f"Sublayers: {sublayer_names}" if sublayer_names else "No sublayers",
   807→                "Component Dict Generation",
   808→                "Generated Code",
   809→            ],
   810→            generated_code=code,
   811→            confidence=1.0,
   812→            parameters={"layer_name": layer.name, "sublayers": str(len(sublayer_names))},
   813→            artifact_names=[artifact_name],
   814→        )
   815→
   816→        return code

   818→    def _generate_run_method(
   819→        self,
   820→        program: AICLProgram,
   821→        dep_order: List[LayerSection],
   822→        layer_attr_map: Dict[str, str],
   823→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   824→        recovery_map: Dict[str, str],
   825→        template_name: str,
   826→        behavior_method_names: Dict[str, str] = None,
   827→    ) -> str:
   828→        """Generate the main run method — uses architecture template."""
   829→        behavior_method_names = behavior_method_names or {}
   830→        lines = []
   831→        lines.append('    def run(self) -> None:')
   832→        lines.append('        """Run the application with full error handling."""')
   833→        lines.append('        self._logger.info("Starting application...")')
   834→        lines.append('        self._running = True')
   835→        lines.append('')
   836→
   837→        if template_name == "game_loop":
   838→            # Game loop: init → update → render → validate → loop
   839→            lines.append('        self._game_active = True')
   840→            lines.append('        try:')
   841→            lines.append('            while self._running and self._game_active:')
   842→            lines.append('                frame_start = time.time()')
   843→            lines.append('')
   844→            lines.append('                # Update phase')
   845→            for behavior in program.behaviors:
   846→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   847→                input_params = ', '.join(
   848→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   849→                    {e.name.lower() for e in program.entities}
   850→                    else inp.name
   851→                    for inp in behavior.inputs
   852→                )
   853→                lines.append(f'                self._behavior_{method_name}({input_params})')
   854→            lines.append('')
   855→            lines.append('                # Check conditions')
   856→            for i, condition in enumerate(program.conditions):
   857→                lines.append(f'                self._handle_condition_{i+1}()')
   858→            lines.append('')
   859→            lines.append('                # Render phase')
   860→            for layer in dep_order:
   861→                attr_name = layer_attr_map[layer.name]
   862→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   863→                    lines.append(f'                if self._{attr_name} is not None:')
   864→                    lines.append(f'                    self._render_frame()')
   865→            lines.append('')
   866→            lines.append('                # Frame rate control')
   867→            lines.append('                elapsed = time.time() - frame_start')
   868→            lines.append('                if elapsed < 1.0 / 60.0:')
   869→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   870→            lines.append('                self._frame_count += 1')
   871→            lines.append('')
   872→
   873→        elif template_name == "event_driven":
   874→            # Event-driven: connect → listen → handle → respond
   875→            lines.append('        try:')
   876→            lines.append('            self._connect()')
   877→            lines.append('            self._connected = True')
   878→            lines.append('            self._logger.info("Connected to server")')
   879→            lines.append('')
   880→            lines.append('            while self._running and self._connected:')
   881→            lines.append('                # Process message queue')
   882→            lines.append('                while self._message_queue:')
   883→            lines.append('                    msg = self._message_queue.pop(0)')
   884→            lines.append('                    self._process_message(msg)')
   885→            lines.append('')
   886→            lines.append('                # Process offline queue if reconnected')
   887→            lines.append('                if self._offline_queue and self._connected:')
   888→            lines.append('                    self._sync_offline_messages()')
   889→            lines.append('')
   890→            lines.append('                # Wait for new messages')
   891→            lines.append('                time.sleep(0.1)')
   892→
   893→        else:
   894→            # Default: linear execution
   895→            lines.append('        try:')
   896→            for layer in dep_order:
   897→                attr_name = layer_attr_map[layer.name]
   898→                lines.append(f'            # Execute {layer.name}')
   899→                lines.append(f'            if self._{attr_name} is not None:')
   900→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   901→                lines.append(f'            else:')
   902→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   903→
   904→        # Validations
   905→        lines.append('')
   906→        lines.append('            # Run validations')
   907→        for i, validation in enumerate(program.validations):
   908→            lines.append(f'            self._validate_{i+1}()')
   909→
   910→        lines.append('')
   911→        lines.append('            self._logger.info("Application completed successfully")')
   912→        lines.append('')
   913→        lines.append('        except Exception as e:')
   914→        lines.append('            self._logger.error(f"Application error: {e}")')
   915→        lines.append('            self._handle_risk(e)')
   916→        lines.append('        finally:')
   917→        lines.append('            self._running = False')
   918→        lines.append('            self._logger.info("Application stopped")')
   919→
   920→        code = '\n'.join(lines)
   921→
   922→        # Register artifact
   923→        artifact_name = "Method_run"
   924→        self._provenance.register_artifact(
   925→            name=artifact_name,
   926→            artifact_type=ArtifactType.METHOD,
   927→            source=f"Architecture Template: {template_name}",
   928→            code_snippet="def run(self) -> None:",
   929→        )
   930→
   931→        # Record run method provenance
   932→        _, run_template_info = ArchitectureTemplateMapper.match(
   933→            program.goals[0].description if program.goals else ""
   934→        )
   935→        self._provenance.record(
   936→            source_type=ProvenanceType.RUN_METHOD,
   937→            source_location="Method run",
   938→            source_text=f"Template: {template_name}, Behaviors: {len(program.behaviors)}, Conditions: {len(program.conditions)}",
   939→            resolution_path=[
   940→                "AICL Source",
   941→                f"Goal → Template: {template_name}",
   942→                f"Structure: {run_template_info.get('structure', 'unknown')}",
   943→                "Run Method Generation",
   944→                "Generated Code",
   945→            ],
   946→            generated_code=code,
   947→            confidence=1.0,
   948→            template_name=template_name,
   949→            artifact_names=[artifact_name],
   950→        )
   951→
   952→        return code

   954→    def _generate_behavior_method(
   955→        self, behavior: BehaviorSection, method_name: str, program: AICLProgram
   956→    ) -> str:
   957→        """Generate a behavior method using pattern matching — no TODOs."""
   958→        lines = []
   959→
   960→        # Build parameter list from inputs
   961→        params = ["self"]
   962→        for inp in behavior.inputs:
   963→            # Check if input references an entity
   964→            entity_names = {e.name for e in program.entities}
   965→            if inp.name in entity_names:
   966→                params.append(f'{inp.name.lower()}: Any')
   967→            elif inp.param_type in entity_names:
   968→                params.append(f'{inp.name.lower()}: Any')
   969→            else:
   970→                params.append(f'{inp.name.lower()}: Any = None')
   971→
   972→        param_str = ', '.join(params)
   973→        lines.append(f'    def _behavior_{method_name}({param_str}) -> Any:')
   974→        lines.append(f'        """Behavior: {behavior.name}"""')
   975→
   976→        # Compile the action using the Behavior Compiler
   977→        entity_context = {}
   978→        for inp in behavior.inputs:
   979→            entity_names = {e.name for e in program.entities}
   980→            if inp.name in entity_names:
   981→                entity_context["entity"] = inp.name.lower()
   982→            elif inp.param_type in entity_names:
   983→                entity_context["entity"] = inp.param_type.lower()
   984→
   985→        action_code, fully_compiled = self._behavior_compiler.compile_action(
   986→            behavior.action, context=entity_context
   987→        )
   988→
   989→        if not fully_compiled:
   990→            self._todo_count += 1
   991→
   992→        # Register artifact
   993→        artifact_name = f"Method__behavior_{method_name}"
   994→        self._provenance.register_artifact(
   995→            name=artifact_name,
   996→            artifact_type=ArtifactType.METHOD,
   997→            source=f"Behavior {behavior.name}",
   998→            code_snippet=f"def _behavior_{method_name}(...):",
   999→        )
  1000→
  1001→        # Record provenance
  1002→        self._record_behavior_provenance(behavior, action_code, fully_compiled, entity_context, artifact_name)
  1003→
  1004→        # Indent the compiled code
  1005→        for code_line in action_code.split('\n'):
  1006→            lines.append(f'        {code_line}')
  1007→
  1008→        return '\n'.join(lines)

  1071→    def _generate_condition_handler(
  1072→        self, condition: ConditionSection, index: int, program: AICLProgram
  1073→    ) -> str:
  1074→        """Generate a condition handler using pattern matching — no TODOs."""
  1075→        lines = []
  1076→        lines.append(f'    def _handle_condition_{index}(self) -> None:')
  1077→        when_desc = condition.when_clause or "unknown condition"
  1078→        then_desc = condition.then_clause or "unknown action"
  1079→        lines.append(f'        """Condition: WHEN {when_desc} THEN {then_desc}"""')
  1080→
  1081→        # Compile the WHEN clause
  1082→        when_code, when_compiled = self._behavior_compiler.compile_action(
  1083→            f"check {when_desc}",
  1084→            context={}
  1085→        )
  1086→
  1087→        # Compile the THEN clause
  1088→        then_code, then_compiled = self._behavior_compiler.compile_action(
  1089→            then_desc,
  1090→            context={}
  1091→        )
  1092→
  1093→        if not when_compiled or not then_compiled:
  1094→            self._todo_count += 1
  1095→
  1096→        # Generate the condition check
  1097→        when_lower = when_desc.lower()
  1098→
  1099→        # Detect common condition patterns
  1100→        if any(w in when_lower for w in ['hit', 'collision', 'collide', 'contact']):
  1101→            lines.append(f'        # When: {when_desc}')
  1102→            lines.append(f'        collision = self._check_collisions()')
  1103→            lines.append(f'        if collision:')
  1104→            lines.append(f'            self._logger.info("Condition triggered: {when_desc}")')
