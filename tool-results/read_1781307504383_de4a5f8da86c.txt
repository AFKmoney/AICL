    77→class Compiler:
    78→    """
    79→    AICL Multi-Stage Compiler v0.5.
    80→
    81→    Compiles AICL source code into deterministic, executable Python code
    82→    using behavior patterns and architecture templates.
    83→
    84→    Every generated artifact is tracked and must have provenance.

    88→        compiler = Compiler()
    89→        result = compiler.compile(source_code)

    91→            print(result.source_code)

    93→
    94→    def __init__(self, target_language: str = "python"):
    95→        self.target_language = target_language
    96→        self.warnings: List[str] = []
    97→        self.errors: List[str] = []
    98→        self.stages_completed: List[str] = []
    99→        self._used_names: Set[str] = set()
   100→        self._behavior_compiler = BehaviorCompiler()
   101→        self._todo_count = 0
   102→        self._provenance = CompilationProvenance()

   104→    def compile(self, source: str) -> CompilationResult:
   105→        """Compile AICL source code into executable Python code."""
   106→        self.warnings = []
   107→        self.errors = []
   108→        self.stages_completed = []
   109→        self._used_names = set()
   110→        self._todo_count = 0
   111→        self._provenance = CompilationProvenance()
   112→
   113→        result = CompilationResult()
   114→
   115→        # Stage 1: Specification Parsing
   116→        program = self._stage_parsing(source)
   117→        if program is None:
   118→            result.success = False
   119→            result.errors = self.errors
   120→            return result
   121→        self.stages_completed.append("Specification Parsing")
   122→
   123→        # Stage 2: Architecture Validation
   124→        validation_warnings = self._stage_architecture_validation(program)
   125→        self.warnings.extend(validation_warnings)
   126→        self.stages_completed.append("Architecture Validation")
   127→
   128→        # Stage 3: Dependency Analysis
   129→        dep_order = self._stage_dependency_analysis(program)
   130→        self.stages_completed.append("Dependency Analysis")
   131→
   132→        # Stage 4: Risk Analysis
   133→        risk_pairs = self._stage_risk_analysis(program)
   134→        self.stages_completed.append("Risk Analysis")
   135→
   136→        # Stage 5: Recovery Synthesis
   137→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   138→        self.stages_completed.append("Recovery Synthesis")
   139→
   140→        # Stage 6: Code Generation
   141→        source_code = self._stage_code_generation(
   142→            program, dep_order, risk_pairs, recovery_map
   143→        )
   144→        self.stages_completed.append("Code Generation")
   145→
   146→        # Stage 7: Test Generation
   147→        test_code = self._stage_test_generation(program, dep_order)
   148→        self.stages_completed.append("Test Generation")
   149→
   150→        # Stage 8: Optimization
   151→        source_code = self._stage_optimization(source_code, program)
   152→        self.stages_completed.append("Optimization")
   153→
   154→        # Stage 9: Final Construction
   155→        source_code = self._stage_final_construction(source_code, program)
   156→        self.stages_completed.append("Final Construction")
   157→
   158→        # Build Architecture Tree string
   159→        tree = ArchitectureTree(program)
   160→        result.architecture_tree_str = tree.to_string()
   161→
   162→        result.success = True
   163→        result.source_code = source_code
   164→        result.test_code = test_code
   165→        result.warnings = self.warnings
   166→        result.errors = self.errors
   167→        result.stages_completed = self.stages_completed
   168→        result.todo_count = self._todo_count
   169→        result.fully_compiled = self._todo_count == 0
   170→        result.provenance = self._provenance
   171→
   172→        return result
   173→
   174→    def compile_to_file(self, source: str, output_dir: str) -> CompilationResult:
   175→        """Compile AICL source code and write the output to files."""
   176→        result = self.compile(source)
   177→
   178→        if result.success:
   179→            os.makedirs(output_dir, exist_ok=True)
   180→
   181→            main_path = os.path.join(output_dir, "main.py")
   182→            with open(main_path, 'w') as f:
   183→                f.write(result.source_code)
   184→
   185→            test_path = os.path.join(output_dir, "test_main.py")
   186→            with open(test_path, 'w') as f:
   187→                f.write(result.test_code)
   188→
   189→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   190→            with open(tree_path, 'w') as f:
   191→                f.write(result.architecture_tree_str)
   192→
   193→        return result

   199→    def _stage_parsing(self, source: str) -> Optional[AICLProgram]:
   200→        """Parse the AICL source code into an AST."""
   201→        try:
   202→            parser = Parser(source)
   203→            program = parser.parse()
   204→            return program
   205→        except (ParseError, Exception) as e:
   206→            self.errors.append(f"Parse error: {str(e)}")
   207→            return None

   209→    # =========================================================================
   210→    # Stage 2: Architecture Validation
   211→    # =========================================================================
   212→
   213→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   214→        """Validate the architectural structure of the program."""
   215→        warnings = program.validate()
   216→
   217→        layer_names = [l.name for l in program.layers]
   218→        seen = set()
   219→        for name in layer_names:
   220→            if name.lower() in seen:
   221→                warnings.append(f"Duplicate layer name: {name}")
   222→            seen.add(name.lower())
   223→
   224→        entity_names = {e.name for e in program.entities}
   225→        for behavior in program.behaviors:
   226→            for inp in behavior.inputs:
   227→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   228→                    warnings.append(
   229→                        f"Behavior '{behavior.name}' references unknown type "
   230→                        f"'{inp.param_type}' in input"
   231→                    )
   232→
   233→        return warnings

   243→    # =========================================================================
   244→    # Stage 4: Risk Analysis
   245→    # =========================================================================
   246→
   247→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   248→        """Analyze risks and pair them with recovery procedures."""
   249→        pairs = []
   250→        for i, risk in enumerate(program.risks):
   251→            recovery = None
   252→            if i < len(program.recoveries):
   253→                recovery = program.recoveries[i]
   254→            pairs.append((risk, recovery))
   255→
   256→        if len(program.recoveries) > len(program.risks):
   257→            for i in range(len(program.risks), len(program.recoveries)):
   258→                generic_risk = RiskSection(
   259→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   260→                )
   261→                pairs.append((generic_risk, program.recoveries[i]))
   262→
   263→        return pairs
   264→
   265→    # =========================================================================
   266→    # Stage 5: Recovery Synthesis
   267→    # =========================================================================
   268→
   269→    def _stage_recovery_synthesis(
   270→        self,
   271→        program: AICLProgram,
   272→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]
   273→    ) -> Dict[str, str]:
   274→        """Synthesize recovery procedures for each risk."""
   275→        recovery_map = {}
   276→        for risk, recovery in risk_pairs:
   277→            if recovery:
   278→                recovery_map[risk.description] = recovery.description
   279→            else:
   280→                recovery_map[risk.description] = f"log and handle: {risk.description}"
   281→        return recovery_map

   287→    def _unique_name(self, base: str) -> str:
   288→        """Generate a unique Python identifier from a base name."""
   289→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   290→        clean = re.sub(r'_+', '_', clean).strip('_')
   291→        if not clean:
   292→            clean = "item"
   293→        if clean[0].isdigit():
   294→            clean = "f_" + clean
   295→
   296→        name = clean
   297→        counter = 2
   298→        while name in self._used_names:
   299→            name = f"{clean}_{counter}"
   300→            counter += 1
   301→
   302→        self._used_names.add(name)
   303→        return name

   305→    def _stage_code_generation(
   306→        self,
   307→        program: AICLProgram,
   308→        dep_order: List[LayerSection],
   309→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   310→        recovery_map: Dict[str, str]
   311→    ) -> str:
   312→        """Generate executable Python code from the AICL program."""
   313→        sections = []
   314→
   315→        goal_desc = program.goals[0].description if program.goals else "AICL Generated Application"
   316→
   317→        # Record architecture template provenance
   318→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   319→        self._provenance.record(
   320→            source_type=ProvenanceType.ARCHITECTURE_TEMPLATE,
   321→            source_location="Goal",
   322→            source_text=goal_desc,
   323→            resolution_path=[
   324→                "AICL Source",
   325→                f"Goal: {goal_desc}",
   326→                "ArchitectureTemplateMapper.match()",
   327→                f"Template: {template_name}",
   328→                f"Structure: {template_info['structure']}",
   329→                "Generated Code",
   330→            ],
   331→            generated_code=f"# Architecture template: {template_name} ({template_info['structure']})",
   332→            confidence=1.0,
   333→            template_name=template_name,
   334→            artifact_names=["architecture_template"],
   335→        )
   336→
   337→        sections.append('"""')
   338→        sections.append(f'AICL Generated Application')
   339→        sections.append(f'Goal: {goal_desc}')
   340→        sections.append(f'')
   341→        sections.append(f'Auto-generated by the AICL compiler v0.5.')
   342→        sections.append(f'Auditable compilation — every artifact has provenance.')
   343→        sections.append(f'"""')
   344→        sections.append('')
   345→
   346→        # Imports — with provenance
   347→        import_code = self._generate_imports(program)
   348→        sections.append(import_code)
   349→        sections.append('')
   350→
   351→        # Entity classes — with provenance (already implemented)
   352→        for entity in program.entities:
   353→            sections.append(self._generate_entity(entity))
   354→            sections.append('')
   355→
   356→        # Main application class
   357→        class_name = self._make_class_name(goal_desc)
   358→        sections.append(self._generate_application_class(
   359→            program, class_name, dep_order, risk_pairs, recovery_map
   360→        ))
   361→        sections.append('')
   362→
   363→        # Entry point — with provenance
   364→        sections.append(self._generate_entry_point(class_name))
   365→
   366→        return '\n'.join(sections)

   368→    def _generate_imports(self, program: AICLProgram) -> str:
   369→        """Generate import statements based on program content."""
   370→        imports = [
   371→            "import sys",
   372→            "import logging",
   373→            "import time",
   374→            "from typing import Optional, List, Dict, Any",
   375→        ]
   376→
   377→        if program.parallels:
   378→            imports.append("import threading")
   379→            imports.append("from concurrent.futures import ThreadPoolExecutor, as_completed")
   380→
   381→        needs_datetime = False
   382→        for entity in program.entities:
   383→            for f in entity.fields:
   384→                if f.field_type == "datetime":
   385→                    needs_datetime = True
   386→                    break
   387→
   388→        if needs_datetime:
   389→            imports.append("from datetime import datetime")
   390→
   391→        imports.append("from dataclasses import dataclass, field")
   392→
   393→        # Check if we need hashlib for security
   394→        if program.securities:
   395→            for sec in program.securities:
   396→                for action in sec.actions:
   397→                    if action.action_type == "encrypt":
   398→                        imports.append("import hashlib")
   399→                        imports.append("import base64")
   400→                        break
   401→
   402→        code = '\n'.join(imports)
   403→
   404→        # Register import artifact and record provenance
   405→        self._provenance.register_artifact(
   406→            name="imports",
   407→            artifact_type=ArtifactType.IMPORT_BLOCK,
   408→            source="Program structure",
   409→            code_snippet="import sys, logging, time, ...",
   410→        )
   411→        self._provenance.record(
   412→            source_type=ProvenanceType.IMPORT_GENERATION,
   413→            source_location="Program Structure",
   414→            source_text="Required imports based on program features",
   415→            resolution_path=[
   416→                "AICL Source",
   417→                "Program Analysis",
   418→                "Feature Detection (entities, security, parallel)",
   419→                "Import Resolution",
   420→                "Generated Code",
   421→            ],
   422→            generated_code=code,
   423→            confidence=1.0,
   424→            artifact_names=["imports"],
   425→        )
   426→
   427→        return code

   429→    def _generate_entity(self, entity: EntitySection) -> str:
   430→        """Generate a Python dataclass for an AICL Entity."""
   431→        class_name = self._make_class_name(entity.name)
   432→        artifact_name = f"Entity_{class_name}"
   433→        lines = []
   434→        lines.append('@dataclass')
   435→        lines.append(f'class {class_name}:')
   436→        lines.append(f'    """Entity: {entity.name}"""')
   437→
   438→        if not entity.fields:
   439→            lines.append('    pass')
   440→        else:
   441→            for f in entity.fields:
   442→                python_type = self._aicl_type_to_python(f.field_type)
   443→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   444→                if default:
   445→                    lines.append(f'    {f.name}: {python_type} = {default}')
   446→                else:
   447→                    lines.append(f'    {f.name}: {python_type} = 0')
   448→
   449→        code = '\n'.join(lines)
   450→
   451→        # Register artifact
   452→        self._provenance.register_artifact(
   453→            name=artifact_name,
   454→            artifact_type=ArtifactType.DATACLASS,
   455→            source=f"Entity {entity.name}",
   456→            code_snippet=f"class {class_name}:",
   457→        )
   458→
   459→        # Record entity generation provenance
   460→        fields_desc = ', '.join(f'{f.name}: {f.field_type}' for f in entity.fields) if entity.fields else 'no fields'
   461→        self._provenance.record(
   462→            source_type=ProvenanceType.ENTITY_GENERATION,
   463→            source_location=f"Entity {entity.name}",
   464→            source_text=fields_desc,
   465→            resolution_path=[
   466→                "AICL Source",
   467→                f"Entity {entity.name}",
   468→                f"Fields: {fields_desc}",
   469→                "Dataclass Generation",
   470→                "Generated Code",
   471→            ],
   472→            generated_code=code,
   473→            confidence=1.0,
   474→            parameters={"class_name": class_name, "fields": str(len(entity.fields))},
   475→            artifact_names=[artifact_name],
   476→        )
   477→
   478→        return code

   480→    def _generate_application_class(
   481→        self,
   482→        program: AICLProgram,
   483→        class_name: str,
   484→        dep_order: List[LayerSection],
   485→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   486→        recovery_map: Dict[str, str]
   487→    ) -> str:
   488→        """Generate the main application class."""
   489→        lines = []
   490→
   491→        # Determine architecture template from Goal
   492→        goal_desc = program.goals[0].description if program.goals else ""
   493→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   494→
   495→        # Register the class artifact
   496→        class_artifact = f"Class_{class_name}"
   497→        self._provenance.register_artifact(
   498→            name=class_artifact,
   499→            artifact_type=ArtifactType.CLASS,
   500→            source=f"Goal: {goal_desc}",
   501→            code_snippet=f"class {class_name}:",
   502→        )
   503→
   504→        lines.append(f'class {class_name}:')
   505→        lines.append(f'    """')
   506→        if program.goals:
   507→            lines.append(f'    Goal: {program.goals[0].description}')
   508→        lines.append(f'    Architecture: {template_info["structure"]}')
   509→        lines.append(f'    """')
   510→        lines.append('')
   511→
   512→        # Class attributes - one per layer
   513→        lines.append('    # Layer instances')
   514→        layer_attr_map = {}
   515→        for layer in dep_order:
   516→            attr_name = self._unique_name(layer.name)
   517→            layer_attr_map[layer.name] = attr_name
   518→            lines.append(f'    _{attr_name}: Any = None')
   519→        lines.append('')
   520→
   521→        # Entity instances
   522→        if program.entities:
   523→            lines.append('    # Entity instances')
   524→            for entity in program.entities:
   525→                attr = self._unique_name(entity.name.lower())
   526→                lines.append(f'    _{attr}: Any = None')
   527→            lines.append('')
   528→
   529→        # Risks as class constants
   530→        if program.risks:
   531→            lines.append('    # Known risks')
   532→            for i, risk in enumerate(program.risks):
   533→                lines.append(f'    RISK_{i+1} = {risk.description!r}')
   534→            lines.append('')
   535→
   536→        # Constraints as class constants
   537→        if program.constraints:
   538→            lines.append('    # Constraints')
   539→            for i, con in enumerate(program.constraints):
   540→                lines.append(f'    CONSTRAINT_{i+1} = {con.description!r}')
   541→            lines.append('')
   542→
   543→        # Game/application state
   544→        lines.append('    # Application state')
   545→        lines.append('    _running: bool = False')
   546→        lines.append('    _initialized: bool = False')
   547→        if template_name == "game_loop":
   548→            lines.append('    _game_active: bool = False')
   549→            lines.append('    _frame_count: int = 0')
   550→            lines.append('    _last_frame_time: float = 0.0')
   551→        elif template_name == "event_driven":
   552→            lines.append('    _connected: bool = False')
   553→            lines.append('    _message_queue: List = field(default_factory=list)')
   554→            lines.append('    _offline_queue: List = field(default_factory=list)')
   555→        lines.append('')
   556→
   557→        # Record class structure provenance
   558→        class_structure_code = '\n'.join(lines)
   559→        self._provenance.record(
   560→            source_type=ProvenanceType.CLASS_STRUCTURE,
   561→            source_location=f"Class {class_name}",
   562→            source_text=f"Layers: {len(dep_order)}, Entities: {len(program.entities)}, Risks: {len(program.risks)}",
   563→            resolution_path=[
   564→                "AICL Source",
   565→                f"Goal: {goal_desc}",
   566→                f"Template: {template_name}",
   567→                "Class Structure Generation",
   568→                "Layer Attributes, Entity Attributes, State Variables",
   569→                "Generated Code",
   570→            ],
   571→            generated_code=class_structure_code,
   572→            confidence=1.0,
   573→            template_name=template_name,
   574→            artifact_names=[class_artifact],
   575→        )
   576→
   577→        # __init__ method
   578→        init_artifact = f"Method_{class_name}.__init__"
   579→        init_lines = []
   580→        init_lines.append('    def __init__(self):')
   581→        init_lines.append('        """Initialize the application and all layers."""')
   582→        init_lines.append('        self._logger = logging.getLogger(self.__class__.__name__)')
   583→        init_lines.append('        self._running = False')
   584→        init_lines.append('        self._initialized = False')
   585→        if template_name == "game_loop":
   586→            init_lines.append('        self._game_active = True')
   587→            init_lines.append('        self._frame_count = 0')
   588→            init_lines.append('        self._last_frame_time = time.time()')
   589→        elif template_name == "event_driven":
   590→            init_lines.append('        self._connected = False')
   591→            init_lines.append('        self._message_queue = []')
   592→            init_lines.append('        self._offline_queue = []')
   593→        init_lines.append('')
   594→
   595→        # Initialize entity instances
   596→        if program.entities:
   597→            init_lines.append('        # Initialize entities')
   598→            for entity in program.entities:
   599→                entity_class = self._make_class_name(entity.name)
   600→                attr = entity.name.lower()
   601→                init_lines.append(f'        self._{attr} = {entity_class}()')
   602→            init_lines.append('')
   603→
   604→        # Initialize each layer
   605→        for layer in dep_order:
   606→            attr_name = layer_attr_map[layer.name]
   607→            init_lines.append(f'        # Initialize {layer.name}')
   608→            init_lines.append(f'        self._{attr_name} = self._init_{attr_name}()')
   609→            init_lines.append('')
   610→
   611→        init_lines.append('        self._initialized = True')
   612→        init_lines.append('        self._logger.info("Application initialized successfully")')
   613→        init_lines.append('')
   614→
   615→        # Register __init__ artifact
   616→        self._provenance.register_artifact(
   617→            name=init_artifact,
   618→            artifact_type=ArtifactType.METHOD,
   619→            source=f"Class {class_name}",
   620→            code_snippet="def __init__(self):",
   621→        )
   622→
   623→        # Record __init__ provenance
   624→        init_code = '\n'.join(init_lines)
   625→        self._provenance.record(
   626→            source_type=ProvenanceType.CLASS_STRUCTURE,
   627→            source_location=f"Method __init__",
   628→            source_text=f"Initialize {len(dep_order)} layers, {len(program.entities)} entities",
   629→            resolution_path=[
   630→                "AICL Source",
   631→                f"Class {class_name}",
   632→                "Constructor Generation",
   633→                f"Template: {template_name}",
   634→                "Entity Init + Layer Init Calls",
   635→                "Generated Code",
   636→            ],
   637→            generated_code=init_code,
   638→            confidence=1.0,
   639→            template_name=template_name,
   640→            artifact_names=[init_artifact],
   641→        )
   642→
   643→        lines.extend(init_lines)
   644→
   645→        # Layer initialization methods
   646→        for layer in dep_order:
   647→            attr_name = layer_attr_map[layer.name]
   648→            lines.append(self._generate_layer_init(layer, attr_name, risk_pairs, recovery_map))
   649→            lines.append('')
   650→
   651→        # Pre-compute behavior method names (must be done before run method generation)
   652→        behavior_method_names = {}
   653→        for i, behavior in enumerate(program.behaviors):
   654→            method_name = self._unique_name(behavior.name).lower()
   655→            behavior_method_names[behavior.name] = method_name
   656→
   657→        # Run method — uses architecture template (with provenance)
   658→        lines.append(self._generate_run_method(program, dep_order, layer_attr_map, risk_pairs, recovery_map, template_name, behavior_method_names))
   659→        lines.append('')
   660→
   661→        # Behavior methods
   662→        for i, behavior in enumerate(program.behaviors):
   663→            method_name = behavior_method_names[behavior.name]
   664→            lines.append(self._generate_behavior_method(behavior, method_name, program))
   665→            lines.append('')
   666→
   667→        # Condition handlers
   668→        for i, condition in enumerate(program.conditions):
   669→            lines.append(self._generate_condition_handler(condition, i + 1, program))
   670→            lines.append('')
   671→
   672→        # Event handlers
   673→        for i, event in enumerate(program.events):
   674→            lines.append(self._generate_event_handler(event, i + 1, program))
   675→            lines.append('')
   676→
   677→        # Parallel execution
   678→        if program.parallels:
   679→            lines.append(self._generate_parallel_execution(program, layer_attr_map))
   680→            lines.append('')
   681→
   682→        # Validation methods
   683→        for i, validation in enumerate(program.validations):
   684→            lines.append(self._generate_validation_method(validation, i + 1, program))
   685→            lines.append('')
   686→
   687→        # Security methods
   688→        if program.securities:
   689→            lines.append(self._generate_security_methods(program))
   690→            lines.append('')
   691→
   692→        # Risk handling method
   693→        lines.append(self._generate_risk_handler(risk_pairs, recovery_map, program))
   694→
   695→        # Helper methods
   696→        lines.append('')
   697→        lines.append(self._generate_helper_methods(program, template_name))
   698→
   699→        return '\n'.join(lines)

   701→    def _generate_layer_init(
   702→        self,
   703→        layer: LayerSection,
   704→        attr_name: str,
   705→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   706→        recovery_map: Dict[str, str]
   707→    ) -> str:
   708→        """Generate initialization method for a layer — deterministic, no TODOs."""
   709→        lines = []
   710→        lines.append(f'    def _init_{attr_name}(self) -> Any:')
   711→        lines.append(f'        """Initialize the {layer.name} layer."""')
   712→        lines.append(f'        self._logger.info("Initializing {layer.name}...")')
   713→        lines.append(f'        try:')
   714→
   715→        # Generate layer component as a dict
   716→        sublayer_names = [s.name for s in layer.sublayers]
   717→        lines.append(f'            component = {{')
   718→        lines.append(f'                "name": "{layer.name}",')
   719→        lines.append(f'                "active": True,')
   720→        lines.append(f'                "sublayers": {sublayer_names!r},')
   721→        lines.append(f'            }}')
   722→
   723→        # Generate sublayer initialization
   724→        for sublayer in layer.sublayers:
   725→            sub_attr = self._unique_name(sublayer.name)
   726→            lines.append(f'            # Sublayer: {sublayer.name}')
   727→            lines.append(f'            component["{sub_attr}"] = {{"name": "{sublayer.name}", "active": True}}')
   728→
   729→        lines.append(f'            self._logger.info("{layer.name} initialized successfully")')
   730→        lines.append(f'            return component')
   731→        lines.append(f'        except Exception as e:')
   732→        lines.append(f'            self._logger.error(f"Failed to initialize {layer.name}: {{e}}")')
   733→
   734→        # Add recovery from risk pairs that match this layer
   735→        for risk, recovery in risk_pairs:
   736→            if recovery and self._risk_matches_layer(risk.description, layer.name):
   737→                recovery_code, _ = self._behavior_compiler.compile_action(
   738→                    recovery.description,
   739→                    context={}
   740→                )
   741→                lines.append(f'            # Recovery: {recovery.description}')
   742→                lines.append(f'            self._logger.info("Applying recovery: {recovery.description}")')
   743→                for code_line in recovery_code.split('\n'):
   744→                    if code_line.strip():
   745→                        lines.append(f'            {code_line}')
   746→
   747→        lines.append(f'            return None')
   748→
   749→        code = '\n'.join(lines)
   750→
   751→        # Register artifact
   752→        artifact_name = f"Method__init_{attr_name}"
   753→        self._provenance.register_artifact(
   754→            name=artifact_name,
   755→            artifact_type=ArtifactType.METHOD,
   756→            source=f"Layer {layer.name}",
   757→            code_snippet=f"def _init_{attr_name}(self):",
   758→        )
   759→
   760→        # Record layer initialization provenance
   761→        self._provenance.record(
   762→            source_type=ProvenanceType.LAYER_INITIALIZATION,
   763→            source_location=f"Layer {layer.name}",
   764→            source_text=f"Sublayers: {', '.join(sublayer_names)}" if sublayer_names else "No sublayers",
   765→            resolution_path=[
   766→                "AICL Source",
   767→                f"Layer {layer.name}",
   768→                f"Sublayers: {sublayer_names}" if sublayer_names else "No sublayers",
   769→                "Component Dict Generation",
   770→                "Generated Code",
   771→            ],
   772→            generated_code=code,
   773→            confidence=1.0,
   774→            parameters={"layer_name": layer.name, "sublayers": str(len(sublayer_names))},
   775→            artifact_names=[artifact_name],
   776→        )
   777→
   778→        return code

   780→    def _generate_run_method(
   781→        self,
   782→        program: AICLProgram,
   783→        dep_order: List[LayerSection],
   784→        layer_attr_map: Dict[str, str],
   785→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   786→        recovery_map: Dict[str, str],
   787→        template_name: str,
   788→        behavior_method_names: Dict[str, str] = None,
   789→    ) -> str:
   790→        """Generate the main run method — uses architecture template."""
   791→        behavior_method_names = behavior_method_names or {}
   792→        lines = []
   793→        lines.append('    def run(self) -> None:')
   794→        lines.append('        """Run the application with full error handling."""')
   795→        lines.append('        self._logger.info("Starting application...")')
   796→        lines.append('        self._running = True')
   797→        lines.append('')
   798→
   799→        if template_name == "game_loop":
   800→            # Game loop: init → update → render → validate → loop
   801→            lines.append('        self._game_active = True')
   802→            lines.append('        try:')
   803→            lines.append('            while self._running and self._game_active:')
   804→            lines.append('                frame_start = time.time()')
   805→            lines.append('')
   806→            lines.append('                # Update phase')
   807→            for behavior in program.behaviors:
   808→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   809→                input_params = ', '.join(
   810→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   811→                    {e.name.lower() for e in program.entities}
   812→                    else inp.name
   813→                    for inp in behavior.inputs
   814→                )
   815→                lines.append(f'                self._behavior_{method_name}({input_params})')
   816→            lines.append('')
   817→            lines.append('                # Check conditions')
   818→            for i, condition in enumerate(program.conditions):
   819→                lines.append(f'                self._handle_condition_{i+1}()')
   820→            lines.append('')
   821→            lines.append('                # Render phase')
   822→            for layer in dep_order:
   823→                attr_name = layer_attr_map[layer.name]
   824→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   825→                    lines.append(f'                if self._{attr_name} is not None:')
   826→                    lines.append(f'                    self._render_frame()')
   827→            lines.append('')
   828→            lines.append('                # Frame rate control')
   829→            lines.append('                elapsed = time.time() - frame_start')
   830→            lines.append('                if elapsed < 1.0 / 60.0:')
   831→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   832→            lines.append('                self._frame_count += 1')
   833→            lines.append('')
   834→
   835→        elif template_name == "event_driven":
   836→            # Event-driven: connect → listen → handle → respond
   837→            lines.append('        try:')
   838→            lines.append('            self._connect()')
   839→            lines.append('            self._connected = True')
   840→            lines.append('            self._logger.info("Connected to server")')
   841→            lines.append('')
   842→            lines.append('            while self._running and self._connected:')
   843→            lines.append('                # Process message queue')
   844→            lines.append('                while self._message_queue:')
   845→            lines.append('                    msg = self._message_queue.pop(0)')
   846→            lines.append('                    self._process_message(msg)')
   847→            lines.append('')
   848→            lines.append('                # Process offline queue if reconnected')
   849→            lines.append('                if self._offline_queue and self._connected:')
   850→            lines.append('                    self._sync_offline_messages()')
   851→            lines.append('')
   852→            lines.append('                # Wait for new messages')
   853→            lines.append('                time.sleep(0.1)')
   854→
   855→        else:
   856→            # Default: linear execution
   857→            lines.append('        try:')
   858→            for layer in dep_order:
   859→                attr_name = layer_attr_map[layer.name]
   860→                lines.append(f'            # Execute {layer.name}')
   861→                lines.append(f'            if self._{attr_name} is not None:')
   862→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   863→                lines.append(f'            else:')
   864→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   865→
   866→        # Validations
   867→        lines.append('')
   868→        lines.append('            # Run validations')
   869→        for i, validation in enumerate(program.validations):
   870→            lines.append(f'            self._validate_{i+1}()')
   871→
   872→        lines.append('')
   873→        lines.append('            self._logger.info("Application completed successfully")')
   874→        lines.append('')
   875→        lines.append('        except Exception as e:')
   876→        lines.append('            self._logger.error(f"Application error: {e}")')
   877→        lines.append('            self._handle_risk(e)')
   878→        lines.append('        finally:')
   879→        lines.append('            self._running = False')
   880→        lines.append('            self._logger.info("Application stopped")')
   881→
   882→        code = '\n'.join(lines)
   883→
   884→        # Register artifact
   885→        artifact_name = "Method_run"
   886→        self._provenance.register_artifact(
   887→            name=artifact_name,
   888→            artifact_type=ArtifactType.METHOD,
   889→            source=f"Architecture Template: {template_name}",
   890→            code_snippet="def run(self) -> None:",
   891→        )
   892→
   893→        # Record run method provenance
   894→        _, run_template_info = ArchitectureTemplateMapper.match(
   895→            program.goals[0].description if program.goals else ""
   896→        )
   897→        self._provenance.record(
   898→            source_type=ProvenanceType.RUN_METHOD,
   899→            source_location="Method run",
   900→            source_text=f"Template: {template_name}, Behaviors: {len(program.behaviors)}, Conditions: {len(program.conditions)}",
   901→            resolution_path=[
   902→                "AICL Source",
   903→                f"Goal → Template: {template_name}",
   904→                f"Structure: {run_template_info.get('structure', 'unknown')}",
   905→                "Run Method Generation",
   906→                "Generated Code",
   907→            ],
   908→            generated_code=code,
   909→            confidence=1.0,
   910→            template_name=template_name,
   911→            artifact_names=[artifact_name],
   912→        )
   913→
   914→        return code

   916→    def _generate_behavior_method(
   917→        self, behavior: BehaviorSection, method_name: str, program: AICLProgram
   918→    ) -> str:
   919→        """Generate a behavior method using pattern matching — no TODOs."""
   920→        lines = []
   921→
   922→        # Build parameter list from inputs
   923→        params = ["self"]
   924→        for inp in behavior.inputs:
   925→            # Check if input references an entity
   926→            entity_names = {e.name for e in program.entities}
   927→            if inp.name in entity_names:
   928→                params.append(f'{inp.name.lower()}: Any')
   929→            elif inp.param_type in entity_names:
   930→                params.append(f'{inp.name.lower()}: Any')
   931→            else:
   932→                params.append(f'{inp.name.lower()}: Any = None')
   933→
   934→        param_str = ', '.join(params)
   935→        lines.append(f'    def _behavior_{method_name}({param_str}) -> Any:')
   936→        lines.append(f'        """Behavior: {behavior.name}"""')
   937→
   938→        # Compile the action using the Behavior Compiler
   939→        entity_context = {}
   940→        for inp in behavior.inputs:
   941→            entity_names = {e.name for e in program.entities}
   942→            if inp.name in entity_names:
   943→                entity_context["entity"] = inp.name.lower()
   944→            elif inp.param_type in entity_names:
   945→                entity_context["entity"] = inp.param_type.lower()
   946→
   947→        action_code, fully_compiled = self._behavior_compiler.compile_action(
   948→            behavior.action, context=entity_context
   949→        )
   950→
   951→        if not fully_compiled:
   952→            self._todo_count += 1
   953→
   954→        # Register artifact
   955→        artifact_name = f"Method__behavior_{method_name}"
   956→        self._provenance.register_artifact(
   957→            name=artifact_name,
   958→            artifact_type=ArtifactType.METHOD,
   959→            source=f"Behavior {behavior.name}",
   960→            code_snippet=f"def _behavior_{method_name}(...):",
   961→        )
   962→
   963→        # Record provenance
   964→        self._record_behavior_provenance(behavior, action_code, fully_compiled, entity_context, artifact_name)
   965→
   966→        # Indent the compiled code
   967→        for code_line in action_code.split('\n'):
   968→            lines.append(f'        {code_line}')
   969→
   970→        return '\n'.join(lines)

  1033→    def _generate_condition_handler(
  1034→        self, condition: ConditionSection, index: int, program: AICLProgram
  1035→    ) -> str:
  1036→        """Generate a condition handler using pattern matching — no TODOs."""
  1037→        lines = []
  1038→        lines.append(f'    def _handle_condition_{index}(self) -> None:')
  1039→        when_desc = condition.when_clause or "unknown condition"
  1040→        then_desc = condition.then_clause or "unknown action"
  1041→        lines.append(f'        """Condition: WHEN {when_desc} THEN {then_desc}"""')
  1042→
  1043→        # Compile the WHEN clause
  1044→        when_code, when_compiled = self._behavior_compiler.compile_action(
  1045→            f"check {when_desc}",
  1046→            context={}
  1047→        )
  1048→
  1049→        # Compile the THEN clause
  1050→        then_code, then_compiled = self._behavior_compiler.compile_action(
  1051→            then_desc,
  1052→            context={}
  1053→        )
  1054→
  1055→        if not when_compiled or not then_compiled:
  1056→            self._todo_count += 1
  1057→
  1058→        # Generate the condition check
  1059→        when_lower = when_desc.lower()
  1060→
  1061→        # Detect common condition patterns
  1062→        if any(w in when_lower for w in ['hit', 'collision', 'collide', 'contact']):
  1063→            lines.append(f'        # When: {when_desc}')
  1064→            lines.append(f'        collision = self._check_collisions()')
  1065→            lines.append(f'        if collision:')
  1066→            lines.append(f'            self._logger.info("Condition triggered: {when_desc}")')
  1067→            for code_line in then_code.split('\n'):
  1068→                lines.append(f'            {code_line}')
  1069→        elif any(w in when_lower for w in ['score', 'reach', 'point']):
  1070→            lines.append(f'        # When: {when_desc}')
  1071→            # Extract threshold from description
  1072→            import re as _re
  1073→            num_match = _re.search(r'\d+', when_desc)
  1074→            threshold = num_match.group() if num_match else "10"
  1075→            lines.append(f'        for player in [self._player]:')
  1076→            lines.append(f'            if player.score >= {threshold}:')
  1077→            lines.append(f'                self._logger.info("Condition triggered: {when_desc}")')
  1078→            for code_line in then_code.split('\n'):
  1079→                lines.append(f'                {code_line}')
  1080→        elif any(w in when_lower for w in ['check', 'checkmate', 'king']):
  1081→            lines.append(f'        # When: {when_desc}')
  1082→            lines.append(f'        if self._is_in_check():')
  1083→            lines.append(f'            self._logger.info("Condition triggered: {when_desc}")')
  1084→            for code_line in then_code.split('\n'):
  1085→                lines.append(f'            {code_line}')
  1086→        elif any(w in when_lower for w in ['disconnect', 'offline', 'unavailable']):
  1087→            lines.append(f'        # When: {when_desc}')
  1088→            lines.append(f'        if not self._connected:')
  1089→            lines.append(f'            self._logger.info("Condition triggered: {when_desc}")')
  1090→            for code_line in then_code.split('\n'):
  1091→                lines.append(f'            {code_line}')
