   104→   104→    def compile(self, source: str) -> CompilationResult:
   105→   105→        """Compile AICL source code into executable Python code."""
   106→   106→        self.warnings = []
   107→   107→        self.errors = []
   108→   108→        self.stages_completed = []
   109→   109→        self._used_names = set()
   110→   110→        self._todo_count = 0
   111→   111→        self._provenance = CompilationProvenance()
   112→   112→
   113→   113→        result = CompilationResult()
   114→   114→
   115→   115→        # Stage 1: Specification Parsing
   116→   116→        program = self._stage_parsing(source)
   117→   117→        if program is None:
   118→   118→            result.success = False
   119→   119→            result.errors = self.errors
   120→   120→            return result
   121→   121→        self.stages_completed.append("Specification Parsing")
   122→   122→
   123→   123→        # Stage 2: Architecture Validation
   124→   124→        validation_warnings = self._stage_architecture_validation(program)
   125→   125→        self.warnings.extend(validation_warnings)
   126→   126→        self.stages_completed.append("Architecture Validation")
   127→   127→
   128→   128→        # Stage 3: Dependency Analysis
   129→   129→        dep_order = self._stage_dependency_analysis(program)
   130→   130→        self.stages_completed.append("Dependency Analysis")
   131→   131→
   132→   132→        # Stage 4: Risk Analysis
   133→   133→        risk_pairs = self._stage_risk_analysis(program)
   134→   134→        self.stages_completed.append("Risk Analysis")
   135→   135→
   136→   136→        # Stage 5: Recovery Synthesis
   137→   137→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   138→   138→        self.stages_completed.append("Recovery Synthesis")
   139→   139→
   140→   140→        # Stage 6: Code Generation
   141→   141→        source_code = self._stage_code_generation(
   142→   142→            program, dep_order, risk_pairs, recovery_map
   143→   143→        )
   144→   144→        self.stages_completed.append("Code Generation")
   145→   145→
   146→   146→        # Stage 7: Test Generation
   147→   147→        test_code = self._stage_test_generation(program, dep_order)
   148→   148→        self.stages_completed.append("Test Generation")
   149→   149→
   150→   150→        # Stage 8: Optimization
   151→   151→        source_code = self._stage_optimization(source_code, program)
   152→   152→        self.stages_completed.append("Optimization")
   153→   153→
   154→   154→        # Stage 9: Final Construction
   155→   155→        source_code = self._stage_final_construction(source_code, program)
   156→   156→        self.stages_completed.append("Final Construction")
   157→   157→
   158→   158→        # Build Architecture Tree string
   159→   159→        tree = ArchitectureTree(program)
   160→   160→        result.architecture_tree_str = tree.to_string()
   161→   161→
   162→   162→        result.success = True
   163→   163→        result.source_code = source_code
   164→   164→        result.test_code = test_code
   165→   165→        result.warnings = self.warnings
   166→   166→        result.errors = self.errors
   167→   167→        result.stages_completed = self.stages_completed
   168→   168→        result.todo_count = self._todo_count
   169→   169→        result.fully_compiled = self._todo_count == 0
   170→   170→        result.provenance = self._provenance
   171→   171→
   172→   172→        return result

   213→   213→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   214→   214→        """Validate the architectural structure of the program."""
   215→   215→        warnings = program.validate()
   216→   216→
   217→   217→        layer_names = [l.name for l in program.layers]
   218→   218→        seen = set()
   219→   219→        for name in layer_names:
   220→   220→            if name.lower() in seen:
   221→   221→                warnings.append(f"Duplicate layer name: {name}")
   222→   222→            seen.add(name.lower())
   223→   223→
   224→   224→        entity_names = {e.name for e in program.entities}
   225→   225→        for behavior in program.behaviors:
   226→   226→            for inp in behavior.inputs:
   227→   227→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   228→   228→                    warnings.append(
   229→   229→                        f"Behavior '{behavior.name}' references unknown type "
   230→   230→                        f"'{inp.param_type}' in input"
   231→   231→                    )
   232→   232→
   233→   233→        return warnings

   247→   247→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   248→   248→        """Analyze risks and pair them with recovery procedures."""
   249→   249→        pairs = []
   250→   250→        for i, risk in enumerate(program.risks):
   251→   251→            recovery = None
   252→   252→            if i < len(program.recoveries):
   253→   253→                recovery = program.recoveries[i]
   254→   254→            pairs.append((risk, recovery))
   255→   255→
   256→   256→        if len(program.recoveries) > len(program.risks):
   257→   257→            for i in range(len(program.risks), len(program.recoveries)):
   258→   258→                generic_risk = RiskSection(
   259→   259→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   260→   260→                )
   261→   261→                pairs.append((generic_risk, program.recoveries[i]))
   262→   262→
   263→   263→        return pairs

   272→   272→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]

   277→   277→            if recovery:
   278→   278→                recovery_map[risk.description] = recovery.description
   279→   279→            else:
   280→   280→                recovery_map[risk.description] = f"log and handle: {risk.description}"

   287→   287→    def _unique_name(self, base: str) -> str:
   288→   288→        """Generate a unique Python identifier from a base name."""
   289→   289→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   290→   290→        clean = re.sub(r'_+', '_', clean).strip('_')
   291→   291→        if not clean:
   292→   292→            clean = "item"
   293→   293→        if clean[0].isdigit():
   294→   294→            clean = "f_" + clean
   295→   295→
   296→   296→        name = clean
   297→   297→        counter = 2
   298→   298→        while name in self._used_names:
   299→   299→            name = f"{clean}_{counter}"
   300→   300→            counter += 1
   301→   301→
   302→   302→        self._used_names.add(name)
   303→   303→        return name

   444→   444→                if default:
   445→   445→                    lines.append(f'    {f.name}: {python_type} = {default}')
   446→   446→                else:

   463→   463→            source_location=f"Entity {entity.name}",

   468→   468→                f"Fields: {fields_desc}",
   469→   469→                "Dataclass Generation",
   470→   470→                "Generated Code",
   471→   471→            ],
   472→   472→            generated_code=code,
   473→   473→            confidence=1.0,
   474→   474→            parameters={"class_name": class_name, "fields": str(len(entity.fields))},
   475→   475→            artifact_names=[artifact_name],
   476→   476→        )
   477→   477→
   478→   478→        return code

   488→   488→        """Generate the main application class."""
   489→   489→        lines = []

   780→   780→    def _generate_run_method(
   781→   781→        self,
   782→   782→        program: AICLProgram,
   783→   783→        dep_order: List[LayerSection],
   784→   784→        layer_attr_map: Dict[str, str],

   786→   786→        recovery_map: Dict[str, str],
   787→   787→        template_name: str,
   788→   788→        behavior_method_names: Dict[str, str] = None,

   790→   790→        """Generate the main run method — uses architecture template."""
   791→   791→        behavior_method_names = behavior_method_names or {}
   792→   792→        lines = []
   793→   793→        lines.append('    def run(self) -> None:')
   794→   794→        lines.append('        """Run the application with full error handling."""')
   795→   795→        lines.append('        self._logger.info("Starting application...")')
   796→   796→        lines.append('        self._running = True')
   797→   797→        lines.append('')
   798→   798→
   799→   799→        if template_name == "game_loop":
   800→   800→            # Game loop: init → update → render → validate → loop
   801→   801→            lines.append('        self._game_active = True')
   802→   802→            lines.append('        try:')
   803→   803→            lines.append('            while self._running and self._game_active:')
   804→   804→            lines.append('                frame_start = time.time()')
   805→   805→            lines.append('')
   806→   806→            lines.append('                # Update phase')
   807→   807→            for behavior in program.behaviors:
   808→   808→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   809→   809→                input_params = ', '.join(
   810→   810→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   811→   811→                    {e.name.lower() for e in program.entities}
   812→   812→                    else inp.name
   813→   813→                    for inp in behavior.inputs
   814→   814→                )
   815→   815→                lines.append(f'                self._behavior_{method_name}({input_params})')
   816→   816→            lines.append('')
   817→   817→            lines.append('                # Check conditions')
   818→   818→            for i, condition in enumerate(program.conditions):
   819→   819→                lines.append(f'                self._handle_condition_{i+1}()')
   820→   820→            lines.append('')
   821→   821→            lines.append('                # Render phase')
   822→   822→            for layer in dep_order:
   823→   823→                attr_name = layer_attr_map[layer.name]
   824→   824→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   825→   825→                    lines.append(f'                if self._{attr_name} is not None:')
   826→   826→                    lines.append(f'                    self._render_frame()')
   827→   827→            lines.append('')
   828→   828→            lines.append('                # Frame rate control')
   829→   829→            lines.append('                elapsed = time.time() - frame_start')
   830→   830→            lines.append('                if elapsed < 1.0 / 60.0:')
   831→   831→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   832→   832→            lines.append('                self._frame_count += 1')
   833→   833→            lines.append('')
   834→   834→
   835→   835→        elif template_name == "event_driven":
   836→   836→            # Event-driven: connect → listen → handle → respond
   837→   837→            lines.append('        try:')
   838→   838→            lines.append('            self._connect()')
   839→   839→            lines.append('            self._connected = True')
   840→   840→            lines.append('            self._logger.info("Connected to server")')
   841→   841→            lines.append('')
   842→   842→            lines.append('            while self._running and self._connected:')
   843→   843→            lines.append('                # Process message queue')
   844→   844→            lines.append('                while self._message_queue:')
   845→   845→            lines.append('                    msg = self._message_queue.pop(0)')
   846→   846→            lines.append('                    self._process_message(msg)')
   847→   847→            lines.append('')
   848→   848→            lines.append('                # Process offline queue if reconnected')
   849→   849→            lines.append('                if self._offline_queue and self._connected:')
   850→   850→            lines.append('                    self._sync_offline_messages()')
   851→   851→            lines.append('')
   852→   852→            lines.append('                # Wait for new messages')
   853→   853→            lines.append('                time.sleep(0.1)')
   854→   854→
   855→   855→        else:
   856→   856→            # Default: linear execution
   857→   857→            lines.append('        try:')
   858→   858→            for layer in dep_order:
   859→   859→                attr_name = layer_attr_map[layer.name]
   860→   860→                lines.append(f'            # Execute {layer.name}')
   861→   861→                lines.append(f'            if self._{attr_name} is not None:')
   862→   862→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   863→   863→                lines.append(f'            else:')
   864→   864→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   865→   865→
   866→   866→        # Validations
   867→   867→        lines.append('')
   868→   868→        lines.append('            # Run validations')
   869→   869→        for i, validation in enumerate(program.validations):
   870→   870→            lines.append(f'            self._validate_{i+1}()')
   871→   871→
   872→   872→        lines.append('')
   873→   873→        lines.append('            self._logger.info("Application completed successfully")')
   874→   874→        lines.append('')
   875→   875→        lines.append('        except Exception as e:')
   876→   876→        lines.append('            self._logger.error(f"Application error: {e}")')
   877→   877→        lines.append('            self._handle_risk(e)')
   878→   878→        lines.append('        finally:')
   879→   879→        lines.append('            self._running = False')
   880→   880→        lines.append('            self._logger.info("Application stopped")')
   881→   881→
   882→   882→        code = '\n'.join(lines)
   883→   883→
   884→   884→        # Register artifact
   885→   885→        artifact_name = "Method_run"
   886→   886→        self._provenance.register_artifact(
   887→   887→            name=artifact_name,
   888→   888→            artifact_type=ArtifactType.METHOD,
   889→   889→            source=f"Architecture Template: {template_name}",
   890→   890→            code_snippet="def run(self) -> None:",
   891→   891→        )
   892→   892→
   893→   893→        # Record run method provenance
   894→   894→        _, run_template_info = ArchitectureTemplateMapper.match(
   895→   895→            program.goals[0].description if program.goals else ""
   896→   896→        )
   897→   897→        self._provenance.record(
   898→   898→            source_type=ProvenanceType.RUN_METHOD,
   899→   899→            source_location="Method run",
   900→   900→            source_text=f"Template: {template_name}, Behaviors: {len(program.behaviors)}, Conditions: {len(program.conditions)}",
   901→   901→            resolution_path=[
   902→   902→                "AICL Source",
   903→   903→                f"Goal → Template: {template_name}",
   904→   904→                f"Structure: {run_template_info.get('structure', 'unknown')}",
   905→   905→                "Run Method Generation",
   906→   906→                "Generated Code",
   907→   907→            ],
   908→   908→            generated_code=code,
   909→   909→            confidence=1.0,
   910→   910→            template_name=template_name,
   911→   911→            artifact_names=[artifact_name],
   912→   912→        )
   913→   913→
