  1590→    def _make_class_name(self, name: str) -> str:
  1591→        """Convert a description to a valid Python class name."""
  1592→        words = name.replace('-', ' ').replace('_', ' ').split()
  1593→        if not words:
  1594→            return "AiclApplication"
  1595→
  1596→        filtered = [w for w in words if w.lower() not in ('a', 'an', 'the', 'on', 'in', 'of')]
  1597→        if not filtered:
  1598→            filtered = words
  1599→
  1600→        return ''.join(w.capitalize() for w in filtered)
