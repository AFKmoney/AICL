    90→    90→    Compiles AICL source code into deterministic, executable Python code
    91→    91→    using behavior patterns and architecture templates.

   103→   103→    def __init__(self, target_language: str = "python"):
   104→   104→        self.target_language = target_language

   106→   106→        self.errors: List[str] = []

   113→   113→    def compile(self, source: str) -> CompilationResult:
   114→   114→        """Compile AICL source code into executable Python code."""
   115→   115→        self.warnings = []
   116→   116→        self.errors = []
   117→   117→        self.stages_completed = []
   118→   118→        self._used_names = set()
   119→   119→        self._todo_count = 0
   120→   120→        self._provenance = CompilationProvenance()
   121→   121→
   122→   122→        result = CompilationResult()
   123→   123→
   124→   124→        # Stage 1: Specification Parsing
   125→   125→        program = self._stage_parsing(source)
   126→   126→        if program is None:
   127→   127→            result.success = False
   128→   128→            result.errors = self.errors
   129→   129→            return result
   130→   130→        self.stages_completed.append("Specification Parsing")
   131→   131→
   132→   132→        # Stage 2: Architecture Validation
   133→   133→        validation_warnings = self._stage_architecture_validation(program)
   134→   134→        self.warnings.extend(validation_warnings)
   135→   135→        self.stages_completed.append("Architecture Validation")
   136→   136→
   137→   137→        # Stage 3: Dependency Analysis
   138→   138→        dep_order = self._stage_dependency_analysis(program)
   139→   139→        self.stages_completed.append("Dependency Analysis")
   140→   140→
   141→   141→        # Stage 4: Risk Analysis
   142→   142→        risk_pairs = self._stage_risk_analysis(program)
   143→   143→        self.stages_completed.append("Risk Analysis")
   144→   144→
   145→   145→        # Stage 5: Recovery Synthesis
   146→   146→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   147→   147→        self.stages_completed.append("Recovery Synthesis")
   148→   148→
   149→   149→        # Stage 6: Code Generation
   150→   150→        source_code = self._stage_code_generation(
   151→   151→            program, dep_order, risk_pairs, recovery_map
   152→   152→        )
   153→   153→        self.stages_completed.append("Code Generation")
   154→   154→
   155→   155→        # Stage 7: Test Generation
   156→   156→        test_code = self._stage_test_generation(program, dep_order)
   157→   157→        self.stages_completed.append("Test Generation")
   158→   158→
   159→   159→        # Stage 8: Optimization
   160→   160→        source_code = self._stage_optimization(source_code, program)
   161→   161→        self.stages_completed.append("Optimization")
   162→   162→
   163→   163→        # Stage 9: Final Construction
   164→   164→        source_code = self._stage_final_construction(source_code, program)
   165→   165→        self.stages_completed.append("Final Construction")
   166→   166→
   167→   167→        # Build Architecture Tree string
   168→   168→        tree = ArchitectureTree(program)
   169→   169→        result.architecture_tree_str = tree.to_string()
   170→   170→
   171→   171→        result.success = True
   172→   172→        result.source_code = source_code
   173→   173→        result.test_code = test_code
   174→   174→        result.warnings = self.warnings
   175→   175→        result.errors = self.errors
   176→   176→        result.stages_completed = self.stages_completed
   177→   177→        result.todo_count = self._todo_count
   178→   178→        result.fully_compiled = self._todo_count == 0
   179→   179→        result.provenance = self._provenance
   180→   180→
   181→   181→        # Generate Proof of Origin — the central artifact
   182→   182→        result.proof = self._provenance.to_proof(
   183→   183→            source_text=source,
   184→   184→            generated_source=source_code,
   185→   185→            generated_tests=test_code,
   186→   186→            compiler_version="0.6.0",
   187→   187→            target_language=self.target_language,
   188→   188→        )
   189→   189→
   190→   190→        return result
   191→   191→
   192→   192→    def compile_to_file(self, source: str, output_dir: str, source_path: str = "") -> CompilationResult:
   193→   193→        """
   194→   194→        Compile AICL source code and write the output to files.
   195→   195→
   196→   196→        Produces:
   197→   197→            - main.py              (executable program)
   198→   198→            - test_main.py         (generated tests)
   199→   199→            - architecture_tree.txt (architecture visualization)
   200→   200→            - main.aicl-proof      (Proof of Origin — first-class artifact)
   201→   201→
   202→   202→        The .aicl-proof file is a self-contained, verifiable record that
   203→   203→        contains ALL information needed to reconstruct explain() and audit()
   204→   204→        without the compiler. It is the central artifact of auditable compilation.
   205→   205→        """
   206→   206→        result = self.compile(source)
   207→   207→
   208→   208→        if result.success:
   209→   209→            os.makedirs(output_dir, exist_ok=True)
   210→   210→
   211→   211→            main_path = os.path.join(output_dir, "main.py")
   212→   212→            with open(main_path, 'w') as f:
   213→   213→                f.write(result.source_code)
   214→   214→
   215→   215→            test_path = os.path.join(output_dir, "test_main.py")
   216→   216→            with open(test_path, 'w') as f:
   217→   217→                f.write(result.test_code)
   218→   218→
   219→   219→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   220→   220→            with open(tree_path, 'w') as f:
   221→   221→                f.write(result.architecture_tree_str)
   222→   222→
   223→   223→            # Write Proof of Origin — the central artifact
   224→   224→            if result.proof:
   225→   225→                # Update source_path in proof if provided
   226→   226→                if source_path:
   227→   227→                    result.proof.source_path = source_path
   228→   228→                proof_path = os.path.join(output_dir, "main.aicl-proof")
   229→   229→                result.proof.to_file(proof_path)
   230→   230→
   231→   231→        return result

   251→   251→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   252→   252→        """Validate the architectural structure of the program."""
   253→   253→        warnings = program.validate()
   254→   254→
   255→   255→        layer_names = [l.name for l in program.layers]
   256→   256→        seen = set()
   257→   257→        for name in layer_names:
   258→   258→            if name.lower() in seen:
   259→   259→                warnings.append(f"Duplicate layer name: {name}")
   260→   260→            seen.add(name.lower())
   261→   261→
   262→   262→        entity_names = {e.name for e in program.entities}
   263→   263→        for behavior in program.behaviors:
   264→   264→            for inp in behavior.inputs:
   265→   265→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   266→   266→                    warnings.append(
   267→   267→                        f"Behavior '{behavior.name}' references unknown type "
   268→   268→                        f"'{inp.param_type}' in input"
   269→   269→                    )
   270→   270→
   271→   271→        return warnings

   285→   285→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   286→   286→        """Analyze risks and pair them with recovery procedures."""
   287→   287→        pairs = []
   288→   288→        for i, risk in enumerate(program.risks):
   289→   289→            recovery = None
   290→   290→            if i < len(program.recoveries):
   291→   291→                recovery = program.recoveries[i]
   292→   292→            pairs.append((risk, recovery))
   293→   293→
   294→   294→        if len(program.recoveries) > len(program.risks):
   295→   295→            for i in range(len(program.risks), len(program.recoveries)):
   296→   296→                generic_risk = RiskSection(
   297→   297→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   298→   298→                )
   299→   299→                pairs.append((generic_risk, program.recoveries[i]))
   300→   300→
   301→   301→        return pairs

   325→   325→    def _unique_name(self, base: str) -> str:
   326→   326→        """Generate a unique Python identifier from a base name."""
   327→   327→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   328→   328→        clean = re.sub(r'_+', '_', clean).strip('_')
   329→   329→        if not clean:
   330→   330→            clean = "item"
   331→   331→        if clean[0].isdigit():
   332→   332→            clean = "f_" + clean
   333→   333→
   334→   334→        name = clean
   335→   335→        counter = 2
   336→   336→        while name in self._used_names:
   337→   337→            name = f"{clean}_{counter}"
   338→   338→            counter += 1
   339→   339→
   340→   340→        self._used_names.add(name)
   341→   341→        return name

   467→   467→    def _generate_entity(self, entity: EntitySection) -> str:
   468→   468→        """Generate a Python dataclass for an AICL Entity."""
   469→   469→        class_name = self._make_class_name(entity.name)
   470→   470→        artifact_name = f"Entity_{class_name}"
   471→   471→        lines = []
   472→   472→        lines.append('@dataclass')
   473→   473→        lines.append(f'class {class_name}:')
   474→   474→        lines.append(f'    """Entity: {entity.name}"""')
   475→   475→
   476→   476→        if not entity.fields:
   477→   477→            lines.append('    pass')
   478→   478→        else:
   479→   479→            for f in entity.fields:
   480→   480→                python_type = self._aicl_type_to_python(f.field_type)
   481→   481→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   482→   482→                if default:
   483→   483→                    lines.append(f'    {f.name}: {python_type} = {default}')
   484→   484→                else:
   485→   485→                    lines.append(f'    {f.name}: {python_type} = 0')
   486→   486→
   487→   487→        code = '\n'.join(lines)

   739→   739→    def _generate_layer_init(

   818→   818→    def _generate_run_method(
   819→   819→        self,
   820→   820→        program: AICLProgram,
   821→   821→        dep_order: List[LayerSection],
   822→   822→        layer_attr_map: Dict[str, str],
   823→   823→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   824→   824→        recovery_map: Dict[str, str],
   825→   825→        template_name: str,
   826→   826→        behavior_method_names: Dict[str, str] = None,
   827→   827→    ) -> str:
   828→   828→        """Generate the main run method — uses architecture template."""
   829→   829→        behavior_method_names = behavior_method_names or {}
   830→   830→        lines = []
   831→   831→        lines.append('    def run(self) -> None:')
   832→   832→        lines.append('        """Run the application with full error handling."""')
   833→   833→        lines.append('        self._logger.info("Starting application...")')
   834→   834→        lines.append('        self._running = True')
   835→   835→        lines.append('')
   836→   836→
   837→   837→        if template_name == "game_loop":
   838→   838→            # Game loop: init → update → render → validate → loop
   839→   839→            lines.append('        self._game_active = True')
   840→   840→            lines.append('        try:')
   841→   841→            lines.append('            while self._running and self._game_active:')
   842→   842→            lines.append('                frame_start = time.time()')
   843→   843→            lines.append('')
   844→   844→            lines.append('                # Update phase')
   845→   845→            for behavior in program.behaviors:
   846→   846→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   847→   847→                input_params = ', '.join(
   848→   848→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   849→   849→                    {e.name.lower() for e in program.entities}
   850→   850→                    else inp.name
   851→   851→                    for inp in behavior.inputs
   852→   852→                )
   853→   853→                lines.append(f'                self._behavior_{method_name}({input_params})')
   854→   854→            lines.append('')
   855→   855→            lines.append('                # Check conditions')
   856→   856→            for i, condition in enumerate(program.conditions):
   857→   857→                lines.append(f'                self._handle_condition_{i+1}()')
   858→   858→            lines.append('')
   859→   859→            lines.append('                # Render phase')
   860→   860→            for layer in dep_order:
   861→   861→                attr_name = layer_attr_map[layer.name]
   862→   862→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   863→   863→                    lines.append(f'                if self._{attr_name} is not None:')
   864→   864→                    lines.append(f'                    self._render_frame()')
   865→   865→            lines.append('')
   866→   866→            lines.append('                # Frame rate control')
   867→   867→            lines.append('                elapsed = time.time() - frame_start')
   868→   868→            lines.append('                if elapsed < 1.0 / 60.0:')
   869→   869→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   870→   870→            lines.append('                self._frame_count += 1')
   871→   871→            lines.append('')
   872→   872→
   873→   873→        elif template_name == "event_driven":
   874→   874→            # Event-driven: connect → listen → handle → respond
   875→   875→            lines.append('        try:')
   876→   876→            lines.append('            self._connect()')
   877→   877→            lines.append('            self._connected = True')
   878→   878→            lines.append('            self._logger.info("Connected to server")')
   879→   879→            lines.append('')
   880→   880→            lines.append('            while self._running and self._connected:')
   881→   881→            lines.append('                # Process message queue')
   882→   882→            lines.append('                while self._message_queue:')
   883→   883→            lines.append('                    msg = self._message_queue.pop(0)')
   884→   884→            lines.append('                    self._process_message(msg)')
   885→   885→            lines.append('')
   886→   886→            lines.append('                # Process offline queue if reconnected')
   887→   887→            lines.append('                if self._offline_queue and self._connected:')
   888→   888→            lines.append('                    self._sync_offline_messages()')
   889→   889→            lines.append('')
   890→   890→            lines.append('                # Wait for new messages')
   891→   891→            lines.append('                time.sleep(0.1)')
   892→   892→
   893→   893→        else:
   894→   894→            # Default: linear execution
   895→   895→            lines.append('        try:')
   896→   896→            for layer in dep_order:
   897→   897→                attr_name = layer_attr_map[layer.name]
   898→   898→                lines.append(f'            # Execute {layer.name}')
   899→   899→                lines.append(f'            if self._{attr_name} is not None:')
   900→   900→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   901→   901→                lines.append(f'            else:')
   902→   902→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   903→   903→
   904→   904→        # Validations
   905→   905→        lines.append('')
   906→   906→        lines.append('            # Run validations')
   907→   907→        for i, validation in enumerate(program.validations):
   908→   908→            lines.append(f'            self._validate_{i+1}()')
   909→   909→
   910→   910→        lines.append('')
   911→   911→        lines.append('            self._logger.info("Application completed successfully")')
   912→   912→        lines.append('')
   913→   913→        lines.append('        except Exception as e:')
   914→   914→        lines.append('            self._logger.error(f"Application error: {e}")')
   915→   915→        lines.append('            self._handle_risk(e)')
   916→   916→        lines.append('        finally:')
   917→   917→        lines.append('            self._running = False')
   918→   918→        lines.append('            self._logger.info("Application stopped")')
