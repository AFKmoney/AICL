     1→     1→"""
     2→     2→AICL Compilation Provenance & Audit System
     3→     3→
     4→     4→Every line of generated code must have a traceable provenance chain:
     5→     5→    AICL Source → Parse → Pattern Match → Template → Generated Code
     6→     6→

     8→     8→and provides an audit system to verify that every generated artifact
     9→     9→is traceable to its originating specification.
    10→    10→

    12→    12→    - Provenance Record: Explains why a specific piece of code was generated
    13→    13→    - Generated Artifact: A named unit of generated code (class, method, function, test)
    14→    14→    - Audit Coverage: Ratio of auditable artifacts to total generated artifacts
    15→    15→    - Orphan Artifact: A generated artifact without provenance
    16→    16→
    17→    17→Formal Property (The Auditability Theorem):
    18→    18→    A program is auditable if every generated artifact
    19→    19→    can be traced to its originating specification
    20→    20→    through a complete provenance chain.
    21→    21→
    22→    22→    Audit Coverage = Auditable Artifacts / Generated Artifacts
    23→    23→    Target: Audit Coverage = 1.0
    24→    24→
    25→    25→Usage:
    26→    26→    aicl explain pong.aicl              # Full compilation trace
    27→    27→    aicl explain pong.aicl --behavior MovePaddle  # Specific behavior trace

    99→    99→    A named unit of generated code.

   117→   117→        return len(self.provenance_indices) > 0

   120→   120→    def is_orphan(self) -> bool:

   125→   125→class CompilationProvenance:
   126→   126→    """
   127→   127→    Tracks the complete provenance of a compilation and provides
   128→   128→    audit capabilities.
   129→   129→
   130→   130→    Records every decision the compiler makes, creating an audit trail
   131→   131→    that can be queried to understand WHY any line of code exists.
   132→   132→
   133→   133→    This is the antidote to compiler complexity hiding. As AICL grows
   134→   134→    (30 patterns → 100 patterns → 500 patterns), the provenance tracker
   135→   135→    ensures every decision remains explainable.
   136→   136→
   137→   137→    v0.5: Added artifact tracking and audit coverage computation.
   138→   138→    """
   139→   139→
   140→   140→    def __init__(self):
   141→   141→        self.records: List[ProvenanceRecord] = []
   142→   142→        self.artifacts: List[GeneratedArtifact] = []
   143→   143→
   144→   144→    def record(
   145→   145→        self,
   146→   146→        source_type: ProvenanceType,
   147→   147→        source_location: str,
   148→   148→        source_text: str,
   149→   149→        resolution_path: List[str],
   150→   150→        generated_code: str,
   151→   151→        confidence: float = 1.0,
   152→   152→        pattern_name: str = "",
   153→   153→        template_name: str = "",
   154→   154→        parameters: Dict[str, str] = None,
   155→   155→        artifact_names: List[str] = None,
   156→   156→    ) -> ProvenanceRecord:
   157→   157→        """Record a compilation decision."""
   158→   158→        rec = ProvenanceRecord(
   159→   159→            source_type=source_type,
   160→   160→            source_location=source_location,
   161→   161→            source_text=source_text,
   162→   162→            resolution_path=resolution_path,
   163→   163→            generated_code=generated_code,
   164→   164→            confidence=confidence,
   165→   165→            pattern_name=pattern_name,
   166→   166→            template_name=template_name,
   167→   167→            parameters=parameters or {},
   168→   168→            artifact_names=artifact_names or [],
   169→   169→        )
   170→   170→        self.records.append(rec)
   171→   171→
   172→   172→        # Link this provenance record to any matching artifacts
   173→   173→        record_index = len(self.records) - 1
   174→   174→        for artifact_name in (artifact_names or []):
   175→   175→            for artifact in self.artifacts:
   176→   176→                if artifact.name == artifact_name:
   177→   177→                    artifact.provenance_indices.append(record_index)
   178→   178→
   179→   179→        return rec

   181→   181→    def register_artifact(
   182→   182→        self,
   183→   183→        name: str,
   184→   184→        artifact_type: ArtifactType,
   185→   185→        source: str,
   186→   186→        code_snippet: str = "",
   187→   187→    ) -> GeneratedArtifact:
   188→   188→        """
   189→   189→        Register a generated artifact.
   190→   190→
   191→   191→        Every method, class, function, and dataclass produced by the
   192→   192→        compiler should be registered. The audit system will verify
   193→   193→        that each artifact has at least one provenance record.
   194→   194→        """
   195→   195→        artifact = GeneratedArtifact(
   196→   196→            name=name,
   197→   197→            artifact_type=artifact_type,
   198→   198→            source=source,
   199→   199→            code_snippet=code_snippet,
   200→   200→        )
   201→   201→        self.artifacts.append(artifact)
   202→   202→
   203→   203→        # Check if any existing provenance records already cover this artifact
   204→   204→        for i, rec in enumerate(self.records):
   205→   205→            if name in rec.artifact_names:
   206→   206→                artifact.provenance_indices.append(i)
   207→   207→
   208→   208→        return artifact

   210→   210→    def link_provenance_to_artifact(self, record_index: int, artifact_name: str):
   211→   211→        """Explicitly link a provenance record to an artifact."""
   212→   212→        if record_index < len(self.records):
   213→   213→            if artifact_name not in self.records[record_index].artifact_names:
   214→   214→                self.records[record_index].artifact_names.append(artifact_name)
   215→   215→
   216→   216→        for artifact in self.artifacts:
   217→   217→            if artifact.name == artifact_name:
   218→   218→                if record_index not in artifact.provenance_indices:
   219→   219→                    artifact.provenance_indices.append(record_index)

   225→   225→    def explain(self, target: str = None) -> str:
   226→   226→        """
   227→   227→        Generate an explanation of the compilation.
   228→   228→
   229→   229→        Args:
   230→   230→            target: Optional filter (behavior name, pattern name, etc.)
   231→   231→
   232→   232→        Returns:
   233→   233→            Human-readable explanation of all compilation decisions.
   234→   234→        """
   235→   235→        lines = []
   236→   236→        lines.append("=" * 70)
   237→   237→        lines.append("AICL COMPILATION PROVENANCE TRACE")
   238→   238→        lines.append("=" * 70)
   239→   239→        lines.append("")
   240→   240→
   241→   241→        # Summary
   242→   242→        total = len(self.records)
   243→   243→        pattern_count = sum(1 for r in self.records if r.source_type == ProvenanceType.PATTERN_MATCH)
   244→   244→        sublang_count = sum(1 for r in self.records if r.source_type == ProvenanceType.SUB_LANGUAGE)
   245→   245→        fallback_count = sum(1 for r in self.records if r.source_type == ProvenanceType.FALLBACK)
   246→   246→        template_count = sum(1 for r in self.records if r.source_type == ProvenanceType.ARCHITECTURE_TEMPLATE)
   247→   247→        direct_count = sum(1 for r in self.records if r.source_type == ProvenanceType.DIRECT_MAPPING)
   248→   248→
   249→   249→        lines.append(f"Total decisions: {total}")
   250→   250→        lines.append(f"  Pattern matches:    {pattern_count}")
   251→   251→        lines.append(f"  Sub-language:       {sublang_count}")
   252→   252→        lines.append(f"  Direct mappings:    {direct_count}")
   253→   253→        lines.append(f"  Architecture tmpl:  {template_count}")
   254→   254→        lines.append(f"  Fallbacks:          {fallback_count}")
   255→   255→        lines.append("")
   256→   256→
   257→   257→        # Confidence score
   258→   258→        if total > 0:
   259→   259→            avg_confidence = sum(r.confidence for r in self.records) / total
   260→   260→            fully_deterministic = sum(1 for r in self.records if r.confidence >= 0.9)
   261→   261→            lines.append(f"Average confidence:    {avg_confidence:.2f}")
   262→   262→            lines.append(f"Fully deterministic:   {fully_deterministic}/{total} ({100*fully_deterministic//max(total,1)}%)")
   263→   263→            lines.append("")
   264→   264→
   265→   265→        # Group by source type
   266→   266→        by_type: Dict[ProvenanceType, List[ProvenanceRecord]] = {}
   267→   267→        for rec in self.records:
   268→   268→            if rec.source_type not in by_type:
   269→   269→                by_type[rec.source_type] = []
   270→   270→            by_type[rec.source_type].append(rec)
   271→   271→
   272→   272→        for source_type, records in by_type.items():
   273→   273→            type_label = source_type.value.replace("_", " ").title()
   274→   274→            lines.append("-" * 70)
   275→   275→            lines.append(f" {type_label} ({len(records)} decisions)")
   276→   276→            lines.append("-" * 70)
   277→   277→
   278→   278→            for i, rec in enumerate(records):
   279→   279→                # Filter by target if specified
   280→   280→                if target and target.lower() not in rec.source_location.lower() and target.lower() not in rec.pattern_name.lower():
   281→   281→                    continue
   282→   282→
   283→   283→                lines.append(f"")
   284→   284→                lines.append(f"  [{i+1}] {rec.source_location}")
   285→   285→                lines.append(f"  Source:  \"{rec.source_text}\"")
   286→   286→                lines.append(f"  Path:    {' → '.join(rec.resolution_path)}")
   287→   287→
   288→   288→                if rec.pattern_name:
   289→   289→                    lines.append(f"  Pattern: {rec.pattern_name}")
   290→   290→                if rec.template_name:
   291→   291→                    lines.append(f"  Template: {rec.template_name}")
   292→   292→                if rec.parameters:
   293→   293→                    params_str = ", ".join(f"{k}={v}" for k, v in rec.parameters.items() if not k.startswith("#"))
   294→   294→                    if params_str:
   295→   295→                        lines.append(f"  Params:  {params_str}")
   296→   296→
   297→   297→                lines.append(f"  Confidence: {rec.confidence:.2f}")
   298→   298→
   299→   299→                # Show generated code (truncated)
   300→   300→                code_lines = rec.generated_code.strip().split('\n')
   301→   301→                if len(code_lines) <= 4:
   302→   302→                    for cl in code_lines:
   303→   303→                        lines.append(f"  Code:    {cl}")
   304→   304→                else:
   305→   305→                    for cl in code_lines[:3]:
   306→   306→                        lines.append(f"  Code:    {cl}")
   307→   307→                    lines.append(f"  Code:    ... ({len(code_lines) - 3} more lines)")
   308→   308→
   309→   309→        lines.append("")
   310→   310→        lines.append("=" * 70)
   311→   311→        lines.append("END OF PROVENANCE TRACE")
   312→   312→        lines.append("=" * 70)
   313→   313→
   314→   314→        return '\n'.join(lines)

   321→   321→            "Explain compilation MovePaddle"

   373→   373→    def compute_explicability_coverage(self, generated_source: str, generated_tests: str = "") -> Dict:
   374→   374→        """
   375→   375→        Compute the explicability coverage ratio.
   376→   376→
   377→   377→        The explicability coverage measures what fraction of generated code
   378→   378→        lines are accounted for by provenance records. The target is 1.0
   379→   379→        (every line has provenance). Any deviation is a gap to investigate.
   380→   380→
   381→   381→        Returns:
   382→   382→            Dict with keys:
   383→   383→                - total_lines: total non-blank, non-comment lines in generated code
   384→   384→                - accounted_lines: lines that appear in at least one provenance record
   385→   385→                - coverage_ratio: accounted_lines / total_lines (target: 1.0)
   386→   386→                - unaccounted_lines: list of line numbers without provenance
   387→   387→                - by_type_coverage: coverage broken down by provenance type
   388→   388→        """
   389→   389→        # Combine source and test code
   390→   390→        all_code = generated_source
   391→   391→        if generated_tests:
   392→   392→            all_code += "\n" + generated_tests
   393→   393→
   394→   394→        # Count meaningful lines (non-blank, non-import, non-pure-comment)
   395→   395→        code_lines = []
   396→   396→        for i, line in enumerate(all_code.split('\n')):
   397→   397→            stripped = line.strip()
   398→   398→            if not stripped:
   399→   399→                continue
   400→   400→            # Skip pure comment lines and import lines for coverage
   401→   401→            if stripped.startswith('#') and not any(kw in stripped for kw in ['When:', 'On:', 'Risk:', 'Recovery:', 'Action:', 'Intent:']):
   402→   402→                continue
   403→   403→            if stripped.startswith(('import ', 'from ')):
   404→   404→                continue
   405→   405→            if stripped in ('"""', "'''", ''):
   406→   406→                continue
   407→   407→            code_lines.append((i + 1, stripped))
   408→   408→
   409→   409→        total_lines = len(code_lines)
   410→   410→        if total_lines == 0:
   411→   411→            return {
   412→   412→                "total_lines": 0,
   413→   413→                "accounted_lines": 0,
   414→   414→                "coverage_ratio": 1.0,
   415→   415→                "unaccounted_lines": [],
   416→   416→                "by_type_coverage": {},
   417→   417→            }
   418→   418→
   419→   419→        # Build set of code fragments from provenance records
   420→   420→        accounted_fragments = set()
   421→   421→        for rec in self.records:
   422→   422→            for code_line in rec.generated_code.strip().split('\n'):
   423→   423→                stripped = code_line.strip()
   424→   424→                if stripped:
   425→   425→                    # Normalize for matching (remove variable values that change)
   426→   426→                    normalized = re.sub(r'[\'"].*?[\'\"]', "'...'", stripped)
   427→   427→                    normalized = re.sub(r'\{[^}]*\}', '{...}', normalized)
   428→   428→                    accounted_fragments.add(normalized)
   429→   429→                    # Also add the original
   430→   430→                    accounted_fragments.add(stripped)
   431→   431→
   432→   432→        # Check which lines are accounted for
   433→   433→        accounted_lines = 0
   434→   434→        unaccounted = []
   435→   435→        for line_num, line_text in code_lines:
   436→   436→            stripped = line_text.strip()
   437→   437→            # Check if this line or a normalized version appears in provenance
   438→   438→            found = False
   439→   439→            if stripped in accounted_fragments:
   440→   440→                found = True
   441→   441→            else:
   442→   442→                # Try normalizing the source line too
   443→   443→                normalized = re.sub(r'[\'"].*?[\'\"]', "'...'", stripped)
   444→   444→                normalized = re.sub(r'\{[^}]*\}', '{...}', normalized)
   445→   445→                if normalized in accounted_fragments:
   446→   446→                    found = True
   447→   447→                else:
   448→   448→                    # Check if any provenance fragment is a prefix of this line
   449→   449→                    for frag in accounted_fragments:
   450→   450→                        if frag and stripped.startswith(frag.split('=')[0].split('(')[0].strip()):
   451→   451→                            found = True
   452→   452→                            break
   453→   453→
   454→   454→            if found:
   455→   455→                accounted_lines += 1
   456→   456→            else:
   457→   457→                # Only flag as unaccounted if it's a significant line
   458→   458→                if not stripped.startswith(('#', '"""', "'''", 'pass', 'else:', 'try:', 'except', 'finally:')):
   459→   459→                    unaccounted.append((line_num, stripped[:80]))
   460→   460→
   461→   461→        # Coverage by provenance type
   462→   462→        by_type_coverage = {}
   463→   463→        for ptype in ProvenanceType:
   464→   464→            count = len([r for r in self.records if r.source_type == ptype])
   465→   465→            if count > 0:
   466→   466→                by_type_coverage[ptype.value] = count
   467→   467→
   468→   468→        return {
   469→   469→            "total_lines": total_lines,
   470→   470→            "accounted_lines": accounted_lines,
   471→   471→            "coverage_ratio": accounted_lines / total_lines if total_lines > 0 else 1.0,
   472→   472→            "unaccounted_lines": unaccounted,
   473→   473→            "by_type_coverage": by_type_coverage,
   474→   474→        }

   476→   476→    def compute_audit_coverage(self) -> Dict:
   477→   477→        """
   478→   478→        Compute the audit coverage of generated artifacts.
   479→   479→
   480→   480→        Audit Coverage = Auditable Artifacts / Generated Artifacts
   481→   481→
   482→   482→        An artifact is "auditable" if it has at least one provenance record.
   483→   483→        An artifact without provenance is an "orphan".
   484→   484→
   485→   485→        This is the key metric for the Auditability property:
   486→   486→            "A program is auditable if every generated artifact
   487→   487→             can be traced to its originating specification
   488→   488→             through a complete provenance chain."
   489→   489→
   490→   490→        Returns:
   491→   491→            Dict with:
   492→   492→                - total_artifacts: total number of generated artifacts
   493→   493→                - auditable_artifacts: artifacts with at least one provenance record
   494→   494→                - orphan_artifacts: list of artifacts without provenance
   495→   495→                - audit_coverage: ratio of auditable to total (target: 1.0)
   496→   496→                - by_type: breakdown by artifact type
   497→   497→        """
   498→   498→        total = len(self.artifacts)
   499→   499→        if total == 0:
   500→   500→            return {
   501→   501→                "total_artifacts": 0,
   502→   502→                "auditable_artifacts": 0,
   503→   503→                "orphan_artifacts": [],
   504→   504→                "audit_coverage": 1.0,
   505→   505→                "by_type": {},
   506→   506→            }
   507→   507→
   508→   508→        auditable = [a for a in self.artifacts if a.has_provenance]
   509→   509→        orphans = [a for a in self.artifacts if a.is_orphan]
   510→   510→
   511→   511→        # Breakdown by type
   512→   512→        by_type = {}
   513→   513→        for atype in ArtifactType:
   514→   514→            type_artifacts = [a for a in self.artifacts if a.artifact_type == atype]
   515→   515→            type_auditable = [a for a in type_artifacts if a.has_provenance]
   516→   516→            if type_artifacts:
   517→   517→                by_type[atype.value] = {
   518→   518→                    "total": len(type_artifacts),
   519→   519→                    "auditable": len(type_auditable),
   520→   520→                    "coverage": len(type_auditable) / len(type_artifacts),
   521→   521→                }
   522→   522→
   523→   523→        return {
   524→   524→            "total_artifacts": total,
   525→   525→            "auditable_artifacts": len(auditable),
   526→   526→            "orphan_artifacts": [{"name": a.name, "type": a.artifact_type.value, "source": a.source} for a in orphans],
   527→   527→            "audit_coverage": len(auditable) / total,
   528→   528→            "by_type": by_type,
   529→   529→        }

   531→   531→    def explain_coverage(self, generated_source: str, generated_tests: str = "") -> str:

   542→   542→        lines.append("")

   565→   565→        else:
   566→   566→            lines.append("All lines have provenance coverage.")
   567→   567→
   568→   568→        lines.append("")
   569→   569→        lines.append("=" * 70)
   570→   570→        return '\n'.join(lines)

   572→   572→    def audit(self, generated_source: str = "", generated_tests: str = "") -> str:
   573→   573→        """
   574→   574→        Generate a complete audit report.
   575→   575→
   576→   576→        This is the primary output of the audit system. It verifies the
   577→   577→        Auditability property:
   578→   578→
   579→   579→            "A program is auditable if every generated artifact
   580→   580→             can be traced to its originating specification
   581→   581→             through a complete provenance chain."
   582→   582→
   583→   583→        The audit report contains:
   584→   584→            1. Artifact summary (total, auditable, orphans)
   585→   585→            2. Audit coverage ratio (target: 1.0)
   586→   586→            3. Orphan artifact list (artifacts without provenance)
   587→   587→            4. Audit status (VERIFIED / PARTIAL / FAILED)
   588→   588→            5. Explicability coverage (line-level)
   589→   589→            6. Breakdown by artifact type
   590→   590→
   591→   591→        Returns:
   592→   592→            Human-readable audit report.
   593→   593→        """
   594→   594→        lines = []
   595→   595→        lines.append("=" * 70)
   596→   596→        lines.append("AICL AUDIT REPORT")
   597→   597→        lines.append("=" * 70)
   598→   598→        lines.append("")
   599→   599→
   600→   600→        # Artifact audit
   601→   601→        audit = self.compute_audit_coverage()
   602→   602→
   603→   603→        lines.append("ARTIFACT AUDIT")
   604→   604→        lines.append("-" * 70)
   605→   605→        lines.append(f"  Generated artifacts:     {audit['total_artifacts']}")
   606→   606→        lines.append(f"  Artifacts with provenance: {audit['auditable_artifacts']}")
   607→   607→        lines.append(f"  Orphan artifacts:        {len(audit['orphan_artifacts'])}")
   608→   608→        lines.append("")
   609→   609→
   610→   610→        coverage_pct = audit['audit_coverage'] * 100
   611→   611→        lines.append(f"  Audit Coverage:          {coverage_pct:.2f}%")
   612→   612→        lines.append("")
   613→   613→
   614→   614→        # Audit status
   615→   615→        if audit['audit_coverage'] >= 1.0:
   616→   616→            lines.append("  Audit Status: VERIFIED")
   617→   617→        elif audit['audit_coverage'] >= 0.90:
   618→   618→            lines.append("  Audit Status: PARTIAL (investigation needed)")
   619→   619→        else:
   620→   620→            lines.append("  Audit Status: FAILED (significant provenance gaps)")
   621→   621→        lines.append("")
   622→   622→
   623→   623→        # Orphan artifacts
   624→   624→        if audit['orphan_artifacts']:
   625→   625→            lines.append("ORPHAN ARTIFACTS (no provenance)")
   626→   626→            lines.append("-" * 70)
   627→   627→            for orphan in audit['orphan_artifacts']:
   628→   628→                lines.append(f"  [{orphan['type']:15s}] {orphan['name']}")
   629→   629→                lines.append(f"    Source: {orphan['source']}")
   630→   630→            lines.append("")
   631→   631→
   632→   632→        # Breakdown by artifact type
   633→   633→        if audit['by_type']:
   634→   634→            lines.append("COVERAGE BY ARTIFACT TYPE")
   635→   635→            lines.append("-" * 70)
   636→   636→            for atype, info in sorted(audit['by_type'].items(), key=lambda x: -x[1]['total']):
   637→   637→                cov_pct = info['coverage'] * 100
   638→   638→                status = "VERIFIED" if info['coverage'] >= 1.0 else "PARTIAL" if info['coverage'] >= 0.9 else "FAILED"
   639→   639→                lines.append(f"  {atype:15s}  {info['auditable']:3d}/{info['total']:3d}  {cov_pct:6.2f}%  [{status}]")
   640→   640→            lines.append("")
   641→   641→
   642→   642→        # Line-level explicability coverage
   643→   643→        if generated_source:
   644→   644→            exp_coverage = self.compute_explicability_coverage(generated_source, generated_tests)
   645→   645→            lines.append("LINE-LEVEL EXPLICABILITY")
   646→   646→            lines.append("-" * 70)
   647→   647→            lines.append(f"  Total meaningful lines:  {exp_coverage['total_lines']}")
   648→   648→            lines.append(f"  Lines with provenance:   {exp_coverage['accounted_lines']}")
   649→   649→            lines.append(f"  Explicability coverage:  {exp_coverage['coverage_ratio']:.2%}")
   650→   650→
   651→   651→            if exp_coverage['coverage_ratio'] >= 0.95:
   652→   652→                lines.append("  Line status: PASS (>= 95%)")
   653→   653→            elif exp_coverage['coverage_ratio'] >= 0.80:
   654→   654→                lines.append("  Line status: PARTIAL (80-95%)")
   655→   655→            else:
   656→   656→                lines.append("  Line status: FAIL (< 80%)")
   657→   657→
   658→   658→            if exp_coverage['unaccounted_lines']:
   659→   659→                lines.append(f"  Unaccounted lines: {len(exp_coverage['unaccounted_lines'])}")
   660→   660→            lines.append("")
   661→   661→
   662→   662→        # Provenance summary
   663→   663→        lines.append("PROVENANCE SUMMARY")
   664→   664→        lines.append("-" * 70)
   665→   665→        total_records = len(self.records)
   666→   666→        lines.append(f"  Total provenance records: {total_records}")
   667→   667→
   668→   668→        if total_records > 0:
   669→   669→            avg_conf = sum(r.confidence for r in self.records) / total_records
   670→   670→            lines.append(f"  Average confidence:       {avg_conf:.2f}")
   671→   671→            deterministic = sum(1 for r in self.records if r.confidence >= 0.9)
   672→   672→            lines.append(f"  Deterministic decisions:  {deterministic}/{total_records} ({100*deterministic//total_records}%)")
   673→   673→
   674→   674→            # By type summary
   675→   675→            by_type = {}
   676→   676→            for rec in self.records:
   677→   677→                label = rec.source_type.value.replace("_", " ").title()
   678→   678→                by_type[label] = by_type.get(label, 0) + 1
   679→   679→
   680→   680→            lines.append("  Provenance by type:")
   681→   681→            for label, count in sorted(by_type.items(), key=lambda x: -x[1]):
   682→   682→                lines.append(f"    {label:30s} {count:4d}")
   683→   683→        lines.append("")
   684→   684→
   685→   685→        # Final verdict
   686→   686→        lines.append("=" * 70)
   687→   687→        if audit['audit_coverage'] >= 1.0 and (not generated_source or exp_coverage['coverage_ratio'] >= 0.95):
   688→   688→            lines.append("AUDIT RESULT: PASSED")
   689→   689→            lines.append("  Every generated artifact has provenance.")
   690→   690→            lines.append("  Compilation is fully auditable.")
   691→   691→        elif audit['audit_coverage'] >= 0.90:
   692→   692→            lines.append("AUDIT RESULT: PARTIAL")
   693→   693→            lines.append("  Most artifacts have provenance.")
   694→   694→            lines.append("  Some orphan artifacts need investigation.")
   695→   695→        else:
   696→   696→            lines.append("AUDIT RESULT: FAILED")
   697→   697→            lines.append("  Significant provenance gaps detected.")
   698→   698→            lines.append("  Compilation is not fully auditable.")
   699→   699→        lines.append("=" * 70)
   700→   700→
   701→   701→        return '\n'.join(lines)
   702→   702→
   703→   703→    def audit_passed(self, generated_source: str = "", generated_tests: str = "") -> bool:
   704→   704→        """
   705→   705→        Check whether the audit passes.
   706→   706→
   707→   707→        Audit passes if:
   708→   708→            1. Audit coverage = 1.0 (no orphan artifacts)
   709→   709→            2. Explicability coverage >= 0.95 (line-level)
   710→   710→
   711→   711→        Used by `aicl audit --strict` to determine exit code.
   712→   712→        """
   713→   713→        audit = self.compute_audit_coverage()
   714→   714→        if audit['audit_coverage'] < 1.0:
   715→   715→            return False
   716→   716→
   717→   717→        if generated_source:
   718→   718→            exp = self.compute_explicability_coverage(generated_source, generated_tests)
   719→   719→            if exp['coverage_ratio'] < 0.95:
   720→   720→                return False
   721→   721→
   722→   722→        return True

   773→   773→class ProofOfOrigin:
   774→   774→    """
   775→   775→    Proof of Origin — the central artifact of auditable compilation.
   776→   776→
   777→   777→    A Proof of Origin is a self-contained, verifiable record that proves:
   778→   778→        1. Every generated artifact has a traceable provenance chain
   779→   779→        2. No artifact exists without justification (No Orphan Property)
   780→   780→        3. Audit coverage = 1.0 (Complete Coverage Property)
   781→   781→        4. The proof is cryptographically bound to the generated code
   782→   782→
   783→   783→    Architecture:
   784→   784→        aicl compile pong.aicl
   785→   785→            ↓
   786→   786→        pong.py              (executable program)
   787→   787→        pong.aicl-proof      (proof of origin)
   788→   788→
   789→   789→        aicl explain --proof pong.aicl-proof    → human-readable explanation
   790→   790→        aicl audit --proof pong.aicl-proof      → audit report
   791→   791→        aicl proof pong.aicl-proof              → inspect/verify proof
   792→   792→
   793→   793→    The proof file is JSON, machine-readable, and contains ALL information
   794→   794→    needed to reconstruct explain() and audit() without the compiler.
   795→   795→
   796→   796→    Formal Properties Verified:
   797→   797→        - No Orphan Artifact Property:
   798→   798→            Every generated artifact has at least one provenance chain.
   799→   799→        - Complete Coverage Property:
   800→   800→            Audit Coverage = Auditable Artifacts / Generated Artifacts = 1.0
   801→   801→        - Hash Binding:
   802→   802→            SHA-256(proof) is bound to SHA-256(generated code).
   803→   803→    """
   804→   804→
   805→   805→    PROOF_FORMAT_VERSION = "2.0"
   806→   806→
   807→   807→    def __init__(self):
   808→   808→        # Metadata
   809→   809→        self.format_version: str = self.PROOF_FORMAT_VERSION
   810→   810→        self.compiler_version: str = ""
   811→   811→        self.timestamp: str = ""
   812→   812→        self.source_path: str = ""
   813→   813→        self.target_language: str = "python"
   814→   814→
   815→   815→        # Source binding
   816→   816→        self.source_hash: str = ""          # SHA-256 of AICL source
   817→   817→        self.source_text: str = ""          # Full AICL source text
   818→   818→
   819→   819→        # Code binding
   820→   820→        self.program_hash: str = ""         # SHA-256 of generated program
   821→   821→        self.test_hash: str = ""            # SHA-256 of generated tests
   822→   822→        self.generated_source: str = ""     # Generated program code
   823→   823→        self.generated_tests: str = ""      # Generated test code
   824→   824→
   825→   825→        # Provenance records
   826→   826→        self.records: List[Dict[str, Any]] = []
   827→   827→
   828→   828→        # Generated artifacts
   829→   829→        self.artifacts: List[Dict[str, Any]] = []
   830→   830→
   831→   831→        # Formal properties
   832→   832→        self.formal_properties: Dict[str, Any] = {}
   833→   833→
   834→   834→        # Audit coverage
   835→   835→        self.audit_coverage: Dict[str, Any] = {}
   836→   836→
   837→   837→        # Explicability coverage
   838→   838→        self.explicability_coverage: Dict[str, Any] = {}
   839→   839→
   840→   840→    @classmethod
   841→   841→    def from_compilation(
   842→   842→        cls,
   843→   843→        provenance: CompilationProvenance,
   844→   844→        source_path: str = "",
   845→   845→        source_text: str = "",
   846→   846→        generated_source: str = "",
   847→   847→        generated_tests: str = "",
   848→   848→        compiler_version: str = "0.6.0",
   849→   849→        target_language: str = "python",
   850→   850→    ) -> 'ProofOfOrigin':
   851→   851→        """
   852→   852→        Create a Proof of Origin from a completed compilation.
   853→   853→
   854→   854→        This is the primary constructor — used by the compiler at the end
   855→   855→        of a successful compilation to produce the proof artifact.
   856→   856→        """
   857→   857→        proof = cls()
   858→   858→        proof.compiler_version = compiler_version
   859→   859→        proof.timestamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
   860→   860→        proof.source_path = source_path
   861→   861→        proof.target_language = target_language
   862→   862→
   863→   863→        # Bind to source
   864→   864→        proof.source_text = source_text
   865→   865→        proof.source_hash = hashlib.sha256(source_text.encode('utf-8')).hexdigest()
   866→   866→
   867→   867→        # Bind to generated code
   868→   868→        proof.generated_source = generated_source
   869→   869→        proof.generated_tests = generated_tests
   870→   870→        proof.program_hash = hashlib.sha256(generated_source.encode('utf-8')).hexdigest()
   871→   871→        proof.test_hash = hashlib.sha256(generated_tests.encode('utf-8')).hexdigest()
   872→   872→
   873→   873→        # Serialize provenance records
   874→   874→        proof.records = []
   875→   875→        for rec in provenance.records:
   876→   876→            proof.records.append({
   877→   877→                "source_type": rec.source_type.value,
   878→   878→                "source_location": rec.source_location,
   879→   879→                "source_text": rec.source_text,
   880→   880→                "resolution_path": rec.resolution_path,
   881→   881→                "generated_code": rec.generated_code,
   882→   882→                "confidence": rec.confidence,
   883→   883→                "pattern_name": rec.pattern_name,
   884→   884→                "template_name": rec.template_name,
   885→   885→                "parameters": rec.parameters,
   886→   886→                "artifact_names": rec.artifact_names,
   887→   887→            })
   888→   888→
   889→   889→        # Serialize artifacts
   890→   890→        proof.artifacts = []
   891→   891→        for art in provenance.artifacts:
   892→   892→            proof.artifacts.append({
   893→   893→                "name": art.name,
   894→   894→                "artifact_type": art.artifact_type.value,
   895→   895→                "source": art.source,
   896→   896→                "provenance_indices": art.provenance_indices,
   897→   897→                "code_snippet": art.code_snippet,
   898→   898→                "has_provenance": art.has_provenance,
   899→   899→                "is_orphan": art.is_orphan,
   900→   900→            })
   901→   901→
   902→   902→        # Compute and store formal properties
   903→   903→        audit = provenance.compute_audit_coverage()

  1051→  1051→    def verify(self, generated_source: str = "", generated_tests: str = "") -> Dict[str, Any]:
  1052→  1052→        """
  1053→  1053→        Verify the integrity of this proof.
  1054→  1054→
  1055→  1055→        Checks:
  1056→  1056→            1. Format version is supported
  1057→  1057→            2. Source hash binding: SHA-256(source_text) matches source_hash
  1058→  1058→            3. Program hash binding: SHA-256(generated code) matches program_hash
  1059→  1059→            4. Test hash binding: SHA-256(test code) matches test_hash
  1060→  1060→            5. No Orphan Property: every artifact has provenance
  1061→  1061→            6. Complete Coverage Property: audit coverage = 1.0
  1062→  1062→            7. Record-artifact linkage integrity
  1063→  1063→            8. Artifact-artifact consistency (no duplicate artifact names with conflicting data)
  1064→  1064→
  1065→  1065→        In format v2.0, the proof contains the generated code and source text
  1066→  1066→        internally. When no external code is provided, verification uses the
  1067→  1067→        embedded code — making the proof fully self-verifiable by an independent
  1068→  1068→        third party without any AICL dependency.
  1069→  1069→
  1070→  1070→        Returns:
  1071→  1071→            Dict with 'valid' (bool) and 'checks' (list of check results).
  1072→  1072→        """
  1073→  1073→        checks = []
  1074→  1074→
  1075→  1075→        # Use embedded code if no external code is provided (v2.0 self-contained)
  1076→  1076→        source_to_check = generated_source or self.generated_source
  1077→  1077→        tests_to_check = generated_tests or self.generated_tests
  1078→  1078→
  1079→  1079→        # Check 1: Format version
  1080→  1080→        checks.append({
  1081→  1081→            "name": "format_version",
  1082→  1082→            "description": "Proof format version is supported",
  1083→  1083→            "passed": self.format_version in ("1.0", "2.0"),
  1084→  1084→        })
  1085→  1085→
  1086→  1086→        # Check 2: Source hash binding
  1087→  1087→        if self.source_text:
  1088→  1088→            actual_hash = hashlib.sha256(self.source_text.encode('utf-8')).hexdigest()
  1089→  1089→            hash_match = actual_hash == self.source_hash
  1090→  1090→            checks.append({
  1091→  1091→                "name": "source_hash_binding",
  1092→  1092→                "description": "AICL source text SHA-256 matches proof",
  1093→  1093→                "passed": hash_match,
  1094→  1094→                "expected": self.source_hash,
  1095→  1095→                "actual": actual_hash,
  1096→  1096→            })
  1097→  1097→        else:
  1098→  1098→            checks.append({
  1099→  1099→                "name": "source_hash_binding",
  1100→  1100→                "description": "No source text embedded in proof",
  1101→  1101→                "passed": True,
  1102→  1102→                "note": "Source hash not verified (no embedded source text)",
  1103→  1103→            })
  1104→  1104→
  1105→  1105→        # Check 3: Program hash binding
  1106→  1106→        if source_to_check:
  1107→  1107→            actual_hash = hashlib.sha256(source_to_check.encode('utf-8')).hexdigest()
  1108→  1108→            hash_match = actual_hash == self.program_hash
  1109→  1109→            checks.append({
  1110→  1110→                "name": "program_hash_binding",
  1111→  1111→                "description": "Generated program SHA-256 matches proof",
  1112→  1112→                "passed": hash_match,
  1113→  1113→                "expected": self.program_hash,
  1114→  1114→                "actual": actual_hash,
  1115→  1115→            })
  1116→  1116→        else:
  1117→  1117→            checks.append({
  1118→  1118→                "name": "program_hash_binding",
  1119→  1119→                "description": "No generated source provided or embedded for hash verification",
  1120→  1120→                "passed": False,
  1121→  1121→                "note": "Cannot verify hash binding — proof is not self-contained",
  1122→  1122→            })
  1123→  1123→
  1124→  1124→        # Check 4: Test hash binding
  1125→  1125→        if tests_to_check:
  1126→  1126→            actual_hash = hashlib.sha256(tests_to_check.encode('utf-8')).hexdigest()
  1127→  1127→            hash_match = actual_hash == self.test_hash
  1128→  1128→            checks.append({
  1129→  1129→                "name": "test_hash_binding",
  1130→  1130→                "description": "Generated tests SHA-256 matches proof",
  1131→  1131→                "passed": hash_match,
  1132→  1132→                "expected": self.test_hash,
  1133→  1133→                "actual": actual_hash,
  1134→  1134→            })
  1135→  1135→        else:
  1136→  1136→            checks.append({
  1137→  1137→                "name": "test_hash_binding",
  1138→  1138→                "description": "No test code provided or embedded for hash verification",
  1139→  1139→                "passed": False,
  1140→  1140→                "note": "Cannot verify hash binding — proof is not self-contained",
  1141→  1141→            })
  1142→  1142→
  1143→  1143→        # Check 5: No Orphan Property
  1144→  1144→        no_orphan = self.formal_properties.get('no_orphan_artifact_property', {})
  1145→  1145→        orphan_count = no_orphan.get('orphan_count',
  1146→  1146→            len(self.audit_coverage.get('orphan_artifacts', [])))
  1147→  1147→        checks.append({
  1148→  1148→            "name": "no_orphan_artifact_property",
  1149→  1149→            "description": "Every generated artifact has at least one provenance chain",
  1150→  1150→            "passed": orphan_count == 0,
  1151→  1151→            "orphan_count": orphan_count,
  1152→  1152→        })
  1153→  1153→
  1154→  1154→        # Check 6: Complete Coverage Property
  1155→  1155→        complete_cov = self.formal_properties.get('complete_coverage_property', {})
  1156→  1156→        coverage = complete_cov.get('coverage',
  1157→  1157→            self.audit_coverage.get('audit_coverage', 0))
  1158→  1158→        checks.append({
  1159→  1159→            "name": "complete_coverage_property",
  1160→  1160→            "description": "Audit Coverage = Auditable Artifacts / Generated Artifacts = 1.0",
  1161→  1161→            "passed": coverage >= 1.0,
  1162→  1162→            "coverage": coverage,
  1163→  1163→        })
  1164→  1164→
  1165→  1165→        # Check 7: Record-artifact linkage integrity
  1166→  1166→        linkage_valid = True
  1167→  1167→        for artifact_data in self.artifacts:
  1168→  1168→            if not artifact_data.get('is_orphan', True):
  1169→  1169→                # Non-orphan artifact should have provenance indices
  1170→  1170→                indices = artifact_data.get('provenance_indices', [])
  1171→  1171→                if not indices:
  1172→  1172→                    linkage_valid = False
  1173→  1173→                    break
  1174→  1174→                # Each index should reference a valid record
  1175→  1175→                for idx in indices:
  1176→  1176→                    if idx < 0 or idx >= len(self.records):
  1177→  1177→                        linkage_valid = False
  1178→  1178→                        break
  1179→  1179→        checks.append({
  1180→  1180→            "name": "record_artifact_linkage",
  1181→  1181→            "description": "All provenance indices reference valid records",
  1182→  1182→            "passed": linkage_valid,
  1183→  1183→        })
  1184→  1184→
  1185→  1185→        # Check 8: Artifact-artifact consistency
  1186→  1186→        # Verify that artifact is_orphan field is consistent with provenance_indices
  1187→  1187→        consistency_valid = True
  1188→  1188→        for artifact_data in self.artifacts:
  1189→  1189→            has_indices = bool(artifact_data.get('provenance_indices', []))
  1190→  1190→            claims_orphan = artifact_data.get('is_orphan', not has_indices)
  1191→  1191→            if has_indices and claims_orphan:
  1192→  1192→                consistency_valid = False
  1193→  1193→                break
  1194→  1194→            if not has_indices and not claims_orphan:
  1195→  1195→                consistency_valid = False
  1196→  1196→                break
  1197→  1197→        checks.append({
  1198→  1198→            "name": "artifact_consistency",
  1199→  1199→            "description": "Artifact orphan status is consistent with provenance linkage",
  1200→  1200→            "passed": consistency_valid,
  1201→  1201→        })
  1202→  1202→
  1203→  1203→        all_passed = all(c['passed'] for c in checks)
  1204→  1204→        return {
  1205→  1205→            "valid": all_passed,
  1206→  1206→            "checks": checks,
  1207→  1207→        }

  1213→  1213→    def explain(self, target: str = None) -> str:
  1214→  1214→        """
  1215→  1215→        Reconstruct a full explanation from this proof.
  1216→  1216→
  1217→  1217→        This proves that the .aicl-proof file contains ALL information
  1218→  1218→        needed to answer "Why did the compiler generate this line?"
  1219→  1219→        without requiring the compiler itself.
  1220→  1220→
  1221→  1221→        Args:
  1222→  1222→            target: Optional filter (behavior name, pattern name, etc.)
  1223→  1223→
  1224→  1224→        Returns:
  1225→  1225→            Human-readable explanation of all compilation decisions.
  1226→  1226→        """
  1227→  1227→        lines = []
  1228→  1228→        lines.append("=" * 70)
  1229→  1229→        lines.append("AICL PROOF OF ORIGIN — EXPLANATION")
  1230→  1230→        lines.append("=" * 70)
  1231→  1231→        lines.append("")
  1232→  1232→        lines.append(f"Source: {self.source_path or '<unknown>'}")
  1233→  1233→        lines.append(f"Compiled: {self.timestamp}")
  1234→  1234→        lines.append(f"Compiler: AICL v{self.compiler_version}")
  1235→  1235→        lines.append(f"Target: {self.target_language}")
  1236→  1236→        lines.append("")
  1237→  1237→
  1238→  1238→        # Summary
  1239→  1239→        total = len(self.records)
  1240→  1240→        type_counts = {}
  1241→  1241→        for rec in self.records:
  1242→  1242→            st = rec.get('source_type', 'unknown')
  1243→  1243→            type_counts[st] = type_counts.get(st, 0) + 1
  1244→  1244→
  1245→  1245→        lines.append(f"Total decisions: {total}")
  1246→  1246→        for st, count in sorted(type_counts.items(), key=lambda x: -x[1]):
  1247→  1247→            label = st.replace("_", " ").title()
  1248→  1248→            lines.append(f"  {label}: {count}")
  1249→  1249→        lines.append("")
  1250→  1250→
  1251→  1251→        # Confidence
  1252→  1252→        if total > 0:
  1253→  1253→            confidences = [r.get('confidence', 1.0) for r in self.records]
  1254→  1254→            avg_conf = sum(confidences) / len(confidences)
  1255→  1255→            fully_det = sum(1 for c in confidences if c >= 0.9)
  1256→  1256→            lines.append(f"Average confidence: {avg_conf:.2f}")
  1257→  1257→            lines.append(f"Fully deterministic: {fully_det}/{total} ({100*fully_det//max(total,1)}%)")
  1258→  1258→            lines.append("")
  1259→  1259→
  1260→  1260→        # Group by source type
  1261→  1261→        by_type: Dict[str, list] = {}
  1262→  1262→        for rec in self.records:
  1263→  1263→            st = rec.get('source_type', 'unknown')
  1264→  1264→            if st not in by_type:
  1265→  1265→                by_type[st] = []
  1266→  1266→            by_type[st].append(rec)
  1267→  1267→
  1268→  1268→        for source_type, records in by_type.items():
  1269→  1269→            type_label = source_type.replace("_", " ").title()
  1270→  1270→            lines.append("-" * 70)
  1271→  1271→            lines.append(f" {type_label} ({len(records)} decisions)")
  1272→  1272→            lines.append("-" * 70)
  1273→  1273→
  1274→  1274→            for i, rec in enumerate(records):
  1275→  1275→                # Filter by target if specified
  1276→  1276→                if target and target.lower() not in rec.get('source_location', '').lower() \
  1277→  1277→                        and target.lower() not in rec.get('pattern_name', '').lower():
  1278→  1278→                    continue
  1279→  1279→
  1280→  1280→                lines.append("")
  1281→  1281→                lines.append(f"  [{i+1}] {rec.get('source_location', '<unknown>')}")
  1282→  1282→                lines.append(f"  Source:  \"{rec.get('source_text', '')}\"")
  1283→  1283→                path = rec.get('resolution_path', [])
  1284→  1284→                lines.append(f"  Path:    {' → '.join(path)}")
  1285→  1285→
  1286→  1286→                if rec.get('pattern_name'):
  1287→  1287→                    lines.append(f"  Pattern: {rec['pattern_name']}")
  1288→  1288→                if rec.get('template_name'):
  1289→  1289→                    lines.append(f"  Template: {rec['template_name']}")
  1290→  1290→                if rec.get('parameters'):
  1291→  1291→                    params = rec['parameters']
  1292→  1292→                    params_str = ", ".join(f"{k}={v}" for k, v in params.items() if not k.startswith("#"))
  1293→  1293→                    if params_str:
  1294→  1294→                        lines.append(f"  Params:  {params_str}")
  1295→  1295→
  1296→  1296→                lines.append(f"  Confidence: {rec.get('confidence', 1.0):.2f}")
  1297→  1297→
  1298→  1298→                # Show generated code (truncated)
  1299→  1299→                code = rec.get('generated_code', '')
  1300→  1300→                code_lines = code.strip().split('\n') if code.strip() else []
  1301→  1301→                if len(code_lines) <= 4:
  1302→  1302→                    for cl in code_lines:
