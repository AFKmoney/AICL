    69→def highlight_aicl(source: str) -> Text:
    70→    """Apply basic syntax highlighting to AICL source."""
    71→    text = Text()
    72→    for line in source.split("\n"):
    73→        stripped = line.lstrip()
    74→        if stripped.startswith("#"):
    75→            text.append(line, style="dim italic")
    76→        elif any(stripped.startswith(kw + ":") or stripped.startswith(kw + " ") for kw in AICL_KEYWORDS):
    77→            # Find the keyword
    78→            for kw in AICL_KEYWORDS:
    79→                if stripped.startswith(kw + ":") or stripped.startswith(kw + " "):
    80→                    indent = len(line) - len(stripped)
    81→                    text.append(" " * indent)
    82→                    text.append(kw, style="bold cyan")
    83→                    rest = stripped[len(kw):]
    84→                    # Color the colon
    85→                    if rest.startswith(":"):
    86→                        text.append(":", style="bold cyan")
    87→                        rest = rest[1:]
    88→                    text.append(rest, style="white")
    89→                    break
    90→            else:
    91→                text.append(line, style="white")
    92→        elif any(stripped.startswith(t + " ") or stripped == t for t in AICL_TYPES):
    93→            text.append(line, style="green")
    94→        else:
    95→            text.append(line, style="white")
    96→        text.append("\n")
    97→    return text

   384→    def _cmd_compile(self, args, output, editor):
   385→        if not self.current_file:
   386→            output.write("[red]No file loaded. Use :open <path> first.[/]")
   387→            return
   388→        target = "python"
   389→        if args.strip() in ("rust", "javascript", "js", "go"):
   390→            target = "javascript" if args.strip() == "js" else args.strip()
   391→
   392→        output.write(f"[bold]Compiling[/] {self.current_file} → {target}...")
   393→        try:
   394→            result = self.compiler.compile(self.current_file, target=target)
   395→            self.last_output_dir = result.get("output_dir", "")
   396→            coverage = result.get("audit_coverage", 0) * 100
   397→            todos = result.get("todos_remaining", 0)
   398→            valid = result.get("proof_valid", False)
   399→
   400→            status = "[green]VALID[/]" if valid else "[red]INVALID[/]"
   401→            output.write(f"""[green]Compilation successful![/]
   402→  Target:        {target}
   403→  Output:        {self.last_output_dir}
   404→  TODOs:         {todos}
   405→  Audit:         {coverage:.1f}%
   406→  Proof:         {status}""")
   407→        except Exception as e:
   408→            output.write(f"[red]Compilation failed:[/] {e}")

   410→    def _cmd_verify(self, args, output, editor):
   411→        if not self.current_file:
   412→            output.write("[red]No file loaded. Use :open <path> first.[/]")
   413→            return
   414→        output.write("[bold]Verifying specification...[/]")
   415→        try:
   416→            with open(self.current_file) as f:
   417→                source = f.read()
   418→            result = verify_source(source)
   419→            checks = result.get("checks", [])
   420→            passed = sum(1 for c in checks if c.get("status") == "pass")
   421→            failed = sum(1 for c in checks if c.get("status") == "fail")
   422→            warned = sum(1 for c in checks if c.get("status") == "warn")
   423→            overall = result.get("overall", "UNKNOWN")
   424→            color = "green" if overall == "PASS" else "yellow" if overall == "WARN" else "red"
   425→            output.write(f"[{color}]Overall: {overall}[/]  ({passed} passed, {warned} warnings, {failed} failed)")
   426→            for check in checks:
   427→                icon = {"pass": "[green]PASS[/]", "warn": "[yellow]WARN[/]", "fail": "[red]FAIL[/]"}
   428→                status = icon.get(check["status"], check["status"])
   429→                output.write(f"  {status}  {check.get('name', '')}")
   430→        except Exception as e:
   431→            output.write(f"[red]Verification failed:[/] {e}")

   647→    def _cmd_verify_proof(self, args, output, editor):
   648→        if not self.last_output_dir:
   649→            output.write("[red]Compile first with :compile[/]")
   650→            return
   651→        proof_path = os.path.join(self.last_output_dir, "main.aicl-proof")
   652→        if not os.path.exists(proof_path):
   653→            output.write("[red]Proof not found[/]")
   654→            return
   655→        # Use independent verifier
   656→        try:
   657→            import subprocess
   658→            verifier = Path(__file__).parent.parent.parent.parent / "tools" / "verify_proof.py"
   659→            if not verifier.exists():
   660→                verifier = Path("tools/verify_proof.py")
   661→            result = subprocess.run(
   662→                [sys.executable, str(verifier), proof_path, "--verbose"],
   663→                capture_output=True, text=True, timeout=30
   664→            )
   665→            output.write(result.stdout if result.stdout else result.stderr)
   666→        except Exception as e:
   667→            output.write(f"[red]Independent verification failed:[/] {e}")

   671→    def _load_file(self, path: str, output=None, editor=None):
   672→        if output is None:
   673→            output = self.query_one("#output-log", RichLog)
   674→        if editor is None:
   675→            editor = self.query_one("#editor-content", RichLog)
   676→        try:
   677→            with open(path) as f:
   678→                source = f.read()
   679→            editor.clear()
   680→            editor.write(highlight_aicl(source))
   681→            self.current_file = path
   682→            self.query_one("#tab-label", Static).update(f" {os.path.basename(path)} ")
   683→            output.write(f"[green]Loaded:[/] {path}")
   684→        except Exception as e:
   685→            output.write(f"[red]Cannot open {path}:[/] {e}")
