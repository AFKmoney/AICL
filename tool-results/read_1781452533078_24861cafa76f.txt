    52→// ============================================================
    53→// AICL Syntax Highlighter
    54→// ============================================================
    55→const AICL_KEYWORDS = [
    56→  'Goal', 'Layer', 'Sublayer', 'Validation', 'Risk', 'Recovery',
    57→  'Constraint', 'Entity', 'Behavior', 'Input', 'Output', 'Action',
    58→  'Condition', 'When', 'Then', 'Event', 'On', 'Parallel', 'Optimize',
    59→  'Priority', 'Learn', 'Adapt', 'Based', 'Security', 'Encrypt',
    60→  'Protect', 'Native', 'Import'
    61→];
    62→
    63→const AICL_TYPES = [
    64→  'string', 'integer', 'float', 'boolean', 'datetime',
    65→  'list', 'dict', 'set', 'any', 'void', 'bytes'
    66→];
    67→
    68→function highlightAICL(code: string): string {
    69→  const lines = code.split('\n');
    70→  return lines.map(line => {
    71→    // Comment
    72→    const commentIdx = line.indexOf('#');
    73→    if (commentIdx !== -1) {
    74→      const before = line.substring(0, commentIdx);
    75→      const comment = line.substring(commentIdx);
    76→      return highlightLine(before) + `<span class="aicl-comment">${escapeHtml(comment)}</span>`;
    77→    }
    78→    return highlightLine(line);
    79→  }).join('\n');
    80→}
    81→
    82→function highlightLine(line: string): string {
    83→  let result = escapeHtml(line);
    84→
    85→  // Keywords at start of line or after whitespace (word boundary)
    86→  for (const kw of AICL_KEYWORDS) {
    87→    const regex = new RegExp(`(^|\\s)(${kw})(?=[:\\s]|$)`, 'g');
    88→    result = result.replace(regex, `$1<span class="aicl-keyword">${kw}</span>`);
    89→  }
    90→
    91→  // Types
    92→  for (const t of AICL_TYPES) {
    93→    const regex = new RegExp(`(\\s|^)(${t})(?=\\s|:|$)`, 'g');
    94→    result = result.replace(regex, `$1<span class="aicl-type">${t}</span>`);
    95→  }
    96→
    97→  return result;
    98→}
    99→
   100→function escapeHtml(str: string): string {
   101→  return str
   102→    .replace(/&/g, '&amp;')
   103→    .replace(/</g, '&lt;')
   104→    .replace(/>/g, '&gt;')
   105→    .replace(/"/g, '&quot;');
   106→}

   659→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   660→    const ta = e.currentTarget;
   661→
   662→    // Tab → insert 4 spaces
   663→    if (e.key === 'Tab') {
   664→      e.preventDefault();
   665→      const start = ta.selectionStart;
   666→      const end = ta.selectionEnd;
   667→      const content = activeFile.content;
   668→      if (e.shiftKey) {
   669→        // Shift+Tab: remove 4 spaces
   670→        const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   671→        const line = content.substring(lineStart, start);
   672→        if (line.startsWith('    ')) {
   673→          const newContent = content.substring(0, lineStart) + content.substring(lineStart + 4);
   674→          updateFileContent(activeFileId, newContent);
   675→          setTimeout(() => {
   676→            ta.selectionStart = ta.selectionEnd = start - 4;
   677→          }, 0);
   678→        }
   679→      } else {
   680→        const newContent = content.substring(0, start) + '    ' + content.substring(end);
   681→        updateFileContent(activeFileId, newContent);
   682→        setTimeout(() => {
   683→          ta.selectionStart = ta.selectionEnd = start + 4;
   684→        }, 0);
   685→      }
   686→      return;
   687→    }
   688→
   689→    // Enter → auto-indent
   690→    if (e.key === 'Enter') {
   691→      e.preventDefault();
   692→      const start = ta.selectionStart;
   693→      const content = activeFile.content;
   694→      const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   695→      const currentLine = content.substring(lineStart, start);
   696→      const indent = currentLine.match(/^\s*/)?.[0] || '';
   697→
   698→      // If the line ends with a colon, add extra indent
   699→      const trimmedLine = currentLine.trimEnd();
   700→      const needsIndent = trimmedLine.endsWith(':');
   701→
   702→      const newIndent = needsIndent ? indent + '    ' : indent;
   703→      const newContent = content.substring(0, start) + '\n' + newIndent + content.substring(ta.selectionEnd);
   704→      updateFileContent(activeFileId, newContent);
   705→      setTimeout(() => {
   706→        ta.selectionStart = ta.selectionEnd = start + 1 + newIndent.length;
   707→      }, 0);
   708→      return;
   709→    }
   710→
   711→    // Ctrl+S → Save
   712→    if ((e.ctrlKey || e.metaKey) && e.key === 's') {
   713→      e.preventDefault();
   714→      saveFile();
   715→      return;
   716→    }
   717→
   718→    // Ctrl+F → Find
