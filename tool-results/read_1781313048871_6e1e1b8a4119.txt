    95→    95→class Compiler:
    96→    96→    """
    97→    97→    AICL Multi-Stage Compiler v0.5.
    98→    98→
    99→    99→    Compiles AICL source code into deterministic, executable Python code
   100→   100→    using behavior patterns and architecture templates.
   101→   101→
   102→   102→    Every generated artifact is tracked and must have provenance.
   103→   103→    The audit system verifies: Audit Coverage = 1.0 (no orphans).
   104→   104→
   105→   105→    Usage:
   106→   106→        compiler = Compiler()
   107→   107→        result = compiler.compile(source_code)
   108→   108→        if result.success:
   109→   109→            print(result.source_code)
   110→   110→    """
   111→   111→
   112→   112→    def __init__(self, target_language: str = "python"):
   113→   113→        self.target_language = target_language
   114→   114→        self.warnings: List[str] = []
   115→   115→        self.errors: List[str] = []
   116→   116→        self.stages_completed: List[str] = []
   117→   117→        self._used_names: Set[str] = set()
   118→   118→        self._behavior_compiler = BehaviorCompiler()
   119→   119→        self._todo_count = 0
   120→   120→        self._provenance = CompilationProvenance()

   122→   122→    def compile(self, source: str) -> CompilationResult:
   123→   123→        """Compile AICL source code into executable Python code."""
   124→   124→        self.warnings = []
   125→   125→        self.errors = []
   126→   126→        self.stages_completed = []
   127→   127→        self._used_names = set()
   128→   128→        self._todo_count = 0
   129→   129→        self._provenance = CompilationProvenance()
   130→   130→
   131→   131→        result = CompilationResult()
   132→   132→
   133→   133→        # Stage 1: Specification Parsing
   134→   134→        program = self._stage_parsing(source)
   135→   135→        if program is None:
   136→   136→            result.success = False
   137→   137→            result.errors = self.errors
   138→   138→            return result
   139→   139→        self.stages_completed.append("Specification Parsing")
   140→   140→
   141→   141→        # Stage 2: Architecture Validation
   142→   142→        validation_warnings = self._stage_architecture_validation(program)
   143→   143→        self.warnings.extend(validation_warnings)
   144→   144→        self.stages_completed.append("Architecture Validation")
   145→   145→
   146→   146→        # Stage 3: Dependency Analysis
   147→   147→        dep_order = self._stage_dependency_analysis(program)
   148→   148→        self.stages_completed.append("Dependency Analysis")
   149→   149→
   150→   150→        # Stage 4: Risk Analysis
   151→   151→        risk_pairs = self._stage_risk_analysis(program)
   152→   152→        self.stages_completed.append("Risk Analysis")
   153→   153→
   154→   154→        # Stage 5: Recovery Synthesis
   155→   155→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   156→   156→        self.stages_completed.append("Recovery Synthesis")
   157→   157→
   158→   158→        # Stage 6: Code Generation
   159→   159→        source_code = self._stage_code_generation(
   160→   160→            program, dep_order, risk_pairs, recovery_map
   161→   161→        )
   162→   162→        self.stages_completed.append("Code Generation")
   163→   163→
   164→   164→        # Stage 7: Test Generation
   165→   165→        test_code = self._stage_test_generation(program, dep_order)
   166→   166→        self.stages_completed.append("Test Generation")
   167→   167→
   168→   168→        # Stage 8: Optimization
   169→   169→        source_code = self._stage_optimization(source_code, program)
   170→   170→        self.stages_completed.append("Optimization")
   171→   171→
   172→   172→        # Stage 9: Final Construction
   173→   173→        source_code = self._stage_final_construction(source_code, program)
   174→   174→        self.stages_completed.append("Final Construction")
   175→   175→
   176→   176→        # Build Architecture Tree string
   177→   177→        tree = ArchitectureTree(program)
   178→   178→        result.architecture_tree_str = tree.to_string()
   179→   179→
   180→   180→        result.success = True
   181→   181→        result.source_code = source_code
   182→   182→        result.test_code = test_code
   183→   183→        result.warnings = self.warnings
   184→   184→        result.errors = self.errors
   185→   185→        result.stages_completed = self.stages_completed
   186→   186→        result.todo_count = self._todo_count
   187→   187→        result.fully_compiled = self._todo_count == 0
   188→   188→        result.provenance = self._provenance
   189→   189→
   190→   190→        # Generate Proof of Origin — the central artifact
   191→   191→        result.proof = self._provenance.to_proof(
   192→   192→            source_text=source,
   193→   193→            generated_source=source_code,
   194→   194→            generated_tests=test_code,
   195→   195→            compiler_version="0.7.0",
   196→   196→            target_language=self.target_language,
   197→   197→        )
   198→   198→
   199→   199→        return result
   200→   200→
   201→   201→    def compile_to_file(self, source: str, output_dir: str, source_path: str = "") -> CompilationResult:
   202→   202→        """
   203→   203→        Compile AICL source code and write the output to files.
   204→   204→
   205→   205→        Produces:
   206→   206→            - main.py              (executable program)
   207→   207→            - test_main.py         (generated tests)
   208→   208→            - architecture_tree.txt (architecture visualization)
   209→   209→            - main.aicl-proof      (Proof of Origin — first-class artifact)
   210→   210→
   211→   211→        The .aicl-proof file is a self-contained, verifiable record that
   212→   212→        contains ALL information needed to reconstruct explain() and audit()
   213→   213→        without the compiler. It is the central artifact of auditable compilation.
   214→   214→        """
   215→   215→        result = self.compile(source)
   216→   216→
   217→   217→        if result.success:
   218→   218→            os.makedirs(output_dir, exist_ok=True)
   219→   219→
   220→   220→            main_path = os.path.join(output_dir, "main.py")
   221→   221→            with open(main_path, 'w') as f:
   222→   222→                f.write(result.source_code)
   223→   223→
   224→   224→            test_path = os.path.join(output_dir, "test_main.py")
   225→   225→            with open(test_path, 'w') as f:
   226→   226→                f.write(result.test_code)
   227→   227→
   228→   228→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   229→   229→            with open(tree_path, 'w') as f:
   230→   230→                f.write(result.architecture_tree_str)
   231→   231→
   232→   232→            # Write Proof of Origin — the central artifact
   233→   233→            if result.proof:
   234→   234→                # Update source_path in proof if provided
   235→   235→                if source_path:
   236→   236→                    result.proof.source_path = source_path
   237→   237→                proof_path = os.path.join(output_dir, "main.aicl-proof")
   238→   238→                result.proof.to_file(proof_path)
   239→   239→
   240→   240→        return result

   260→   260→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   261→   261→        """Validate the architectural structure of the program."""
   262→   262→        warnings = program.validate()
   263→   263→
   264→   264→        layer_names = [l.name for l in program.layers]
   265→   265→        seen = set()
   266→   266→        for name in layer_names:
   267→   267→            if name.lower() in seen:
   268→   268→                warnings.append(f"Duplicate layer name: {name}")
   269→   269→            seen.add(name.lower())
   270→   270→
   271→   271→        entity_names = {e.name for e in program.entities}
   272→   272→        for behavior in program.behaviors:
   273→   273→            for inp in behavior.inputs:
   274→   274→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   275→   275→                    warnings.append(
   276→   276→                        f"Behavior '{behavior.name}' references unknown type "
   277→   277→                        f"'{inp.param_type}' in input"
   278→   278→                    )
   279→   279→
   280→   280→        return warnings

   294→   294→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   295→   295→        """Analyze risks and pair them with recovery procedures."""
   296→   296→        pairs = []
   297→   297→        for i, risk in enumerate(program.risks):
   298→   298→            recovery = None
   299→   299→            if i < len(program.recoveries):
   300→   300→                recovery = program.recoveries[i]
   301→   301→            pairs.append((risk, recovery))
   302→   302→
   303→   303→        if len(program.recoveries) > len(program.risks):
   304→   304→            for i in range(len(program.risks), len(program.recoveries)):
   305→   305→                generic_risk = RiskSection(
   306→   306→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   307→   307→                )
   308→   308→                pairs.append((generic_risk, program.recoveries[i]))
   309→   309→
   310→   310→        return pairs

   334→   334→    def _unique_name(self, base: str) -> str:
   335→   335→        """Generate a unique Python identifier from a base name."""
   336→   336→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   337→   337→        clean = re.sub(r'_+', '_', clean).strip('_')
   338→   338→        if not clean:
   339→   339→            clean = "item"
   340→   340→        if clean[0].isdigit():
   341→   341→            clean = "f_" + clean
   342→   342→
   343→   343→        name = clean
   344→   344→        counter = 2
   345→   345→        while name in self._used_names:
   346→   346→            name = f"{clean}_{counter}"
   347→   347→            counter += 1
   348→   348→
   349→   349→        self._used_names.add(name)
   350→   350→        return name

   476→   476→    def _generate_entity(self, entity: EntitySection) -> str:
   477→   477→        """Generate a Python dataclass for an AICL Entity."""
   478→   478→        class_name = self._make_class_name(entity.name)
   479→   479→        artifact_name = f"Entity_{class_name}"
   480→   480→        lines = []
   481→   481→        lines.append('@dataclass')
   482→   482→        lines.append(f'class {class_name}:')
   483→   483→        lines.append(f'    """Entity: {entity.name}"""')
   484→   484→
   485→   485→        if not entity.fields:

   748→   748→    def _generate_layer_init(

   827→   827→    def _generate_run_method(
   828→   828→        self,
   829→   829→        program: AICLProgram,
   830→   830→        dep_order: List[LayerSection],
   831→   831→        layer_attr_map: Dict[str, str],
   832→   832→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   833→   833→        recovery_map: Dict[str, str],
   834→   834→        template_name: str,
   835→   835→        behavior_method_names: Dict[str, str] = None,
   836→   836→    ) -> str:
   837→   837→        """Generate the main run method — uses architecture template."""
   838→   838→        behavior_method_names = behavior_method_names or {}
   839→   839→        lines = []
   840→   840→        lines.append('    def run(self) -> None:')
   841→   841→        lines.append('        """Run the application with full error handling."""')
   842→   842→        lines.append('        self._logger.info("Starting application...")')
   843→   843→        lines.append('        self._running = True')
   844→   844→        lines.append('')
   845→   845→
   846→   846→        if template_name == "game_loop":
   847→   847→            # Game loop: init → update → render → validate → loop
   848→   848→            lines.append('        self._game_active = True')
   849→   849→            lines.append('        try:')
   850→   850→            lines.append('            while self._running and self._game_active:')
   851→   851→            lines.append('                frame_start = time.time()')
   852→   852→            lines.append('')
   853→   853→            lines.append('                # Update phase')
   854→   854→            for behavior in program.behaviors:
   855→   855→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   856→   856→                input_params = ', '.join(
   857→   857→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   858→   858→                    {e.name.lower() for e in program.entities}
   859→   859→                    else inp.name
   860→   860→                    for inp in behavior.inputs
   861→   861→                )
   862→   862→                lines.append(f'                self._behavior_{method_name}({input_params})')
   863→   863→            lines.append('')
   864→   864→            lines.append('                # Check conditions')
   865→   865→            for i, condition in enumerate(program.conditions):
   866→   866→                lines.append(f'                self._handle_condition_{i+1}()')
   867→   867→            lines.append('')
   868→   868→            lines.append('                # Render phase')
   869→   869→            for layer in dep_order:
   870→   870→                attr_name = layer_attr_map[layer.name]
   871→   871→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   872→   872→                    lines.append(f'                if self._{attr_name} is not None:')
   873→   873→                    lines.append(f'                    self._render_frame()')
   874→   874→            lines.append('')
   875→   875→            lines.append('                # Frame rate control')
   876→   876→            lines.append('                elapsed = time.time() - frame_start')
   877→   877→            lines.append('                if elapsed < 1.0 / 60.0:')
   878→   878→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   879→   879→            lines.append('                self._frame_count += 1')
   880→   880→            lines.append('')
   881→   881→
   882→   882→        elif template_name == "event_driven":
   883→   883→            # Event-driven: connect → listen → handle → respond
   884→   884→            lines.append('        try:')
   885→   885→            lines.append('            self._connect()')
   886→   886→            lines.append('            self._connected = True')
   887→   887→            lines.append('            self._logger.info("Connected to server")')
   888→   888→            lines.append('')
   889→   889→            lines.append('            while self._running and self._connected:')
   890→   890→            lines.append('                # Process message queue')
   891→   891→            lines.append('                while self._message_queue:')
   892→   892→            lines.append('                    msg = self._message_queue.pop(0)')
   893→   893→            lines.append('                    self._process_message(msg)')
   894→   894→            lines.append('')
   895→   895→            lines.append('                # Process offline queue if reconnected')
   896→   896→            lines.append('                if self._offline_queue and self._connected:')
   897→   897→            lines.append('                    self._sync_offline_messages()')
   898→   898→            lines.append('')
   899→   899→            lines.append('                # Wait for new messages')
   900→   900→            lines.append('                time.sleep(0.1)')
   901→   901→
   902→   902→        else:
   903→   903→            # Default: linear execution
   904→   904→            lines.append('        try:')
   905→   905→            for layer in dep_order:
   906→   906→                attr_name = layer_attr_map[layer.name]
   907→   907→                lines.append(f'            # Execute {layer.name}')
   908→   908→                lines.append(f'            if self._{attr_name} is not None:')
   909→   909→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   910→   910→                lines.append(f'            else:')
   911→   911→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   912→   912→
   913→   913→        # Validations
   914→   914→        lines.append('')
   915→   915→        lines.append('            # Run validations')
   916→   916→        for i, validation in enumerate(program.validations):
