     1→"""
     2→AICL Compiler - Multi-Stage Compilation Pipeline
     3→
     4→The AICL compiler transforms architectural specifications into executable
     5→source code through a multi-stage reasoning pipeline. Unlike conventional
     6→compilers, the AICL compiler treats compilation as a reasoning process
     7→where architecture, risks, validations, and constraints all contribute
     8→to the generated code.
     9→
    10→Compilation Stages:
    11→  1. Specification Parsing (handled by Parser)
    12→  2. Architecture Validation
    13→  3. Dependency Analysis
    14→  4. Risk Analysis
    15→  5. Recovery Synthesis
    16→  6. Code Generation
    17→  7. Test Generation
    18→  8. Optimization
    19→  9. Final Output Construction
    20→"""
    21→
    22→import os
    23→import re
    24→import textwrap
    25→from typing import List, Dict, Optional, Tuple, Set
    26→from dataclasses import dataclass, field
    27→
    28→from .ast_nodes import (
    29→    AICLProgram, GoalSection, ConstraintSection, RiskSection,
    30→    RecoverySection, LayerSection, ValidationSection, EntitySection,
    31→    BehaviorSection, ConditionSection, EventSection, ParallelSection,
    32→    OptimizeSection, LearnSection, AdaptSection, SecuritySection,
    33→    NativeSection, EntityField,
    34→)
    35→from .architecture_tree import ArchitectureTree, ArchitectureNode
    36→from .parser import Parser, ParseError
    37→
    38→
    39→@dataclass
    40→class CompilationResult:
    41→    """Result of a compilation process."""
    42→    success: bool = True
    43→    source_code: str = ""
    44→    test_code: str = ""
    45→    warnings: List[str] = field(default_factory=list)
    46→    errors: List[str] = field(default_factory=list)
    47→    architecture_tree_str: str = ""
    48→    stages_completed: List[str] = field(default_factory=list)
    49→
    50→
    51→class CompilerError(Exception):
    52→    """Raised when compilation fails."""
    53→
    54→    def __init__(self, message: str, stage: str = ""):
    55→        self.stage = stage
    56→        super().__init__(f"Compiler error in stage '{stage}': {message}" if stage else message)
    57→
    58→
    59→class Compiler:
    60→    """
    61→    AICL Multi-Stage Compiler.
    62→
    63→    Compiles AICL source code into executable Python code.
    64→    The compilation pipeline includes architecture validation,
    65→    dependency analysis, risk analysis, recovery synthesis,
    66→    code generation, test generation, and optimization.
    67→
    68→    Usage:
    69→        compiler = Compiler()
    70→        result = compiler.compile(source_code)
    71→        if result.success:
    72→            print(result.source_code)
    73→    """
    74→
    75→    def __init__(self, target_language: str = "python"):
    76→        self.target_language = target_language
    77→        self.warnings: List[str] = []
    78→        self.errors: List[str] = []
    79→        self.stages_completed: List[str] = []
    80→        self._used_names: Set[str] = set()
    81→
    82→    def compile(self, source: str) -> CompilationResult:
    83→        """
    84→        Compile AICL source code into executable Python code.
    85→
    86→        Args:
    87→            source: AICL source code string.
    88→
    89→        Returns:
    90→            CompilationResult containing the generated code and metadata.
    91→        """
    92→        self.warnings = []
    93→        self.errors = []
    94→        self.stages_completed = []
    95→        self._used_names = set()
    96→
    97→        result = CompilationResult()
    98→
    99→        # Stage 1: Specification Parsing
   100→        program = self._stage_parsing(source)
   101→        if program is None:
   102→            result.success = False
   103→            result.errors = self.errors
   104→            return result
   105→        self.stages_completed.append("Specification Parsing")
   106→
   107→        # Stage 2: Architecture Validation
   108→        validation_warnings = self._stage_architecture_validation(program)
   109→        self.warnings.extend(validation_warnings)
   110→        self.stages_completed.append("Architecture Validation")
   111→
   112→        # Stage 3: Dependency Analysis
   113→        dep_order = self._stage_dependency_analysis(program)
   114→        self.stages_completed.append("Dependency Analysis")
   115→
   116→        # Stage 4: Risk Analysis
   117→        risk_pairs = self._stage_risk_analysis(program)
   118→        self.stages_completed.append("Risk Analysis")
   119→
   120→        # Stage 5: Recovery Synthesis
   121→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   122→        self.stages_completed.append("Recovery Synthesis")
   123→
   124→        # Stage 6: Code Generation
   125→        source_code = self._stage_code_generation(
   126→            program, dep_order, risk_pairs, recovery_map
   127→        )
   128→        self.stages_completed.append("Code Generation")
   129→
   130→        # Stage 7: Test Generation
   131→        test_code = self._stage_test_generation(program, dep_order)
   132→        self.stages_completed.append("Test Generation")
   133→
   134→        # Stage 8: Optimization
   135→        source_code = self._stage_optimization(source_code, program)
   136→        self.stages_completed.append("Optimization")
   137→
   138→        # Stage 9: Final Construction
   139→        source_code = self._stage_final_construction(source_code, program)
   140→        self.stages_completed.append("Final Construction")
   141→
   142→        # Build Architecture Tree string
   143→        tree = ArchitectureTree(program)
   144→        result.architecture_tree_str = tree.to_string()
   145→
   146→        result.success = True
   147→        result.source_code = source_code
   148→        result.test_code = test_code
   149→        result.warnings = self.warnings
   150→        result.errors = self.errors
   151→        result.stages_completed = self.stages_completed
   152→
   153→        return result
   154→
   155→    def compile_to_file(self, source: str, output_dir: str) -> CompilationResult:
   156→        """
   157→        Compile AICL source code and write the output to files.
   158→
   159→        Args:
   160→            source: AICL source code string.
   161→            output_dir: Directory to write output files.
   162→
   163→        Returns:
   164→            CompilationResult with file paths in the metadata.
   165→        """
   166→        result = self.compile(source)
   167→
   168→        if result.success:
   169→            os.makedirs(output_dir, exist_ok=True)
   170→
   171→            # Write main source file
   172→            main_path = os.path.join(output_dir, "main.py")
   173→            with open(main_path, 'w') as f:
   174→                f.write(result.source_code)
   175→
   176→            # Write test file
   177→            test_path = os.path.join(output_dir, "test_main.py")
   178→            with open(test_path, 'w') as f:
   179→                f.write(result.test_code)
   180→
   181→            # Write architecture tree
   182→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   183→            with open(tree_path, 'w') as f:
   184→                f.write(result.architecture_tree_str)
   185→
   186→        return result

   188→    # =========================================================================
   189→    # Stage 1: Specification Parsing
   190→    # =========================================================================
   191→
   192→    def _stage_parsing(self, source: str) -> Optional[AICLProgram]:
   193→        """Parse the AICL source code into an AST."""
   194→        try:
   195→            parser = Parser(source)
   196→            program = parser.parse()
   197→            return program
   198→        except (ParseError, Exception) as e:
   199→            self.errors.append(f"Parse error: {str(e)}")
   200→            return None

   202→    # =========================================================================
   203→    # Stage 2: Architecture Validation
   204→    # =========================================================================
   205→
   206→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   207→        """Validate the architectural structure of the program."""
   208→        warnings = program.validate()
   209→
   210→        # Check for duplicate layer names
   211→        layer_names = [l.name for l in program.layers]
   212→        seen = set()
   213→        for name in layer_names:
   214→            if name.lower() in seen:
   215→                warnings.append(f"Duplicate layer name: {name}")
   216→            seen.add(name.lower())
   217→
   218→        # Check entity field references
   219→        entity_names = {e.name for e in program.entities}
   220→        for behavior in program.behaviors:
   221→            for inp in behavior.inputs:
   222→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   223→                    warnings.append(
   224→                        f"Behavior '{behavior.name}' references unknown type "
   225→                        f"'{inp.param_type}' in input"
   226→                    )
   227→
   228→        return warnings

   238→    # =========================================================================
   239→    # Stage 4: Risk Analysis
   240→    # =========================================================================
   241→
   242→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   243→        """Analyze risks and pair them with recovery procedures."""
   244→        pairs = []
   245→        for i, risk in enumerate(program.risks):
   246→            recovery = None
   247→            if i < len(program.recoveries):
   248→                recovery = program.recoveries[i]
   249→            pairs.append((risk, recovery))
   250→
   251→        # Handle orphaned recoveries
   252→        if len(program.recoveries) > len(program.risks):
   253→            for i in range(len(program.risks), len(program.recoveries)):
   254→                generic_risk = RiskSection(
   255→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   256→                )
   257→                pairs.append((generic_risk, program.recoveries[i]))
   258→
   259→        return pairs
   260→
   261→    # =========================================================================
   262→    # Stage 5: Recovery Synthesis
   263→    # =========================================================================
   264→
   265→    def _stage_recovery_synthesis(
   266→        self,
   267→        program: AICLProgram,
   268→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]
   269→    ) -> Dict[str, str]:
   270→        """Synthesize recovery procedures for each risk."""
   271→        recovery_map = {}
   272→        for risk, recovery in risk_pairs:
   273→            if recovery:
   274→                recovery_map[risk.description] = recovery.description
   275→            else:
   276→                recovery_map[risk.description] = f"log and handle: {risk.description}"
   277→        return recovery_map

   283→    def _unique_name(self, base: str) -> str:
   284→        """Generate a unique Python identifier from a base name."""
   285→        # Clean the base name
   286→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   287→        clean = re.sub(r'_+', '_', clean).strip('_')
   288→        if not clean:
   289→            clean = "item"
   290→        if clean[0].isdigit():
   291→            clean = "f_" + clean
   292→
   293→        # Make unique
   294→        name = clean
   295→        counter = 2
   296→        while name in self._used_names:
   297→            name = f"{clean}_{counter}"
   298→            counter += 1
   299→
   300→        self._used_names.add(name)
   301→        return name

   303→    def _stage_code_generation(
   304→        self,
   305→        program: AICLProgram,
   306→        dep_order: List[LayerSection],
   307→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   308→        recovery_map: Dict[str, str]
   309→    ) -> str:
   310→        """Generate executable Python code from the AICL program."""
   311→        sections = []
   312→
   313→        # Module docstring
   314→        goal_desc = program.goals[0].description if program.goals else "AICL Generated Application"
   315→        sections.append('"""')
   316→        sections.append(f'AICL Generated Application')
   317→        sections.append(f'Goal: {goal_desc}')
   318→        sections.append(f'')
   319→        sections.append(f'This file was automatically generated by the AICL compiler.')
   320→        sections.append(f'Do not edit manually - regenerate from .aicl source.')
   321→        sections.append(f'"""')
   322→        sections.append('')
   323→
   324→        # Imports
   325→        sections.append(self._generate_imports(program))
   326→        sections.append('')
   327→
   328→        # Entity classes
   329→        for entity in program.entities:
   330→            sections.append(self._generate_entity(entity))
   331→            sections.append('')
   332→
   333→        # Main application class
   334→        class_name = self._make_class_name(goal_desc)
   335→        sections.append(self._generate_application_class(
   336→            program, class_name, dep_order, risk_pairs, recovery_map
   337→        ))
   338→        sections.append('')
   339→
   340→        # Entry point
   341→        sections.append(self._generate_entry_point(class_name))
   342→
   343→        return '\n'.join(sections)

   345→    def _generate_imports(self, program: AICLProgram) -> str:
   346→        """Generate import statements based on program content."""
   347→        imports = [
   348→            "import sys",
   349→            "import logging",
   350→            "import time",
   351→            "from typing import Optional, List, Dict, Any",
   352→        ]
   353→
   354→        # Check if we need threading for Parallel sections
   355→        if program.parallels:
   356→            imports.append("import threading")
   357→            imports.append("from concurrent.futures import ThreadPoolExecutor, as_completed")
   358→
   359→        # Check if we need datetime for Entity fields
   360→        needs_datetime = False
   361→        for entity in program.entities:
   362→            for f in entity.fields:
   363→                if f.field_type == "datetime":
   364→                    needs_datetime = True
   365→                    break
   366→
   367→        if needs_datetime:
   368→            imports.append("from datetime import datetime")
   369→
   370→        # Always need dataclass for entities
   371→        imports.append("from dataclasses import dataclass, field")
   372→
   373→        return '\n'.join(imports)

   375→    def _generate_entity(self, entity: EntitySection) -> str:
   376→        """Generate a Python dataclass for an AICL Entity."""
   377→        class_name = self._make_class_name(entity.name)
   378→        lines = []
   379→        lines.append('@dataclass')
   380→        lines.append(f'class {class_name}:')
   381→        lines.append(f'    """Entity: {entity.name}"""')
   382→
   383→        if not entity.fields:
   384→            lines.append('    pass')
   385→        else:
   386→            for f in entity.fields:
   387→                python_type = self._aicl_type_to_python(f.field_type)
   388→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   389→                if default:
   390→                    lines.append(f'    {f.name}: {python_type} = {default}')
   391→                else:
   392→                    lines.append(f'    {f.name}: {python_type} = 0')
   393→
   394→        return '\n'.join(lines)

   396→    def _generate_application_class(
   397→        self,
   398→        program: AICLProgram,
   399→        class_name: str,
   400→        dep_order: List[LayerSection],
   401→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   402→        recovery_map: Dict[str, str]
   403→    ) -> str:
   404→        """Generate the main application class."""
   405→        lines = []
   406→
   407→        lines.append(f'class {class_name}:')
   408→        lines.append(f'    """')
   409→
   410→        if program.goals:
   411→            lines.append(f'    Goal: {program.goals[0].description}')
   412→        lines.append(f'    """')
   413→        lines.append('')
   414→
   415→        # Class attributes - one per layer with unique names
   416→        lines.append('    # Layer instances')
   417→        layer_attr_map = {}  # Map from layer name to attribute name
   418→        for layer in dep_order:
   419→            attr_name = self._unique_name(layer.name)
   420→            layer_attr_map[layer.name] = attr_name
   421→            lines.append(f'    _{attr_name}: Any = None')
   422→        lines.append('')
   423→
   424→        # Risks as class constants
   425→        if program.risks:
   426→            lines.append('    # Known risks')
   427→            for i, risk in enumerate(program.risks):
   428→                const_name = self._unique_name(f"risk_{i+1}")
   429→                lines.append(f'    RISK_{i+1} = {risk.description!r}')
   430→            lines.append('')
   431→
   432→        # Constraints as class constants
   433→        if program.constraints:
   434→            lines.append('    # Constraints')
   435→            for i, con in enumerate(program.constraints):
   436→                lines.append(f'    CONSTRAINT_{i+1} = {con.description!r}')
   437→            lines.append('')
   438→
   439→        # __init__ method
   440→        lines.append('    def __init__(self):')
   441→        lines.append('        """Initialize the application and all layers."""')
   442→        lines.append('        self._logger = logging.getLogger(self.__class__.__name__)')
   443→        lines.append('        self._running = False')
   444→        lines.append('        self._initialized = False')
   445→        lines.append('')
   446→
   447→        # Initialize each layer with error handling
   448→        for layer in dep_order:
   449→            attr_name = layer_attr_map[layer.name]
   450→            lines.append(f'        # Initialize {layer.name}')
   451→            lines.append(f'        self._{attr_name} = self._init_{attr_name}()')
   452→            lines.append('')
   453→
   454→        lines.append('        self._initialized = True')
   455→        lines.append('        self._logger.info("Application initialized successfully")')
   456→        lines.append('')
   457→
   458→        # Layer initialization methods
   459→        for layer in dep_order:
   460→            attr_name = layer_attr_map[layer.name]
   461→            lines.append(self._generate_layer_init(layer, attr_name, risk_pairs, recovery_map))
   462→            lines.append('')
   463→
   464→        # Run method
   465→        lines.append(self._generate_run_method(program, dep_order, layer_attr_map, risk_pairs, recovery_map))
   466→        lines.append('')
   467→
   468→        # Validation methods
   469→        for i, validation in enumerate(program.validations):
   470→            lines.append(self._generate_validation_method(validation, i + 1))
   471→            lines.append('')
   472→
   473→        # Condition handlers
   474→        for i, condition in enumerate(program.conditions):
   475→            lines.append(self._generate_condition_handler(condition, i + 1))
   476→            lines.append('')
   477→
   478→        # Event handlers
   479→        for i, event in enumerate(program.events):
   480→            lines.append(self._generate_event_handler(event, i + 1))
   481→            lines.append('')
   482→
   483→        # Parallel execution
   484→        if program.parallels:
   485→            lines.append(self._generate_parallel_execution(program, layer_attr_map))
   486→            lines.append('')
   487→
   488→        # Security methods
   489→        if program.securities:
   490→            lines.append(self._generate_security_methods(program))
   491→            lines.append('')
   492→
   493→        # Risk handling method
   494→        lines.append(self._generate_risk_handler(risk_pairs, recovery_map))
   495→
   496→        return '\n'.join(lines)

   498→    def _generate_layer_init(
   499→        self,
   500→        layer: LayerSection,
   501→        attr_name: str,
   502→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   503→        recovery_map: Dict[str, str]
   504→    ) -> str:
   505→        """Generate initialization method for a layer."""
   506→        lines = []
   507→        lines.append(f'    def _init_{attr_name}(self) -> Any:')
   508→        lines.append(f'        """Initialize the {layer.name} layer."""')
   509→        lines.append(f'        self._logger.info("Initializing {layer.name}...")')
   510→        lines.append(f'        try:')
   511→        lines.append(f'            # TODO: Implement {layer.name} initialization')
   512→        lines.append(f'            component = None  # Placeholder')
   513→
   514→        # Generate sublayer initialization
   515→        for sublayer in layer.sublayers:
   516→            sub_attr = self._unique_name(sublayer.name)
   517→            lines.append(f'            # Sublayer: {sublayer.name}')
   518→            lines.append(f'            _{sub_attr} = None  # Placeholder for {sublayer.name}')
   519→
   520→        lines.append(f'            self._logger.info("{layer.name} initialized successfully")')
   521→        lines.append(f'            return component')
   522→        lines.append(f'        except Exception as e:')
   523→        lines.append(f'            self._logger.error(f"Failed to initialize {layer.name}: {{e}}")')
   524→
   525→        # Add recovery from risk pairs that match this layer
   526→        for risk, recovery in risk_pairs:
   527→            if recovery and self._risk_matches_layer(risk.description, layer.name):
   528→                lines.append(f'            # Recovery: {recovery.description}')
   529→                lines.append(f'            self._logger.info("Applying recovery: {recovery.description}")')
   530→
   531→        lines.append(f'            return None')
   532→        return '\n'.join(lines)

   534→    def _generate_run_method(
   535→        self,
   536→        program: AICLProgram,
   537→        dep_order: List[LayerSection],
   538→        layer_attr_map: Dict[str, str],
   539→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   540→        recovery_map: Dict[str, str]
   541→    ) -> str:
   542→        """Generate the main run method."""
   543→        lines = []
   544→        lines.append('    def run(self) -> None:')
   545→        lines.append('        """Run the application with full error handling."""')
   546→        lines.append('        self._logger.info("Starting application...")')
   547→        lines.append('        self._running = True')
   548→        lines.append('')
   549→        lines.append('        try:')
   550→
   551→        # Execute layers in order
   552→        for layer in dep_order:
   553→            attr_name = layer_attr_map[layer.name]
   554→            lines.append(f'            # Execute {layer.name}')
   555→            lines.append(f'            if self._{attr_name} is not None:')
   556→            lines.append(f'                self._logger.info("Running {layer.name}...")')
   557→            lines.append(f'            else:')
   558→            lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   559→
   560→        lines.append('')
   561→        lines.append('            # Run validations')
   562→        for i, validation in enumerate(program.validations):
   563→            lines.append(f'            self._validate_{i+1}()')
   564→
   565→        lines.append('')
   566→        lines.append('            self._logger.info("Application completed successfully")')
   567→        lines.append('')
   568→        lines.append('        except Exception as e:')
   569→        lines.append('            self._logger.error(f"Application error: {e}")')
   570→        lines.append('            self._handle_risk(e)')
   571→        lines.append('        finally:')
   572→        lines.append('            self._running = False')
   573→        lines.append('            self._logger.info("Application stopped")')
   574→
   575→        return '\n'.join(lines)

   577→    def _generate_validation_method(self, validation: ValidationSection, index: int) -> str:
   578→        """Generate a validation method."""
   579→        lines = []
   580→        lines.append(f'    def _validate_{index}(self) -> bool:')
   581→        lines.append(f'        """Validation: {validation.description}"""')
   582→        lines.append(f'        self._logger.info("Validating: {validation.description}")')
   583→        lines.append(f'        try:')
   584→        lines.append(f'            # TODO: Implement validation for: {validation.description}')
   585→        lines.append(f'            result = True  # Placeholder - always passes until implemented')

   847→    def _make_class_name(self, name: str) -> str:
   848→        """Convert a description to a valid Python class name."""
   849→        words = name.replace('-', ' ').replace('_', ' ').split()
   850→        if not words:
   851→            return "AiclApplication"
   852→
   853→        # Filter out common articles and prepositions
   854→        filtered = [w for w in words if w.lower() not in ('a', 'an', 'the', 'on', 'in', 'of')]
   855→        if not filtered:
   856→            filtered = words
   857→
   858→        return ''.join(w.capitalize() for w in filtered)
