    51→// ============================================================
    52→// AICL Syntax Highlighter
    53→// ============================================================
    54→const AICL_KEYWORDS = [
    55→  'Goal', 'Layer', 'Sublayer', 'Validation', 'Risk', 'Recovery',
    56→  'Constraint', 'Entity', 'Behavior', 'Input', 'Output', 'Action',
    57→  'Condition', 'When', 'Then', 'Event', 'On', 'Parallel', 'Optimize',
    58→  'Priority', 'Learn', 'Adapt', 'Based', 'Security', 'Encrypt',
    59→  'Protect', 'Native', 'Import'
    60→];
    61→
    62→const AICL_TYPES = [
    63→  'string', 'integer', 'float', 'boolean', 'datetime',
    64→  'list', 'dict', 'set', 'any', 'void', 'bytes'
    65→];
    66→
    67→function highlightAICL(code: string): string {
    68→  const lines = code.split('\n');
    69→  return lines.map(line => {
    70→    // Comment
    71→    const commentIdx = line.indexOf('#');
    72→    if (commentIdx !== -1) {
    73→      const before = line.substring(0, commentIdx);
    74→      const comment = line.substring(commentIdx);
    75→      return highlightLine(before) + `<span class="aicl-comment">${escapeHtml(comment)}</span>`;
    76→    }
    77→    return highlightLine(line);
    78→  }).join('\n');
    79→}
    80→
    81→function highlightLine(line: string): string {
    82→  let result = escapeHtml(line);
    83→
    84→  // Keywords at start of line or after whitespace (word boundary)
    85→  for (const kw of AICL_KEYWORDS) {
    86→    const regex = new RegExp(`(^|\\s)(${kw})(?=[:\\s]|$)`, 'g');
    87→    result = result.replace(regex, `$1<span class="aicl-keyword">${kw}</span>`);
    88→  }
    89→
    90→  // Types
    91→  for (const t of AICL_TYPES) {
    92→    const regex = new RegExp(`(\\s|^)(${t})(?=\\s|:|$)`, 'g');
    93→    result = result.replace(regex, `$1<span class="aicl-type">${t}</span>`);
    94→  }
    95→
    96→  return result;
    97→}
    98→
    99→function escapeHtml(str: string): string {
   100→  return str
   101→    .replace(/&/g, '&amp;')
   102→    .replace(/</g, '&lt;')
   103→    .replace(/>/g, '&gt;')
   104→    .replace(/"/g, '&quot;');
   105→}

   636→  // --- Editor key handling ---
   637→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   638→    const ta = e.currentTarget;
   639→
   640→    // Tab → insert 4 spaces
   641→    if (e.key === 'Tab') {
   642→      e.preventDefault();
   643→      const start = ta.selectionStart;
   644→      const end = ta.selectionEnd;
   645→      const content = activeFile.content;
   646→      if (e.shiftKey) {
   647→        // Shift+Tab: remove 4 spaces
   648→        const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   649→        const line = content.substring(lineStart, start);
   650→        if (line.startsWith('    ')) {
   651→          const newContent = content.substring(0, lineStart) + content.substring(lineStart + 4);
   652→          updateFileContent(activeFileId, newContent);
   653→          setTimeout(() => {
   654→            ta.selectionStart = ta.selectionEnd = start - 4;
   655→          }, 0);
   656→        }
   657→      } else {
   658→        const newContent = content.substring(0, start) + '    ' + content.substring(end);
   659→        updateFileContent(activeFileId, newContent);
   660→        setTimeout(() => {
   661→          ta.selectionStart = ta.selectionEnd = start + 4;
   662→        }, 0);
   663→      }
   664→      return;
   665→    }
   666→
   667→    // Enter → auto-indent
   668→    if (e.key === 'Enter') {
   669→      e.preventDefault();
   670→      const start = ta.selectionStart;
   671→      const content = activeFile.content;
   672→      const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   673→      const currentLine = content.substring(lineStart, start);
   674→      const indent = currentLine.match(/^\s*/)?.[0] || '';
   675→
   676→      // If the line ends with a colon, add extra indent
   677→      const trimmedLine = currentLine.trimEnd();
   678→      const needsIndent = trimmedLine.endsWith(':');
   679→
   680→      const newIndent = needsIndent ? indent + '    ' : indent;
   681→      const newContent = content.substring(0, start) + '\n' + newIndent + content.substring(ta.selectionEnd);
   682→      updateFileContent(activeFileId, newContent);
   683→      setTimeout(() => {
   684→        ta.selectionStart = ta.selectionEnd = start + 1 + newIndent.length;
   685→      }, 0);
   686→      return;
   687→    }
   688→
   689→    // Ctrl+S → Save
   690→    if ((e.ctrlKey || e.metaKey) && e.key === 's') {
   691→      e.preventDefault();
   692→      saveFile();
   693→      return;
   694→    }
   695→
   696→    // Ctrl+F → Find
   697→    if ((e.ctrlKey || e.metaKey) && e.key === 'f') {
   698→      e.preventDefault();
   699→      setFindOpen(true);
   700→      setTimeout(() => findInputRef.current?.focus(), 50);
   701→      return;
   702→    }
   703→
   704→    // Ctrl+H → Replace
   705→    if ((e.ctrlKey || e.metaKey) && e.key === 'h') {
   706→      e.preventDefault();
   707→      setFindOpen(true);
   708→      return;
   709→    }
   710→
   711→    // Escape → Close find
   712→    if (e.key === 'Escape') {
   713→      setFindOpen(false);
   714→    }
   715→  }, [activeFile, activeFileId, updateFileContent, saveFile]);
