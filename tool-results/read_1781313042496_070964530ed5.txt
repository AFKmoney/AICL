   150→def make_table(headers, rows, col_widths=None):
   151→    """Create a styled table."""
   152→    data = [headers] + rows
   153→    if col_widths is None:
   154→        n = len(headers)
   155→        col_widths = [CONTENT_W / n] * n
   156→    t = Table(data, colWidths=col_widths, repeatRows=1)
   157→    style_cmds = [
   158→        ('BACKGROUND', (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   159→        ('TEXTCOLOR', (0, 0), (-1, 0), TABLE_HEADER_TEXT),
   160→        ('FONTNAME', (0, 0), (-1, 0), 'CarlitoBold'),
   161→        ('FONTSIZE', (0, 0), (-1, 0), 9.5),
   162→        ('FONTNAME', (0, 1), (-1, -1), 'Carlito'),
   163→        ('FONTSIZE', (0, 1), (-1, -1), 9),
   164→        ('LEADING', (0, 0), (-1, -1), 13),
   165→        ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
   166→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   167→        ('TOPPADDING', (0, 0), (-1, -1), 4),
   168→        ('BOTTOMPADDING', (0, 0), (-1, -1), 4),
   169→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   170→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   171→        ('GRID', (0, 0), (-1, -1), 0.4, colors.HexColor('#cccccc')),
   172→    ]
   173→    for i in range(1, len(data)):
   174→        bg = TABLE_ROW_EVEN if i % 2 == 1 else TABLE_ROW_ODD
   175→        style_cmds.append(('BACKGROUND', (0, i), (-1, i), bg))
   176→    t.setStyle(TableStyle(style_cmds))
   177→    return t

   860→    story.append(p(
   861→        "This hash binding creates a chain of trust: the source hash ties the proof to the "
   862→        "original specification, the program hash ties the proof to the generated code, and "
   863→        "the test hash ties the proof to the test suite. Any break in this chain invalidates "
   864→        "the entire proof."
   865→    ))

   872→        "AICL's auditable compilation system establishes three formal properties that are "
   873→        "verified in every compilation. These are not aspirational goals but checkable "

   878→    story.append(Paragraph("9.1 No Orphan Artifact Property", h2_style))
   879→    story.append(quote(
   880→        "For any compilation C, if artifact A is generated, then A has at least one "
   881→        "provenance chain linking it to its originating specification."
   882→    ))
   883→    story.append(p(
   884→        "This is the foundational property of auditable compilation. An orphan artifact is "
   885→        "a generated piece of code (method, class, function, or test) that exists in the "
   886→        "output but has no provenance record explaining why it was generated. The No Orphan "
   887→        "Property states that such artifacts must not exist. In the current implementation, "

   889→        "a total of 204 generated artifacts."

   894→        "AI code generator, or a model-driven architecture tool — be able to demonstrate "
   895→        "that every line of code it produces has a traceable origin? The No Orphan Property "
   896→        "provides a formal framework for answering this question."
   897→    ))
   898→
   899→    story.append(Paragraph("9.2 Complete Coverage Property", h2_style))
   900→    story.append(quote(
   901→        "Coverage = |{A in Artifacts : A has provenance chain}| / |Artifacts| = 1.0"
   902→    ))
   903→    story.append(p(
   904→        "The Complete Coverage Property is the quantitative expression of the No Orphan "
   905→        "Property. It states that the fraction of artifacts with provenance must equal "
   906→        "exactly 1.0. Anything less means at least one artifact lacks justification. "
   907→        "The independent verifier computes this ratio independently from the proof file's "
   908→        "artifact list and provenance records, then compares it to the declared value. "
   909→        "Both must equal 1.0 for the proof to be valid."
   910→    ))
   911→
   912→    story.append(Paragraph("9.3 Hash Binding Property", h2_style))
   913→    story.append(quote(
   914→        "SHA-256(generated_code) = program_hash stored in Proof of Origin"
   915→    ))
   916→    story.append(p(
   917→        "The Hash Binding Property ensures that the proof is cryptographically bound to "
   918→        "the generated code. Without this property, a proof file could be detached from "
   919→        "its code artifact and attached to different code, defeating the purpose of "
   920→        "provenance tracking. The SHA-256 hash creates a tamper-evident link: any "
   921→        "modification to the generated code, the source text, or the proof itself will "
   922→        "cause a hash mismatch, immediately invalidating the proof."
   923→    ))
