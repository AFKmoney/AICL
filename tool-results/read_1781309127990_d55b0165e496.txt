     1→"""
     2→AICL Compilation Provenance & Audit System
     3→
     4→Every line of generated code must have a traceable provenance chain:
     5→    AICL Source → Parse → Pattern Match → Template → Generated Code
     6→
     7→This module records and reports WHY each line of code was generated,
     8→and provides an audit system to verify that every generated artifact
     9→is traceable to its originating specification.
    10→
    11→Core Concepts:
    12→    - Provenance Record: Explains why a specific piece of code was generated
    13→    - Generated Artifact: A named unit of generated code (class, method, function, test)
    14→    - Audit Coverage: Ratio of auditable artifacts to total generated artifacts
    15→    - Orphan Artifact: A generated artifact without provenance
    16→
    17→Formal Property (The Auditability Theorem):
    18→    A program is auditable if every generated artifact
    19→    can be traced to its originating specification
    20→    through a complete provenance chain.
    21→
    22→    Audit Coverage = Auditable Artifacts / Generated Artifacts
    23→    Target: Audit Coverage = 1.0
    24→
    25→Usage:
    26→    aicl explain pong.aicl              # Full compilation trace
    27→    aicl explain pong.aicl --behavior MovePaddle  # Specific behavior trace
    28→    aicl audit pong.aicl                # Audit report with coverage
    29→    aicl audit pong.aicl --strict       # Fail if coverage < 100%
    30→"""

    76→@dataclass
    77→class ProvenanceRecord:
    78→    """
    79→    A single provenance record explaining why a piece of code was generated.
    80→
    81→    This is the audit trail of the compiler. Every generated method,
    82→    every line of behavior code, every recovery handler must have one.
    83→    """
    84→    source_type: ProvenanceType
    85→    source_location: str          # Where in the AICL source (e.g., "Behavior MovePaddle")
    86→    source_text: str              # The original AICL text
    87→    resolution_path: List[str]    # Chain: ["AICL Source", "Parser", "Pattern Match", "MOVE", "Template"]
    88→    generated_code: str           # The code that was generated
    89→    confidence: float = 1.0       # Confidence of the match (1.0 = deterministic)
    90→    pattern_name: str = ""        # Pattern that matched (if applicable)
    91→    template_name: str = ""       # Architecture template (if applicable)
    92→    parameters: Dict[str, str] = field(default_factory=dict)  # Parameters used
    93→    artifact_names: List[str] = field(default_factory=list)    # Which artifacts this record covers

    95→
    96→@dataclass
    97→class GeneratedArtifact:
    98→    """
    99→    A named unit of generated code.
   100→
   101→    Every method, class, function, and dataclass produced by the compiler
   102→    is registered as an artifact. The audit system verifies that each
   103→    artifact has at least one provenance record explaining its existence.
   104→
   105→    An artifact without provenance is an "orphan" — it exists in the
   106→    generated code but the compiler cannot explain why.
   107→    """
   108→    name: str                           # Human-readable name (e.g., "_behavior_move_paddle")
   109→    artifact_type: ArtifactType         # Type of artifact
   110→    source: str                         # Which AICL section triggered generation
   111→    provenance_indices: List[int] = field(default_factory=list)  # Indices into records list
   112→    code_snippet: str = ""              # Brief snippet of the generated code (first line)
   113→
   114→    @property
   115→    def has_provenance(self) -> bool:
   116→        """Whether this artifact has at least one provenance record."""
   117→        return len(self.provenance_indices) > 0
   118→
   119→    @property
   120→    def is_orphan(self) -> bool:
   121→        """Whether this artifact is an orphan (no provenance)."""
   122→        return not self.has_provenance

   125→class CompilationProvenance:
   126→    """
   127→    Tracks the complete provenance of a compilation and provides
   128→    audit capabilities.
   129→
   130→    Records every decision the compiler makes, creating an audit trail
   131→    that can be queried to understand WHY any line of code exists.
   132→
   133→    This is the antidote to compiler complexity hiding. As AICL grows
   134→    (30 patterns → 100 patterns → 500 patterns), the provenance tracker
   135→    ensures every decision remains explainable.
   136→
   137→    v0.5: Added artifact tracking and audit coverage computation.
   138→    """
   139→
   140→    def __init__(self):
   141→        self.records: List[ProvenanceRecord] = []
   142→        self.artifacts: List[GeneratedArtifact] = []
   143→
   144→    def record(
   145→        self,
   146→        source_type: ProvenanceType,
   147→        source_location: str,
   148→        source_text: str,
   149→        resolution_path: List[str],
   150→        generated_code: str,
   151→        confidence: float = 1.0,
   152→        pattern_name: str = "",
   153→        template_name: str = "",
   154→        parameters: Dict[str, str] = None,
   155→        artifact_names: List[str] = None,
   156→    ) -> ProvenanceRecord:
   157→        """Record a compilation decision."""
   158→        rec = ProvenanceRecord(
   159→            source_type=source_type,
   160→            source_location=source_location,
   161→            source_text=source_text,
   162→            resolution_path=resolution_path,
   163→            generated_code=generated_code,
   164→            confidence=confidence,
   165→            pattern_name=pattern_name,
   166→            template_name=template_name,
   167→            parameters=parameters or {},
   168→            artifact_names=artifact_names or [],
   169→        )
   170→        self.records.append(rec)
   171→
   172→        # Link this provenance record to any matching artifacts
   173→        record_index = len(self.records) - 1
   174→        for artifact_name in (artifact_names or []):
   175→            for artifact in self.artifacts:
   176→                if artifact.name == artifact_name:
   177→                    artifact.provenance_indices.append(record_index)
   178→
   179→        return rec
   180→
   181→    def register_artifact(
   182→        self,
   183→        name: str,
   184→        artifact_type: ArtifactType,
   185→        source: str,
   186→        code_snippet: str = "",
   187→    ) -> GeneratedArtifact:
   188→        """
   189→        Register a generated artifact.
   190→
   191→        Every method, class, function, and dataclass produced by the
   192→        compiler should be registered. The audit system will verify
   193→        that each artifact has at least one provenance record.
   194→        """
   195→        artifact = GeneratedArtifact(
   196→            name=name,
   197→            artifact_type=artifact_type,
   198→            source=source,
   199→            code_snippet=code_snippet,
   200→        )
   201→        self.artifacts.append(artifact)
   202→
   203→        # Check if any existing provenance records already cover this artifact
   204→        for i, rec in enumerate(self.records):
   205→            if name in rec.artifact_names:
   206→                artifact.provenance_indices.append(i)
   207→
   208→        return artifact
   209→
   210→    def link_provenance_to_artifact(self, record_index: int, artifact_name: str):
   211→        """Explicitly link a provenance record to an artifact."""
   212→        if record_index < len(self.records):
   213→            if artifact_name not in self.records[record_index].artifact_names:
   214→                self.records[record_index].artifact_names.append(artifact_name)
   215→
   216→        for artifact in self.artifacts:
   217→            if artifact.name == artifact_name:
   218→                if record_index not in artifact.provenance_indices:
   219→                    artifact.provenance_indices.append(record_index)

   225→    def explain(self, target: str = None) -> str:
   226→        """
   227→        Generate an explanation of the compilation.
   228→
   229→        Args:
   230→            target: Optional filter (behavior name, pattern name, etc.)
   231→
   232→        Returns:
   233→            Human-readable explanation of all compilation decisions.
   234→        """
   235→        lines = []
   236→        lines.append("=" * 70)
   237→        lines.append("AICL COMPILATION PROVENANCE TRACE")
   238→        lines.append("=" * 70)
   239→        lines.append("")
   240→
   241→        # Summary
   242→        total = len(self.records)
   243→        pattern_count = sum(1 for r in self.records if r.source_type == ProvenanceType.PATTERN_MATCH)
   244→        sublang_count = sum(1 for r in self.records if r.source_type == ProvenanceType.SUB_LANGUAGE)
   245→        fallback_count = sum(1 for r in self.records if r.source_type == ProvenanceType.FALLBACK)
   246→        template_count = sum(1 for r in self.records if r.source_type == ProvenanceType.ARCHITECTURE_TEMPLATE)
   247→        direct_count = sum(1 for r in self.records if r.source_type == ProvenanceType.DIRECT_MAPPING)
   248→
   249→        lines.append(f"Total decisions: {total}")
   250→        lines.append(f"  Pattern matches:    {pattern_count}")
   251→        lines.append(f"  Sub-language:       {sublang_count}")
   252→        lines.append(f"  Direct mappings:    {direct_count}")
   253→        lines.append(f"  Architecture tmpl:  {template_count}")
   254→        lines.append(f"  Fallbacks:          {fallback_count}")
   255→        lines.append("")
   256→
   257→        # Confidence score
   258→        if total > 0:
   259→            avg_confidence = sum(r.confidence for r in self.records) / total
   260→            fully_deterministic = sum(1 for r in self.records if r.confidence >= 0.9)
   261→            lines.append(f"Average confidence:    {avg_confidence:.2f}")
   262→            lines.append(f"Fully deterministic:   {fully_deterministic}/{total} ({100*fully_deterministic//max(total,1)}%)")
   263→            lines.append("")
   264→
   265→        # Group by source type
   266→        by_type: Dict[ProvenanceType, List[ProvenanceRecord]] = {}
   267→        for rec in self.records:
   268→            if rec.source_type not in by_type:
   269→                by_type[rec.source_type] = []
   270→            by_type[rec.source_type].append(rec)
   271→
   272→        for source_type, records in by_type.items():
   273→            type_label = source_type.value.replace("_", " ").title()
   274→            lines.append("-" * 70)
   275→            lines.append(f" {type_label} ({len(records)} decisions)")
   276→            lines.append("-" * 70)
   277→
   278→            for i, rec in enumerate(records):
   279→                # Filter by target if specified
   280→                if target and target.lower() not in rec.source_location.lower() and target.lower() not in rec.pattern_name.lower():
   281→                    continue
   282→
   283→                lines.append(f"")
   284→                lines.append(f"  [{i+1}] {rec.source_location}")
   285→                lines.append(f"  Source:  \"{rec.source_text}\"")
   286→                lines.append(f"  Path:    {' → '.join(rec.resolution_path)}")
   287→
   288→                if rec.pattern_name:
   289→                    lines.append(f"  Pattern: {rec.pattern_name}")
   290→                if rec.template_name:
   291→                    lines.append(f"  Template: {rec.template_name}")
   292→                if rec.parameters:
   293→                    params_str = ", ".join(f"{k}={v}" for k, v in rec.parameters.items() if not k.startswith("#"))
   294→                    if params_str:
   295→                        lines.append(f"  Params:  {params_str}")
   296→
   297→                lines.append(f"  Confidence: {rec.confidence:.2f}")
   298→
   299→                # Show generated code (truncated)
   300→                code_lines = rec.generated_code.strip().split('\n')
   301→                if len(code_lines) <= 4:
   302→                    for cl in code_lines:
   303→                        lines.append(f"  Code:    {cl}")
   304→                else:
   305→                    for cl in code_lines[:3]:
   306→                        lines.append(f"  Code:    {cl}")
   307→                    lines.append(f"  Code:    ... ({len(code_lines) - 3} more lines)")
   308→
   309→        lines.append("")
   310→        lines.append("=" * 70)
   311→        lines.append("END OF PROVENANCE TRACE")
   312→        lines.append("=" * 70)
   313→
   314→        return '\n'.join(lines)

   329→        if not matching:

   332→        for rec in matching:
   333→            lines.append(f"Source type: {rec.source_type.value}")
   334→            lines.append(f"Source text: \"{rec.source_text}\"")
   335→            lines.append(f"")
   336→            lines.append(f"Resolution chain:")
   337→            for step in rec.resolution_path:
   338→                lines.append(f"  ↓ {step}")
   339→            lines.append(f"")
   340→            lines.append(f"Generated code:")
   341→            for cl in rec.generated_code.strip().split('\n'):
   342→                lines.append(f"  {cl}")
   343→            lines.append(f"")
   344→            lines.append(f"Confidence: {rec.confidence:.2f}")
   345→            if rec.pattern_name:
   346→                lines.append(f"Pattern: {rec.pattern_name}")
   347→            if rec.parameters:
   348→                lines.append(f"Parameters used:")
   349→                for k, v in rec.parameters.items():
   350→                    if not k.startswith("#"):
   351→                        lines.append(f"  {k} = {v}")
   352→
   353→        return '\n'.join(lines)

   373→    def compute_explicability_coverage(self, generated_source: str, generated_tests: str = "") -> Dict:
   374→        """
   375→        Compute the explicability coverage ratio.
   376→
   377→        The explicability coverage measures what fraction of generated code
   378→        lines are accounted for by provenance records. The target is 1.0
   379→        (every line has provenance). Any deviation is a gap to investigate.
   380→
   381→        Returns:
   382→            Dict with keys:
   383→                - total_lines: total non-blank, non-comment lines in generated code
   384→                - accounted_lines: lines that appear in at least one provenance record
   385→                - coverage_ratio: accounted_lines / total_lines (target: 1.0)
   386→                - unaccounted_lines: list of line numbers without provenance
   387→                - by_type_coverage: coverage broken down by provenance type
   388→        """
   389→        # Combine source and test code
   390→        all_code = generated_source
   391→        if generated_tests:
   392→            all_code += "\n" + generated_tests
   393→
   394→        # Count meaningful lines (non-blank, non-import, non-pure-comment)
   395→        code_lines = []
   396→        for i, line in enumerate(all_code.split('\n')):
   397→            stripped = line.strip()
   398→            if not stripped:
   399→                continue
   400→            # Skip pure comment lines and import lines for coverage
   401→            if stripped.startswith('#') and not any(kw in stripped for kw in ['When:', 'On:', 'Risk:', 'Recovery:', 'Action:', 'Intent:']):
   402→                continue
   403→            if stripped.startswith(('import ', 'from ')):
   404→                continue
   405→            if stripped in ('"""', "'''", ''):
   406→                continue
   407→            code_lines.append((i + 1, stripped))
   408→
   409→        total_lines = len(code_lines)
   410→        if total_lines == 0:
   411→            return {
   412→                "total_lines": 0,
   413→                "accounted_lines": 0,
   414→                "coverage_ratio": 1.0,
   415→                "unaccounted_lines": [],
   416→                "by_type_coverage": {},
   417→            }
   418→
   419→        # Build set of code fragments from provenance records
   420→        accounted_fragments = set()
   421→        for rec in self.records:
   422→            for code_line in rec.generated_code.strip().split('\n'):
   423→                stripped = code_line.strip()
   424→                if stripped:
   425→                    # Normalize for matching (remove variable values that change)
   426→                    normalized = re.sub(r'[\'"].*?[\'\"]', "'...'", stripped)
   427→                    normalized = re.sub(r'\{[^}]*\}', '{...}', normalized)
   428→                    accounted_fragments.add(normalized)
   429→                    # Also add the original
   430→                    accounted_fragments.add(stripped)
   431→
   432→        # Check which lines are accounted for
   433→        accounted_lines = 0
   434→        unaccounted = []
   435→        for line_num, line_text in code_lines:
   436→            stripped = line_text.strip()
   437→            # Check if this line or a normalized version appears in provenance
   438→            found = False
   439→            if stripped in accounted_fragments:
   440→                found = True
   441→            else:
   442→                # Try normalizing the source line too
   443→                normalized = re.sub(r'[\'"].*?[\'\"]', "'...'", stripped)
   444→                normalized = re.sub(r'\{[^}]*\}', '{...}', normalized)
   445→                if normalized in accounted_fragments:
   446→                    found = True
   447→                else:
   448→                    # Check if any provenance fragment is a prefix of this line
   449→                    for frag in accounted_fragments:
   450→                        if frag and stripped.startswith(frag.split('=')[0].split('(')[0].strip()):
   451→                            found = True
   452→                            break
   453→
   454→            if found:
   455→                accounted_lines += 1
   456→            else:
   457→                # Only flag as unaccounted if it's a significant line
   458→                if not stripped.startswith(('#', '"""', "'''", 'pass', 'else:', 'try:', 'except', 'finally:')):
   459→                    unaccounted.append((line_num, stripped[:80]))
   460→
   461→        # Coverage by provenance type
   462→        by_type_coverage = {}
   463→        for ptype in ProvenanceType:
   464→            count = len([r for r in self.records if r.source_type == ptype])
   465→            if count > 0:
   466→                by_type_coverage[ptype.value] = count
   467→
   468→        return {
   469→            "total_lines": total_lines,
   470→            "accounted_lines": accounted_lines,
   471→            "coverage_ratio": accounted_lines / total_lines if total_lines > 0 else 1.0,
   472→            "unaccounted_lines": unaccounted,
   473→            "by_type_coverage": by_type_coverage,
   474→        }

   476→    def compute_audit_coverage(self) -> Dict:
   477→        """
   478→        Compute the audit coverage of generated artifacts.
   479→
   480→        Audit Coverage = Auditable Artifacts / Generated Artifacts
   481→
   482→        An artifact is "auditable" if it has at least one provenance record.
   483→        An artifact without provenance is an "orphan".
   484→
   485→        This is the key metric for the Auditability property:
   486→            "A program is auditable if every generated artifact
   487→             can be traced to its originating specification
   488→             through a complete provenance chain."
   489→
   490→        Returns:
   491→            Dict with:
   492→                - total_artifacts: total number of generated artifacts
   493→                - auditable_artifacts: artifacts with at least one provenance record
   494→                - orphan_artifacts: list of artifacts without provenance
   495→                - audit_coverage: ratio of auditable to total (target: 1.0)
   496→                - by_type: breakdown by artifact type
   497→        """
   498→        total = len(self.artifacts)
   499→        if total == 0:
   500→            return {
   501→                "total_artifacts": 0,
   502→                "auditable_artifacts": 0,
   503→                "orphan_artifacts": [],
   504→                "audit_coverage": 1.0,
   505→                "by_type": {},
   506→            }
   507→
   508→        auditable = [a for a in self.artifacts if a.has_provenance]
   509→        orphans = [a for a in self.artifacts if a.is_orphan]
   510→
   511→        # Breakdown by type
   512→        by_type = {}
   513→        for atype in ArtifactType:
   514→            type_artifacts = [a for a in self.artifacts if a.artifact_type == atype]
   515→            type_auditable = [a for a in type_artifacts if a.has_provenance]
   516→            if type_artifacts:
   517→                by_type[atype.value] = {
   518→                    "total": len(type_artifacts),
   519→                    "auditable": len(type_auditable),
   520→                    "coverage": len(type_auditable) / len(type_artifacts),
   521→                }
   522→
   523→        return {
   524→            "total_artifacts": total,
   525→            "auditable_artifacts": len(auditable),
   526→            "orphan_artifacts": [{"name": a.name, "type": a.artifact_type.value, "source": a.source} for a in orphans],
   527→            "audit_coverage": len(auditable) / total,
   528→            "by_type": by_type,
   529→        }
   530→
   531→    def explain_coverage(self, generated_source: str, generated_tests: str = "") -> str:
   532→        """Generate a coverage report for the explicable compilation property."""
   533→        coverage = self.compute_explicability_coverage(generated_source, generated_tests)
   534→        lines = []
   535→        lines.append("=" * 70)
   536→        lines.append("EXPLICABILITY COVERAGE REPORT")
   537→        lines.append("=" * 70)
   538→        lines.append("")
   539→        lines.append(f"Total meaningful lines:   {coverage['total_lines']}")
   540→        lines.append(f"Lines with provenance:    {coverage['accounted_lines']}")
   541→        lines.append(f"Coverage ratio:           {coverage['coverage_ratio']:.3f}")
   542→        lines.append("")
   543→
   544→        if coverage['coverage_ratio'] >= 0.95:
   545→            lines.append("Status: PASS (coverage >= 95%)")
   546→        elif coverage['coverage_ratio'] >= 0.80:
   547→            lines.append("Status: PARTIAL (coverage 80-95%, investigation needed)")
   548→        else:
   549→            lines.append("Status: FAIL (coverage < 80%, provenance gaps detected)")
   550→
   551→        lines.append("")
   552→
   553→        if coverage['by_type_coverage']:
   554→            lines.append("Provenance by type:")
   555→            for ptype, count in sorted(coverage['by_type_coverage'].items(), key=lambda x: -x[1]):
   556→                lines.append(f"  {ptype:30s} {count:4d} records")
   557→            lines.append("")
   558→
   559→        if coverage['unaccounted_lines']:
   560→            lines.append(f"Unaccounted lines ({len(coverage['unaccounted_lines'])}):")
   561→            for line_num, text in coverage['unaccounted_lines'][:20]:
   562→                lines.append(f"  L{line_num:4d}: {text}")
   563→            if len(coverage['unaccounted_lines']) > 20:
   564→                lines.append(f"  ... and {len(coverage['unaccounted_lines']) - 20} more")
   565→        else:
   566→            lines.append("All lines have provenance coverage.")
   567→
   568→        lines.append("")
   569→        lines.append("=" * 70)
   570→        return '\n'.join(lines)
   571→
   572→    def audit(self, generated_source: str = "", generated_tests: str = "") -> str:
   573→        """
   574→        Generate a complete audit report.
   575→
   576→        This is the primary output of the audit system. It verifies the
   577→        Auditability property:
   578→
   579→            "A program is auditable if every generated artifact
   580→             can be traced to its originating specification
   581→             through a complete provenance chain."
   582→
   583→        The audit report contains:
   584→            1. Artifact summary (total, auditable, orphans)
   585→            2. Audit coverage ratio (target: 1.0)
   586→            3. Orphan artifact list (artifacts without provenance)
   587→            4. Audit status (VERIFIED / PARTIAL / FAILED)
   588→            5. Explicability coverage (line-level)
   589→            6. Breakdown by artifact type
   590→
   591→        Returns:
   592→            Human-readable audit report.
   593→        """
   594→        lines = []
   595→        lines.append("=" * 70)
   596→        lines.append("AICL AUDIT REPORT")
   597→        lines.append("=" * 70)
   598→        lines.append("")
   599→
   600→        # Artifact audit
   601→        audit = self.compute_audit_coverage()
   602→
   603→        lines.append("ARTIFACT AUDIT")
   604→        lines.append("-" * 70)
   605→        lines.append(f"  Generated artifacts:     {audit['total_artifacts']}")
   606→        lines.append(f"  Artifacts with provenance: {audit['auditable_artifacts']}")
   607→        lines.append(f"  Orphan artifacts:        {len(audit['orphan_artifacts'])}")
   608→        lines.append("")
   609→
   610→        coverage_pct = audit['audit_coverage'] * 100
   611→        lines.append(f"  Audit Coverage:          {coverage_pct:.2f}%")
   612→        lines.append("")
   613→
   614→        # Audit status
   615→        if audit['audit_coverage'] >= 1.0:
   616→            lines.append("  Audit Status: VERIFIED")
   617→        elif audit['audit_coverage'] >= 0.90:
   618→            lines.append("  Audit Status: PARTIAL (investigation needed)")
   619→        else:
   620→            lines.append("  Audit Status: FAILED (significant provenance gaps)")
   621→        lines.append("")
   622→
   623→        # Orphan artifacts
   624→        if audit['orphan_artifacts']:
   625→            lines.append("ORPHAN ARTIFACTS (no provenance)")
   626→            lines.append("-" * 70)
   627→            for orphan in audit['orphan_artifacts']:
   628→                lines.append(f"  [{orphan['type']:15s}] {orphan['name']}")
   629→                lines.append(f"    Source: {orphan['source']}")
   630→            lines.append("")
   631→
   632→        # Breakdown by artifact type
   633→        if audit['by_type']:
   634→            lines.append("COVERAGE BY ARTIFACT TYPE")
   635→            lines.append("-" * 70)
   636→            for atype, info in sorted(audit['by_type'].items(), key=lambda x: -x[1]['total']):
   637→                cov_pct = info['coverage'] * 100
   638→                status = "VERIFIED" if info['coverage'] >= 1.0 else "PARTIAL" if info['coverage'] >= 0.9 else "FAILED"
   639→                lines.append(f"  {atype:15s}  {info['auditable']:3d}/{info['total']:3d}  {cov_pct:6.2f}%  [{status}]")
   640→            lines.append("")
   641→
   642→        # Line-level explicability coverage
   643→        if generated_source:
   644→            exp_coverage = self.compute_explicability_coverage(generated_source, generated_tests)
   645→            lines.append("LINE-LEVEL EXPLICABILITY")
   646→            lines.append("-" * 70)
   647→            lines.append(f"  Total meaningful lines:  {exp_coverage['total_lines']}")
   648→            lines.append(f"  Lines with provenance:   {exp_coverage['accounted_lines']}")
   649→            lines.append(f"  Explicability coverage:  {exp_coverage['coverage_ratio']:.2%}")
   650→
   651→            if exp_coverage['coverage_ratio'] >= 0.95:
   652→                lines.append("  Line status: PASS (>= 95%)")
   653→            elif exp_coverage['coverage_ratio'] >= 0.80:
   654→                lines.append("  Line status: PARTIAL (80-95%)")
   655→            else:
   656→                lines.append("  Line status: FAIL (< 80%)")
   657→
   658→            if exp_coverage['unaccounted_lines']:
   659→                lines.append(f"  Unaccounted lines: {len(exp_coverage['unaccounted_lines'])}")
   660→            lines.append("")
   661→
   662→        # Provenance summary
   663→        lines.append("PROVENANCE SUMMARY")
   664→        lines.append("-" * 70)
   665→        total_records = len(self.records)
   666→        lines.append(f"  Total provenance records: {total_records}")
   667→
   668→        if total_records > 0:
   669→            avg_conf = sum(r.confidence for r in self.records) / total_records
   670→            lines.append(f"  Average confidence:       {avg_conf:.2f}")
   671→            deterministic = sum(1 for r in self.records if r.confidence >= 0.9)
   672→            lines.append(f"  Deterministic decisions:  {deterministic}/{total_records} ({100*deterministic//total_records}%)")
   673→
   674→            # By type summary
   675→            by_type = {}
   676→            for rec in self.records:
   677→                label = rec.source_type.value.replace("_", " ").title()
   678→                by_type[label] = by_type.get(label, 0) + 1
   679→
   680→            lines.append("  Provenance by type:")
   681→            for label, count in sorted(by_type.items(), key=lambda x: -x[1]):
   682→                lines.append(f"    {label:30s} {count:4d}")
   683→        lines.append("")
   684→
   685→        # Final verdict
   686→        lines.append("=" * 70)
   687→        if audit['audit_coverage'] >= 1.0 and (not generated_source or exp_coverage['coverage_ratio'] >= 0.95):
   688→            lines.append("AUDIT RESULT: PASSED")
   689→            lines.append("  Every generated artifact has provenance.")
   690→            lines.append("  Compilation is fully auditable.")
   691→        elif audit['audit_coverage'] >= 0.90:
   692→            lines.append("AUDIT RESULT: PARTIAL")
   693→            lines.append("  Most artifacts have provenance.")
   694→            lines.append("  Some orphan artifacts need investigation.")
   695→        else:
   696→            lines.append("AUDIT RESULT: FAILED")
   697→            lines.append("  Significant provenance gaps detected.")
   698→            lines.append("  Compilation is not fully auditable.")
   699→        lines.append("=" * 70)
   700→
   701→        return '\n'.join(lines)
   702→
   703→    def audit_passed(self, generated_source: str = "", generated_tests: str = "") -> bool:
   704→        """
   705→        Check whether the audit passes.
   706→
   707→        Audit passes if:
   708→            1. Audit coverage = 1.0 (no orphan artifacts)
   709→            2. Explicability coverage >= 0.95 (line-level)
   710→
   711→        Used by `aicl audit --strict` to determine exit code.
   712→        """
   713→        audit = self.compute_audit_coverage()
   714→        if audit['audit_coverage'] < 1.0:
   715→            return False
   716→
   717→        if generated_source:
   718→            exp = self.compute_explicability_coverage(generated_source, generated_tests)
   719→            if exp['coverage_ratio'] < 0.95:
   720→                return False
   721→
   722→        return True

   728→    def to_proof(
   729→        self,
   730→        source_path: str = "",
   731→        source_text: str = "",
   732→        generated_source: str = "",
   733→        generated_tests: str = "",
   734→        compiler_version: str = "0.6.0",
   735→        target_language: str = "python",
   736→    ) -> 'ProofOfOrigin':
   737→        """
   738→        Generate a Proof of Origin from this compilation.
   739→
   740→        The Proof of Origin is the central artifact of auditable compilation.
   741→        It is a self-contained, verifiable record that binds:
   742→            - The original AICL specification
   743→            - Every generated artifact
   744→            - Every provenance record
   745→            - The formal properties verified (No Orphan, Complete Coverage)
   746→            - Cryptographic hashes binding proof to generated code
   747→
   748→        If this file exists, `aicl explain` and `aicl audit` can be

   753→            source_text: The original AICL source text
   754→            generated_source: The generated program code
   755→            generated_tests: The generated test code
   756→            compiler_version: Version of the AICL compiler
   757→            target_language: Target language of the compilation
   758→
   759→        Returns:
   760→            ProofOfOrigin instance ready for serialization.
   761→        """
   762→        return ProofOfOrigin.from_compilation(
   763→            provenance=self,
   764→            source_path=source_path,
   765→            source_text=source_text,
   766→            generated_source=generated_source,
   767→            generated_tests=generated_tests,
   768→            compiler_version=compiler_version,
   769→            target_language=target_language,
   770→        )

   772→
   773→class ProofOfOrigin:
   774→    """
   775→    Proof of Origin — the central artifact of auditable compilation.
   776→
   777→    A Proof of Origin is a self-contained, verifiable record that proves:
   778→        1. Every generated artifact has a traceable provenance chain
   779→        2. No artifact exists without justification (No Orphan Property)
   780→        3. Audit coverage = 1.0 (Complete Coverage Property)
   781→        4. The proof is cryptographically bound to the generated code
   782→
   783→    Architecture:
   784→        aicl compile pong.aicl
   785→            ↓
   786→        pong.py              (executable program)
   787→        pong.aicl-proof      (proof of origin)
   788→
   789→        aicl explain --proof pong.aicl-proof    → human-readable explanation
   790→        aicl audit --proof pong.aicl-proof      → audit report
   791→        aicl proof pong.aicl-proof              → inspect/verify proof
   792→
   793→    The proof file is JSON, machine-readable, and contains ALL information
   794→    needed to reconstruct explain() and audit() without the compiler.
   795→
   796→    Formal Properties Verified:
   797→        - No Orphan Artifact Property:
   798→            Every generated artifact has at least one provenance chain.
   799→        - Complete Coverage Property:
   800→            Audit Coverage = Auditable Artifacts / Generated Artifacts = 1.0
   801→        - Hash Binding:
   802→            SHA-256(proof) is bound to SHA-256(generated code).
   803→    """
   804→
   805→    PROOF_FORMAT_VERSION = "1.0"
   806→
   807→    def __init__(self):
   808→        # Metadata
   809→        self.format_version: str = self.PROOF_FORMAT_VERSION
   810→        self.compiler_version: str = ""
   811→        self.timestamp: str = ""
   812→        self.source_path: str = ""
   813→        self.target_language: str = "python"
   814→
   815→        # Source binding
   816→        self.source_hash: str = ""          # SHA-256 of AICL source
   817→        self.source_text: str = ""          # Full AICL source text
   818→
   819→        # Code binding
   820→        self.program_hash: str = ""         # SHA-256 of generated program
   821→        self.test_hash: str = ""            # SHA-256 of generated tests
   822→        self.generated_source: str = ""     # Generated program code
   823→        self.generated_tests: str = ""      # Generated test code
   824→
   825→        # Provenance records
   826→        self.records: List[Dict[str, Any]] = []
   827→
   828→        # Generated artifacts
   829→        self.artifacts: List[Dict[str, Any]] = []
   830→
   831→        # Formal properties
   832→        self.formal_properties: Dict[str, Any] = {}
   833→
   834→        # Audit coverage
   835→        self.audit_coverage: Dict[str, Any] = {}
   836→
   837→        # Explicability coverage
   838→        self.explicability_coverage: Dict[str, Any] = {}
   839→
   840→    @classmethod
   841→    def from_compilation(
   842→        cls,
   843→        provenance: CompilationProvenance,
   844→        source_path: str = "",
   845→        source_text: str = "",
   846→        generated_source: str = "",
   847→        generated_tests: str = "",
   848→        compiler_version: str = "0.6.0",
   849→        target_language: str = "python",
   850→    ) -> 'ProofOfOrigin':
   851→        """
   852→        Create a Proof of Origin from a completed compilation.
   853→
   854→        This is the primary constructor — used by the compiler at the end
   855→        of a successful compilation to produce the proof artifact.
   856→        """
   857→        proof = cls()
   858→        proof.compiler_version = compiler_version
   859→        proof.timestamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
   860→        proof.source_path = source_path
   861→        proof.target_language = target_language
   862→
   863→        # Bind to source
   864→        proof.source_text = source_text
   865→        proof.source_hash = hashlib.sha256(source_text.encode('utf-8')).hexdigest()
   866→
   867→        # Bind to generated code
   868→        proof.generated_source = generated_source
   869→        proof.generated_tests = generated_tests
   870→        proof.program_hash = hashlib.sha256(generated_source.encode('utf-8')).hexdigest()
   871→        proof.test_hash = hashlib.sha256(generated_tests.encode('utf-8')).hexdigest()
   872→
   873→        # Serialize provenance records
   874→        proof.records = []
   875→        for rec in provenance.records:
   876→            proof.records.append({
   877→                "source_type": rec.source_type.value,
   878→                "source_location": rec.source_location,
   879→                "source_text": rec.source_text,
   880→                "resolution_path": rec.resolution_path,
   881→                "generated_code": rec.generated_code,
   882→                "confidence": rec.confidence,
   883→                "pattern_name": rec.pattern_name,
   884→                "template_name": rec.template_name,
   885→                "parameters": rec.parameters,
   886→                "artifact_names": rec.artifact_names,
   887→            })
   888→
   889→        # Serialize artifacts
   890→        proof.artifacts = []
   891→        for art in provenance.artifacts:
   892→            proof.artifacts.append({
   893→                "name": art.name,
   894→                "artifact_type": art.artifact_type.value,
   895→                "source": art.source,
   896→                "provenance_indices": art.provenance_indices,
   897→                "code_snippet": art.code_snippet,
   898→                "has_provenance": art.has_provenance,
   899→                "is_orphan": art.is_orphan,
   900→            })
   901→
   902→        # Compute and store formal properties
   903→        audit = provenance.compute_audit_coverage()
   904→        exp_coverage = provenance.compute_explicability_coverage(
   905→            generated_source, generated_tests
   906→        ) if generated_source else {"coverage_ratio": 1.0, "total_lines": 0, "accounted_lines": 0}
   907→
   908→        proof.formal_properties = {
   909→            "no_orphan_artifact_property": {
   910→                "statement": "Every generated artifact has at least one provenance chain.",
   911→                "verified": len(audit['orphan_artifacts']) == 0,
   912→                "orphan_count": len(audit['orphan_artifacts']),
   913→            },
   914→            "complete_coverage_property": {
   915→                "statement": "Audit Coverage = Auditable Artifacts / Generated Artifacts = 1.0",
   916→                "verified": audit['audit_coverage'] >= 1.0,
   917→                "coverage": audit['audit_coverage'],
   918→            },
   919→            "hash_binding_property": {
   920→                "statement": "Proof is cryptographically bound to generated code via SHA-256.",
   921→                "verified": True,  # Hashes are computed at proof creation time
   922→                "program_hash": proof.program_hash,
   923→                "test_hash": proof.test_hash,
   924→                "source_hash": proof.source_hash,
   925→            },
   926→        }
   927→
   928→        proof.audit_coverage = audit
   929→        proof.explicability_coverage = exp_coverage
   930→
   931→        return proof
   932→

  1036→    def verify(self, generated_source: str = "", generated_tests: str = "") -> Dict[str, Any]:
  1037→        """
  1038→        Verify the integrity of this proof.
  1039→
  1040→        Checks:
  1041→            1. Format version is supported
  1042→            2. Hash binding: proof hashes match the provided code
  1043→            3. No Orphan Property: every artifact has provenance
  1044→            4. Complete Coverage Property: audit coverage = 1.0
  1045→            5. Record-artifact linkage integrity
  1046→
  1047→        Returns:
  1048→            Dict with 'valid' (bool) and 'checks' (list of check results).
  1049→        """
  1050→        checks = []
  1051→
  1052→        # Check 1: Format version
  1053→        checks.append({
  1054→            "name": "format_version",
  1055→            "description": "Proof format version is supported",
  1056→            "passed": self.format_version == self.PROOF_FORMAT_VERSION,
  1057→        })
  1058→
  1059→        # Check 2: Hash binding — verify generated code matches proof
  1060→        if generated_source:
  1061→            actual_hash = hashlib.sha256(generated_source.encode('utf-8')).hexdigest()
  1062→            hash_match = actual_hash == self.program_hash
  1063→            checks.append({
  1064→                "name": "program_hash_binding",

  1146→    def explain(self, target: str = None) -> str:
  1147→        """
  1148→        Reconstruct a full explanation from this proof.
  1149→
  1150→        This proves that the .aicl-proof file contains ALL information
  1151→        needed to answer "Why did the compiler generate this line?"
  1152→        without requiring the compiler itself.
  1153→
  1154→        Args:
  1155→            target: Optional filter (behavior name, pattern name, etc.)
  1156→
  1157→        Returns:
  1158→            Human-readable explanation of all compilation decisions.
  1159→        """
  1160→        lines = []
  1161→        lines.append("=" * 70)
  1162→        lines.append("AICL PROOF OF ORIGIN — EXPLANATION")
  1163→        lines.append("=" * 70)
  1164→        lines.append("")
  1165→        lines.append(f"Source: {self.source_path or '<unknown>'}")
  1166→        lines.append(f"Compiled: {self.timestamp}")
  1167→        lines.append(f"Compiler: AICL v{self.compiler_version}")
  1168→        lines.append(f"Target: {self.target_language}")
  1169→        lines.append("")
  1170→
  1171→        # Summary
  1172→        total = len(self.records)
  1173→        type_counts = {}
  1174→        for rec in self.records:
  1175→            st = rec.get('source_type', 'unknown')
  1176→            type_counts[st] = type_counts.get(st, 0) + 1
  1177→
  1178→        lines.append(f"Total decisions: {total}")
  1179→        for st, count in sorted(type_counts.items(), key=lambda x: -x[1]):
  1180→            label = st.replace("_", " ").title()
  1181→            lines.append(f"  {label}: {count}")
  1182→        lines.append("")
  1183→
  1184→        # Confidence
  1185→        if total > 0:
  1186→            confidences = [r.get('confidence', 1.0) for r in self.records]
  1187→            avg_conf = sum(confidences) / len(confidences)
  1188→            fully_det = sum(1 for c in confidences if c >= 0.9)
  1189→            lines.append(f"Average confidence: {avg_conf:.2f}")
  1190→            lines.append(f"Fully deterministic: {fully_det}/{total} ({100*fully_det//max(total,1)}%)")
  1191→            lines.append("")
  1192→
  1193→        # Group by source type
  1194→        by_type: Dict[str, list] = {}
  1195→        for rec in self.records:
  1196→            st = rec.get('source_type', 'unknown')
  1197→            if st not in by_type:
  1198→                by_type[st] = []
  1199→            by_type[st].append(rec)
  1200→
  1201→        for source_type, records in by_type.items():
  1202→            type_label = source_type.replace("_", " ").title()
  1203→            lines.append("-" * 70)
  1204→            lines.append(f" {type_label} ({len(records)} decisions)")
  1205→            lines.append("-" * 70)
  1206→
  1207→            for i, rec in enumerate(records):
  1208→                # Filter by target if specified
  1209→                if target and target.lower() not in rec.get('source_location', '').lower() \
  1210→                        and target.lower() not in rec.get('pattern_name', '').lower():
  1211→                    continue
  1212→
  1213→                lines.append("")
  1214→                lines.append(f"  [{i+1}] {rec.get('source_location', '<unknown>')}")
  1215→                lines.append(f"  Source:  \"{rec.get('source_text', '')}\"")
  1216→                path = rec.get('resolution_path', [])
  1217→                lines.append(f"  Path:    {' → '.join(path)}")
  1218→
  1219→                if rec.get('pattern_name'):
  1220→                    lines.append(f"  Pattern: {rec['pattern_name']}")
  1221→                if rec.get('template_name'):
  1222→                    lines.append(f"  Template: {rec['template_name']}")
  1223→                if rec.get('parameters'):
  1224→                    params = rec['parameters']
  1225→                    params_str = ", ".join(f"{k}={v}" for k, v in params.items() if not k.startswith("#"))
  1226→                    if params_str:
  1227→                        lines.append(f"  Params:  {params_str}")
  1228→
  1229→                lines.append(f"  Confidence: {rec.get('confidence', 1.0):.2f}")
  1230→
  1231→                # Show generated code (truncated)
  1232→                code = rec.get('generated_code', '')
  1233→                code_lines = code.strip().split('\n') if code.strip() else []
  1234→                if len(code_lines) <= 4:
  1235→                    for cl in code_lines:
  1236→                        lines.append(f"  Code:    {cl}")
  1237→                elif code_lines:
  1238→                    for cl in code_lines[:3]:
  1239→                        lines.append(f"  Code:    {cl}")
  1240→                    lines.append(f"  Code:    ... ({len(code_lines) - 3} more lines)")
  1241→
  1242→                # Artifacts covered
  1243→                art_names = rec.get('artifact_names', [])
  1244→                if art_names:
  1245→                    lines.append(f"  Covers:  {', '.join(art_names)}")
  1246→
  1247→        lines.append("")
  1248→        lines.append("=" * 70)
  1249→        lines.append("END OF EXPLANATION (from Proof of Origin)")
  1250→        lines.append("=" * 70)
  1251→
  1252→        return '\n'.join(lines)
