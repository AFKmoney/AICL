   187→def code(text):
   188→    """Render code block. Escape XML special chars first."""
   189→    safe = (text
   190→        .replace('&', '&amp;')
   191→        .replace('<', '&lt;')
   192→        .replace('>', '&gt;')
   193→    )
   194→    lines = safe.strip().split('\n')
   195→    formatted = '<br/>'.join(lines)
   196→    return Paragraph(formatted, sCode)

   248→class AccentRule(Flowable):
   249→    """A thin horizontal accent-colored rule."""
   250→    def __init__(self, width, thickness=1.5):
   251→        Flowable.__init__(self)
   252→        self.width = width
   253→        self.thickness = thickness
   254→        self.height = thickness + 4
   255→    def draw(self):
   256→        self.canv.setStrokeColor(ACCENT)
   257→        self.canv.setLineWidth(self.thickness)
   258→        self.canv.line(0, 2, self.width, 2)
