     1→"""
     2→AICL Behavior Pattern Library - Deterministic Compilation Engine
     3→
     4→This module is the heart of AICL's deterministic compilation. It maps
     5→behavioral descriptions to concrete code through two mechanisms:
     6→
     7→1. PATTERN MATCHING: Action descriptions like "Update paddle position"
     8→   are matched to known patterns (MOVE, BROADCAST, CREATE, etc.)
     9→   which produce deterministic, parameterized code.
    10→
    11→2. SUB-LANGUAGE: When no pattern matches, users can write explicit
    12→   action statements using AICL's action sub-language:
    13→     assign x += direction * speed
    14→     clamp position between 0 and max
    15→     check score >= 10
    16→     send message to channel
    17→
    18→Together, these eliminate TODOs from compiled output.
    19→
    20→Design Principle:
    21→    The compiler must be 100% deterministic. Same input -> same output. Always.
    22→    AI-assisted code filling (--ai-fill) is a separate, optional mode.
    23→"""
    24→
    25→import re
    26→from dataclasses import dataclass, field
    27→from typing import List, Dict, Optional, Tuple, Callable
    28→from enum import Enum
    29→
    30→

    68→
    69→# =============================================================================
    70→# Pattern Match Result
    71→# =============================================================================
    72→
    73→@dataclass
    74→class PatternMatch:
    75→    """Result of matching an action description to a pattern."""
    76→    pattern_name: str
    77→    category: PatternCategory
    78→    confidence: float  # 0.0 to 1.0
    79→    parameters: Dict[str, str] = field(default_factory=dict)
    80→    code_template: str = ""  # Python code template with {param} placeholders
    81→
    82→
    83→# =============================================================================
    84→# Sub-Language Statement
    85→# =============================================================================
    86→
    87→class SubLangType(Enum):
    88→    ASSIGN = "assign"
    89→    CLAMP = "clamp"
    90→    CHECK = "check"
    91→    SEND = "send"
    92→    RETURN = "return"
    93→    CALL = "call"
    94→    LOG = "log"
    95→    RAISE = "raise"
    96→
    97→
    98→@dataclass
    99→class SubLangStatement:
   100→    """A parsed sub-language statement."""
   101→    statement_type: SubLangType
   102→    raw: str = ""
   103→    target: str = ""       # variable/attribute being operated on
   104→    operator: str = ""     # +=, -=, =, etc.
   105→    value: str = ""        # right-hand side expression
   106→    min_val: str = ""      # for clamp
   107→    max_val: str = ""      # for clamp
   108→    condition: str = ""    # for check
   109→    destination: str = ""  # for send
   110→    message: str = ""      # for send/log
   111→

   677→    def match(self, action_description: str) -> Optional[PatternMatch]:
   678→        """
   679→        Match an action description to a known pattern.
   680→
   681→        Uses keyword matching (not AI). The matching algorithm:
   682→        1. Normalize the description to lowercase
   683→        2. Check each pattern's keywords against the description
   684→        3. Return the best match (most keyword hits)
   685→
   686→        Returns None if no pattern matches with confidence > 0.3.
   687→        """
   688→        if not action_description or not action_description.strip():
   689→            return None
   690→
   691→        desc_lower = action_description.lower().strip()
   692→        best_match = None
   693→        best_score = 0.0
   694→
   695→        for pattern in self._patterns:
   696→            score = self._score_pattern(desc_lower, pattern['keywords'])
   697→            if score > best_score:
   698→                best_score = score
   699→                best_match = pattern
   700→
   701→        if best_match and best_score > 0.3:
   702→            # Extract parameters from context
   703→            params = self._extract_parameters(
   704→                action_description, best_match['parameters']
   705→            )
   706→            return PatternMatch(
   707→                pattern_name=best_match['name'],
   708→                category=best_match['category'],
   709→                confidence=best_score,
   710→                parameters=params,
   711→                code_template=best_match['template'],
   712→            )
   713→
   714→        return None

   716→    def _word_in_text(self, word: str, text: str) -> bool:
   717→        """Check if a word appears as a whole word in text (not as substring)."""
   718→        pattern = r'\b' + re.escape(word) + r'\b'
   719→        return bool(re.search(pattern, text))

   721→    def _score_pattern(self, desc_lower: str, keywords: List[str]) -> float:
   722→        """
   723→        Score how well a description matches a pattern's keywords.
   724→
   725→        Uses word-boundary matching to avoid false positives
   726→        (e.g., "play" matching inside "multiplayer").
   727→        """
   728→        if not keywords:
   729→            return 0.0
   730→
   731→        matches = 0
   732→        for kw in keywords:
   733→            kw_lower = kw.lower()
   734→            # For multi-word keywords, check if the phrase appears
   735→            if ' ' in kw_lower:
   736→                if kw_lower in desc_lower:
   737→                    word_count = len(kw_lower.split())
   738→                    matches += word_count
   739→                elif any(self._word_in_text(w, desc_lower) for w in kw_lower.split() if len(w) > 3):
   740→                    matches += 0.3
   741→            else:
   742→                # Single-word keyword: use word-boundary matching
   743→                if self._word_in_text(kw_lower, desc_lower):
   744→                    matches += 1
   745→
   746→        # Normalize: best single keyword match gets 1.0
   747→        max_possible = max(len(kw.split()) for kw in keywords)
   748→        score = min(matches / max(max_possible, 1), 1.0)
   749→
   750→        # Bonus for exact phrase match
   751→        for kw in keywords:
   752→            if ' ' in kw.lower() and kw.lower() in desc_lower:
   753→                score = min(score + 0.2, 1.0)
   754→                break
   755→            elif ' ' not in kw.lower() and self._word_in_text(kw.lower(), desc_lower):
   756→                score = min(score + 0.2, 1.0)
   757→                break
   758→
   759→        return score

   928→# =============================================================================
   929→# Sub-Language Parser
   930→# =============================================================================
   931→
   932→class SubLanguageParser:
   933→    """
   934→    Parser for AICL's action sub-language.
   935→
   936→    The sub-language allows explicit, deterministic specification of
   937→    behavior when no pattern matches. It provides a small set of
   938→    composable statements:
   939→
   940→        assign <target> <op> <expr>     # e.g., assign x += direction * speed
   941→        clamp <target> between <min> and <max>
   942→        check <condition>               # e.g., check score >= 10
   943→        send <message> to <destination>
   944→        return <expr>
   945→        call <method>(<args>)
   946→        log <message>
   947→        raise <exception>
   948→
   949→    Grammar (BNF):
   950→        <stmt_list>  ::= <stmt> (NEWLINE <stmt>)*
   951→        <stmt>       ::= <assign_stmt> | <clamp_stmt> | <check_stmt>
   952→                       | <send_stmt> | <return_stmt> | <call_stmt>
   953→                       | <log_stmt> | <raise_stmt>
   954→        <assign_stmt>::= "assign" <target> <assign_op> <expr>
   955→        <clamp_stmt> ::= "clamp" <target> "between" <expr> "and" <expr>
   956→        <check_stmt> ::= "check" <condition>
   957→        <send_stmt>  ::= "send" <expr> "to" <expr>
   958→        <return_stmt>::= "return" <expr>
   959→        <call_stmt>  ::= "call" <identifier> "(" <args>? ")"
   960→        <log_stmt>   ::= "log" <expr>
   961→        <raise_stmt> ::= "raise" <expr>
   962→        <assign_op>  ::= "=" | "+=" | "-=" | "*=" | "/="
   963→        <expr>       ::= <any Python expression>
   964→        <target>     ::= <identifier> ("." <identifier>)*
   965→    """
   966→
   967→    KEYWORDS = {'assign', 'clamp', 'check', 'send', 'return', 'call', 'log', 'raise'}
   968→
   969→    def parse(self, action_text: str) -> List[SubLangStatement]:
   970→        """
   971→        Parse action text into sub-language statements.
   972→
   973→        Returns a list of SubLangStatement. If the text doesn't contain
   974→        any sub-language keywords, returns an empty list (indicating the
   975→        text should be passed to the pattern matcher instead).
   976→        """
   977→        statements = []
   978→        lines = action_text.strip().split('\n')
   979→
   980→        for line in lines:
   981→            line = line.strip()
   982→            if not line:
   983→                continue
   984→
   985→            stmt = self._parse_line(line)
   986→            if stmt:
   987→                statements.append(stmt)
   988→
   989→        return statements
   990→
   991→    def _parse_line(self, line: str) -> Optional[SubLangStatement]:
   992→        """Parse a single line of sub-language."""
   993→        line_lower = line.lower()
   994→
   995→        if line_lower.startswith('assign '):
   996→            return self._parse_assign(line[7:].strip())
   997→        elif line_lower.startswith('clamp '):
   998→            return self._parse_clamp(line[6:].strip())
   999→        elif line_lower.startswith('check '):
  1000→            return self._parse_check(line[6:].strip())
  1001→        elif line_lower.startswith('send '):
  1002→            return self._parse_send(line[5:].strip())
  1003→        elif line_lower.startswith('return '):
  1004→            return self._parse_return(line[7:].strip())
  1005→        elif line_lower.startswith('call '):
  1006→            return self._parse_call(line[5:].strip())
  1007→        elif line_lower.startswith('log '):
  1008→            return self._parse_log(line[4:].strip())
  1009→        elif line_lower.startswith('raise '):
  1010→            return self._parse_raise(line[6:].strip())
  1011→
  1012→        return None
  1013→
  1014→    def _parse_assign(self, text: str) -> SubLangStatement:
  1015→        """Parse: assign target op expr"""
  1016→        # Try compound assignment operators first
  1017→        for op in ['+=', '-=', '*=', '/=', '//=', '%=', '**=', '=']:
  1018→            if op in text:
  1019→                parts = text.split(op, 1)
  1020→                if len(parts) == 2:
  1021→                    return SubLangStatement(
  1022→                        statement_type=SubLangType.ASSIGN,
  1023→                        raw=text,
  1024→                        target=parts[0].strip(),
  1025→                        operator=op,
  1026→                        value=parts[1].strip(),
  1027→                    )
  1028→        return SubLangStatement(
  1029→            statement_type=SubLangType.ASSIGN,
  1030→            raw=text,
  1031→            target=text,
  1032→            operator="=",
  1033→            value="None",
  1034→        )
  1035→
  1036→    def _parse_clamp(self, text: str) -> SubLangStatement:
  1037→        """Parse: clamp target between min and max"""
  1038→        # "paddle_position between 0 and screen_height"
  1039→        match = re.match(
  1040→            r'(\S+)\s+between\s+(.+?)\s+and\s+(.+)',
  1041→            text, re.IGNORECASE
  1042→        )
  1043→        if match:
  1044→            return SubLangStatement(
  1045→                statement_type=SubLangType.CLAMP,
  1046→                raw=text,
  1047→                target=match.group(1).strip(),
  1048→                min_val=match.group(2).strip(),
  1049→                max_val=match.group(3).strip(),
  1050→            )
  1051→        return SubLangStatement(
  1052→            statement_type=SubLangType.CLAMP,
  1053→            raw=text,
  1054→            target=text,
  1055→            min_val="0",
  1056→            max_val="MAX_VALUE",
  1057→        )
  1058→
  1059→    def _parse_check(self, text: str) -> SubLangStatement:
  1060→        """Parse: check condition"""
  1061→        return SubLangStatement(
  1062→            statement_type=SubLangType.CHECK,
  1063→            raw=text,
  1064→            condition=text,
  1065→        )
  1066→
  1067→    def _parse_send(self, text: str) -> SubLangStatement:
  1068→        """Parse: send message to destination"""
  1069→        match = re.match(
  1070→            r'(.+?)\s+to\s+(.+)',
  1071→            text, re.IGNORECASE
  1072→        )
  1073→        if match:
  1074→            return SubLangStatement(
  1075→                statement_type=SubLangType.SEND,
  1076→                raw=text,
  1077→                message=match.group(1).strip(),
  1078→                destination=match.group(2).strip(),
  1079→            )
  1080→        return SubLangStatement(
  1081→            statement_type=SubLangType.SEND,
  1082→            raw=text,
  1083→            message=text,
  1084→            destination="recipient",
  1085→        )
  1086→
  1087→    def _parse_return(self, text: str) -> SubLangStatement:
  1088→        """Parse: return expr"""
  1089→        return SubLangStatement(
  1090→            statement_type=SubLangType.RETURN,
  1091→            raw=text,
  1092→            value=text,
  1093→        )
  1094→
  1095→    def _parse_call(self, text: str) -> SubLangStatement:
  1096→        """Parse: call method(args)"""
  1097→        return SubLangStatement(
  1098→            statement_type=SubLangType.CALL,
  1099→            raw=text,
  1100→            target=text,
  1101→        )
  1102→
  1103→    def _parse_log(self, text: str) -> SubLangStatement:
  1104→        """Parse: log message"""
  1105→        return SubLangStatement(
  1106→            statement_type=SubLangType.LOG,
  1107→            raw=text,
  1108→            message=text,
  1109→        )
  1110→
  1111→    def _parse_raise(self, text: str) -> SubLangStatement:
  1112→        """Parse: raise exception"""
  1113→        return SubLangStatement(
  1114→            statement_type=SubLangType.RAISE,
  1115→            raw=text,
  1116→            message=text,
  1117→        )
  1118→
  1119→    def is_sub_language(self, action_text: str) -> bool:
  1120→        """
  1121→        Check if action text contains sub-language statements.
  1122→
  1123→        A sub-language statement must be a deliberate instruction, not a
  1124→        natural language description that happens to start with a keyword.
  1125→        We require the keyword to be followed by a valid sub-language
  1126→        construct (not just any text).
  1127→
  1128→        For 'assign': must contain an assignment operator (=, +=, -=, etc.)
  1129→        For 'clamp': must contain the word 'between'
  1130→        For 'check': must contain a comparison operator (>=, <=, ==, !=, >, <)
  1131→        For 'send': must contain the word 'to'
  1132→        For 'return', 'call', 'log', 'raise': always treated as sub-language
  1133→        """
  1134→        for line in action_text.strip().split('\n'):
  1135→            line_stripped = line.strip().lower()
  1136→            if not line_stripped:
  1137→                continue
  1138→
  1139→            if line_stripped.startswith('assign ') and any(op in line_stripped for op in ['+=', '-=', '*=', '/=', '//=', '%=', '**=', ' = ']):
  1140→                return True
  1141→            elif line_stripped.startswith('clamp ') and 'between' in line_stripped:
  1142→                return True
  1143→            elif line_stripped.startswith('check ') and any(op in line_stripped for op in ['>=', '<=', '==', '!=', '>', '<']):
  1144→                return True
  1145→            elif line_stripped.startswith('send ') and ' to ' in line_stripped:
  1146→                return True
  1147→            elif line_stripped.startswith('return '):
  1148→                return True
  1149→            elif line_stripped.startswith('call '):
  1150→                return True
  1151→            elif line_stripped.startswith('log '):
  1152→                return True
  1153→            elif line_stripped.startswith('raise '):
  1154→                return True
  1155→
  1156→        return False

  1163→class BehaviorCompiler:
  1164→    """
  1165→    Compiles AICL Behavior sections into deterministic Python code.
  1166→
  1167→    Resolution order:
  1168→    1. Sub-language statements (explicit, highest priority)
  1169→    2. Pattern matching (keyword-based, deterministic)
  1170→    3. Fallback: structured skeleton with semantic comments
  1171→       (NOT a bare TODO — a commented implementation outline)
  1172→    """
  1173→
  1174→    def __init__(self):
  1175→        self.pattern_library = BehaviorPatternLibrary()
  1176→        self.sub_lang_parser = SubLanguageParser()
  1177→
  1178→    def compile_action(
  1179→        self,
  1180→        action_description: str,
  1181→        context: Optional[Dict] = None,
  1182→    ) -> Tuple[str, bool]:
  1183→        """
  1184→        Compile an action description into Python code.
  1185→
  1186→        Args:
  1187→            action_description: The Action: text from an AICL Behavior
  1188→            context: Optional context (entity names, field names, etc.)
  1189→
  1190→        Returns:
  1191→            Tuple of (generated_code, is_fully_compiled)
  1192→            is_fully_compiled is True if no TODOs remain.
  1193→        """
  1194→        if not action_description or not action_description.strip():
  1195→            return "pass", True
  1196→
  1197→        context = context or {}
  1198→
  1199→        # 1. Try sub-language first
  1200→        if self.sub_lang_parser.is_sub_language(action_description):
  1201→            stmts = self.sub_lang_parser.parse(action_description)
  1202→            if stmts:
  1203→                code = self._compile_sub_language(stmts, context)
  1204→                return code, True
  1205→
  1206→        # 2. Try pattern matching
  1207→        match = self.pattern_library.match(action_description)
  1208→        if match and match.confidence > 0.3:
  1209→            # Merge context parameters over defaults
  1210→            params = match.parameters.copy()
  1211→            if context:
  1212→                for key, value in context.items():
  1213→                    if key in params:
  1214→                        params[key] = value
  1215→
  1216→            code = match.code_template.format(**params)
  1217→            return code, True
  1218→
  1219→        # 3. Fallback: structured skeleton with semantic guidance
  1220→        code = self._compile_fallback(action_description, context)
  1221→        return code, False
  1222→
  1223→    def _compile_sub_language(
  1224→        self, stmts: List[SubLangStatement], context: Dict
  1225→    ) -> str:
  1226→        """Compile sub-language statements to Python code."""
  1227→        lines = []
  1228→        for stmt in stmts:
  1229→            if stmt.statement_type == SubLangType.ASSIGN:
  1230→                if stmt.operator == '=':
  1231→                    lines.append(f"{stmt.target} = {stmt.value}")
  1232→                else:
  1233→                    lines.append(f"{stmt.target} {stmt.operator} {stmt.value}")
  1234→
  1235→            elif stmt.statement_type == SubLangType.CLAMP:
  1236→                lines.append(f"if {stmt.target} < {stmt.min_val}:")
  1237→                lines.append(f"    {stmt.target} = {stmt.min_val}")
  1238→                lines.append(f"elif {stmt.target} > {stmt.max_val}:")
  1239→                lines.append(f"    {stmt.target} = {stmt.max_val}")
  1240→
  1241→            elif stmt.statement_type == SubLangType.CHECK:
  1242→                lines.append(f"if not ({stmt.condition}):")
  1243→                lines.append(f'    self._logger.warning("Check failed: {stmt.condition}")')
  1244→                lines.append(f"    return False")
  1245→
  1246→            elif stmt.statement_type == SubLangType.SEND:
  1247→                lines.append(f"self._send_to({stmt.destination}, {stmt.message})")
  1248→                lines.append(f'self._logger.info(f"Sent to {stmt.destination}")')
  1249→
  1250→            elif stmt.statement_type == SubLangType.RETURN:
  1251→                lines.append(f"return {stmt.value}")
  1252→
  1253→            elif stmt.statement_type == SubLangType.CALL:
  1254→                lines.append(f"self.{stmt.target}")
  1255→
  1256→            elif stmt.statement_type == SubLangType.LOG:
  1257→                lines.append(f'self._logger.info(f"{stmt.message}")')
  1258→
  1259→            elif stmt.statement_type == SubLangType.RAISE:
  1260→                lines.append(f"raise {stmt.message}")
  1261→
  1262→        return '\n'.join(lines)
  1263→
  1264→    def _compile_fallback(
  1265→        self, action_description: str, context: Dict
  1266→    ) -> str:
  1267→        """
  1268→        Generate a structured fallback when no pattern matches.
  1269→
  1270→        This is NOT a bare TODO. It produces:
  1271→        - A comment describing the intent
  1272→        - A structured code skeleton
  1273→        - Parameter extraction hints
  1274→        - A logging statement
  1275→        """
  1276→        desc = action_description.strip()
  1277→        # Extract verb and object from the description
  1278→        words = desc.split()
  1279→        verb = words[0].lower() if words else "process"
  1280→        obj = ' '.join(words[1:]) if len(words) > 1 else "data"
  1281→
  1282→        lines = [
  1283→            f"# Intent: {desc}",
  1284→            f"# Verb: {verb}, Object: {obj}",
  1285→            f"self._logger.info(f\"Executing: {desc}\")",
  1286→        ]
  1287→
  1288→        # Generate verb-specific skeleton
  1289→        if verb in ("update", "change", "modify", "set"):
  1290→            lines.append(f"# Update: {obj}")
  1291→            lines.append(f"# target.{obj.replace(' ', '_')} = new_value")
  1292→            lines.append("pass")
  1293→        elif verb in ("create", "initialize", "setup", "init"):
  1294→            lines.append(f"# Create: {obj}")
  1295→            lines.append(f"# instance = {obj.title().replace(' ', '')}()")
  1296→            lines.append("pass")
  1297→        elif verb in ("delete", "remove", "destroy"):
  1298→            lines.append(f"# Delete: {obj}")
  1299→            lines.append("pass")
  1300→        elif verb in ("validate", "check", "verify"):
  1301→            lines.append(f"# Validate: {obj}")
  1302→            lines.append("return True")
  1303→        elif verb in ("display", "show", "render", "draw"):
  1304→            lines.append(f"# Display: {obj}")
  1305→            lines.append("pass")
  1306→        else:
  1307→            lines.append(f"# Process: {desc}")
  1308→            lines.append("pass")
  1309→
  1310→        return '\n'.join(lines)
