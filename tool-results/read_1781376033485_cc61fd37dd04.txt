    94→
    95→class Compiler:
    96→    """
    97→    AICL Multi-Stage Compiler v0.5.
    98→
    99→    Compiles AICL source code into deterministic, executable Python code
   100→    using behavior patterns and architecture templates.
   101→
   102→    Every generated artifact is tracked and must have provenance.
   103→    The audit system verifies: Audit Coverage = 1.0 (no orphans).
   104→
   105→    Usage:
   106→        compiler = Compiler()
   107→        result = compiler.compile(source_code)
   108→        if result.success:
   109→            print(result.source_code)
   110→    """
   111→
   112→    def __init__(self, target_language: str = "python"):
   113→        self.target_language = target_language
   114→        self.warnings: List[str] = []
   115→        self.errors: List[str] = []
   116→        self.stages_completed: List[str] = []
   117→        self._used_names: Set[str] = set()
   118→        self._behavior_compiler = BehaviorCompiler()
   119→        self._todo_count = 0
   120→        self._provenance = CompilationProvenance()
   121→
   122→    def compile(self, source: str) -> CompilationResult:
   123→        """Compile AICL source code into executable Python code."""
   124→        self.warnings = []
   125→        self.errors = []
   126→        self.stages_completed = []
   127→        self._used_names = set()
   128→        self._todo_count = 0
   129→        self._provenance = CompilationProvenance()
   130→
   131→        result = CompilationResult()
   132→
   133→        # Stage 1: Specification Parsing
   134→        program = self._stage_parsing(source)
   135→        if program is None:
   136→            result.success = False
   137→            result.errors = self.errors
   138→            return result
   139→        self.stages_completed.append("Specification Parsing")
   140→
   141→        # Stage 2: Architecture Validation
   142→        validation_warnings = self._stage_architecture_validation(program)
   143→        self.warnings.extend(validation_warnings)
   144→        self.stages_completed.append("Architecture Validation")
   145→
   146→        # Stage 3: Dependency Analysis
   147→        dep_order = self._stage_dependency_analysis(program)
   148→        self.stages_completed.append("Dependency Analysis")
   149→
   150→        # Stage 4: Risk Analysis
   151→        risk_pairs = self._stage_risk_analysis(program)
   152→        self.stages_completed.append("Risk Analysis")
   153→
   154→        # Stage 5: Recovery Synthesis
   155→        recovery_map = self._stage_recovery_synthesis(program, risk_pairs)
   156→        self.stages_completed.append("Recovery Synthesis")
   157→
   158→        # Stage 6: Code Generation
   159→        source_code = self._stage_code_generation(
   160→            program, dep_order, risk_pairs, recovery_map
   161→        )
   162→        self.stages_completed.append("Code Generation")
   163→
   164→        # Stage 7: Test Generation
   165→        test_code = self._stage_test_generation(program, dep_order)
   166→        self.stages_completed.append("Test Generation")
   167→
   168→        # Stage 8: Optimization
   169→        source_code = self._stage_optimization(source_code, program)
   170→        self.stages_completed.append("Optimization")
   171→
   172→        # Stage 9: Final Construction
   173→        source_code = self._stage_final_construction(source_code, program)
   174→        self.stages_completed.append("Final Construction")
   175→
   176→        # Build Architecture Tree string
   177→        tree = ArchitectureTree(program)
   178→        result.architecture_tree_str = tree.to_string()
   179→
   180→        result.success = True
   181→        result.source_code = source_code
   182→        result.test_code = test_code
   183→        result.warnings = self.warnings
   184→        result.errors = self.errors
   185→        result.stages_completed = self.stages_completed
   186→        result.todo_count = self._todo_count
   187→        result.fully_compiled = self._todo_count == 0
   188→        result.provenance = self._provenance
   189→
   190→        # Generate Proof of Origin — the central artifact
   191→        result.proof = self._provenance.to_proof(
   192→            source_text=source,
   193→            generated_source=source_code,
   194→            generated_tests=test_code,
   195→            compiler_version="1.0.0",
   196→            target_language=self.target_language,
   197→        )
   198→
   199→        return result
   200→
   201→    def compile_to_file(self, source: str, output_dir: str, source_path: str = "") -> CompilationResult:
   202→        """
   203→        Compile AICL source code and write the output to files.
   204→
   205→        Produces:
   206→            - main.py              (executable program)
   207→            - test_main.py         (generated tests)
   208→            - architecture_tree.txt (architecture visualization)
   209→            - main.aicl-proof      (Proof of Origin — first-class artifact)
   210→
   211→        The .aicl-proof file is a self-contained, verifiable record that
   212→        contains ALL information needed to reconstruct explain() and audit()
   213→        without the compiler. It is the central artifact of auditable compilation.
   214→        """
   215→        result = self.compile(source)
   216→
   217→        if result.success:
   218→            os.makedirs(output_dir, exist_ok=True)
   219→
   220→            main_path = os.path.join(output_dir, "main.py")
   221→            with open(main_path, 'w') as f:
   222→                f.write(result.source_code)
   223→
   224→            test_path = os.path.join(output_dir, "test_main.py")
   225→            with open(test_path, 'w') as f:
   226→                f.write(result.test_code)
   227→
   228→            tree_path = os.path.join(output_dir, "architecture_tree.txt")
   229→            with open(tree_path, 'w') as f:
   230→                f.write(result.architecture_tree_str)
   231→
   232→            # Write Proof of Origin — the central artifact
   233→            if result.proof:
   234→                # Update source_path in proof if provided
   235→                if source_path:
   236→                    result.proof.source_path = source_path
   237→                proof_path = os.path.join(output_dir, "main.aicl-proof")
   238→                result.proof.to_file(proof_path)
   239→
   240→        return result

   246→    def _stage_parsing(self, source: str) -> Optional[AICLProgram]:
   247→        """Parse the AICL source code into an AST."""
   248→        try:
   249→            parser = Parser(source)

   253→            self.errors.append(f"Parse error: {str(e)}")

   259→
   260→    def _stage_architecture_validation(self, program: AICLProgram) -> List[str]:
   261→        """Validate the architectural structure of the program."""
   262→        warnings = program.validate()
   263→
   264→        layer_names = [l.name for l in program.layers]
   265→        seen = set()
   266→        for name in layer_names:
   267→            if name.lower() in seen:
   268→                warnings.append(f"Duplicate layer name: {name}")
   269→            seen.add(name.lower())
   270→
   271→        entity_names = {e.name for e in program.entities}
   272→        for behavior in program.behaviors:
   273→            for inp in behavior.inputs:
   274→                if inp.param_type not in entity_names and inp.param_type not in TYPE_NAMES_SET:
   275→                    warnings.append(
   276→                        f"Behavior '{behavior.name}' references unknown type "
   277→                        f"'{inp.param_type}' in input"
   278→                    )
   279→
   280→        return warnings

   291→    # Stage 4: Risk Analysis
   292→    # =========================================================================
   293→
   294→    def _stage_risk_analysis(self, program: AICLProgram) -> List[Tuple[RiskSection, Optional[RecoverySection]]]:
   295→        """Analyze risks and pair them with recovery procedures."""
   296→        pairs = []
   297→        for i, risk in enumerate(program.risks):
   298→            recovery = None
   299→            if i < len(program.recoveries):
   300→                recovery = program.recoveries[i]
   301→            pairs.append((risk, recovery))
   302→
   303→        if len(program.recoveries) > len(program.risks):
   304→            for i in range(len(program.risks), len(program.recoveries)):
   305→                generic_risk = RiskSection(
   306→                    description=f"Unspecified risk (for recovery: {program.recoveries[i].description})"
   307→                )
   308→                pairs.append((generic_risk, program.recoveries[i]))
   309→
   310→        return pairs
   311→
   312→    # =========================================================================
   313→    # Stage 5: Recovery Synthesis
   314→    # =========================================================================
   315→
   316→    def _stage_recovery_synthesis(
   317→        self,
   318→        program: AICLProgram,
   319→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]]
   320→    ) -> Dict[str, str]:
   321→        """Synthesize recovery procedures for each risk."""
   322→        recovery_map = {}
   323→        for risk, recovery in risk_pairs:
   324→            if recovery:
   325→                recovery_map[risk.description] = recovery.description
   326→            else:
   327→                recovery_map[risk.description] = f"log and handle: {risk.description}"
   328→        return recovery_map

   334→    def _unique_name(self, base: str) -> str:
   335→        """Generate a unique Python identifier from a base name."""
   336→        clean = re.sub(r'[^a-zA-Z0-9_]', '_', base.lower())
   337→        clean = re.sub(r'_+', '_', clean).strip('_')
   338→        if not clean:
   339→            clean = "item"
   340→        if clean[0].isdigit():
   341→            clean = "f_" + clean
   342→
   343→        name = clean
   344→        counter = 2
   345→        while name in self._used_names:
   346→            name = f"{clean}_{counter}"
   347→            counter += 1
   348→
   349→        self._used_names.add(name)
   350→        return name

   352→    def _stage_code_generation(
   353→        self,
   354→        program: AICLProgram,
   355→        dep_order: List[LayerSection],
   356→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   357→        recovery_map: Dict[str, str]
   358→    ) -> str:
   359→        """Generate executable Python code from the AICL program."""
   360→        sections = []
   361→
   362→        goal_desc = program.goals[0].description if program.goals else "AICL Generated Application"
   363→
   364→        # Record architecture template provenance
   365→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   366→        self._provenance.record(
   367→            source_type=ProvenanceType.ARCHITECTURE_TEMPLATE,
   368→            source_location="Goal",
   369→            source_text=goal_desc,
   370→            resolution_path=[
   371→                "AICL Source",
   372→                f"Goal: {goal_desc}",
   373→                "ArchitectureTemplateMapper.match()",
   374→                f"Template: {template_name}",
   375→                f"Structure: {template_info['structure']}",
   376→                "Generated Code",
   377→            ],
   378→            generated_code=f"# Architecture template: {template_name} ({template_info['structure']})",
   379→            confidence=1.0,
   380→            template_name=template_name,
   381→            artifact_names=["architecture_template"],
   382→        )
   383→
   384→        sections.append('"""')
   385→        sections.append(f'AICL Generated Application')
   386→        sections.append(f'Goal: {goal_desc}')
   387→        sections.append(f'')
   388→        sections.append(f'Auto-generated by the AICL compiler v0.5.')
   389→        sections.append(f'Auditable compilation — every artifact has provenance.')
   390→        sections.append(f'"""')
   391→        sections.append('')
   392→
   393→        # Imports — with provenance
   394→        import_code = self._generate_imports(program)
   395→        sections.append(import_code)
   396→        sections.append('')
   397→
   398→        # Entity classes — with provenance (already implemented)
   399→        for entity in program.entities:
   400→            sections.append(self._generate_entity(entity))
   401→            sections.append('')
   402→
   403→        # Main application class
   404→        class_name = self._make_class_name(goal_desc)
   405→        sections.append(self._generate_application_class(
   406→            program, class_name, dep_order, risk_pairs, recovery_map
   407→        ))
   408→        sections.append('')
   409→
   410→        # Entry point — with provenance
   411→        sections.append(self._generate_entry_point(class_name))
   412→
   413→        return '\n'.join(sections)

   415→    def _generate_imports(self, program: AICLProgram) -> str:
   416→        """Generate import statements based on program content."""
   417→        imports = [
   418→            "import sys",
   419→            "import logging",
   420→            "import time",
   421→            "from typing import Optional, List, Dict, Any",
   422→        ]
   423→
   424→        if program.parallels:
   425→            imports.append("import threading")
   426→            imports.append("from concurrent.futures import ThreadPoolExecutor, as_completed")
   427→
   428→        needs_datetime = False
   429→        for entity in program.entities:
   430→            for f in entity.fields:
   431→                if f.field_type == "datetime":
   432→                    needs_datetime = True
   433→                    break
   434→
   435→        if needs_datetime:
   436→            imports.append("from datetime import datetime")
   437→
   438→        imports.append("from dataclasses import dataclass, field")
   439→
   440→        # Check if we need hashlib for security
   441→        if program.securities:
   442→            for sec in program.securities:
   443→                for action in sec.actions:
   444→                    if action.action_type == "encrypt":
   445→                        imports.append("import hashlib")
   446→                        imports.append("import base64")
   447→                        break
   448→
   449→        code = '\n'.join(imports)
   450→
   451→        # Register import artifact and record provenance
   452→        self._provenance.register_artifact(
   453→            name="imports",
   454→            artifact_type=ArtifactType.IMPORT_BLOCK,
   455→            source="Program structure",
   456→            code_snippet="import sys, logging, time, ...",
   457→        )
   458→        self._provenance.record(
   459→            source_type=ProvenanceType.IMPORT_GENERATION,
   460→            source_location="Program Structure",
   461→            source_text="Required imports based on program features",
   462→            resolution_path=[
   463→                "AICL Source",
   464→                "Program Analysis",
   465→                "Feature Detection (entities, security, parallel)",
   466→                "Import Resolution",
   467→                "Generated Code",
   468→            ],
   469→            generated_code=code,
   470→            confidence=1.0,
   471→            artifact_names=["imports"],
   472→        )
   473→
   474→        return code

   476→    def _generate_entity(self, entity: EntitySection) -> str:
   477→        """Generate a Python dataclass for an AICL Entity."""
   478→        class_name = self._make_class_name(entity.name)
   479→        artifact_name = f"Entity_{class_name}"
   480→        lines = []
   481→        lines.append('@dataclass')
   482→        lines.append(f'class {class_name}:')
   483→        lines.append(f'    """Entity: {entity.name}"""')
   484→
   485→        if not entity.fields:
   486→            lines.append('    pass')
   487→        else:
   488→            for f in entity.fields:
   489→                python_type = self._aicl_type_to_python(f.field_type)
   490→                default = 'None' if python_type not in ('int', 'float', 'bool') else None
   491→                if default:
   492→                    lines.append(f'    {f.name}: {python_type} = {default}')
   493→                else:
   494→                    lines.append(f'    {f.name}: {python_type} = 0')
   495→
   496→        code = '\n'.join(lines)
   497→
   498→        # Register artifact
   499→        self._provenance.register_artifact(
   500→            name=artifact_name,
   501→            artifact_type=ArtifactType.DATACLASS,
   502→            source=f"Entity {entity.name}",
   503→            code_snippet=f"class {class_name}:",
   504→        )
   505→
   506→        # Record entity generation provenance
   507→        fields_desc = ', '.join(f'{f.name}: {f.field_type}' for f in entity.fields) if entity.fields else 'no fields'
   508→        self._provenance.record(
   509→            source_type=ProvenanceType.ENTITY_GENERATION,
   510→            source_location=f"Entity {entity.name}",
   511→            source_text=fields_desc,
   512→            resolution_path=[
   513→                "AICL Source",
   514→                f"Entity {entity.name}",
   515→                f"Fields: {fields_desc}",
   516→                "Dataclass Generation",
   517→                "Generated Code",
   518→            ],
   519→            generated_code=code,
   520→            confidence=1.0,
   521→            parameters={"class_name": class_name, "fields": str(len(entity.fields))},
   522→            artifact_names=[artifact_name],
   523→        )
   524→
   525→        return code

   527→    def _generate_application_class(
   528→        self,
   529→        program: AICLProgram,
   530→        class_name: str,
   531→        dep_order: List[LayerSection],
   532→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   533→        recovery_map: Dict[str, str]
   534→    ) -> str:
   535→        """Generate the main application class."""
   536→        lines = []
   537→
   538→        # Determine architecture template from Goal
   539→        goal_desc = program.goals[0].description if program.goals else ""
   540→        template_name, template_info = ArchitectureTemplateMapper.match(goal_desc)
   541→
   542→        # Register the class artifact
   543→        class_artifact = f"Class_{class_name}"
   544→        self._provenance.register_artifact(
   545→            name=class_artifact,
   546→            artifact_type=ArtifactType.CLASS,
   547→            source=f"Goal: {goal_desc}",
   548→            code_snippet=f"class {class_name}:",
   549→        )
   550→
   551→        lines.append(f'class {class_name}:')
   552→        lines.append(f'    """')
   553→        if program.goals:
   554→            lines.append(f'    Goal: {program.goals[0].description}')
   555→        lines.append(f'    Architecture: {template_info["structure"]}')
   556→        lines.append(f'    """')
   557→        lines.append('')
   558→
   559→        # Class attributes - one per layer
   560→        lines.append('    # Layer instances')
   561→        layer_attr_map = {}
   562→        for layer in dep_order:
   563→            attr_name = self._unique_name(layer.name)
   564→            layer_attr_map[layer.name] = attr_name
   565→            lines.append(f'    _{attr_name}: Any = None')
   566→        lines.append('')
   567→
   568→        # Entity instances
   569→        if program.entities:
   570→            lines.append('    # Entity instances')
   571→            for entity in program.entities:
   572→                attr = self._unique_name(entity.name.lower())
   573→                lines.append(f'    _{attr}: Any = None')
   574→            lines.append('')
   575→
   576→        # Risks as class constants
   577→        if program.risks:
   578→            lines.append('    # Known risks')
   579→            for i, risk in enumerate(program.risks):
   580→                lines.append(f'    RISK_{i+1} = {risk.description!r}')
   581→            lines.append('')
   582→
   583→        # Constraints as class constants
   584→        if program.constraints:
   585→            lines.append('    # Constraints')
   586→            for i, con in enumerate(program.constraints):
   587→                lines.append(f'    CONSTRAINT_{i+1} = {con.description!r}')
   588→            lines.append('')
   589→
   590→        # Game/application state
   591→        lines.append('    # Application state')
   592→        lines.append('    _running: bool = False')
   593→        lines.append('    _initialized: bool = False')
   594→        if template_name == "game_loop":
   595→            lines.append('    _game_active: bool = False')
   596→            lines.append('    _frame_count: int = 0')

   601→            lines.append('    _offline_queue: List = field(default_factory=list)')
   602→        lines.append('')

   604→        # Record class structure provenance
   605→        class_structure_code = '\n'.join(lines)
   606→        self._provenance.record(
   607→            source_type=ProvenanceType.CLASS_STRUCTURE,
   608→            source_location=f"Class {class_name}",
   609→            source_text=f"Layers: {len(dep_order)}, Entities: {len(program.entities)}, Risks: {len(program.risks)}",
   610→            resolution_path=[
   611→                "AICL Source",
   612→                f"Goal: {goal_desc}",
   613→                f"Template: {template_name}",
   614→                "Class Structure Generation",
   615→                "Layer Attributes, Entity Attributes, State Variables",
   616→                "Generated Code",
   617→            ],
   618→            generated_code=class_structure_code,
   619→            confidence=1.0,
   620→            template_name=template_name,
   621→            artifact_names=[class_artifact],
   622→        )
   623→
   624→        # __init__ method
   625→        init_artifact = f"Method_{class_name}.__init__"
   626→        init_lines = []
   627→        init_lines.append('    def __init__(self):')
   628→        init_lines.append('        """Initialize the application and all layers."""')
   629→        init_lines.append('        self._logger = logging.getLogger(self.__class__.__name__)')
   630→        init_lines.append('        self._running = False')
   631→        init_lines.append('        self._initialized = False')
   632→        if template_name == "game_loop":
   633→            init_lines.append('        self._game_active = True')
   634→            init_lines.append('        self._frame_count = 0')
   635→            init_lines.append('        self._last_frame_time = time.time()')
   636→        elif template_name == "event_driven":
   637→            init_lines.append('        self._connected = False')
   638→            init_lines.append('        self._message_queue = []')
   639→            init_lines.append('        self._offline_queue = []')
   640→        init_lines.append('')
   641→
   642→        # Initialize entity instances
   643→        if program.entities:
   644→            init_lines.append('        # Initialize entities')
   645→            for entity in program.entities:
   646→                entity_class = self._make_class_name(entity.name)
   647→                attr = entity.name.lower()
   648→                init_lines.append(f'        self._{attr} = {entity_class}()')
   649→            init_lines.append('')
   650→
   651→        # Initialize each layer
   652→        for layer in dep_order:
   653→            attr_name = layer_attr_map[layer.name]
   654→            init_lines.append(f'        # Initialize {layer.name}')
   655→            init_lines.append(f'        self._{attr_name} = self._init_{attr_name}()')
   656→            init_lines.append('')
   657→
   658→        init_lines.append('        self._initialized = True')
   659→        init_lines.append('        self._logger.info("Application initialized successfully")')
   660→        init_lines.append('')
   661→
   662→        # Register __init__ artifact
   663→        self._provenance.register_artifact(
   664→            name=init_artifact,
   665→            artifact_type=ArtifactType.METHOD,
   666→            source=f"Class {class_name}",
   667→            code_snippet="def __init__(self):",
   668→        )
   669→
   670→        # Record __init__ provenance
   671→        init_code = '\n'.join(init_lines)
   672→        self._provenance.record(
   673→            source_type=ProvenanceType.CLASS_STRUCTURE,
   674→            source_location=f"Method __init__",
   675→            source_text=f"Initialize {len(dep_order)} layers, {len(program.entities)} entities",
   676→            resolution_path=[
   677→                "AICL Source",
   678→                f"Class {class_name}",
   679→                "Constructor Generation",
   680→                f"Template: {template_name}",
   681→                "Entity Init + Layer Init Calls",
   682→                "Generated Code",
   683→            ],
   684→            generated_code=init_code,
   685→            confidence=1.0,
   686→            template_name=template_name,
   687→            artifact_names=[init_artifact],
   688→        )
   689→
   690→        lines.extend(init_lines)
   691→
   692→        # Layer initialization methods
   693→        for layer in dep_order:
   694→            attr_name = layer_attr_map[layer.name]
   695→            lines.append(self._generate_layer_init(layer, attr_name, risk_pairs, recovery_map))
   696→            lines.append('')
   697→
   698→        # Pre-compute behavior method names (must be done before run method generation)
   699→        behavior_method_names = {}
   700→        for i, behavior in enumerate(program.behaviors):
   701→            method_name = self._unique_name(behavior.name).lower()
   702→            behavior_method_names[behavior.name] = method_name
   703→
   704→        # Run method — uses architecture template (with provenance)
   705→        lines.append(self._generate_run_method(program, dep_order, layer_attr_map, risk_pairs, recovery_map, template_name, behavior_method_names))
   706→        lines.append('')
   707→
   708→        # Behavior methods
   709→        for i, behavior in enumerate(program.behaviors):
   710→            method_name = behavior_method_names[behavior.name]
   711→            lines.append(self._generate_behavior_method(behavior, method_name, program))
   712→            lines.append('')
   713→
   714→        # Condition handlers

   717→            lines.append('')

   721→            lines.append(self._generate_event_handler(event, i + 1, program))

   732→            lines.append('')
   733→
   734→        # Security methods
   735→        if program.securities:
   736→            lines.append(self._generate_security_methods(program))
   737→            lines.append('')
   738→
   739→        # Risk handling method
   740→        lines.append(self._generate_risk_handler(risk_pairs, recovery_map, program))
   741→
   742→        # Helper methods
   743→        lines.append('')
   744→        lines.append(self._generate_helper_methods(program, template_name))

   748→    def _generate_layer_init(
   749→        self,
   750→        layer: LayerSection,
   751→        attr_name: str,
   752→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   753→        recovery_map: Dict[str, str]
   754→    ) -> str:
   755→        """Generate initialization method for a layer — deterministic, no TODOs."""
   756→        lines = []
   757→        lines.append(f'    def _init_{attr_name}(self) -> Any:')
   758→        lines.append(f'        """Initialize the {layer.name} layer."""')
   759→        lines.append(f'        self._logger.info("Initializing {layer.name}...")')
   760→        lines.append(f'        try:')
   761→
   762→        # Generate layer component as a dict
   763→        sublayer_names = [s.name for s in layer.sublayers]
   764→        lines.append(f'            component = {{')
   765→        lines.append(f'                "name": "{layer.name}",')
   766→        lines.append(f'                "active": True,')
   767→        lines.append(f'                "sublayers": {sublayer_names!r},')
   768→        lines.append(f'            }}')
   769→
   770→        # Generate sublayer initialization
   771→        for sublayer in layer.sublayers:
   772→            sub_attr = self._unique_name(sublayer.name)
   773→            lines.append(f'            # Sublayer: {sublayer.name}')
   774→            lines.append(f'            component["{sub_attr}"] = {{"name": "{sublayer.name}", "active": True}}')
   775→
   776→        lines.append(f'            self._logger.info("{layer.name} initialized successfully")')
   777→        lines.append(f'            return component')
   778→        lines.append(f'        except Exception as e:')
   779→        lines.append(f'            self._logger.error(f"Failed to initialize {layer.name}: {{e}}")')
   780→
   781→        # Add recovery from risk pairs that match this layer
   782→        for risk, recovery in risk_pairs:
   783→            if recovery and self._risk_matches_layer(risk.description, layer.name):
   784→                recovery_code, _ = self._behavior_compiler.compile_action(
   785→                    recovery.description,
   786→                    context={}
   787→                )
   788→                lines.append(f'            # Recovery: {recovery.description}')
   789→                lines.append(f'            self._logger.info("Applying recovery: {recovery.description}")')
   790→                for code_line in recovery_code.split('\n'):
   791→                    if code_line.strip():
   792→                        lines.append(f'            {code_line}')
   793→
   794→        lines.append(f'            return None')
   795→
   796→        code = '\n'.join(lines)
   797→
   798→        # Register artifact
   799→        artifact_name = f"Method__init_{attr_name}"
   800→        self._provenance.register_artifact(
   801→            name=artifact_name,
   802→            artifact_type=ArtifactType.METHOD,
   803→            source=f"Layer {layer.name}",
   804→            code_snippet=f"def _init_{attr_name}(self):",
   805→        )
   806→
   807→        # Record layer initialization provenance
   808→        self._provenance.record(
   809→            source_type=ProvenanceType.LAYER_INITIALIZATION,
   810→            source_location=f"Layer {layer.name}",
   811→            source_text=f"Sublayers: {', '.join(sublayer_names)}" if sublayer_names else "No sublayers",
   812→            resolution_path=[
   813→                "AICL Source",
   814→                f"Layer {layer.name}",
   815→                f"Sublayers: {sublayer_names}" if sublayer_names else "No sublayers",
   816→                "Component Dict Generation",
   817→                "Generated Code",
   818→            ],
   819→            generated_code=code,
   820→            confidence=1.0,
   821→            parameters={"layer_name": layer.name, "sublayers": str(len(sublayer_names))},
   822→            artifact_names=[artifact_name],
   823→        )
   824→
   825→        return code

   827→    def _generate_run_method(
   828→        self,
   829→        program: AICLProgram,
   830→        dep_order: List[LayerSection],
   831→        layer_attr_map: Dict[str, str],
   832→        risk_pairs: List[Tuple[RiskSection, Optional[RecoverySection]]],
   833→        recovery_map: Dict[str, str],
   834→        template_name: str,
   835→        behavior_method_names: Dict[str, str] = None,
   836→    ) -> str:
   837→        """Generate the main run method — uses architecture template."""
   838→        behavior_method_names = behavior_method_names or {}
   839→        lines = []
   840→        lines.append('    def run(self) -> None:')
   841→        lines.append('        """Run the application with full error handling."""')
   842→        lines.append('        self._logger.info("Starting application...")')
   843→        lines.append('        self._running = True')
   844→        lines.append('')
   845→
   846→        if template_name == "game_loop":
   847→            # Game loop: init → update → render → validate → loop
   848→            lines.append('        self._game_active = True')
   849→            lines.append('        try:')
   850→            lines.append('            while self._running and self._game_active:')
   851→            lines.append('                frame_start = time.time()')
   852→            lines.append('')
   853→            lines.append('                # Update phase')
   854→            for behavior in program.behaviors:
   855→                method_name = behavior_method_names.get(behavior.name, behavior.name.lower())
   856→                input_params = ', '.join(
   857→                    f'self._{inp.name.lower()}' if inp.name.lower() in
   858→                    {e.name.lower() for e in program.entities}
   859→                    else inp.name
   860→                    for inp in behavior.inputs
   861→                )
   862→                lines.append(f'                self._behavior_{method_name}({input_params})')
   863→            lines.append('')
   864→            lines.append('                # Check conditions')
   865→            for i, condition in enumerate(program.conditions):
   866→                lines.append(f'                self._handle_condition_{i+1}()')
   867→            lines.append('')
   868→            lines.append('                # Render phase')
   869→            for layer in dep_order:
   870→                attr_name = layer_attr_map[layer.name]
   871→                if 'render' in layer.name.lower() or 'window' in layer.name.lower() or 'display' in layer.name.lower():
   872→                    lines.append(f'                if self._{attr_name} is not None:')
   873→                    lines.append(f'                    self._render_frame()')
   874→            lines.append('')
   875→            lines.append('                # Frame rate control')
   876→            lines.append('                elapsed = time.time() - frame_start')
   877→            lines.append('                if elapsed < 1.0 / 60.0:')
   878→            lines.append('                    time.sleep(1.0 / 60.0 - elapsed)')
   879→            lines.append('                self._frame_count += 1')
   880→            lines.append('')
   881→
   882→        elif template_name == "event_driven":
   883→            # Event-driven: connect → listen → handle → respond
   884→            lines.append('        try:')
   885→            lines.append('            self._connect()')
   886→            lines.append('            self._connected = True')
   887→            lines.append('            self._logger.info("Connected to server")')
   888→            lines.append('')
   889→            lines.append('            while self._running and self._connected:')
   890→            lines.append('                # Process message queue')
   891→            lines.append('                while self._message_queue:')
   892→            lines.append('                    msg = self._message_queue.pop(0)')
   893→            lines.append('                    self._process_message(msg)')
   894→            lines.append('')
   895→            lines.append('                # Process offline queue if reconnected')
   896→            lines.append('                if self._offline_queue and self._connected:')
   897→            lines.append('                    self._sync_offline_messages()')
   898→            lines.append('')
   899→            lines.append('                # Wait for new messages')
   900→            lines.append('                time.sleep(0.1)')
   901→
   902→        else:
   903→            # Default: linear execution
   904→            lines.append('        try:')
   905→            for layer in dep_order:
   906→                attr_name = layer_attr_map[layer.name]
   907→                lines.append(f'            # Execute {layer.name}')
   908→                lines.append(f'            if self._{attr_name} is not None:')
   909→                lines.append(f'                self._logger.info("Running {layer.name}...")')
   910→                lines.append(f'            else:')
   911→                lines.append(f'                self._logger.warning("{layer.name} not initialized, skipping")')
   912→
   913→        # Validations
   914→        lines.append('')
   915→        lines.append('            # Run validations')
   916→        for i, validation in enumerate(program.validations):
   917→            lines.append(f'            self._validate_{i+1}()')
   918→
   919→        lines.append('')
   920→        lines.append('            self._logger.info("Application completed successfully")')
   921→        lines.append('')
   922→        lines.append('        except Exception as e:')
   923→        lines.append('            self._logger.error(f"Application error: {e}")')
   924→        lines.append('            self._handle_risk(e)')
   925→        lines.append('        finally:')
   926→        lines.append('            self._running = False')
   927→        lines.append('            self._logger.info("Application stopped")')
   928→
   929→        code = '\n'.join(lines)
   930→
   931→        # Register artifact
   932→        artifact_name = "Method_run"
   933→        self._provenance.register_artifact(
   934→            name=artifact_name,
   935→            artifact_type=ArtifactType.METHOD,
   936→            source=f"Architecture Template: {template_name}",
   937→            code_snippet="def run(self) -> None:",
   938→        )
   939→
   940→        # Record run method provenance
   941→        _, run_template_info = ArchitectureTemplateMapper.match(
   942→            program.goals[0].description if program.goals else ""
   943→        )
   944→        self._provenance.record(
   945→            source_type=ProvenanceType.RUN_METHOD,
   946→            source_location="Method run",
   947→            source_text=f"Template: {template_name}, Behaviors: {len(program.behaviors)}, Conditions: {len(program.conditions)}",
   948→            resolution_path=[
   949→                "AICL Source",
   950→                f"Goal → Template: {template_name}",
   951→                f"Structure: {run_template_info.get('structure', 'unknown')}",
   952→                "Run Method Generation",
   953→                "Generated Code",
   954→            ],
   955→            generated_code=code,
   956→            confidence=1.0,
   957→            template_name=template_name,
   958→            artifact_names=[artifact_name],
   959→        )
   960→
   961→        return code

   963→    def _generate_behavior_method(
   964→        self, behavior: BehaviorSection, method_name: str, program: AICLProgram
   965→    ) -> str:
   966→        """Generate a behavior method using pattern matching — no TODOs."""
   967→        lines = []
   968→
   969→        # Build parameter list from inputs
   970→        params = ["self"]
   971→        for inp in behavior.inputs:
   972→            # Check if input references an entity
   973→            entity_names = {e.name for e in program.entities}
   974→            if inp.name in entity_names:
   975→                params.append(f'{inp.name.lower()}: Any')
   976→            elif inp.param_type in entity_names:
   977→                params.append(f'{inp.name.lower()}: Any')
   978→            else:
   979→                params.append(f'{inp.name.lower()}: Any = None')
   980→
   981→        param_str = ', '.join(params)
   982→        lines.append(f'    def _behavior_{method_name}({param_str}) -> Any:')
   983→        lines.append(f'        """Behavior: {behavior.name}"""')
   984→
   985→        # Compile the action using the Behavior Compiler
   986→        entity_context = {}
   987→        for inp in behavior.inputs:
   988→            entity_names = {e.name for e in program.entities}
   989→            if inp.name in entity_names:
   990→                entity_context["entity"] = inp.name.lower()
   991→            elif inp.param_type in entity_names:
   992→                entity_context["entity"] = inp.param_type.lower()
   993→
   994→        action_code, fully_compiled = self._behavior_compiler.compile_action(
   995→            behavior.action, context=entity_context
   996→        )
   997→
   998→        if not fully_compiled:
   999→            self._todo_count += 1
  1000→
  1001→        # Register artifact
  1002→        artifact_name = f"Method__behavior_{method_name}"
  1003→        self._provenance.register_artifact(
  1004→            name=artifact_name,
  1005→            artifact_type=ArtifactType.METHOD,
  1006→            source=f"Behavior {behavior.name}",
  1007→            code_snippet=f"def _behavior_{method_name}(...):",
  1008→        )
  1009→
  1010→        # Record provenance
  1011→        self._record_behavior_provenance(behavior, action_code, fully_compiled, entity_context, artifact_name)
  1012→
  1013→        # Indent the compiled code
  1014→        for code_line in action_code.split('\n'):
  1015→            lines.append(f'        {code_line}')
  1016→
  1017→        return '\n'.join(lines)

  1080→    def _generate_condition_handler(
  1081→        self, condition: ConditionSection, index: int, program: AICLProgram
  1082→    ) -> str:
  1083→        """Generate a condition handler using pattern matching — no TODOs."""
  1084→        lines = []
  1085→        lines.append(f'    def _handle_condition_{index}(self) -> None:')
  1086→        when_desc = condition.when_clause or "unknown condition"
  1087→        then_desc = condition.then_clause or "unknown action"
  1088→        lines.append(f'        """Condition: WHEN {when_desc} THEN {then_desc}"""')
  1089→
  1090→        # Compile the WHEN clause
  1091→        when_code, when_compiled = self._behavior_compiler.compile_action(
  1092→            f"check {when_desc}",
  1093→            context={}
  1094→        )
  1095→
  1096→        # Compile the THEN clause
  1097→        then_code, then_compiled = self._behavior_compiler.compile_action(
  1098→            then_desc,
  1099→            context={}
  1100→        )
  1101→
  1102→        if not when_compiled or not then_compiled:
  1103→            self._todo_count += 1
  1104→
  1105→        # Generate the condition check
  1106→        when_lower = when_desc.lower()
