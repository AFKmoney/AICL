    51→// ============================================================
    52→// AICL Syntax Highlighter
    53→// ============================================================
    54→const AICL_KEYWORDS = [
    55→  'Goal', 'Layer', 'Sublayer', 'Validation', 'Risk', 'Recovery',
    56→  'Constraint', 'Entity', 'Behavior', 'Input', 'Output', 'Action',
    57→  'Condition', 'When', 'Then', 'Event', 'On', 'Parallel', 'Optimize',
    58→  'Priority', 'Learn', 'Adapt', 'Based', 'Security', 'Encrypt',
    59→  'Protect', 'Native', 'Import'
    60→];
    61→
    62→const AICL_TYPES = [
    63→  'string', 'integer', 'float', 'boolean', 'datetime',
    64→  'list', 'dict', 'set', 'any', 'void', 'bytes'
    65→];
    66→
    67→function highlightAICL(code: string): string {
    68→  const lines = code.split('\n');
    69→  return lines.map(line => {
    70→    // Comment
    71→    const commentIdx = line.indexOf('#');
    72→    if (commentIdx !== -1) {
    73→      const before = line.substring(0, commentIdx);
    74→      const comment = line.substring(commentIdx);
    75→      return highlightLine(before) + `<span class="aicl-comment">${escapeHtml(comment)}</span>`;
    76→    }
    77→    return highlightLine(line);
    78→  }).join('\n');
    79→}
    80→
    81→function highlightLine(line: string): string {
    82→  let result = escapeHtml(line);
    83→
    84→  // Keywords at start of line or after whitespace (word boundary)
    85→  for (const kw of AICL_KEYWORDS) {
    86→    const regex = new RegExp(`(^|\\s)(${kw})(?=[:\\s]|$)`, 'g');
    87→    result = result.replace(regex, `$1<span class="aicl-keyword">${kw}</span>`);
    88→  }
    89→
    90→  // Types
    91→  for (const t of AICL_TYPES) {
    92→    const regex = new RegExp(`(\\s|^)(${t})(?=\\s|:|$)`, 'g');
    93→    result = result.replace(regex, `$1<span class="aicl-type">${t}</span>`);
    94→  }
    95→
    96→  return result;
    97→}
    98→
    99→function escapeHtml(str: string): string {
   100→  return str
   101→    .replace(/&/g, '&amp;')
   102→    .replace(/</g, '&lt;')
   103→    .replace(/>/g, '&gt;')
   104→    .replace(/"/g, '&quot;');
   105→}

   639→  // --- Editor key handling ---
   640→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   641→    const ta = e.currentTarget;
   642→
   643→    // Tab → insert 4 spaces
   644→    if (e.key === 'Tab') {
   645→      e.preventDefault();
   646→      const start = ta.selectionStart;
   647→      const end = ta.selectionEnd;
   648→      const content = activeFile.content;
   649→      if (e.shiftKey) {
   650→        // Shift+Tab: remove 4 spaces
   651→        const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   652→        const line = content.substring(lineStart, start);
   653→        if (line.startsWith('    ')) {
   654→          const newContent = content.substring(0, lineStart) + content.substring(lineStart + 4);
   655→          updateFileContent(activeFileId, newContent);
   656→          setTimeout(() => {
   657→            ta.selectionStart = ta.selectionEnd = start - 4;
   658→          }, 0);
   659→        }
   660→      } else {
   661→        const newContent = content.substring(0, start) + '    ' + content.substring(end);
   662→        updateFileContent(activeFileId, newContent);
   663→        setTimeout(() => {
   664→          ta.selectionStart = ta.selectionEnd = start + 4;
   665→        }, 0);
   666→      }
   667→      return;
   668→    }
   669→
   670→    // Enter → auto-indent
   671→    if (e.key === 'Enter') {
   672→      e.preventDefault();
   673→      const start = ta.selectionStart;
   674→      const content = activeFile.content;
   675→      const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   676→      const currentLine = content.substring(lineStart, start);
   677→      const indent = currentLine.match(/^\s*/)?.[0] || '';
   678→
   679→      // If the line ends with a colon, add extra indent
   680→      const trimmedLine = currentLine.trimEnd();
   681→      const needsIndent = trimmedLine.endsWith(':');
   682→
   683→      const newIndent = needsIndent ? indent + '    ' : indent;
   684→      const newContent = content.substring(0, start) + '\n' + newIndent + content.substring(ta.selectionEnd);
   685→      updateFileContent(activeFileId, newContent);
   686→      setTimeout(() => {
   687→        ta.selectionStart = ta.selectionEnd = start + 1 + newIndent.length;
   688→      }, 0);
   689→      return;
   690→    }
   691→
   692→    // Ctrl+S → Save
   693→    if ((e.ctrlKey || e.metaKey) && e.key === 's') {
   694→      e.preventDefault();
   695→      saveFile();
   696→      return;
   697→    }
   698→
   699→    // Ctrl+F → Find
   700→    if ((e.ctrlKey || e.metaKey) && e.key === 'f') {
   701→      e.preventDefault();
   702→      setFindOpen(true);
   703→      setTimeout(() => findInputRef.current?.focus(), 50);
   704→      return;
   705→    }
   706→
   707→    // Ctrl+H → Replace
   708→    if ((e.ctrlKey || e.metaKey) && e.key === 'h') {
   709→      e.preventDefault();
   710→      setFindOpen(true);
   711→      return;
   712→    }
   713→
   714→    // Escape → Close find
   715→    if (e.key === 'Escape') {
   716→      setFindOpen(false);
   717→    }
   718→  }, [activeFile, activeFileId, updateFileContent, saveFile]);
