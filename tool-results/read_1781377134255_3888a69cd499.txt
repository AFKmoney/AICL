    59→// AICL Syntax Highlighter
    60→// ============================================================
    61→const AICL_KEYWORDS = [
    62→  'Goal', 'Layer', 'Sublayer', 'Validation', 'Risk', 'Recovery',
    63→  'Constraint', 'Entity', 'Behavior', 'Input', 'Output', 'Action',
    64→  'Condition', 'When', 'Then', 'Event', 'On', 'Parallel', 'Optimize',
    65→  'Priority', 'Learn', 'Adapt', 'Based', 'Security', 'Encrypt',
    66→  'Protect', 'Native', 'Import'
    67→];
    68→
    69→const AICL_TYPES = [
    70→  'string', 'integer', 'float', 'boolean', 'datetime',
    71→  'list', 'dict', 'set', 'any', 'void', 'bytes'
    72→];
    73→
    74→function highlightAICL(code: string): string {
    75→  const lines = code.split('\n');
    76→  return lines.map(line => {
    77→    // Comment
    78→    const commentIdx = line.indexOf('#');
    79→    if (commentIdx !== -1) {
    80→      const before = line.substring(0, commentIdx);
    81→      const comment = line.substring(commentIdx);
    82→      return highlightLine(before) + `<span class="aicl-comment">${escapeHtml(comment)}</span>`;
    83→    }
    84→    return highlightLine(line);
    85→  }).join('\n');
    86→}
    87→
    88→function highlightLine(line: string): string {
    89→  let result = escapeHtml(line);
    90→
    91→  // Keywords at start of line or after whitespace (word boundary)
    92→  for (const kw of AICL_KEYWORDS) {
    93→    const regex = new RegExp(`(^|\\s)(${kw})(?=[:\\s]|$)`, 'g');
    94→    result = result.replace(regex, `$1<span class="aicl-keyword">${kw}</span>`);
    95→  }
    96→
    97→  // Types
    98→  for (const t of AICL_TYPES) {
    99→    const regex = new RegExp(`(\\s|^)(${t})(?=\\s|:|$)`, 'g');
   100→    result = result.replace(regex, `$1<span class="aicl-type">${t}</span>`);
   101→  }
   102→
   103→  return result;
   104→}
   105→
   106→function escapeHtml(str: string): string {
   107→  return str
   108→    .replace(/&/g, '&amp;')
   109→    .replace(/</g, '&lt;')
   110→    .replace(/>/g, '&gt;')
   111→    .replace(/"/g, '&quot;');
   112→}

   757→    const ta = e.currentTarget;
   758→
   759→    // Tab → insert 4 spaces
   760→    if (e.key === 'Tab') {
   761→      e.preventDefault();
   762→      const start = ta.selectionStart;
   763→      const end = ta.selectionEnd;
   764→      const content = activeFile.content;
   765→      if (e.shiftKey) {
   766→        // Shift+Tab: remove 4 spaces
   767→        const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   768→        const line = content.substring(lineStart, start);

   772→          setTimeout(() => {

   775→        }

   779→        setTimeout(() => {
   780→          ta.selectionStart = ta.selectionEnd = start + 4;
   781→        }, 0);
   782→      }
   783→      return;
   784→    }
   785→
   786→    // Enter → auto-indent
   787→    if (e.key === 'Enter') {
   788→      e.preventDefault();
   789→      const start = ta.selectionStart;
   790→      const content = activeFile.content;
   791→      const lineStart = content.lastIndexOf('\n', start - 1) + 1;
   792→      const currentLine = content.substring(lineStart, start);
   793→      const indent = currentLine.match(/^\s*/)?.[0] || '';
   794→
   795→      // If the line ends with a colon, add extra indent
   796→      const trimmedLine = currentLine.trimEnd();
   797→      const needsIndent = trimmedLine.endsWith(':');
   798→
   799→      const newIndent = needsIndent ? indent + '    ' : indent;
   800→      const newContent = content.substring(0, start) + '\n' + newIndent + content.substring(ta.selectionEnd);
   801→      updateFileContent(activeFileId, newContent);
   802→      setTimeout(() => {
   803→        ta.selectionStart = ta.selectionEnd = start + 1 + newIndent.length;
   804→      }, 0);
   805→      return;
   806→    }
   807→
   808→    // Ctrl+S → Save
   809→    if ((e.ctrlKey || e.metaKey) && e.key === 's') {
   810→      e.preventDefault();
   811→      saveFile();
   812→      return;
   813→    }
   814→
   815→    // Ctrl+F → Find

   818→      setFindOpen(true);
